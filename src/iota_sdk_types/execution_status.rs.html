9/// The status of an executed Transaction
10///
11/// # BCS
12///
13/// The BCS serialized form of this type is specified in
14/// [`bcs-schema.abnf`](https://github.com/iotaledger/iota-rust-sdk/blob/develop/crates/iota-sdk-types/bcs-schema.abnf).
15#[derive(Clone, Debug, Eq, PartialEq)]
16#[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
17#[cfg_attr(feature = "bcs-schema", derive(iota_bcs_schema::BcsSchema))]
18#[non_exhaustive]
19pub enum ExecutionStatus {
20    /// The Transaction successfully executed.
21    Success,
22    /// The Transaction didn't execute successfully.
23    ///
24    /// Failed transactions are still committed to the blockchain but any
25    /// intended effects are rolled back to prior to this transaction
26    /// executing with the caveat that gas objects are still smashed and gas
27    /// usage is still charged.
28    Failure {
29        /// The error encountered during execution.
30        error: ExecutionError,
31        /// The command, if any, during which the error occurred.
32        #[cfg_attr(feature = "proptest", map(|x: Option<u16>| x.map(Into::into)))]
33        command: Option<u64>,
34    },
35}
36
37impl ExecutionStatus {
38    crate::def_is!(Success, Failure);
39
40    pub fn new_failure(error: ExecutionError, command: Option<u64>) -> Self {
41        Self::Failure { error, command }
42    }
43
44    pub fn unwrap(&self) {
45        match self {
46            Self::Success => {}
47            Self::Failure { .. } => {
48                panic!("Unable to unwrap() on {self:?}");
49            }
50        }
51    }
52
53    pub fn unwrap_err(self) -> (ExecutionError, Option<u64>) {
54        match self {
55            Self::Success => {
56                panic!("Unable to unwrap_err() on {self:?}");
57            }
58            Self::Failure { error, command } => (error, command),
59        }
60    }
61
62    /// The error encountered during execution.
63    pub fn error(&self) -> Option<&ExecutionError> {
64        if let Self::Failure { error, .. } = self {
65            Some(error)
66        } else {
67            None
68        }
69    }
70
71    /// The command, if any, during which the error occurred.
72    pub fn error_command(&self) -> Option<u64> {
73        if let Self::Failure { command, .. } = self {
74            *command
75        } else {
76            None
77        }
78    }
79}

81impl crate::TreeDisplay for ExecutionStatus {
82    fn fmt_tree(&self, w: &mut crate::TreeWriter<'_, '_>) -> std::fmt::Result {
83        w.enum_name("Execution Status");
84        match self {
85            ExecutionStatus::Success => w.header("Success"),
86            ExecutionStatus::Failure { error, command } => {
87                w.header("Failure")?;
88                w.leaf("Error", error, false)?;
89                w.option_leaf("Command", command, true)
90            }
91        }
92    }
93}

97fn display_move_location_opt(location: &Option<MoveLocation>) -> impl core::fmt::Display + '_ {
98    struct W<'a>(&'a Option<MoveLocation>);
99    impl core::fmt::Display for W<'_> {
100        fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
101            match &self.0 {
102                None => write!(f, "UNKNOWN"),

110fn display_congested_objects(objects: &[ObjectId]) -> impl core::fmt::Display + '_ {
111    struct W<'a>(&'a [ObjectId]);
112    impl core::fmt::Display for W<'_> {
113        fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
114            let mut iter = self.0.iter();
115            if let Some(first) = iter.next() {
116                write!(f, "{first}")?;
117                for obj in iter {
118                    write!(f, ", {obj}")?;
119                }
120            }
121            Ok(())
122        }
123    }
124    W(objects)
125}

432/// Location in move bytecode where an error occurred
433///
434/// # BCS
435///
436/// The BCS serialized form of this type is specified in
437/// [`bcs-schema.abnf`](https://github.com/iotaledger/iota-rust-sdk/blob/develop/crates/iota-sdk-types/bcs-schema.abnf).
438#[derive(Clone, Debug, Eq, Hash, PartialEq)]
439#[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]
440#[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
441#[cfg_attr(feature = "bcs-schema", derive(iota_bcs_schema::BcsSchema))]
442pub struct MoveLocation {
443    /// The package id
444    pub package: ObjectId,
445    /// The module name
446    pub module: Identifier,
447    /// The function index
448    pub function: u16,
449    /// Index into the code stream for a jump. The offset is relative to the
450    /// beginning of the instruction stream.
451    pub instruction: u16,
452    /// The name of the function if available
453    pub function_name: Option<Identifier>,
454}
455
456impl core::fmt::Display for MoveLocation {
457    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> std::fmt::Result {
458        let Self {
459            package,
460            module,
461            function,
462            instruction,
463            function_name,
464        } = self;
465        if let Some(fname) = function_name {
466            write!(
467                f,
468                "{package}::{module}::{fname} (function index {function}) at offset {instruction}"
469            )
470        } else {
471            write!(
472                f,
473                "{package}::{module} in function definition {function} at offset {instruction}"
474            )
475        }
476    }
477}

619        IncompatibleUpgrade,

621        UnknownUpgradePolicy,
622        PackageIdDoesNotMatch,
623    );

654#[cfg(feature = "serde")]
655#[cfg_attr(doc_cfg, doc(cfg(feature = "serde")))]
656mod serialization {
657    use serde::{Deserialize, Deserializer, Serialize, Serializer};
658
659    use super::*;
660
661    #[derive(serde::Deserialize, serde::Serialize)]
662    #[serde(rename = "ExecutionStatus")]
663    struct ReadableExecutionStatus {
664        success: bool,
665        #[serde(skip_serializing_if = "Option::is_none")]
666        status: Option<FailureStatus>,
667    }
668
669    #[derive(serde::Deserialize, serde::Serialize)]
670    struct FailureStatus {
671        error: ExecutionError,
672        #[serde(skip_serializing_if = "Option::is_none")]
673        command: Option<u16>,
674    }
675
676    #[derive(serde::Deserialize, serde::Serialize)]
677    #[serde(rename = "ExecutionStatus")]
678    enum BinaryExecutionStatus {
679        Success,
680        Failure {
681            error: ExecutionError,
682            command: Option<u64>,
683        },
684    }
685
686    impl Serialize for ExecutionStatus {
687        fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
688        where
689            S: Serializer,
690        {
691            if serializer.is_human_readable() {
692                let readable = match self.clone() {
693                    ExecutionStatus::Success => ReadableExecutionStatus {
694                        success: true,
695                        status: None,
696                    },
697                    ExecutionStatus::Failure { error, command } => ReadableExecutionStatus {
698                        success: false,
699                        status: Some(FailureStatus {
700                            error,
701                            command: command.map(|c| c as u16),
702                        }),
703                    },
704                };
705                readable.serialize(serializer)
706            } else {
707                let binary = match self.clone() {
708                    ExecutionStatus::Success => BinaryExecutionStatus::Success,
709                    ExecutionStatus::Failure { error, command } => {
710                        BinaryExecutionStatus::Failure { error, command }
711                    }
712                };
713                binary.serialize(serializer)
714            }
715        }
716    }
717
718    impl<'de> Deserialize<'de> for ExecutionStatus {
719        fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
720        where
721            D: Deserializer<'de>,
722        {
723            if deserializer.is_human_readable() {
724                let ReadableExecutionStatus { success, status } =
725                    Deserialize::deserialize(deserializer)?;
726                match (success, status) {
727                    (true, None) => Ok(ExecutionStatus::Success),
728                    (false, Some(FailureStatus { error, command })) => {
729                        Ok(ExecutionStatus::Failure {
730                            error,
731                            command: command.map(Into::into),
732                        })
733                    }
734                    // invalid cases
735                    (true, Some(_)) | (false, None) => {
736                        Err(serde::de::Error::custom("invalid execution status"))
737                    }
738                }
739            } else {
740                BinaryExecutionStatus::deserialize(deserializer).map(|readable| match readable {
741                    BinaryExecutionStatus::Success => Self::Success,
742                    BinaryExecutionStatus::Failure { error, command } => {
743                        Self::Failure { error, command }
744                    }
745                })
746            }
747        }
748    }
749}
