5use crate::{StructTag, TypeParseError};

7/// Type of a move value
8///
9/// # BCS
10///
11/// The BCS serialized form of this type is specified in
12/// [`bcs-schema.abnf`](https://github.com/iotaledger/iota-rust-sdk/blob/develop/crates/iota-sdk-types/bcs-schema.abnf).
13#[derive(Clone, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
14#[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
15pub enum TypeTag {
16    U8,
17    U16,
18    U32,
19    U64,
20    U128,
21    U256,
22    Bool,
23    Address,
24    Signer,
25    #[cfg_attr(feature = "proptest", weight(0))]
26    Vector(Box<TypeTag>),
27    Struct(Box<StructTag>),
28}
29
30impl TypeTag {
31    crate::def_is!(U8, U16, U32, U64, U128, U256, Bool, Address, Signer);
32
33    /// Checks if this type tag is a vector.
34    pub fn is_vector(&self) -> bool {
35        matches!(self, Self::Vector(_))
36    }
37
38    /// Returns a reference to the inner type tag if this type tag is a vector,
39    /// or `None` otherwise.
40    pub fn as_opt_vector_type_tag(&self) -> Option<&TypeTag> {
41        if let Self::Vector(inner) = self {
42            Some(inner)
43        } else {
44            None
45        }
46    }
47
48    /// Returns a reference to the inner type tag if this type tag is a vector,
49    /// or panics otherwise.
50    pub fn as_vector_type_tag(&self) -> &TypeTag {
51        self.as_opt_vector_type_tag().expect("not a Vector")
52    }
53
54    /// Converts this type tag into the inner type tag of a vector, if it is
55    /// one, or returns `None` otherwise.
56    pub fn into_opt_vector_type_tag(self) -> Option<TypeTag> {
57        if let Self::Vector(inner) = self {
58            Some(*inner)
59        } else {
60            None
61        }
62    }
63
64    /// Converts this type tag into the inner type tag of a vector, if it is
65    /// one, or panics otherwise.
66    pub fn into_vector_type_tag(self) -> TypeTag {
67        self.into_opt_vector_type_tag().expect("not a Vector")
68    }
69
70    /// Checks if this type tag is a struct.
71    pub fn is_struct(&self) -> bool {
72        matches!(self, Self::Struct(_))
73    }
74
75    /// Returns a reference to the struct tag if this type tag is a struct, or
76    /// `None` otherwise.
77    pub fn as_opt_struct_tag(&self) -> Option<&StructTag> {
78        if let Self::Struct(inner) = self {
79            Some(inner)
80        } else {
81            None
82        }
83    }
84
85    /// Returns a reference to the struct tag if this type tag is a struct, or
86    /// panics otherwise.
87    pub fn as_struct_tag(&self) -> &StructTag {
88        self.as_opt_struct_tag().expect("not a Struct")
89    }
90
91    /// Converts this type tag into a struct tag, if it is one, or returns
92    /// `None` otherwise.
93    pub fn into_opt_struct_tag(self) -> Option<StructTag> {
94        if let Self::Struct(inner) = self {
95            Some(*inner)
96        } else {
97            None
98        }
99    }
100
101    /// Converts this type tag into a struct tag, if it is one, or panics
102    /// otherwise.
103    pub fn into_struct_tag(self) -> StructTag {
104        self.into_opt_struct_tag().expect("not a Struct")
105    }
106
107    pub fn u8() -> Self {
108        Self::U8
109    }
110
111    pub fn u16() -> Self {
112        Self::U16
113    }
114
115    pub fn u32() -> Self {
116        Self::U32
117    }
118
119    pub fn u64() -> Self {
120        Self::U64
121    }
122
123    pub fn u128() -> Self {
124        Self::U128
125    }
126
127    pub fn u256() -> Self {
128        Self::U256
129    }
130
131    pub fn bool() -> Self {
132        Self::Bool
133    }
134
135    pub fn address() -> Self {
136        Self::Address
137    }
138
139    pub fn signer() -> Self {
140        Self::Signer
141    }
142
143    /// Returns the string representation of this type tag using the
144    /// canonical display, with or without a `0x` prefix.
145    pub fn to_canonical_string(&self, with_prefix: bool) -> String {
146        match self {
147            TypeTag::U8 => "u8".to_owned(),
148            TypeTag::U16 => "u16".to_owned(),
149            TypeTag::U32 => "u32".to_owned(),
150            TypeTag::U64 => "u64".to_owned(),
151            TypeTag::U128 => "u128".to_owned(),
152            TypeTag::U256 => "u256".to_owned(),
153            TypeTag::Bool => "bool".to_owned(),
154            TypeTag::Address => "address".to_owned(),
155            TypeTag::Signer => "signer".to_owned(),
156            TypeTag::Vector(t) => {
157                format!("vector<{}>", t.to_canonical_string(with_prefix))
158            }
159            TypeTag::Struct(s) => s.to_canonical_string(with_prefix),
160        }
161    }
162}

164impl std::fmt::Display for TypeTag {
165    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
166        match self {
167            TypeTag::U8 => write!(f, "u8"),
168            TypeTag::U16 => write!(f, "u16"),
169            TypeTag::U32 => write!(f, "u32"),
170            TypeTag::U64 => write!(f, "u64"),
171            TypeTag::U128 => write!(f, "u128"),
172            TypeTag::U256 => write!(f, "u256"),
173            TypeTag::Bool => write!(f, "bool"),
174            TypeTag::Address => write!(f, "address"),
175            TypeTag::Signer => write!(f, "signer"),
176            TypeTag::Vector(t) => write!(f, "vector<{t}>"),
177            TypeTag::Struct(s) => write!(f, "{s}"),
178        }
179    }
180}

182impl std::str::FromStr for TypeTag {
183    type Err = TypeParseError;
184
185    fn from_str(s: &str) -> Result<Self, Self::Err> {
186        use winnow::Parser;
187        crate::move_core::parse::parse_type_tag
188            .parse(s)
189            .map_err(|e| e.into_inner())
190    }
191}
192
193impl From<StructTag> for TypeTag {
194    fn from(value: StructTag) -> Self {
195        Self::Struct(Box::new(value))
196    }
