15/// Maximum length in bytes of a Move [`Identifier`].
16pub const MAX_IDENTIFIER_LENGTH: usize = 128;
17/// Maximum nesting depth allowed when parsing a [`TypeTag`].
18pub const MAX_TYPE_TAG_NESTING: usize = 16;
19
20/// ALLOWED_IDENTIFIERS = r"(?:[a-zA-Z][a-zA-Z0-9_]*)|(?:_[a-zA-Z0-9_]+)";
21pub(crate) fn parse_identifier(input: &mut &str) -> ModalResult<Identifier, TypeParseError> {
22    alt((
23        "<SELF>",
24        (one_of(|c: char| c.is_alpha()), valid_remainder(0)).take(),
25        ('_', valid_remainder(1)).take(),
26    ))
27    .parse_next(input)
28    .and_then(|s| {
29        if s.len() > MAX_IDENTIFIER_LENGTH {
30            return Err(winnow::error::ErrMode::Cut(
31                TypeParseError::IdentifierMaxLengthExceeded { actual: s.len() },
32            ));
33        }
34        Ok(s)
35    })
36    .map(Identifier::new_unchecked)
37}
38
39fn valid_remainder<'a>(
40    minimum: usize,
41) -> impl FnMut(&mut &'a str) -> ModalResult<&'a str, TypeParseError> {
42    move |input: &mut &'a str| {
43        take_while(
44            // Use .. instead of ..= since we've already processed a single character
45            minimum..MAX_IDENTIFIER_LENGTH,
46            (b'_', b'a'..=b'z', b'A'..=b'Z', b'0'..=b'9'),
47        )
48        .parse_next(input)
49    }
50}

52pub(crate) fn parse_address(input: &mut &str) -> ModalResult<Address, TypeParseError> {
53    ("0x", take_while(1..=64, AsChar::is_hex_digit))
54        .take()
55        .try_map(Address::from_prefixed_short_hex)
56        .parse_next(input)
57}

59pub(crate) fn parse_type_tag(input: &mut &str) -> ModalResult<TypeTag, TypeParseError> {
60    parse_type_tag_impl(0).parse_next(input)
61}
62
63fn parse_type_tag_impl(
64    depth: usize,
65) -> impl FnMut(&mut &str) -> ModalResult<TypeTag, TypeParseError> {
66    move |input: &mut &str| {
67        if depth > MAX_TYPE_TAG_NESTING {
68            return Err(winnow::error::ErrMode::Cut(
69                TypeParseError::NestingLimitExceeded,
70            ));
71        }
72        alt((
73            "u8".value(TypeTag::U8),
74            "u16".value(TypeTag::U16),
75            "u32".value(TypeTag::U32),
76            "u64".value(TypeTag::U64),
77            "u128".value(TypeTag::U128),
78            "u256".value(TypeTag::U256),
79            "bool".value(TypeTag::Bool),
80            "address".value(TypeTag::Address),
81            "signer".value(TypeTag::Signer),
82            delimited(
83                ("vector", multispace0, '<', multispace0),
84                parse_type_tag_impl(depth + 1),
85                (multispace0, '>'),
86            )
87            .map(|ty| TypeTag::Vector(Box::new(ty))),
88            parse_struct_tag_impl(depth).map(|s| TypeTag::Struct(Box::new(s))),
89        ))
90        .parse_next(input)
91    }
92}
93
94pub(crate) fn parse_struct_tag(input: &mut &str) -> ModalResult<StructTag, TypeParseError> {
95    parse_struct_tag_impl(0).parse_next(input)
96}
97
98fn parse_struct_tag_impl(
99    depth: usize,
100) -> impl FnMut(&mut &str) -> ModalResult<StructTag, TypeParseError> {
101    move |input: &mut &str| {
102        let (address, _, module, _, name) = (
103            parse_address,
104            "::",
105            parse_identifier,
106            "::",
107            parse_identifier,
108        )
109            .parse_next(input)?;
110
111        // optional generic
112        let generics = opt(delimited(
113            (multispace0, '<', multispace0),
114            parse_generics(depth),
115            (multispace0, '>'),
116        ))
117        .parse_next(input)?
118        .unwrap_or_default();
119
120        Ok(StructTag::new(address, module, name, generics))
121    }
122}
123
124fn parse_generics(
125    depth: usize,
126) -> impl FnMut(&mut &str) -> ModalResult<Vec<TypeTag>, TypeParseError> {
127    move |input: &mut &str| {
128        separated(
129            1..,
130            delimited(multispace0, parse_type_tag_impl(depth + 1), multispace0),
131            ",",
132        )
133        .parse_next(input)
134    }
135}

314    fn test_type_tag_max_nesting_depth() {
315        // Generate a type tag with exactly MAX_TYPE_TAG_NESTING levels (should succeed)
316        let mut valid_nested = "u8".to_string();
317        for _ in 0..MAX_TYPE_TAG_NESTING {
318            valid_nested = format!("vector<{valid_nested}>");
319        }
320        assert!(
321            valid_nested.parse::<TypeTag>().is_ok(),
322            "Should parse type tag with exactly {MAX_TYPE_TAG_NESTING} levels of nesting"
323        );
324
325        // Generate a type tag with MAX_TYPE_TAG_NESTING + 1 levels (should fail)
326        let mut invalid_nested = "u8".to_string();
327        for _ in 0..=MAX_TYPE_TAG_NESTING {
328            invalid_nested = format!("vector<{invalid_nested}>");
329        }
330        assert!(
331            invalid_nested.parse::<TypeTag>().is_err(),
332            "Should reject type tag with more than {MAX_TYPE_TAG_NESTING} levels of nesting"
333        );
334    }

336    #[test]
337    fn test_struct_tag_nesting_depth() {
338        // Test deeply nested struct generics
339        // Each struct generic adds one nesting level
340        let mut valid_struct = "0x1::A::B".to_string();
341        // Start at depth 0, each generic wrapper adds 1 to the depth of inner types
342        // With MAX_TYPE_TAG_NESTING iterations, the innermost type reaches depth
343        // MAX_TYPE_TAG_NESTING
344        for i in 0..MAX_TYPE_TAG_NESTING {
345            valid_struct = format!("0x{}::A::B<{valid_struct}>", i + 2);
346        }
347        let parsed = valid_struct.parse::<StructTag>();
348        assert!(
349            parsed.is_ok(),
350            "Should parse struct tag ({valid_struct}) within nesting limit: {}",
351            parsed.unwrap_err()
352        );
353
354        // Add one more level to exceed the limit
355        let invalid_struct = format!("0xff::A::B<{valid_struct}>");
356        assert!(
357            invalid_struct.parse::<StructTag>().is_err(),
358            "Should reject struct tag ({invalid_struct}) exceeding nesting limit"
359        );
360    }

362    // Custom strategies that exercise the full TypeTag grammar, including vectors
363    // and nested generics. The existing `Arbitrary` derives intentionally skip
364    // these (Vector has weight 0, type_params is always empty) to keep BCS
365    // serialization fuzz tests fast.
366
367    /// Strategy producing valid Move identifiers, including both
368    /// letter-prefixed (`Foo`, `a1`) and underscore-prefixed (`_bar`, `__x1`).
369    fn arb_identifier() -> impl Strategy<Value = Identifier> {
370        prop_oneof!["[a-zA-Z][a-zA-Z0-9_]{0,127}", "_[a-zA-Z0-9_]{1,127}",]
371            .prop_map(|s| Identifier::new(&s).unwrap())
372    }
373
374    /// Strategy producing a StructTag with empty type_params.
375    fn arb_struct_tag_base() -> impl Strategy<Value = StructTag> {
376        (any::<Address>(), arb_identifier(), arb_identifier())
377            .prop_map(|(addr, module, name)| StructTag::new(addr, module, name, Vec::new()))
378    }
379
380    /// Strategy producing valid TypeTags including vectors, structs with
381    /// generics, and nested combinations. Recursion is bounded to depth 4.
382    fn arb_type_tag() -> impl Strategy<Value = TypeTag> {
383        let leaf = prop_oneof![
384            Just(TypeTag::U8),
385            Just(TypeTag::U16),
386            Just(TypeTag::U32),
387            Just(TypeTag::U64),
388            Just(TypeTag::U128),
389            Just(TypeTag::U256),
390            Just(TypeTag::Bool),
391            Just(TypeTag::Address),
392            Just(TypeTag::Signer),
393        ];
394
395        leaf.prop_recursive(
396            4,  // max depth of recursion
397            64, // desired_size (target number of nodes)
398            3,  // expected_branch_size per recursive step
399            |inner| {
400                let vector_strategy = inner.clone().prop_map(|t| TypeTag::Vector(Box::new(t)));
401
402                let struct_no_params =
403                    arb_struct_tag_base().prop_map(|s| TypeTag::Struct(Box::new(s)));
404
405                let struct_with_params = (
406                    any::<Address>(),
407                    arb_identifier(),
408                    arb_identifier(),
409                    proptest::collection::vec(inner, 1..=3),
410                )
411                    .prop_map(|(addr, module, name, params)| {
412                        TypeTag::Struct(Box::new(StructTag::new(addr, module, name, params)))
413                    });
414
415                prop_oneof![vector_strategy, struct_no_params, struct_with_params]
416            },
417        )
418    }
419
420    /// Reduced set of primitive TypeTag leaves for depth-focused tests.
421    fn arb_simple_leaf() -> impl Strategy<Value = TypeTag> {
422        prop_oneof![Just(TypeTag::U8), Just(TypeTag::U64), Just(TypeTag::Bool)]
423    }

439    #[proptest]
440    fn nested_vector_parsing(
441        #[strategy(0u32..=8)] depth: u32,
442        #[strategy(arb_simple_leaf())] leaf: TypeTag,
443    ) {
444        let mut ty = leaf;
445        for _ in 0..depth {
446            ty = TypeTag::Vector(Box::new(ty));
447        }
448        let s = ty.to_string();
449        let parsed = s.parse::<TypeTag>().unwrap();
450        assert_eq!(ty, parsed);
451    }
