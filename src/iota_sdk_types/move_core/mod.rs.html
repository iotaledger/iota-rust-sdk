18#[derive(Debug, thiserror::Error)]
19#[non_exhaustive]
20pub enum TypeParseError {
21    #[error("failed to parse {input}: {}", source.as_ref().map(|s| s.to_string()).unwrap_or_else(|| "unknown error".to_string()))]
22    Parse {
23        input: String,
24        source: Option<Box<dyn std::error::Error + Send + Sync>>,
25    },
26    #[error(
27        "nesting exceeded limit of {}",
28        crate::move_core::parse::MAX_TYPE_TAG_NESTING
29    )]
30    NestingLimitExceeded,
31    #[error(
32        "identifier length {actual} exceeded limit of {}",
33        crate::move_core::parse::MAX_IDENTIFIER_LENGTH
34    )]
35    IdentifierMaxLengthExceeded { actual: usize },
36    #[error(transparent)]
37    Address(#[from] crate::AddressParseError),
38}
