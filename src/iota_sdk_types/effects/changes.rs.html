4//! Derivation of a transaction's balance and object changes from its effects and the objects it
5//! read and wrote.
6//!
7//! Effects say which objects changed and how, but not what they contain, so the caller supplies the
8//! objects: `input_objects` at the versions [`TransactionEffectsV1::modified_at_versions`] gives,
9//! and `output_objects` as the transaction wrote them. Operates purely on that in-memory data.
10//!
11//! An object missing from those sets is an error rather than a skipped entry: a missing input coin
12//! would make a balance delta numerically wrong, and a missing object would drop a change from the
13//! result, with no way for the caller to tell the result was incomplete.
14
15use std::{collections::BTreeMap, ops::Neg};
16
17use crate::{
18    Address, ExecutionStatus, Object, ObjectDigest, ObjectId, ObjectRemoveKind,
19    OwnedObjectReference, Owner, StructTag, TransactionEffectsV1, TypeTag, Version, WriteKind,
20    framework::Coin, object::ObjectReference,
21};

23/// Error deriving balance or object changes from a transaction's effects.
24#[derive(Clone, Copy, Debug, Eq, PartialEq, thiserror::Error)]
25#[non_exhaustive]
26pub enum DeriveChangesError {
27    /// An object the effects name was not among the supplied objects.
28    #[error("object {object_id} at version {version} is unavailable (possibly pruned)")]
29    MissingObject {
30        object_id: ObjectId,
31        version: Version,
32    },
33    /// An object whose type is a coin had contents that are not a valid coin.
34    #[error("coin object {object_id} at version {version} has malformed contents")]
35    MalformedCoin {
36        object_id: ObjectId,
37        version: Version,
38    },
39}

41/// The net change in balance of one coin type for one owner, summed over every coin of that type
42/// the transaction touched.
43///
44/// Not a wire type: this is the output of [`TransactionEffectsV1::balance_changes`], computed from
45/// the effects and the objects they name.
46#[derive(Clone, Debug, Eq, PartialEq)]
47pub struct BalanceChange {
48    pub(crate) owner: Owner,
49    pub(crate) coin_type: TypeTag,
50    pub(crate) amount: i128,
51}
52
53impl BalanceChange {
54    /// Records the net change in balance of `coin_type` for `owner`.
55    ///
56    /// A negative `amount` means the net flow of value is away from the owner.
57    pub const fn new(owner: Owner, coin_type: TypeTag, amount: i128) -> Self {
58        Self {
59            owner,
60            coin_type,
61            amount,
62        }
63    }
64
65    /// The owner whose balance changed.
66    pub const fn owner(&self) -> &Owner {
67        &self.owner
68    }
69
70    /// The coin type whose balance changed.
71    pub const fn coin_type(&self) -> &TypeTag {
72        &self.coin_type
73    }
74
75    /// The net change in balance.
76    ///
77    /// Negative means the net flow of value is away from the owner.
78    pub const fn amount(&self) -> i128 {
79        self.amount
80    }
81}

87        w.leaf("Coin Type", &self.coin_type, false)?;

92/// What a transaction did to one object, named as a kind.
93///
94/// Not a wire type: this is the output of [`TransactionEffectsV1::object_changes`], computed from
95/// the effects and the objects they name.
96///
97/// The effects name every object a transaction changed but carry no contents, so `object_type` has
98/// to come from the object itself. Where only the effects are at hand,
99/// [`ChangedObject`](crate::ChangedObject) carries each side's version, digest and owner, and the
100/// object sets report the version each changed object ends at.
101///
102/// There is no kind for a mutated package, so a system package upgrade — which keeps its id and is
103/// therefore a mutate — is not reported at all.
104#[derive(Clone, Debug, Eq, PartialEq)]
105#[non_exhaustive]
106pub enum ObjectChange {
107    Published {
108        package_id: ObjectId,
109        version: Version,
110        digest: ObjectDigest,
111        modules: Vec<String>,
112    },
113    Mutated {
114        sender: Address,
115        owner: Owner,
116        object_type: StructTag,
117        object_id: ObjectId,
118        version: Version,
119        previous_version: Version,
120        digest: ObjectDigest,
121    },
122    Deleted {
123        sender: Address,
124        object_type: StructTag,
125        object_id: ObjectId,
126        version: Version,
127    },
128    Wrapped {
129        sender: Address,
130        object_type: StructTag,
131        object_id: ObjectId,
132        version: Version,
133    },
134    Unwrapped {
135        sender: Address,
136        owner: Owner,
137        object_type: StructTag,
138        object_id: ObjectId,
139        version: Version,
140        digest: ObjectDigest,
141    },
142    Created {
143        sender: Address,
144        owner: Owner,
145        object_type: StructTag,
146        object_id: ObjectId,
147        version: Version,
148        digest: ObjectDigest,
149    },
150}

239impl TransactionEffectsV1 {
240    /// The balance changes of this transaction, from these effects and the objects it read and
241    /// wrote.
242    ///
243    /// For a failed transaction only the gas charge is reported, which needs no objects; a
244    /// transaction that names no gas object reports nothing. `mocked_object` excludes an object
245    /// mocked during simulation, which is in neither object set — today that is the gas coin a
246    /// local run mints. Object changes have no equivalent, so a mocked object that is not a coin
247    /// still errors there.
248    ///
249    /// Errors if any object these effects name is missing from the sets — a missing input coin
250    /// would corrupt the delta of any coin whose output is present.
251    pub fn balance_changes<'a>(
252        &self,
253        input_objects: impl IntoIterator<Item = &'a Object>,
254        output_objects: impl IntoIterator<Item = &'a Object>,
255        mocked_object: Option<ObjectId>,
256    ) -> Result<Vec<BalanceChange>, DeriveChangesError> {
257        // Only charge gas when the transaction fails, skipping all object parsing.
258        if self.status != ExecutionStatus::Success {
259            return Ok(self
260                .gas_object()
261                .map(|gas| BalanceChange {
262                    owner: gas.owner,
263                    coin_type: TypeTag::from(StructTag::new_gas()),
264                    amount: (self.gas_cost_summary.net_gas_usage() as i128).neg(),
265                })
266                .into_iter()
267                .collect());
268        }
269
270        let objects: BTreeMap<(ObjectId, Version), &Object> = input_objects
271            .into_iter()
272            .chain(output_objects)
273            .map(|o| ((o.id(), o.version()), o))
274            .collect();
275
276        let mut balances = BTreeMap::<(Owner, TypeTag), i128>::new();
277
278        // 1. subtract all input coins
279        for modified in self.modified_at_versions() {
280            let (id, version) = (modified.object_id, modified.version);
281            // The mocked object is in neither object set.
282            if mocked_object == Some(id) {
283                continue;
284            }
285            if let Some((owner, coin_type, amount)) = coin_owner_type_value(&objects, id, version)?
286            {
287                *balances.entry((owner, coin_type)).or_default() -= amount as i128;
288            }
289        }
290
291        // 2. add all mutated coins
292        for (changed, _) in self.all_changed_objects() {
293            let object_ref = changed.reference;
294            if mocked_object == Some(object_ref.object_id) {
295                continue;
296            }
297            if let Some((owner, coin_type, amount)) =
298                coin_owner_type_value(&objects, object_ref.object_id, object_ref.version)?
299            {
300                *balances.entry((owner, coin_type)).or_default() += amount as i128;
301            }
302        }
303
304        Ok(balances
305            .into_iter()
306            .filter(|(_, amount)| *amount != 0)
307            .map(|((owner, coin_type), amount)| BalanceChange {
308                owner,
309                coin_type,
310                amount,
311            })
312            .collect())
313    }
314
315    /// The object changes of this transaction, from these effects and the objects it read and
316    /// wrote.
317    ///
318    /// `sender` is the transaction's, which the effects do not carry. `input_objects` provides the
319    /// types of deleted and wrapped objects, so it must hold them at the versions
320    /// [`TransactionEffectsV1::modified_at_versions`] gives.
321    ///
322    /// Errors if an object these effects name is missing from the sets — skipping it would drop a
323    /// change from the result.
324    pub fn object_changes<'a>(
325        &self,
326        sender: Address,
327        input_objects: impl IntoIterator<Item = &'a Object>,
328        output_objects: impl IntoIterator<Item = &'a Object>,
329    ) -> Result<Vec<ObjectChange>, DeriveChangesError> {
330        let mut object_changes = vec![];
331
332        let modified_at_versions = self
333            .modified_at_versions()
334            .into_iter()
335            .map(|modified| (modified.object_id, modified.version))
336            .collect::<BTreeMap<_, _>>();
337
338        let outputs: BTreeMap<(ObjectId, Version), &Object> = output_objects
339            .into_iter()
340            .map(|o| ((o.id(), o.version()), o))
341            .collect();
342
343        // Input objects are the objects at their modified-at versions, so they are unique per id
344        // and provide the pre-transaction state of removed objects.
345        let inputs_by_id: BTreeMap<ObjectId, &Object> =
346            input_objects.into_iter().map(|o| (o.id(), o)).collect();
347
348        for (changed, kind) in self.all_changed_objects() {
349            let OwnedObjectReference { reference, owner } = changed;
350            let ObjectReference {
351                object_id,
352                version,
353                digest,
354            } = reference;
355            let Some(object) = outputs.get(&(object_id, version)) else {
356                return Err(DeriveChangesError::MissingObject { object_id, version });
357            };
358            if let Some(move_object_type) = object.data.opt_object_type() {
359                let object_type: StructTag = move_object_type.clone().into();
360
361                match kind {
362                    WriteKind::Mutate => object_changes.push(ObjectChange::Mutated {
363                        sender,
364                        owner,
365                        object_type,
366                        object_id,
367                        version,
368                        // A mutated object was read at some version, so the effects always name one
369                        // for it.
370                        previous_version: *modified_at_versions
371                            .get(&object_id)
372                            .expect("a mutated object has a modified-at version"),
373                        digest,
374                    }),
375                    WriteKind::Create => object_changes.push(ObjectChange::Created {
376                        sender,
377                        owner,
378                        object_type,
379                        object_id,
380                        version,
381                        digest,
382                    }),
383                    WriteKind::Unwrap => object_changes.push(ObjectChange::Unwrapped {
384                        sender,
385                        owner,
386                        object_type,
387                        object_id,
388                        version,
389                        digest,
390                    }),
391                }
392            } else if let Some(package) = object.data.as_opt_package()
393                && kind == WriteKind::Create
394            {
395                object_changes.push(ObjectChange::Published {
396                    package_id: package.id(),
397                    version: package.version(),
398                    digest,
399                    modules: package
400                        .serialized_module_map()
401                        .keys()
402                        .map(|k| k.to_string())
403                        .collect(),
404                })
405            }
406        }
407
408        for (removed_object, kind) in self.all_removed_objects() {
409            let object_id = removed_object.object_id;
410            let version = removed_object.version;
411            let Some(object) = inputs_by_id.get(&object_id) else {
412                // The input object lives at its modified-at version, not at the removed reference's
413                // (tombstone) version.
414                return Err(DeriveChangesError::MissingObject {
415                    object_id,
416                    version: modified_at_versions
417                        .get(&object_id)
418                        .copied()
419                        .unwrap_or(version),
420                });
421            };
422            // Packages cannot be removed; skip non-Move objects.
423            if let Some(move_object_type) = object.data.opt_object_type() {
424                let object_type: StructTag = move_object_type.clone().into();
425                match kind {
426                    ObjectRemoveKind::Delete => object_changes.push(ObjectChange::Deleted {
427                        sender,
428                        object_type,
429                        object_id,
430                        version,
431                    }),
432                    ObjectRemoveKind::Wrap => object_changes.push(ObjectChange::Wrapped {
433                        sender,
434                        object_type,
435                        object_id,
436                        version,
437                    }),
438                }
439            }
440        }
441
442        Ok(object_changes)
443    }
444}

446/// Look up an object and return its owner, coin type and balance if it is a coin. Returns `None`
447/// for non-coins; errors if the object is missing from the set or its coin contents are malformed.
448fn coin_owner_type_value(
449    objects: &BTreeMap<(ObjectId, Version), &Object>,
450    id: ObjectId,
451    version: Version,
452) -> Result<Option<(Owner, TypeTag, u64)>, DeriveChangesError> {
453    let Some(object) = objects.get(&(id, version)) else {
454        return Err(DeriveChangesError::MissingObject {
455            object_id: id,
456            version,
457        });
458    };
459    let Some(move_object_type) = object.data.opt_object_type() else {
460        return Ok(None);
461    };
462    let Some(coin_type) = move_object_type.opt_coin_type().cloned() else {
463        return Ok(None);
464    };
465    let value = Coin::try_from_object(object)
466        .map_err(|_| DeriveChangesError::MalformedCoin {
467            object_id: id,
468            version,
469        })?
470        .balance();
471    Ok(Some((object.owner, coin_type, value)))
472}

853    /// Every write and removal kind reaches the matching variant. The result is grouped by kind
854    /// rather than following the order of the effects, since it walks the object sets in turn.
855    #[test]
856    fn object_changes_report_each_kind() {
857        let sender = address(1);
858        let owner = Owner::Address(sender);
859        let created_id = object_id(10);
860        let mutated_id = object_id(11);
861        let deleted_id = object_id(12);
862        let wrapped_id = object_id(13);
863
864        let effects = effects(vec![
865            created(created_id, owner),
866            mutated(mutated_id, owner),
867            deleted(deleted_id, owner),
868            wrapped(wrapped_id, owner),
869        ]);
870        let inputs = vec![
871            gas_coin(mutated_id, INPUT_VERSION, owner, 1),
872            gas_coin(deleted_id, INPUT_VERSION, owner, 1),
873            gas_coin(wrapped_id, INPUT_VERSION, owner, 1),
874        ];
875        let outputs = vec![
876            gas_coin(created_id, LAMPORT, owner, 1),
877            gas_coin(mutated_id, LAMPORT, owner, 1),
878        ];
879
880        let changes = effects.object_changes(sender, &inputs, &outputs).unwrap();
881        let object_type = StructTag::new_gas_coin();

883            changes,
884            vec![

886                    sender,
887                    owner,

895                    sender,
896                    owner,
897                    object_type: object_type.clone(),
898                    object_id: created_id,
899                    version: Version::from_u64(LAMPORT),
900                    digest: digest(1),
901                },
902                // A removed object is reported at the version this transaction assigned, not the
903                // version it was read at.

946    /// A newly published package is reported with its module names. So is a user package upgrade,
947    /// which lands at a fresh id. A system package upgrade keeps its id and so is a mutate, which
948    /// packages do not report.
949    #[test]
950    fn a_published_package_is_reported_with_its_modules() {
951        use std::collections::BTreeMap;
952
953        use crate::{Identifier, MovePackage};
954
955        let sender = address(1);
956        let package_id = object_id(10);
957        let upgraded_id = object_id(11);
958
959        let package = |id: ObjectId, version: u64| Object {
960            data: ObjectData::Package(MovePackage {
961                id,
962                version: Version::from_u64(version),
963                modules: BTreeMap::from([(Identifier::from_static("counter"), vec![0u8])]),
964                type_origin_table: Vec::new(),
965                linkage_table: BTreeMap::new(),
966            }),
967            owner: Owner::Immutable,
968            previous_transaction: TransactionDigest::default(),
969            storage_rebate: 0,
970        };
971        // A publish or a user upgrade: the id is new, so the write creates it.
972        let published = |id| ChangedObject {
973            object_id: id,
974            input_state: ObjectIn::Missing,
975            output_state: ObjectOut::PackageWrite {
976                version: Version::from_u64(1),
977                digest: digest(1),
978            },
979            id_operation: IdOperation::Created,
980        };
981        // A system upgrade: the id already exists, at the previous version.
982        let system_upgrade = ChangedObject {
983            object_id: ObjectId::FRAMEWORK,
984            input_state: ObjectIn::Data {
985                version: Version::from_u64(1),
986                digest: digest(0),
987                owner: Owner::Immutable,
988            },
989            output_state: ObjectOut::PackageWrite {
990                version: Version::from_u64(2),
991                digest: digest(1),
992            },
993            id_operation: IdOperation::None,
994        };
995
996        let effects = effects(vec![
997            published(package_id),
998            published(upgraded_id),
999            system_upgrade,
1000        ]);
1001        let outputs = vec![
1002            package(package_id, 1),
1003            package(upgraded_id, 1),
1004            package(ObjectId::FRAMEWORK, 2),
1005        ];
1006        let inputs = vec![package(ObjectId::FRAMEWORK, 1)];
1007
1008        let changes = effects.object_changes(sender, &inputs, &outputs).unwrap();
1009        let published_ids: Vec<_> = changes
1010            .iter()
1011            .map(|change| match change {
1012                ObjectChange::Published {
1013                    package_id,
1014                    modules,
1015                    ..
1016                } => {
1017                    assert_eq!(modules, &["counter".to_string()]);
1018                    *package_id
1019                }
1020                other => panic!("expected a published package, got {other:?}"),
1021            })
1022            .collect();
1023        assert_eq!(
1024            published_ids,
1025            vec![package_id, upgraded_id],
1026            "the system upgrade is a mutate, so it is not reported"
1027        );
1028    }
