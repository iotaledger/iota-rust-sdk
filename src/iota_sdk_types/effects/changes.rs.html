4//! Derivation of a transaction's balance and object changes from its effects
5//! and the objects it read and wrote.
6//!
7//! Effects say which objects changed and how, but not what they contain, so
8//! the caller supplies the objects: `input_objects` at the versions
9//! [`TransactionEffectsV1::modified_at_versions`] gives, and `output_objects`
10//! as the transaction wrote them. Operates purely on that in-memory data.
11//!
12//! An object missing from those sets is an error rather than a skipped entry:
13//! a missing input coin would make a balance delta numerically wrong, and a
14//! missing object would drop a change from the result, with no way for the
15//! caller to tell the result was incomplete.
16

25/// Error deriving balance or object changes from a transaction's effects.
26#[derive(Clone, Copy, Debug, Eq, PartialEq, thiserror::Error)]
27#[non_exhaustive]
28pub enum DeriveChangesError {
29    /// An object the effects name was not among the supplied objects.
30    #[error("object {object_id} at version {version} is unavailable (possibly pruned)")]
31    MissingObject {
32        object_id: ObjectId,
33        version: Version,
34    },
35    /// An object whose type is a coin had contents that are not a valid coin.
36    #[error("coin object {object_id} at version {version} has malformed contents")]
37    MalformedCoin {
38        object_id: ObjectId,
39        version: Version,
40    },
41}

43/// The net change in balance of one coin type for one owner, summed over
44/// every coin of that type the transaction touched.
45///
46/// Not a wire type: this is the output of
47/// [`TransactionEffectsV1::balance_changes`], computed from the effects and
48/// the objects they name.
49#[derive(Clone, Debug, Eq, PartialEq)]
50pub struct BalanceChange {
51    pub owner: Owner,
52    pub coin_type: TypeTag,
53    /// Negative amount means the net flow of value is away from the owner.
54    pub amount: i128,
55}

59        w.header("Balance Change")?;
60        w.leaf("Owner", &self.owner, false)?;
61        w.leaf("Coin Type", &self.coin_type, false)?;
62        w.leaf("Amount", &self.amount, true)
63    }

66/// What a transaction did to one object, named as a kind.
67///
68/// Not a wire type: this is the output of
69/// [`TransactionEffectsV1::object_changes`], computed from the effects and the
70/// objects they name.
71///
72/// The effects name every object a transaction changed but carry no contents,
73/// so `object_type` has to come from the object itself. Where only the effects
74/// are at hand, [`ChangedObject`](crate::ChangedObject) carries each side's
75/// version, digest and owner, and the object sets report the version each
76/// changed object ends at.
77///
78/// There is no kind for a mutated package, so a system package upgrade — which
79/// keeps its id and is therefore a mutate — is not reported at all.
80#[derive(Clone, Debug, Eq, PartialEq)]
81#[non_exhaustive]
82pub enum ObjectChange {
83    Published {
84        package_id: ObjectId,
85        version: Version,
86        digest: ObjectDigest,
87        modules: Vec<String>,
88    },
89    Mutated {
90        sender: Address,
91        owner: Owner,
92        object_type: StructTag,
93        object_id: ObjectId,
94        version: Version,
95        previous_version: Version,
96        digest: ObjectDigest,
97    },
98    Deleted {
99        sender: Address,
100        object_type: StructTag,
101        object_id: ObjectId,
102        version: Version,
103    },
104    Wrapped {
105        sender: Address,
106        object_type: StructTag,
107        object_id: ObjectId,
108        version: Version,
109    },
110    Unwrapped {
111        sender: Address,
112        owner: Owner,
113        object_type: StructTag,
114        object_id: ObjectId,
115        version: Version,
116        digest: ObjectDigest,
117    },
118    Created {
119        sender: Address,
120        owner: Owner,
121        object_type: StructTag,
122        object_id: ObjectId,
123        version: Version,
124        digest: ObjectDigest,
125    },
126}

215impl TransactionEffectsV1 {
216    /// The balance changes of this transaction, from these effects and the
217    /// objects it read and wrote.
218    ///
219    /// For a failed transaction only the gas charge is reported, which needs no
220    /// objects; a transaction that names no gas object reports nothing.
221    /// `mocked_object` excludes an object mocked during simulation, which is
222    /// in neither object set — today that is the gas coin a local run mints.
223    /// Object changes have no equivalent, so a mocked object that is not a coin
224    /// still errors there.
225    ///
226    /// Errors if any object these effects name is missing from the sets — a
227    /// missing input coin would corrupt the delta of any coin whose output is
228    /// present.
229    pub fn balance_changes<'a>(
230        &self,
231        input_objects: impl IntoIterator<Item = &'a Object>,
232        output_objects: impl IntoIterator<Item = &'a Object>,
233        mocked_object: Option<ObjectId>,
234    ) -> Result<Vec<BalanceChange>, DeriveChangesError> {
235        // Only charge gas when the transaction fails, skipping all object parsing.
236        if self.status != ExecutionStatus::Success {
237            return Ok(self
238                .gas_object()
239                .map(|gas| BalanceChange {
240                    owner: gas.owner,
241                    coin_type: TypeTag::from(StructTag::new_gas()),
242                    amount: (self.gas_cost_summary.net_gas_usage() as i128).neg(),
243                })
244                .into_iter()
245                .collect());
246        }
247
248        let objects: BTreeMap<(ObjectId, Version), &Object> = input_objects
249            .into_iter()
250            .chain(output_objects)
251            .map(|o| ((o.id(), o.version()), o))
252            .collect();
253
254        let mut balances = BTreeMap::<(Owner, TypeTag), i128>::new();
255
256        // 1. subtract all input coins
257        for modified in self.modified_at_versions() {
258            let (id, version) = (modified.object_id, modified.version);
259            // The mocked object is in neither object set.
260            if mocked_object == Some(id) {
261                continue;
262            }
263            if let Some((owner, coin_type, amount)) = coin_owner_type_value(&objects, id, version)?
264            {
265                *balances.entry((owner, coin_type)).or_default() -= amount as i128;
266            }
267        }
268
269        // 2. add all mutated coins
270        for (changed, _) in self.all_changed_objects() {
271            let object_ref = changed.reference;
272            if mocked_object == Some(object_ref.object_id) {
273                continue;
274            }
275            if let Some((owner, coin_type, amount)) =
276                coin_owner_type_value(&objects, object_ref.object_id, object_ref.version)?
277            {
278                *balances.entry((owner, coin_type)).or_default() += amount as i128;
279            }
280        }
281
282        Ok(balances
283            .into_iter()
284            .filter(|(_, amount)| *amount != 0)
285            .map(|((owner, coin_type), amount)| BalanceChange {
286                owner,
287                coin_type,
288                amount,
289            })
290            .collect())
291    }
292
293    /// The object changes of this transaction, from these effects and the
294    /// objects it read and wrote.
295    ///
296    /// `sender` is the transaction's, which the effects do not carry.
297    /// `input_objects` provides the types of deleted and wrapped objects, so it
298    /// must hold them at the versions
299    /// [`TransactionEffectsV1::modified_at_versions`] gives.
300    ///
301    /// Errors if an object these effects name is missing from the sets —
302    /// skipping it would drop a change from the result.
303    pub fn object_changes<'a>(
304        &self,
305        sender: Address,
306        input_objects: impl IntoIterator<Item = &'a Object>,
307        output_objects: impl IntoIterator<Item = &'a Object>,
308    ) -> Result<Vec<ObjectChange>, DeriveChangesError> {
309        let mut object_changes = vec![];
310
311        let modified_at_versions = self
312            .modified_at_versions()
313            .into_iter()
314            .map(|modified| (modified.object_id, modified.version))
315            .collect::<BTreeMap<_, _>>();
316
317        let outputs: BTreeMap<(ObjectId, Version), &Object> = output_objects
318            .into_iter()
319            .map(|o| ((o.id(), o.version()), o))
320            .collect();
321
322        // Input objects are the objects at their modified-at versions, so they are
323        // unique per id and provide the pre-transaction state of removed objects.
324        let inputs_by_id: BTreeMap<ObjectId, &Object> =
325            input_objects.into_iter().map(|o| (o.id(), o)).collect();
326
327        for (changed, kind) in self.all_changed_objects() {
328            let OwnedObjectReference { reference, owner } = changed;
329            let ObjectReference {
330                object_id,
331                version,
332                digest,
333            } = reference;
334            let Some(object) = outputs.get(&(object_id, version)) else {
335                return Err(DeriveChangesError::MissingObject { object_id, version });
336            };
337            if let Some(move_object_type) = object.data.opt_object_type() {
338                let object_type: StructTag = move_object_type.clone().into();
339
340                match kind {
341                    WriteKind::Mutate => object_changes.push(ObjectChange::Mutated {
342                        sender,
343                        owner,
344                        object_type,
345                        object_id,
346                        version,
347                        // A mutated object was read at some version, so the
348                        // effects always name one for it.
349                        previous_version: *modified_at_versions
350                            .get(&object_id)
351                            .expect("a mutated object has a modified-at version"),
352                        digest,
353                    }),
354                    WriteKind::Create => object_changes.push(ObjectChange::Created {
355                        sender,
356                        owner,
357                        object_type,
358                        object_id,
359                        version,
360                        digest,
361                    }),
362                    WriteKind::Unwrap => object_changes.push(ObjectChange::Unwrapped {
363                        sender,
364                        owner,
365                        object_type,
366                        object_id,
367                        version,
368                        digest,
369                    }),
370                }
371            } else if let Some(package) = object.data.as_opt_package()
372                && kind == WriteKind::Create
373            {
374                object_changes.push(ObjectChange::Published {
375                    package_id: package.id(),
376                    version: package.version(),
377                    digest,
378                    modules: package
379                        .serialized_module_map()
380                        .keys()
381                        .map(|k| k.to_string())
382                        .collect(),
383                })
384            }
385        }
386
387        for (removed_object, kind) in self.all_removed_objects() {
388            let object_id = removed_object.object_id;
389            let version = removed_object.version;
390            let Some(object) = inputs_by_id.get(&object_id) else {
391                // The input object lives at its modified-at version, not at the
392                // removed reference's (tombstone) version.
393                return Err(DeriveChangesError::MissingObject {
394                    object_id,
395                    version: modified_at_versions
396                        .get(&object_id)
397                        .copied()
398                        .unwrap_or(version),
399                });
400            };
401            // Packages cannot be removed; skip non-Move objects.
402            if let Some(move_object_type) = object.data.opt_object_type() {
403                let object_type: StructTag = move_object_type.clone().into();
404                match kind {
405                    ObjectRemoveKind::Delete => object_changes.push(ObjectChange::Deleted {
406                        sender,
407                        object_type,
408                        object_id,
409                        version,
410                    }),
411                    ObjectRemoveKind::Wrap => object_changes.push(ObjectChange::Wrapped {
412                        sender,
413                        object_type,
414                        object_id,
415                        version,
416                    }),
417                }
418            }
419        }
420
421        Ok(object_changes)
422    }
423}

425/// Look up an object and return its owner, coin type and balance if it is a
426/// coin. Returns `None` for non-coins; errors if the object is missing from
427/// the set or its coin contents are malformed.
428fn coin_owner_type_value(
429    objects: &BTreeMap<(ObjectId, Version), &Object>,
430    id: ObjectId,
431    version: Version,
432) -> Result<Option<(Owner, TypeTag, u64)>, DeriveChangesError> {
433    let Some(object) = objects.get(&(id, version)) else {
434        return Err(DeriveChangesError::MissingObject {
435            object_id: id,
436            version,
437        });
438    };
439    let Some(move_object_type) = object.data.opt_object_type() else {
440        return Ok(None);
441    };
442    let Some(coin_type) = move_object_type.opt_coin_type().cloned() else {
443        return Ok(None);
444    };
445    let value = Coin::try_from_object(object)
446        .map_err(|_| DeriveChangesError::MalformedCoin {
447            object_id: id,
448            version,
449        })?
450        .balance();
451    Ok(Some((object.owner, coin_type, value)))
452}

847        let effects = effects(vec![

929    /// A newly published package is reported with its module names. So is a
930    /// user package upgrade, which lands at a fresh id. A system package
931    /// upgrade keeps its id and so is a mutate, which packages do not report.
932    #[test]
933    fn a_published_package_is_reported_with_its_modules() {
934        use std::collections::BTreeMap;
935
936        use crate::{Identifier, MovePackage};
937
938        let sender = address(1);
939        let package_id = object_id(10);
940        let upgraded_id = object_id(11);
941
942        let package = |id: ObjectId, version: u64| Object {
943            data: ObjectData::Package(MovePackage {
944                id,
945                version: Version::from_u64(version),
946                modules: BTreeMap::from([(Identifier::from_static("counter"), vec![0u8])]),
947                type_origin_table: Vec::new(),
948                linkage_table: BTreeMap::new(),
949            }),
950            owner: Owner::Immutable,
951            previous_transaction: TransactionDigest::default(),
952            storage_rebate: 0,
953        };
954        // A publish or a user upgrade: the id is new, so the write creates it.
955        let published = |id| ChangedObject {
956            object_id: id,
957            input_state: ObjectIn::Missing,
958            output_state: ObjectOut::PackageWrite {
959                version: Version::from_u64(1),
960                digest: digest(1),
961            },
962            id_operation: IdOperation::Created,
963        };
964        // A system upgrade: the id already exists, at the previous version.
965        let system_upgrade = ChangedObject {
966            object_id: ObjectId::FRAMEWORK,
967            input_state: ObjectIn::Data {
968                version: Version::from_u64(1),
969                digest: digest(0),
970                owner: Owner::Immutable,
971            },
972            output_state: ObjectOut::PackageWrite {
973                version: Version::from_u64(2),
974                digest: digest(1),
975            },
976            id_operation: IdOperation::None,
977        };
978
979        let effects = effects(vec![
980            published(package_id),
981            published(upgraded_id),
982            system_upgrade,
983        ]);
984        let outputs = vec![
985            package(package_id, 1),
986            package(upgraded_id, 1),
987            package(ObjectId::FRAMEWORK, 2),
988        ];
989        let inputs = vec![package(ObjectId::FRAMEWORK, 1)];
990
991        let changes = effects.object_changes(sender, &inputs, &outputs).unwrap();
992        let published_ids: Vec<_> = changes
993            .iter()
994            .map(|change| match change {
995                ObjectChange::Published {
996                    package_id,
997                    modules,
998                    ..
999                } => {
1000                    assert_eq!(modules, &["counter".to_string()]);
1001                    *package_id
1002                }
1003                other => panic!("expected a published package, got {other:?}"),
1004            })
1005            .collect();
1006        assert_eq!(
1007            published_ids,
1008            vec![package_id, upgraded_id],
1009            "the system upgrade is a mutate, so it is not reported"
1010        );
1011    }
