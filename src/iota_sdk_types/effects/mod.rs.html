16/// The output or effects of executing a transaction
17///
18/// # BCS
19///
20/// The BCS serialized form for this type is defined by the following ABNF:
21///
22/// ```text
23/// transaction-effects = %d00 transaction-effects-v1   ; V1
24/// ```
25#[derive(Clone, Debug, Eq, PartialEq)]
26#[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]
27#[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
28#[cfg_attr(feature = "bcs-schema", derive(iota_bcs_schema::BcsSchema))]
29#[non_exhaustive]
30pub enum TransactionEffects {
31    V1(Box<TransactionEffectsV1>),
32}
33
34impl TransactionEffects {
35    crate::def_is_as_into_opt!(V1(Box<TransactionEffectsV1>));
36}
37
38impl crate::TreeDisplay for TransactionEffects {
39    fn fmt_tree(&self, w: &mut crate::TreeWriter<'_, '_>) -> std::fmt::Result {
40        w.enum_name("Transaction Effects");
41        match self {
42            Self::V1(v1) => v1.fmt_tree(w),
43        }
44    }
45}

47/// A shared object an executed transaction took as input.
48///
49/// Not a wire type: this is the effects' view of the shared objects a
50/// transaction was sequenced against, drawn from both the objects it changed
51/// and those it left unchanged.
52#[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
53pub enum InputSharedObject {
54    /// Taken mutably, and written back by the transaction.
55    Mutate(ObjectReference),
56    /// Read without being mutated.
57    ReadOnly(ObjectReference),
58    /// Read, but already deleted by an earlier transaction.
59    ReadDeleted(ObjectVersion),
60    /// Taken mutably, but already deleted by an earlier transaction.
61    MutateDeleted(ObjectVersion),
62    /// Taken by a transaction that consensus canceled; the version carries the
63    /// cancellation reason.
64    Canceled(ObjectVersion),
65}
66
67impl InputSharedObject {
68    /// The object's reference. A shared object that no longer exists is
69    /// reported at the version it was last known at, with the tombstone digest
70    /// for why it is gone.
71    pub fn object_reference(&self) -> ObjectReference {
72        match self {
73            Self::Mutate(reference) | Self::ReadOnly(reference) => *reference,
74            Self::ReadDeleted(object) | Self::MutateDeleted(object) => ObjectReference::new(
75                object.object_id,
76                object.version,
77                ObjectDigest::OBJECT_DELETED,
78            ),
79            Self::Canceled(object) => ObjectReference::new(
80                object.object_id,
81                object.version,
82                ObjectDigest::OBJECT_CANCELED,
83            ),
84        }
85    }
86}
87
88impl crate::TreeDisplay for InputSharedObject {
89    fn fmt_tree(&self, w: &mut crate::TreeWriter<'_, '_>) -> std::fmt::Result {
90        w.header("Input Shared Object")?;
91        let kind = match self {
92            Self::Mutate(_) => "Mutate",
93            Self::ReadOnly(_) => "Read Only",
94            Self::ReadDeleted(_) => "Read Deleted",
95            Self::MutateDeleted(_) => "Mutate Deleted",
96            Self::Canceled(_) => "Canceled",
97        };
98        w.leaf("Kind", &kind, false)?;
99        w.child("Reference", &self.object_reference(), true)
100    }
101}

103/// How an object came to be in the store after a transaction wrote it.
104///
105/// Not a wire type: this tags an object reported by
106/// [`TransactionEffectsV1::all_changed_objects`] with which of the object sets
107/// it came from.
108#[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
109pub enum WriteKind {
110    /// The object existed already and the transaction changed its contents.
111    Mutate,
112    /// The transaction created the object.
113    Create,
114    /// The object was wrapped inside another object, and the transaction
115    /// restored it to the store.
116    Unwrap,
117}
118
119impl std::fmt::Display for WriteKind {
120    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
121        let text = match self {
122            Self::Mutate => "Mutate",
123            Self::Create => "Create",
124            Self::Unwrap => "Unwrap",
125        };
126        f.write_str(text)
127    }
128}

130/// Why an object is no longer in the store after a transaction.
131///
132/// Not a wire type: this tags an object reported by
133/// [`TransactionEffectsV1::all_removed_objects`] with which of the object sets
134/// it came from.
135#[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
136pub enum ObjectRemoveKind {
137    /// The transaction deleted the object.
138    Delete,
139    /// The transaction wrapped the object inside another one.
140    Wrap,
141}
142
143impl std::fmt::Display for ObjectRemoveKind {
144    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
145        let text = match self {
146            Self::Delete => "Delete",
147            Self::Wrap => "Wrap",
148        };
149        f.write_str(text)
150    }
151}

155/// Defines what happened to an ObjectId during execution
156///
157/// # BCS
158///
159/// The BCS serialized form for this type is defined by the following ABNF:
160///
161/// ```text
162/// id-operation = %d00   ; None

168    feature = "serde",

173#[cfg_attr(feature = "bcs-schema", derive(iota_bcs_schema::BcsSchema))]
174#[non_exhaustive]
175pub enum IdOperation {
176    None,
177    Created,
178    Deleted,
179}
180
181impl IdOperation {
182    crate::def_is!(None, Created, Deleted);
183}

400    /// Every object set and both tagged unions, over effects that contain one
401    /// object of each kind. The fixtures hold only created and mutated objects,
402    /// so this is what covers the other four.
403    #[test]
404    fn object_sets_report_each_kind() {
405        use crate::{
406            ChangedObject, IdOperation, ObjectDigest, ObjectId, ObjectIn, ObjectOut,
407            ObjectReference, ObjectRemoveKind, Owner, TransactionEffectsV1, Version, WriteKind,
408        };
409
410        let lamport_version = Version::from_u64(9);
411        let old_version = Version::from_u64(4);
412        let digest = ObjectDigest::new([7; 32]);
413        let owner = Owner::Address(crate::Address::ZERO);
414
415        let created = ObjectId::new([1; 32]);
416        let mutated = ObjectId::new([2; 32]);
417        let unwrapped = ObjectId::new([3; 32]);
418        let deleted = ObjectId::new([4; 32]);
419        let wrapped = ObjectId::new([5; 32]);
420        let unwrapped_then_deleted = ObjectId::new([6; 32]);
421
422        let entry = |object_id, input_state, output_state, id_operation| ChangedObject {
423            object_id,
424            input_state,
425            output_state,
426            id_operation,
427        };
428        let existed = || ObjectIn::Data {
429            version: old_version,
430            digest,
431            owner,
432        };
433        let written = || ObjectOut::ObjectWrite { digest, owner };
434
435        let effects = TransactionEffectsV1 {
436            status: crate::ExecutionStatus::Success,
437            epoch: 0,
438            gas_cost_summary: crate::GasCostSummary::default(),
439            transaction_digest: crate::TransactionDigest::default(),
440            gas_object_index: None,
441            events_digest: None,
442            dependencies: Vec::new(),
443            lamport_version,
444            changed_objects: vec![
445                entry(created, ObjectIn::Missing, written(), IdOperation::Created),
446                entry(mutated, existed(), written(), IdOperation::None),
447                entry(unwrapped, ObjectIn::Missing, written(), IdOperation::None),
448                entry(deleted, existed(), ObjectOut::Missing, IdOperation::Deleted),
449                entry(wrapped, existed(), ObjectOut::Missing, IdOperation::None),
450                entry(
451                    unwrapped_then_deleted,
452                    ObjectIn::Missing,
453                    ObjectOut::Missing,
454                    IdOperation::Deleted,
455                ),
456            ],
457            unchanged_shared_objects: Vec::new(),
458            auxiliary_data_digest: None,
459        };
460
461        let ids = |objects: Vec<crate::OwnedObjectReference>| {
462            objects
463                .into_iter()
464                .map(|owned| owned.reference.object_id)
465                .collect::<Vec<_>>()
466        };
467        let removed_ids = |references: Vec<ObjectReference>| {
468            references
469                .into_iter()
470                .map(|reference| reference.object_id)
471                .collect::<Vec<_>>()
472        };
473
474        assert_eq!(ids(effects.created()), vec![created]);
475        assert_eq!(ids(effects.mutated()), vec![mutated]);
476        assert_eq!(ids(effects.unwrapped()), vec![unwrapped]);
477        assert_eq!(removed_ids(effects.deleted()), vec![deleted]);
478        assert_eq!(removed_ids(effects.wrapped()), vec![wrapped]);
479        assert_eq!(
480            removed_ids(effects.unwrapped_then_deleted()),
481            vec![unwrapped_then_deleted]
482        );
483
484        // The tags, and the order the sets are drawn in.
485        assert_eq!(
486            effects
487                .all_changed_objects()
488                .into_iter()
489                .map(|(object, kind)| (object.reference.object_id, kind))
490                .collect::<Vec<_>>(),
491            vec![
492                (mutated, WriteKind::Mutate),
493                (created, WriteKind::Create),
494                (unwrapped, WriteKind::Unwrap),
495            ]
496        );
497
498        // An object unwrapped and then deleted was never in the store, so it is
499        // not a removal.
500        assert_eq!(
501            effects
502                .all_removed_objects()
503                .into_iter()
504                .map(|(reference, kind)| (reference.object_id, kind))
505                .collect::<Vec<_>>(),
506            vec![
507                (deleted, ObjectRemoveKind::Delete),
508                (wrapped, ObjectRemoveKind::Wrap),
509            ]
510        );
511
512        // Removals carry the tombstone digest for why the object is gone, at the
513        // version this transaction assigned.
514        for (reference, kind) in effects.all_removed_objects() {
515            assert_eq!(reference.version, lamport_version);
516            match kind {
517                ObjectRemoveKind::Delete => assert!(reference.digest.is_deleted()),
518                ObjectRemoveKind::Wrap => assert!(reference.digest.is_wrapped()),
519            }
520        }
521        for reference in effects.unwrapped_then_deleted() {
522            assert!(reference.digest.is_deleted());
523        }
524    }

526    /// The tagged unions are exactly the sets they are drawn from, so nothing
527    /// is dropped or double-counted, and each object carries the right tag.
528    #[test]
529    fn tagged_unions_cover_the_object_sets() {
530        use crate::{ObjectRemoveKind, WriteKind};
531
532        for fixture in [GENESIS_EFFECTS, SPONSOR_TX_EFFECTS] {
533            let effects: TransactionEffects =
534                bcs::from_bytes(&Base64::decode_vec(fixture.trim()).unwrap()).unwrap();
535            let fx = effects.as_v1();
536
537            let changed = fx.all_changed_objects();
538            assert_eq!(
539                changed.len(),
540                fx.mutated().len() + fx.created().len() + fx.unwrapped().len()
541            );
542            assert!(!changed.is_empty(), "the fixture must change objects");
543            let of_kind = |kind| {
544                changed
545                    .iter()
546                    .filter(|(_, k)| *k == kind)
547                    .map(|(object, _)| *object)
548                    .collect::<Vec<_>>()
549            };
550            assert_eq!(of_kind(WriteKind::Mutate), fx.mutated());
551            assert_eq!(of_kind(WriteKind::Create), fx.created());
552            assert_eq!(of_kind(WriteKind::Unwrap), fx.unwrapped());
553
554            let removed = fx.all_removed_objects();
555            assert_eq!(removed.len(), fx.deleted().len() + fx.wrapped().len());
556            let removed_of_kind = |kind| {
557                removed
558                    .iter()
559                    .filter(|(_, k)| *k == kind)
560                    .map(|(reference, _)| *reference)
561                    .collect::<Vec<_>>()
562            };
563            assert_eq!(removed_of_kind(ObjectRemoveKind::Delete), fx.deleted());
564            assert_eq!(removed_of_kind(ObjectRemoveKind::Wrap), fx.wrapped());
565
566            // An object unwrapped and then deleted was never in the store, so it
567            // is not a removal.
568            for reference in fx.unwrapped_then_deleted() {
569                assert!(!removed.iter().any(|(removed, _)| *removed == reference));
570            }
571        }
572    }

574    /// Every changed object falls into exactly one of the reported sets, so the
575    /// sets partition `changed_objects` and never report an object twice.
576    #[test]
577    fn object_sets_partition_the_changed_objects() {
578        for fixture in [GENESIS_EFFECTS, SPONSOR_TX_EFFECTS] {
579            let effects: TransactionEffects =
580                bcs::from_bytes(&Base64::decode_vec(fixture.trim()).unwrap()).unwrap();
581            let fx = effects.as_v1();
582
583            let owned = fx
584                .created()
585                .into_iter()
586                .chain(fx.mutated())
587                .chain(fx.unwrapped())
588                .map(|owned| owned.reference.object_id);
589            let removed = fx
590                .deleted()
591                .into_iter()
592                .chain(fx.unwrapped_then_deleted())
593                .chain(fx.wrapped())
594                .map(|object_ref| object_ref.object_id);
595            let reported: Vec<_> = owned.chain(removed).collect();
596
597            assert!(!reported.is_empty(), "the fixture must change objects");
598            let unique: std::collections::BTreeSet<_> = reported.iter().collect();
599            assert_eq!(unique.len(), reported.len(), "an object was reported twice");
600            assert_eq!(
601                unique,
602                fx.changed_objects
603                    .iter()
604                    .map(|changed| &changed.object_id)
605                    .collect(),
606            );
607        }
608    }
