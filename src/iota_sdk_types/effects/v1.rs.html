11/// Version 1 of TransactionEffects
12///
13/// # BCS
14///
15/// The BCS serialized form of this type is specified in
16/// [`bcs-schema.abnf`](https://github.com/iotaledger/iota-rust-sdk/blob/develop/crates/iota-sdk-types/bcs-schema.abnf).
17#[derive(Clone, Debug, Eq, PartialEq)]
18#[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]
19#[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
20#[cfg_attr(feature = "bcs-schema", derive(iota_bcs_schema::BcsSchema))]
21pub struct TransactionEffectsV1 {
22    /// The status of the execution
23    pub status: ExecutionStatus,
24    /// The epoch when this transaction was executed.
25    #[cfg_attr(feature = "serde", serde(with = "crate::_serde::ReadableDisplay"))]
26    #[cfg_attr(feature = "bcs-schema", bcs_schema(as_type = "u64"))]
27    pub epoch: EpochId,
28    /// The gas used by this transaction
29    pub gas_cost_summary: GasCostSummary,
30    /// The transaction digest
31    pub transaction_digest: TransactionDigest,
32    /// The updated gas object reference, as an index into the `changed_objects`
33    /// vector. Having a dedicated field for convenient access.
34    /// System transaction that don't require gas will leave this as None.
35    pub gas_object_index: Option<u32>,
36    /// The digest of the events emitted during execution,
37    /// can be None if the transaction does not emit any event.
38    pub events_digest: Option<TransactionEventsDigest>,
39    /// The set of transaction digests this transaction depends on.
40    #[cfg_attr(feature = "proptest", any(proptest::collection::size_range(0..=5).lift()))]
41    pub dependencies: Vec<TransactionDigest>,
42    /// The version number of all the written Move objects by this transaction.
43    pub lamport_version: Version,
44    /// Objects whose state are changed in the object store.
45    #[cfg_attr(feature = "proptest", any(proptest::collection::size_range(0..=2).lift()))]
46    pub changed_objects: Vec<ChangedObject>,
47    /// Shared objects that are not mutated in this transaction. Unlike owned
48    /// objects, read-only shared objects' version are not committed in the
49    /// transaction, and in order for a node to catch up and execute it
50    /// without consensus sequencing, the version needs to be committed in
51    /// the effects.
52    #[cfg_attr(feature = "proptest", any(proptest::collection::size_range(0..=2).lift()))]
53    pub unchanged_shared_objects: Vec<UnchangedSharedObject>,
54    /// Auxiliary data that are not protocol-critical, generated as part of the
55    /// effects but are stored separately. Storing it separately allows us
56    /// to avoid bloating the effects with data that are not critical.
57    /// It also provides more flexibility on the format and type of the data.
58    pub auxiliary_data_digest: Option<EffectsAuxDataDigest>,
59}

147/// Type of unchanged shared object
148///
149/// # BCS
150///
151/// The BCS serialized form of this type is specified in
152/// [`bcs-schema.abnf`](https://github.com/iotaledger/iota-rust-sdk/blob/develop/crates/iota-sdk-types/bcs-schema.abnf).

171    /// cancellation reason.

173    /// Read of a per-epoch config object that should remain the same during an
174    /// epoch.
175    PerEpochConfig,
176}

179    crate::def_is!(
180        ReadOnlyRoot,
181        MutateDeleted,
182        ReadDeleted,
183        Canceled,

186}

393impl TransactionEffectsV1 {
394    /// The id and pre-transaction version of every object that existed before
395    /// this transaction and was modified by it (mutated, wrapped or deleted).
396    pub fn modified_at_versions(&self) -> Vec<ObjectVersion> {
397        self.changed_objects
398            .iter()
399            .filter_map(|changed| {
400                changed
401                    .input_state
402                    .opt_version()
403                    .map(|version| ObjectVersion::new(changed.object_id, version))
404            })
405            .collect()
406    }
407
408    /// The shared objects this transaction was sequenced against, whether or
409    /// not it changed them. Excludes per-epoch config objects, which need no
410    /// sequencing.
411    pub fn input_shared_objects(&self) -> Vec<InputSharedObject> {
412        self.changed_objects
413            .iter()
414            .filter_map(|changed| match changed.input_state {
415                ObjectIn::Data {
416                    version,
417                    digest,
418                    owner: Owner::Shared { .. },
419                } => Some(InputSharedObject::Mutate(ObjectReference::new(
420                    changed.object_id,
421                    version,
422                    digest,
423                ))),
424                _ => None,
425            })
426            .chain(
427                self.unchanged_shared_objects
428                    .iter()
429                    .filter_map(|unchanged| {
430                        let object = |version| ObjectVersion::new(unchanged.object_id, version);
431                        match unchanged.kind {
432                            UnchangedSharedKind::ReadOnlyRoot { version, digest } => {
433                                Some(InputSharedObject::ReadOnly(ObjectReference::new(
434                                    unchanged.object_id,
435                                    version,
436                                    digest,
437                                )))
438                            }
439                            UnchangedSharedKind::ReadDeleted { version } => {
440                                Some(InputSharedObject::ReadDeleted(object(version)))
441                            }
442                            UnchangedSharedKind::MutateDeleted { version } => {
443                                Some(InputSharedObject::MutateDeleted(object(version)))
444                            }
445                            UnchangedSharedKind::Canceled { version } => {
446                                Some(InputSharedObject::Canceled(object(version)))
447                            }
448                            // A per-epoch config object is read without being
449                            // sequenced, so it is not an input in this sense.
450                            UnchangedSharedKind::PerEpochConfig => None,
451                        }
452                    }),
453            )
454            .collect()
455    }
456
457    /// The reference and owner, before this transaction, of every object it
458    /// modified.
459    pub fn old_object_metadata(&self) -> Vec<OwnedObjectReference> {
460        self.changed_objects
461            .iter()
462            .filter_map(|changed| match changed.input_state {
463                ObjectIn::Data {
464                    version,
465                    digest,
466                    owner,
467                } => Some(OwnedObjectReference::new(
468                    ObjectReference::new(changed.object_id, version, digest),
469                    owner,
470                )),
471                _ => None,
472            })
473            .collect()
474    }
475
476    /// Objects (Move objects and packages) newly created by this transaction,
477    /// paired with their owner. Excludes objects created and then wrapped
478    /// within the same transaction.
479    pub fn created(&self) -> Vec<OwnedObjectReference> {
480        self.changed_objects
481            .iter()
482            .filter(|changed| {
483                changed.input_state.is_missing() && changed.id_operation == IdOperation::Created
484            })
485            .filter_map(|changed| self.output_reference(changed))
486            .collect()
487    }
488
489    /// Objects that existed before this transaction and whose contents it
490    /// updated (in-place mutations and system package upgrades), at their
491    /// post-transaction reference and owner.
492    pub fn mutated(&self) -> Vec<OwnedObjectReference> {
493        self.changed_objects
494            .iter()
495            .filter(|changed| changed.input_state.is_data())
496            .filter_map(|changed| self.output_reference(changed))
497            .collect()
498    }
499
500    /// Objects that were wrapped inside another object before this transaction
501    /// and that it promoted back to top-level objects in the store.
502    pub fn unwrapped(&self) -> Vec<OwnedObjectReference> {
503        self.changed_objects
504            .iter()
505            .filter(|changed| {
506                changed.input_state.is_missing()
507                    && changed.id_operation == IdOperation::None
508                    // A package is never wrapped, so never unwrapped either.
509                    && changed.output_state.is_object_write()
510            })
511            .filter_map(|changed| self.output_reference(changed))
512            .collect()
513    }
514
515    /// Objects that existed before this transaction and that it deleted.
516    /// References carry the version this transaction assigned and the
517    /// [`ObjectDigest::OBJECT_DELETED`] tombstone digest.
518    pub fn deleted(&self) -> Vec<ObjectReference> {
519        self.removed_references(
520            |changed| changed.input_state.is_data() && changed.id_operation == IdOperation::Deleted,
521            ObjectDigest::OBJECT_DELETED,
522        )
523    }
524
525    /// Objects unwrapped and then deleted within this same transaction, so
526    /// that they existed as top-level objects neither before nor after it.
527    /// References carry the version this transaction assigned and the
528    /// [`ObjectDigest::OBJECT_DELETED`] tombstone digest.
529    pub fn unwrapped_then_deleted(&self) -> Vec<ObjectReference> {
530        self.removed_references(
531            |changed| {
532                changed.input_state.is_missing() && changed.id_operation == IdOperation::Deleted
533            },
534            ObjectDigest::OBJECT_DELETED,
535        )
536    }
537
538    /// Objects that existed as top-level objects before this transaction and
539    /// that it wrapped inside another object, so they are no longer visible in
540    /// the object store. References carry the version this transaction assigned
541    /// and the [`ObjectDigest::OBJECT_WRAPPED`] tombstone digest.
542    pub fn wrapped(&self) -> Vec<ObjectReference> {
543        self.removed_references(
544            |changed| changed.input_state.is_data() && changed.id_operation == IdOperation::None,
545            ObjectDigest::OBJECT_WRAPPED,
546        )
547    }
548
549    /// Every object still in the store after this transaction, tagged with how
550    /// it got there: the created, mutated and unwrapped objects together.
551    /// Excludes the objects the transaction removed.
552    pub fn all_changed_objects(&self) -> Vec<(OwnedObjectReference, WriteKind)> {
553        let tagged = |kind| move |object| (object, kind);
554        self.mutated()
555            .into_iter()
556            .map(tagged(WriteKind::Mutate))
557            .chain(self.created().into_iter().map(tagged(WriteKind::Create)))
558            .chain(self.unwrapped().into_iter().map(tagged(WriteKind::Unwrap)))
559            .collect()
560    }
561
562    /// Every object that was in the store before this transaction and is not
563    /// after it, tagged with why: the deleted and wrapped objects together.
564    /// Excludes an object the transaction unwrapped and then deleted, which was
565    /// not in the store to begin with.
566    pub fn all_removed_objects(&self) -> Vec<(ObjectReference, ObjectRemoveKind)> {
567        let tagged = |kind| move |reference| (reference, kind);
568        self.deleted()
569            .into_iter()
570            .map(tagged(ObjectRemoveKind::Delete))
571            .chain(
572                self.wrapped()
573                    .into_iter()
574                    .map(tagged(ObjectRemoveKind::Wrap)),
575            )
576            .collect()
577    }
578
579    /// The post-transaction reference and owner of the gas object, or `None`
580    /// for a system transaction, which pays no gas and so names none.
581    pub fn gas_object(&self) -> Option<OwnedObjectReference> {
582        let changed = self.changed_objects.get(self.gas_object_index? as usize)?;
583        // Gas is paid in coins, so a gas object is never a package.
584        changed
585            .output_state
586            .is_object_write()
587            .then(|| self.output_reference(changed))?
588    }
589
590    /// The post-transaction reference and owner of a changed object, or `None`
591    /// if this transaction removed it from the store. A package carries its own
592    /// version; every other object takes the version this transaction assigned.
593    fn output_reference(&self, changed: &ChangedObject) -> Option<OwnedObjectReference> {
594        match changed.output_state {
595            ObjectOut::ObjectWrite { digest, owner } => Some(OwnedObjectReference::new(
596                ObjectReference::new(changed.object_id, self.lamport_version, digest),
597                owner,
598            )),
599            ObjectOut::PackageWrite { version, digest } => Some(OwnedObjectReference::new(
600                ObjectReference::new(changed.object_id, version, digest),
601                Owner::Immutable,
602            )),
603            _ => None,
604        }
605    }
606
607    /// References, carrying `digest` as their tombstone, to the objects this
608    /// transaction removed from the store that `select` accepts.
609    fn removed_references(
610        &self,
611        select: impl Fn(&ChangedObject) -> bool,
612        digest: ObjectDigest,
613    ) -> Vec<ObjectReference> {
614        self.changed_objects
615            .iter()
616            .filter(|changed| changed.output_state.is_missing() && select(changed))
617            .map(|changed| ObjectReference::new(changed.object_id, self.lamport_version, digest))
618            .collect()
619    }
