17/// A commitment made by a checkpoint.
18///
19/// # BCS
20///
21/// The BCS serialized form for this type is defined by the following ABNF:
22///
23/// ```text
24/// ; CheckpointCommitment is an enum and each variant is prefixed with its index
25/// checkpoint-commitment = ecmh-live-object-set
26/// ecmh-live-object-set = %d00 digest
27/// ```
28#[derive(Clone, Debug, Eq, PartialEq)]
29#[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]
30#[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
31#[cfg_attr(feature = "bcs-schema", derive(iota_bcs_schema::BcsSchema))]
32#[non_exhaustive]
33pub enum CheckpointCommitment {
34    /// An Elliptic Curve Multiset Hash attesting to the set of Objects that
35    /// compose the live state of the IOTA blockchain.
36    EcmhLiveObjectSet { digest: Digest },
37    // Other commitment types (e.g. merkle roots) go here.
38}
39
40impl CheckpointCommitment {
41    crate::def_is!(EcmhLiveObjectSet);
42
43    pub fn as_ecmh_live_object_set_digest(&self) -> Digest {
44        let Self::EcmhLiveObjectSet { digest } = self;
45        *digest
46    }
47}

50    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
51        match self {
52            CheckpointCommitment::EcmhLiveObjectSet { digest } => {
53                write!(f, "EcmhLiveObjectSet({digest})")
54            }
55        }
56    }
57}

59/// Data, which when included in a [`CheckpointSummary`], signals the end of an
60/// `Epoch`.
61///
62/// # BCS
63///
64/// The BCS serialized form for this type is defined by the following ABNF:
65///
66/// ```text
67/// end-of-epoch-data = (vector validator-committee-member)   ; next-epoch-committee
68///                     u64                                   ; next-epoch-protocol-version
69///                     (vector checkpoint-commitment)        ; epoch-commitments
70///                     i64                                   ; epoch-supply-change
71/// ```
72#[derive(Clone, Debug, Eq, PartialEq)]
73#[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]

76pub struct EndOfEpochData {
77    /// The set of Validators that will be in the ValidatorCommittee for the
78    /// next epoch.
79    pub next_epoch_committee: Vec<ValidatorCommitteeMember>,
80    /// The protocol version that is in effect during the next epoch.
81    #[cfg_attr(feature = "serde", serde(with = "crate::_serde::ReadableDisplay"))]
82    #[cfg_attr(feature = "bcs-schema", bcs_schema(as_type = "u64"))]
83    pub next_epoch_protocol_version: ProtocolVersion,
84    /// Commitments to epoch specific state (e.g. live object set)
85    pub epoch_commitments: Vec<CheckpointCommitment>,
86    /// The number of tokens that were minted (if positive) or burnt (if
87    /// negative) in this epoch.
88    #[cfg_attr(feature = "serde", serde(with = "crate::_serde::ReadableDisplay"))]
89    pub epoch_supply_change: i64,
90}

106/// A header for a Checkpoint on the IOTA blockchain.
107///
108/// On the IOTA network, checkpoints define the history of the blockchain. They
109/// are quite similar to the concept of blocks used by other blockchains like
110/// Bitcoin or Ethereum. The IOTA blockchain, however, forms checkpoints after
111/// transaction execution has already happened to provide a certified history of
112/// the chain, instead of being formed before execution.
113///
114/// Checkpoints commit to a variety of state including but not limited to:
115/// - The hash of the previous checkpoint.
116/// - The set of transaction digests, their corresponding effects digests, as
117///   well as the set of user signatures which authorized its execution.
118/// - The object's produced by a transaction.
119/// - The set of live objects that make up the current state of the chain.
120/// - On epoch transitions, the next validator committee.
121///
122/// `CheckpointSummary`s themselves don't directly include all of the above
123/// information but they are the top-level type by which all the above are
124/// committed to transitively via cryptographic hashes included in the summary.
125/// `CheckpointSummary`s are signed and certified by a quorum of the validator
126/// committee in a given epoch in order to allow verification of the chain's
127/// state.
128///
129/// # BCS
130///
131/// The BCS serialized form for this type is defined by the following ABNF:
132///
133/// ```text
134/// checkpoint-summary = u64                            ; epoch
135///                      u64                            ; sequence_number
136///                      u64                            ; network_total_transactions
137///                      checkpoint-contents-digest     ; contents_digest
138///                      (option checkpoint-digest)     ; previous_digest
139///                      gas-cost-summary               ; epoch_rolling_gas_cost_summary
140///                      u64                            ; timestamp_ms
141///                      (vector checkpoint-commitment) ; checkpoint_commitments
142///                      (option end-of-epoch-data)     ; end_of_epoch_data
143///                      bytes                          ; version_specific_data
144/// ```
145#[derive(Clone, derive_more::Debug, Eq, PartialEq)]
146#[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]
147#[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
148#[cfg_attr(feature = "bcs-schema", derive(iota_bcs_schema::BcsSchema))]
149pub struct CheckpointSummary {
150    /// Epoch that this checkpoint belongs to.
151    #[cfg_attr(feature = "serde", serde(with = "crate::_serde::ReadableDisplay"))]
152    #[cfg_attr(feature = "bcs-schema", bcs_schema(as_type = "u64"))]
153    pub epoch: EpochId,
154    /// The height of this checkpoint.
155    #[cfg_attr(feature = "serde", serde(with = "crate::_serde::ReadableDisplay"))]
156    #[cfg_attr(feature = "bcs-schema", bcs_schema(as_type = "u64"))]
157    pub sequence_number: CheckpointSequenceNumber,
158    /// Total number of transactions committed since genesis, including those in
159    /// this checkpoint.
160    #[cfg_attr(feature = "serde", serde(with = "crate::_serde::ReadableDisplay"))]
161    pub network_total_transactions: u64,
162    /// The hash of the [`CheckpointContents`] for this checkpoint.
163    pub contents_digest: CheckpointContentsDigest,
164    /// The hash of the previous `CheckpointSummary`.
165    ///
166    /// This will be only be `None` for the first, or genesis checkpoint.
167    #[cfg_attr(feature = "serde", serde(default))]
168    pub previous_digest: Option<CheckpointDigest>,
169    /// The running total gas costs of all transactions included in the current
170    /// epoch so far until this checkpoint.
171    pub epoch_rolling_gas_cost_summary: GasCostSummary,
172    /// Timestamp of the checkpoint - number of milliseconds from the Unix epoch
173    /// Checkpoint timestamps are monotonic, but not strongly monotonic -
174    /// subsequent checkpoints can have same timestamp if they originate
175    /// from the same underlining consensus commit
176    #[cfg_attr(feature = "serde", serde(with = "crate::_serde::ReadableDisplay"))]
177    #[cfg_attr(feature = "bcs-schema", bcs_schema(as_type = "u64"))]
178    pub timestamp_ms: CheckpointTimestamp,
179    /// Commitments to checkpoint-specific state.
180    #[cfg_attr(feature = "serde", serde(default))]
181    pub checkpoint_commitments: Vec<CheckpointCommitment>,
182    /// Extra data only present in the final checkpoint of an epoch.
183    #[cfg_attr(feature = "serde", serde(default))]
184    pub end_of_epoch_data: Option<EndOfEpochData>,
185    /// CheckpointSummary is not an evolvable structure - it must be readable by
186    /// any version of the code. Therefore, in order to allow extensions to
187    /// be added to CheckpointSummary, we allow opaque data to be added to
188    /// checkpoints which can be deserialized based on the current
189    /// protocol version.
190    #[cfg_attr(
191        feature = "serde",
192        serde(default, with = "crate::_serde::ReadableBase64Encoded")
193    )]
194    #[debug(
195        "{:?}",
196        <base64ct::Base64 as base64ct::Encoding>::encode_string(version_specific_data)
197    )]
198    pub version_specific_data: Vec<u8>,
199}

400/// CheckpointContents are the transactions included in an upcoming checkpoint.
401/// They must have already been causally ordered. Since the causal order
402/// algorithm is the same among validators, we expect all honest validators to
403/// come up with the same order for each checkpoint content.
404///
405/// # BCS
406///
407/// The BCS serialized form for this type is defined by the following ABNF:
408///
409/// ```text
410/// checkpoint-contents-v1 = (vector execution-digests)      ; transaction and effect digests
411///                          (vector (vector user-signature)) ; set of user signatures for each
412///                                                           ; transaction. MUST be the same
413///                                                           ; length as the vector of digests
414///
415/// execution-digests = transaction-digest transaction-effects-digest   ; transaction, effects
416/// ```
417#[derive(Clone, Debug, Eq, PartialEq)]
418#[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
419pub struct CheckpointContentsV1 {
420    #[cfg_attr(feature = "proptest", any(proptest::collection::size_range(0..=2).lift()))]
421    transactions: Vec<CheckpointTransactionInfo>,
422}
423
424impl CheckpointContentsV1 {
425    pub fn new(transactions: Vec<CheckpointTransactionInfo>) -> Self {
426        Self { transactions }
427    }
428
429    /// Returns a reference to the list of transactions in this checkpoint.
430    pub fn transactions(&self) -> &[CheckpointTransactionInfo] {
431        &self.transactions
432    }
433
434    /// Consumes the `CheckpointContentsV1` and returns the list of
435    /// transactions.
436    pub fn into_transactions(self) -> Vec<CheckpointTransactionInfo> {
437        self.transactions
438    }
439
440    /// The number of transactions in this checkpoint.
441    pub fn len(&self) -> usize {
442        self.transactions.len()
443    }
444
445    /// Whether this checkpoint has no transactions.
446    pub fn is_empty(&self) -> bool {
447        self.transactions.is_empty()
448    }
449}

557#[cfg(feature = "serde")]
558#[cfg_attr(doc_cfg, doc(cfg(feature = "serde")))]
559mod serialization {
560    use serde::{Deserialize, Deserializer, Serialize, Serializer};
561
562    use super::*;
563
564    impl Serialize for CheckpointContentsV1 {
565        fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
566        where
567            S: Serializer,
568        {
569            use serde::ser::{SerializeSeq, SerializeTuple};
570
571            if serializer.is_human_readable() {
572                serializer.serialize_newtype_struct("CheckpointContentsV1", &self.transactions)
573            } else {
574                #[derive(serde::Serialize)]
575                struct Digests<'a> {
576                    transaction: &'a TransactionDigest,
577                    effects: &'a TransactionEffectsDigest,
578                }
579
580                struct DigestSeq<'a>(&'a CheckpointContentsV1);
581                impl Serialize for DigestSeq<'_> {
582                    fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
583                    where
584                        S: Serializer,
585                    {
586                        let mut seq = serializer.serialize_seq(Some(self.0.len()))?;
587                        for txn in &self.0.transactions {
588                            let digests = Digests {
589                                transaction: &txn.transaction,
590                                effects: &txn.effects,
591                            };
592                            seq.serialize_element(&digests)?;
593                        }
594                        seq.end()
595                    }
596                }
597
598                struct SignatureSeq<'a>(&'a CheckpointContentsV1);
599                impl Serialize for SignatureSeq<'_> {
600                    fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
601                    where
602                        S: Serializer,
603                    {
604                        let mut seq = serializer.serialize_seq(Some(self.0.len()))?;
605                        for txn in &self.0.transactions {
606                            seq.serialize_element(&txn.signatures)?;
607                        }
608                        seq.end()
609                    }
610                }
611
612                let mut s = serializer.serialize_tuple(2)?;
613                s.serialize_element(&DigestSeq(self))?;
614                s.serialize_element(&SignatureSeq(self))?;
615                s.end()
616            }
617        }
618    }

628    #[cfg_attr(
629        feature = "bcs-schema",
630        derive(iota_bcs_schema::BcsSchema),
631        bcs_schema(name = "checkpoint-contents-v1")
632    )]

634        digests: Vec<ExecutionDigests>,
635        signatures: Vec<Vec<UserSignature>>,
636    }

638    impl<'de> Deserialize<'de> for CheckpointContentsV1 {
639        fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
640        where
641            D: Deserializer<'de>,
642        {
643            if deserializer.is_human_readable() {
644                let transactions: Vec<CheckpointTransactionInfo> =
645                    Deserialize::deserialize(deserializer)?;
646                Ok(Self { transactions })
647            } else {
648                let BinaryContentsV1 {
649                    digests,
650                    signatures,
651                } = Deserialize::deserialize(deserializer)?;
652
653                if digests.len() != signatures.len() {
654                    return Err(serde::de::Error::custom(
655                        "must have same number of signatures as transactions",
656                    ));
657                }
658
659                Ok(Self {
660                    transactions: digests
661                        .into_iter()
662                        .zip(signatures)
663                        .map(
664                            |(
665                                ExecutionDigests {
666                                    transaction,
667                                    effects,
668                                },
669                                signatures,
670                            )| CheckpointTransactionInfo {
671                                transaction,
672                                effects,
673                                signatures,
674                            },
675                        )
676                        .collect(),
677                })
678            }
679        }
680    }
