9/// Reference to an object
10///
11/// Contains sufficient information to uniquely identify a specific object.
12///
13/// # BCS
14///
15/// The BCS serialized form for this type is defined by the following ABNF:
16///
17/// ```text
18/// object-reference = object-id u64 object-digest
19/// ```
20#[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
21#[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]
22#[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
23#[cfg_attr(feature = "bcs-schema", derive(iota_bcs_schema::BcsSchema))]
24pub struct ObjectReference {
25    /// The object id of this object.
26    pub object_id: ObjectId,
27    /// The version of this object.
28    pub version: Version,
29    /// The digest of this object.
30    pub digest: ObjectDigest,
31}
32
33impl ObjectReference {
34    /// Creates a new object reference from the object's id, version, and
35    /// digest.
36    pub const fn new(object_id: ObjectId, version: Version, digest: ObjectDigest) -> Self {
37        Self {
38            object_id,
39            version,
40            digest,
41        }
42    }
43
44    /// Returns a reference to the object id that this ObjectReference is
45    /// referring to.
46    pub fn object_id(&self) -> &ObjectId {
47        &self.object_id
48    }
49
50    /// Returns the version of the object that this ObjectReference is referring
51    /// to.
52    pub fn version(&self) -> Version {
53        self.version
54    }
55
56    /// Returns the digest of the object that this ObjectReference is referring
57    /// to.
58    pub fn digest(&self) -> &ObjectDigest {
59        &self.digest
60    }
61
62    /// Returns a 3-tuple containing the object id, version, and digest.
63    pub fn into_parts(self) -> (ObjectId, Version, ObjectDigest) {
64        let Self {
65            object_id,
66            version,
67            digest,
68        } = self;
69
70        (object_id, version, digest)
71    }
72}
73
74impl crate::TreeDisplay for ObjectReference {
75    fn fmt_tree(&self, w: &mut crate::TreeWriter<'_, '_>) -> std::fmt::Result {
76        w.header("Object Reference")?;
77        w.leaf("Object ID", &self.object_id, false)?;
78        w.leaf("Version", &self.version, false)?;
79        w.leaf("Digest", &self.digest, true)
80    }
81}

137/// Enum of different types of ownership for an object.
138///
139/// # BCS
140///
141/// The BCS serialized form for this type is defined by the following ABNF:
142///
143/// ```text
144/// owner = owner-address / owner-object / owner-shared / owner-immutable
145///
146/// owner-address   = %d00 address
147/// owner-object    = %d01 object-id
148/// owner-shared    = %d02 u64
149/// owner-immutable = %d03
150/// ```
151#[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
152#[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]
153#[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
154#[cfg_attr(feature = "bcs-schema", derive(iota_bcs_schema::BcsSchema))]
155#[non_exhaustive]
156pub enum Owner {
157    /// Object is exclusively owned by a single address, and is mutable.
158    Address(Address),
159    /// Object is exclusively owned by a single object, and is mutable.
160    Object(ObjectId),
161    /// Object is shared, can be used by any address, and is mutable.
162    Shared(
163        /// The version at which the object became shared
164        Version,
165    ),
166    /// Object is immutable, and hence ownership doesn't matter.
167    Immutable,
168}
169
170impl Owner {
171    crate::def_is!(Immutable);
172
173    crate::def_is_as_into_opt!(Address, Object(ObjectId), Shared(Version));
174
175    /// Returns an `Address` if this object is owned by an address or
176    /// object, and None if it is shared or immutable.
177    pub fn address_or_object(&self) -> Option<&Address> {
178        Some(match self {
179            Self::Address(address) => address,
180            Self::Object(object_id) => object_id.as_address(),
181            _ => return None,
182        })
183    }
184}
185
186impl PartialEq<Address> for Owner {
187    fn eq(&self, other: &Address) -> bool {
188        self.as_opt_address() == Some(other)
189    }
190}
191
192impl PartialEq<ObjectId> for Owner {
193    fn eq(&self, other: &ObjectId) -> bool {
194        self.as_opt_object() == Some(other)
195    }
196}
197
198impl std::fmt::Display for Owner {
199    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
200        match self {
201            Owner::Address(address) => write!(f, "Address({address})"),
202            Owner::Object(object_id) => write!(f, "Object({object_id})"),
203            Owner::Shared(version) => write!(f, "Shared({version})"),
204            Owner::Immutable => write!(f, "Immutable"),
205        }
206    }
207}

209/// Object data, either a package or struct
210///
211/// # BCS
212///
213/// The BCS serialized form for this type is defined by the following ABNF:
214///
215/// ```text
216/// object-data = object-data-struct / object-data-package
217///
218/// object-data-struct  = %d00 object-move-struct
219/// object-data-package = %d01 object-move-package
220/// ```
221#[derive(Clone, Debug, Eq, Hash, PartialEq)]
222#[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]
223#[allow(clippy::large_enum_variant)]
224#[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
225#[cfg_attr(feature = "bcs-schema", derive(iota_bcs_schema::BcsSchema))]
226// TODO think about hiding this type and not exposing it
227pub enum ObjectData {
228    /// An object whose governing logic lives in a published Move module
229    Struct(MoveStruct),
230    /// Map from each module name to raw serialized Move module bytes
231    Package(MovePackage),
232    // ... IOTA "native" types go here
233}
234
235impl ObjectData {
236    crate::def_is_as_into_opt!(Struct(MoveStruct), Package(MovePackage));
237
238    pub fn opt_object_type(&self) -> Option<&MoveObjectType> {
239        match self {
240            Self::Struct(m) => Some(m.object_type()),
241            Self::Package(_) => None,
242        }
243    }
244
245    pub fn opt_struct_tag(&self) -> Option<StructTag> {
246        match self {
247            Self::Struct(m) => Some(m.struct_tag().clone()),
248            Self::Package(_) => None,
249        }
250    }
251
252    pub fn id(&self) -> ObjectId {
253        match self {
254            Self::Struct(v) => v.id(),
255            Self::Package(m) => m.id(),
256        }
257    }
258}

263        match self {
264            ObjectData::Struct(s) => s.fmt_tree(w),
265            ObjectData::Package(p) => p.fmt_tree(w),
266        }
267    }

270/// A [`StructTag`] with optimized BCS serialization for object types.
271///
272/// GasCoin, StakedIota, and Coin variants use compact enum encoding
273/// instead of the full StructTag representation. The Other variant
274/// carries the full StructTag inline.
275///
276/// # BCS
277///
278/// ```text
279/// compressed-struct-tag = other-struct-type / gas-coin-type / staked-iota-type / coin-type
280/// other-struct-type     = %x00 struct-tag
281/// gas-coin-type         = %x01
282/// staked-iota-type      = %x02
283/// coin-type             = %x03 type-tag
284/// ```
285#[derive(Clone, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
286#[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
287pub struct MoveObjectType(StructTag);
288
289impl MoveObjectType {
290    pub fn new(tag: StructTag) -> Self {
291        Self(tag)
292    }
293
294    pub fn struct_tag(&self) -> &StructTag {
295        &self.0
296    }
297
298    pub fn into_struct_tag(self) -> StructTag {
299        self.0
300    }
301}
302
303impl std::ops::Deref for MoveObjectType {
304    type Target = StructTag;
305
306    fn deref(&self) -> &Self::Target {
307        &self.0
308    }
309}
310
311impl From<StructTag> for MoveObjectType {
312    fn from(tag: StructTag) -> Self {
313        Self(tag)
314    }
315}
316
317impl From<MoveObjectType> for StructTag {
318    fn from(obj_type: MoveObjectType) -> Self {
319        obj_type.0
320    }
321}
322
323impl PartialEq<StructTag> for MoveObjectType {
324    fn eq(&self, other: &StructTag) -> bool {
325        &self.0 == other
326    }
327}
328
329impl std::fmt::Display for MoveObjectType {
330    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {

332    }
333}

343/// A move struct
344///
345/// # BCS
346///
347/// The BCS serialized form for this type is defined by the following ABNF:
348///
349/// ```text
350/// move-struct = compressed-struct-tag u64 bytes
351///
352/// compressed-struct-tag = other-struct-type / gas-coin-type / staked-iota-type / coin-type
353/// other-struct-type     = %d00 struct-tag
354/// gas-coin-type         = %d01
355/// staked-iota-type      = %d02
356/// coin-type             = %d03 type-tag
357///
358/// ; The first 32 bytes of the `bytes` contents are the object's object-id.
359/// ```
360#[derive(Clone, derive_more::Debug, Eq, Hash, PartialEq)]
361#[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]
362#[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
363#[cfg_attr(feature = "bcs-schema", derive(iota_bcs_schema::BcsSchema))]
364pub struct MoveStruct {
365    /// The type of this object. Uses optimized BCS serialization.
366    #[cfg_attr(feature = "bcs-schema", bcs_schema(as_type = "compressed-struct-tag"))]
367    object_type: MoveObjectType,
368    /// Number that increases each time a tx takes this object as a mutable
369    /// input This is a lamport timestamp, not a sequentially increasing
370    /// version
371    version: Version,
372    /// BCS bytes of a Move struct value.
373    ///
374    /// The first [`ObjectId::LENGTH`] bytes are always the object's
375    /// [`ObjectId`].
376    #[cfg_attr(
377        feature = "serde",
378        serde(with = "crate::_serde::ReadableBase64Encoded")
379    )]
380    #[cfg_attr(feature = "proptest", any(proptest::collection::size_range(32..=1024).lift()))]
381    #[debug("{:?}", <base64ct::Base64 as base64ct::Encoding>::encode_string(contents))]
382    contents: Vec<u8>,
383}
384
385impl MoveStruct {
386    /// Creates a new `MoveStruct`.
387    ///
388    /// # Errors
389    ///
390    /// Returns an error if `contents` is shorter than [`ObjectId::LENGTH`]
391    /// bytes, since every Move object must contain its [`ObjectId`] as the
392    /// leading bytes.
393    pub fn new(
394        object_type: MoveObjectType,
395        version: Version,
396        contents: Vec<u8>,
397    ) -> Result<Self, MoveStructContentsError> {
398        if contents.len() < ObjectId::LENGTH {
399            return Err(MoveStructContentsError {
400                actual: contents.len(),
401            });
402        }
403        Ok(Self {
404            object_type,
405            version,
406            contents,
407        })
408    }
409
410    /// Returns the type of this Move object.
411    pub fn object_type(&self) -> &MoveObjectType {
412        &self.object_type
413    }
414
415    /// Returns the object type as a [`StructTag`] reference.
416    pub fn struct_tag(&self) -> &StructTag {
417        &self.object_type
418    }
419
420    /// Returns `true` if the object's type matches the given [`StructTag`].
421    pub fn is_struct_tag(&self, s: &StructTag) -> bool {
422        &self.object_type == s
423    }
424
425    /// Returns the object's ID, extracted from the BCS-encoded contents.
426    ///
427    /// This is always valid because the constructor guarantees that `contents`
428    /// is at least [`ObjectId::LENGTH`] bytes long.
429    pub fn id(&self) -> ObjectId {
430        ObjectId::from_bytes(&self.contents[..ObjectId::LENGTH]).unwrap()
431    }
432
433    /// Returns the version (lamport timestamp) of this object.
434    pub fn version(&self) -> Version {
435        self.version
436    }
437
438    /// Sets the version (lamport timestamp) of this object.
439    pub fn set_version(&mut self, version: Version) {
440        self.version = version;
441    }
442
443    /// Sets the type of this object.
444    ///
445    /// The caller must ensure the existing [`contents`](Self::contents) are a
446    /// valid BCS encoding of the new `object_type`; this is not verified.
447    pub fn set_object_type(&mut self, object_type: MoveObjectType) {
448        self.object_type = object_type;
449    }
450
451    /// Returns the raw BCS-encoded contents of this object.
452    pub fn contents(&self) -> &[u8] {
453        &self.contents
454    }
455
456    /// Replaces the BCS-encoded contents of this object.
457    ///
458    /// The caller must ensure the new contents are a valid BCS encoding of the
459    /// object's [`object_type`](Self::object_type); this is not verified.
460    ///
461    /// # Errors
462    ///
463    /// Returns an error if `contents` is shorter than [`ObjectId::LENGTH`]
464    /// bytes.
465    pub fn set_contents(&mut self, contents: Vec<u8>) -> Result<(), MoveStructContentsError> {
466        if contents.len() < ObjectId::LENGTH {
467            return Err(MoveStructContentsError {
468                actual: contents.len(),
469            });
470        }
471        self.contents = contents;
472        Ok(())
473    }
474
475    /// Consumes the object and returns the raw BCS-encoded contents.
476    pub fn into_contents(self) -> Vec<u8> {
477        self.contents
478    }
479
480    /// Returns the object type as a [`TypeTag`].
481    pub fn type_tag(&self) -> TypeTag {
482        TypeTag::Struct(Box::new(self.struct_tag().clone()))
483    }
484
485    /// Consumes the object and returns its type, version, and raw contents.
486    pub fn into_parts(self) -> (MoveObjectType, Version, Vec<u8>) {
487        (self.object_type, self.version, self.contents)
488    }
489
490    /// Deserializes the BCS-encoded contents into a Rust type.
491    #[cfg(feature = "serde")]
492    pub fn to_rust<'de, T: serde::Deserialize<'de>>(&'de self) -> Result<T, bcs::Error> {
493        bcs::from_bytes(self.contents())
494    }
495}

497impl crate::TreeDisplay for MoveStruct {
498    fn fmt_tree(&self, w: &mut crate::TreeWriter<'_, '_>) -> std::fmt::Result {
499        w.header("Move Struct")?;
500        w.leaf("Type", &self.object_type, false)?;
501        w.leaf("Version", &self.version, false)?;
502        w.leaf("Contents", &hex::encode(&self.contents), true)
503    }
504}

509#[error(
510    "MoveStruct contents must be at least {} bytes to contain an ObjectId, got {actual}",
511    ObjectId::LENGTH

513pub struct MoveStructContentsError {

515}

541/// An object on the IOTA blockchain
542///
543/// # BCS
544///
545/// The BCS serialized form for this type is defined by the following ABNF:
546///
547/// ```text
548/// object = object-data owner digest u64
549/// ```
550#[derive(Clone, Debug, Eq, Hash, PartialEq)]
551#[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]
552#[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
553#[cfg_attr(feature = "bcs-schema", derive(iota_bcs_schema::BcsSchema))]
554pub struct Object {
555    /// The meat of the object
556    pub data: ObjectData,
557    /// The owner that unlocks this object
558    pub owner: Owner,
559    /// The digest of the transaction that created or last mutated this object
560    pub previous_transaction: TransactionDigest,
561    /// The amount of IOTA we would rebate if this object gets deleted.
562    /// This number is re-calculated each time the object is mutated based on
563    /// the present storage gas price.
564    #[cfg_attr(feature = "serde", serde(with = "crate::_serde::ReadableDisplay"))]
565    pub storage_rebate: u64,
566}
567
568impl Object {
569    /// Build an object
570    pub fn new(
571        data: ObjectData,
572        owner: Owner,
573        previous_transaction: TransactionDigest,
574        storage_rebate: u64,
575    ) -> Self {
576        Self {
577            data,
578            owner,
579            previous_transaction,
580            storage_rebate,
581        }
582    }
583
584    /// Return this object's id
585    pub fn id(&self) -> ObjectId {
586        match &self.data {
587            ObjectData::Struct(struct_) => struct_.id(),
588            ObjectData::Package(package) => package.id,
589        }
590    }
591
592    /// Return this object's reference
593    #[cfg(all(feature = "hash", feature = "serde"))]
594    pub fn object_ref(&self) -> ObjectReference {
595        ObjectReference {
596            object_id: self.id(),
597            version: self.version(),
598            digest: self.digest(),
599        }
600    }
601
602    /// Return this object's version
603    pub fn version(&self) -> Version {
604        match &self.data {
605            ObjectData::Struct(struct_) => struct_.version(),
606            ObjectData::Package(package) => package.version,
607        }
608    }
609
610    /// Return this object's type
611    pub fn object_type(&self) -> ObjectType {
612        match &self.data {
613            ObjectData::Struct(struct_) => ObjectType::Struct(struct_.struct_tag().clone()),
614            ObjectData::Package(_) => ObjectType::Package,
615        }
616    }
617
618    /// Try to interpret this object as a move struct
619    pub fn as_opt_struct(&self) -> Option<&MoveStruct> {
620        match &self.data {
621            ObjectData::Struct(struct_) => Some(struct_),
622            _ => None,
623        }
624    }
625
626    /// Interpret this object as a move struct
627    pub fn as_struct(&self) -> &MoveStruct {
628        self.as_opt_struct().expect("not a move struct")
629    }
630
631    /// Try to interpret this object as a move package
632    pub fn as_opt_package(&self) -> Option<&MovePackage> {
633        match &self.data {
634            ObjectData::Package(package) => Some(package),
635            _ => None,
636        }
637    }
638
639    /// Interpret this object as a move package
640    pub fn as_package(&self) -> &MovePackage {
641        self.as_opt_package().expect("not a move package")
642    }
643
644    /// Return this object's owner
645    pub fn owner(&self) -> &Owner {
646        &self.owner
647    }
648
649    /// Return this object's data
650    pub fn data(&self) -> &ObjectData {
651        &self.data
652    }
653
654    /// Return the digest of the transaction that last modified this object
655    pub fn previous_transaction(&self) -> TransactionDigest {
656        self.previous_transaction
657    }
658
659    /// Return the storage rebate locked in this object
660    ///
661    /// Storage rebates are credited to the gas coin used in a transaction that
662    /// deletes this object.
663    pub fn storage_rebate(&self) -> u64 {
664        self.storage_rebate
665    }
666
667    #[cfg(feature = "serde")]
668    pub fn to_rust<'de, T: serde::Deserialize<'de>>(
669        &'de self,
670    ) -> Result<T, Box<dyn std::error::Error + Send + Sync>> {
671        let contents = self.as_opt_struct().ok_or("not a struct")?.contents();
672        Ok(bcs::from_bytes::<T>(contents)?)
673    }
674
675    /// Returns true if the object is immutable.
676    pub fn is_immutable(&self) -> bool {
677        self.owner.is_immutable()
678    }
679
680    /// Returns true if the object is owned by an address.
681    pub fn is_address_owned(&self) -> bool {
682        self.owner.is_address()
683    }
684
685    /// Returns true if the object is owned by another object.
686    pub fn is_child_object(&self) -> bool {
687        self.owner.is_object()
688    }
689
690    /// Returns true if the object is shared.
691    pub fn is_shared(&self) -> bool {
692        self.owner.is_shared()
693    }
694
695    /// Returns true if this object is a Move package rather than a Move value.
696    pub fn is_package(&self) -> bool {
697        self.data.is_package()
698    }
699
700    /// Returns true if the object is a system package.
701    pub fn is_system_package(&self) -> bool {
702        self.is_package() && self.id().is_system_package()
703    }
704
705    /// Returns the struct tag of this object if it is a Move struct.
706    pub fn struct_tag(&self) -> Option<StructTag> {
707        self.data.opt_struct_tag()
708    }
709
710    /// Returns true if this object is a gas coin.
711    pub fn is_gas_coin(&self) -> bool {
712        self.as_opt_struct()
713            .is_some_and(|move_object| move_object.struct_tag().is_gas_coin())
714    }

746///
747/// `GenesisObject`'s are included as a part of genesis, the initial
748/// checkpoint/transaction, that initializes the state of the blockchain.
749///
750/// # BCS
751///
752/// The BCS serialized form for this type is defined by the following ABNF:
753///
754/// ```text
755/// genesis-object = %d00 object-data owner   ; RawObject
756/// ```
757#[derive(Clone, Debug, Eq, Hash, PartialEq)]
758#[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
759pub struct GenesisObject {
760    pub data: ObjectData,
761    pub owner: Owner,
762}
763
764impl GenesisObject {
765    pub fn new(data: ObjectData, owner: Owner) -> Self {
766        Self { data, owner }
767    }
768
769    pub fn object_id(&self) -> ObjectId {
770        match &self.data {
771            ObjectData::Struct(struct_) => struct_.id(),
772            ObjectData::Package(package) => package.id,
773        }
774    }
775
776    pub fn version(&self) -> Version {
777        match &self.data {
778            ObjectData::Struct(struct_) => struct_.version(),
779            ObjectData::Package(package) => package.version,
780        }
781    }
782
783    pub fn object_type(&self) -> ObjectType {
784        match &self.data {
785            ObjectData::Struct(struct_) => ObjectType::Struct(struct_.struct_tag().clone()),
786            ObjectData::Package(_) => ObjectType::Package,
787        }
788    }
789
790    pub fn owner(&self) -> &Owner {
791        &self.owner
792    }
793
794    pub fn data(&self) -> &ObjectData {
795        &self.data
796    }
797
798    pub fn id(&self) -> ObjectId {
799        self.data.id()
800    }
801}

831    /// Wrapper around StructTag with a space-efficient representation for
832    /// common types like coins The StructTag for a gas coin is 84 bytes, so
833    /// using 1 byte instead is a win. The inner representation is private
834    /// to prevent incorrectly constructing an `Other` instead of one of the
835    /// specialized variants, e.g. `Other(GasCoin::type_())` instead of
836    /// `GasCoin`
837    #[derive(serde::Deserialize)]
838    #[serde(rename = "MoveObjectType")]
839    #[cfg_attr(
840        feature = "bcs-schema",
841        derive(iota_bcs_schema::BcsSchema),
842        bcs_schema(name = "compressed-struct-tag")
843    )]
844    enum MoveObjectTypeWrapper {
845        /// A type that is not `0x2::coin::Coin<T>`
846        Other(StructTag),
847        /// An IOTA coin (i.e., `0x2::coin::Coin<0x2::iota::IOTA>`)
848        GasCoin,
849        /// A record of a staked IOTA coin (i.e.,
850        /// `0x3::staking_pool::StakedIota`)
851        StakedIota,
852        /// A non-IOTA coin type (i.e., `0x2::coin::Coin<T> where T !=
853        /// 0x2::iota::IOTA`)
854        Coin(TypeTag),
855        // NOTE: if adding a new type here, and there are existing on-chain objects of that
856        // type with Other(_), that is ok, but you must hand-roll PartialEq/Eq/Ord/maybe Hash
857        // to make sure the new type and Other(_) are interpreted consistently.
858    }

860    /// See `MoveObjectType`
861    #[derive(serde::Serialize)]
862    #[serde(rename = "MoveObjectType")]
863    enum MoveObjectTypeRef<'a> {
864        /// A type that is not `0x2::coin::Coin<T>`
865        Other(&'a StructTag),
866        /// An IOTA coin (i.e., `0x2::coin::Coin<0x2::iota::IOTA>`)
867        GasCoin,
868        /// A record of a staked IOTA coin (i.e.,
869        /// `0x3::staking_pool::StakedIota`)
870        StakedIota,
871        /// A non-IOTA coin type (i.e., `0x2::coin::Coin<T> where T !=
872        /// 0x2::iota::IOTA`)
873        Coin(&'a TypeTag),
874        // NOTE: if adding a new type here, and there are existing on-chain objects of that
875        // type with Other(_), that is ok, but you must hand-roll PartialEq/Eq/Ord/maybe Hash
876        // to make sure the new type and Other(_) are interpreted consistently.
877    }

879    impl MoveObjectTypeWrapper {
880        fn into_struct_tag(self) -> StructTag {
881            match self {
882                MoveObjectTypeWrapper::Other(tag) => tag,
883                MoveObjectTypeWrapper::GasCoin => StructTag::new_gas_coin(),
884                MoveObjectTypeWrapper::StakedIota => StructTag::new_staked_iota(),
885                MoveObjectTypeWrapper::Coin(type_tag) => StructTag::new_coin(type_tag),
886            }
887        }
888    }

890    impl<'a> MoveObjectTypeRef<'a> {
891        fn from_struct_tag(s: &'a StructTag) -> Self {
892            if let Some(coin_type) = s.opt_coin_type() {
893                if let TypeTag::Struct(s_inner) = coin_type
894                    && s_inner.address() == Address::FRAMEWORK
895                    && s_inner.module() == "iota"
896                    && s_inner.name() == "IOTA"
897                    && s_inner.type_params().is_empty()
898                {
899                    return Self::GasCoin;
900                }
901
902                Self::Coin(coin_type)
903            } else if s.address() == Address::SYSTEM
904                && s.module() == "staking_pool"
905                && s.name() == "StakedIota"
906                && s.type_params().is_empty()
907            {
908                Self::StakedIota
909            } else {
910                Self::Other(s)
911            }
912        }
913    }

921                self.0.serialize(serializer)
