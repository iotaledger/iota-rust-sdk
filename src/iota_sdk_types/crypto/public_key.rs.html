25/// Enum of valid public keys for the signature schemes supported by IOTA.
26///
27/// # BCS
28///
29/// The BCS serialized form for this type is defined by the following ABNF:
30///
31/// ```text
32/// public-key = %d00 ed25519-public-key /
33///              %d01 secp256k1-public-key /
34///              %d02 secp256r1-public-key /
35///              %d04 passkey-public-key
36/// ```
37///
38/// The gap in the flag values is intentional, as not all signature scheme
39/// support public keys.
40///
41/// There is also a base64 encoding for this type, used by [`Self::to_base64`]
42/// and [`Self::from_base64`], defined as:
43///
44/// ```text
45/// base64-public-key = string ; which is valid base64 encoded
46///                            ; and the decoded bytes are defined
47///                            ; by flagged-public-key
48/// flagged-public-key = (ed25519-flag ed25519-public-key) /
49///                      (secp256k1-flag secp256k1-public-key) /
50///                      (secp256r1-flag secp256r1-public-key) /
51///                      (passkey-flag passkey-public-key)
52/// ```
53#[derive(Clone, Debug, derive_more::Display, derive_more::From, Eq, Hash, PartialEq)]
54#[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
55#[non_exhaustive]
56pub enum PublicKey {
57    #[display("Ed25519PublicKey({_0})")]
58    Ed25519(Ed25519PublicKey),
59    #[display("Secp256k1PublicKey({_0})")]
60    Secp256k1(Secp256k1PublicKey),
61    #[display("Secp256r1PublicKey({_0})")]
62    Secp256r1(Secp256r1PublicKey),
63    #[display("PasskeyPublicKey({_0})")]
64    Passkey(PasskeyPublicKey),
65}
66
67impl PublicKey {
68    crate::def_is_as_into_opt!(
69        Ed25519(Ed25519PublicKey),
70        Secp256k1(Secp256k1PublicKey),
71        Secp256r1(Secp256r1PublicKey),
72        Passkey(PasskeyPublicKey),
73    );
74
75    /// Return the flag for this signature scheme
76    pub fn scheme(&self) -> SignatureScheme {
77        match self {
78            PublicKey::Ed25519(pk) => pk.scheme(),
79            PublicKey::Secp256k1(pk) => pk.scheme(),
80            PublicKey::Secp256r1(pk) => pk.scheme(),
81            PublicKey::Passkey(pk) => pk.scheme(),
82        }
83    }
84
85    /// Encode this public key as a base64 string of its scheme-flagged byte
86    /// representation
87    pub fn to_base64(&self) -> String {
88        let mut bytes: Vec<u8> = Vec::new();
89
90        bytes.extend_from_slice(&[self.scheme() as u8]);
91        bytes.extend_from_slice(self.as_ref());
92
93        Base64::encode_string(&bytes)
94    }
95
96    /// Decode a public key from a base64 string of its scheme-flagged byte
97    /// representation
98    pub fn from_base64(s: &str) -> Result<Self, PublicKeyError> {
99        let bytes = Base64::decode_vec(s)?;
100
101        match bytes.split_first() {
102            Some((flag, tail)) => match SignatureScheme::from_byte(*flag)? {
103                SignatureScheme::Ed25519 => {
104                    let pk = Ed25519PublicKey::from_bytes(tail)?;
105                    Ok(Self::Ed25519(pk))
106                }
107                SignatureScheme::Secp256k1 => {
108                    let pk = Secp256k1PublicKey::from_bytes(tail)?;
109                    Ok(Self::Secp256k1(pk))
110                }
111                SignatureScheme::Secp256r1 => {
112                    let pk = Secp256r1PublicKey::from_bytes(tail)?;
113                    Ok(Self::Secp256r1(pk))
114                }
115                SignatureScheme::PasskeyAuthenticator => {
116                    let pk = PasskeyPublicKey::new(Secp256r1PublicKey::from_bytes(tail)?);
117                    Ok(Self::Passkey(pk))
118                }
119                _ => Err(PublicKeyError::InvalidInput),
120            },
121            None => Err(PublicKeyError::InvalidInput),
122        }
123    }
124}
125
126impl From<PasskeyAuthenticator> for PublicKey {
127    fn from(passkey_authenticator: PasskeyAuthenticator) -> Self {
128        Self::Passkey(passkey_authenticator.public_key())
129    }
130}
131
132impl AsRef<[u8]> for PublicKey {
133    fn as_ref(&self) -> &[u8] {
134        match self {
135            Self::Ed25519(pk) => pk.as_ref(),
136            Self::Secp256k1(pk) => pk.as_ref(),
137            Self::Secp256r1(pk) => pk.as_ref(),
138            Self::Passkey(pk) => pk.as_ref(),
139        }
140    }
141}

145mod serialization {
146    use std::str::FromStr;
147
148    use serde::{Deserialize, Deserializer, Serialize, Serializer};
149
150    use super::*;
151    use crate::{Ed25519PublicKey, PasskeyPublicKey, Secp256k1PublicKey, Secp256r1PublicKey};
152
153    /// Flat wire shape for `PublicKey`.
154    ///
155    /// The BCS variant tags are `PublicKey`'s own historical flag values
156    /// (0x00 Ed25519, 0x01 Secp256k1, 0x02 Secp256r1, 0x04 Passkey) — note
157    /// that passkey public keys use tag `0x04` here even though the passkey
158    /// *signature scheme* flag is `0x06`. The `ZkLoginDeprecated` placeholder
159    /// holds the `0x03` slot and is rejected by the deserializer.
160    #[derive(Deserialize, Serialize)]
161    #[cfg_attr(
162        feature = "bcs-schema",
163        derive(iota_bcs_schema::BcsSchema),
164        bcs_schema(name = "public-key")
165    )]
166    enum PublicKeyBody {
167        Ed25519(Ed25519PublicKey),
168        Secp256k1(Secp256k1PublicKey),
169        Secp256r1(Secp256r1PublicKey),
170        #[cfg_attr(feature = "bcs-schema", bcs_schema(skip))]
171        ZkLoginDeprecated,
172        Passkey(PasskeyPublicKey),
173    }
174
175    #[derive(Deserialize, Serialize)]
176    #[serde(tag = "scheme", rename_all = "lowercase")]
177    #[serde(rename = "PublicKey")]
178    enum ReadablePublicKey {
179        Ed25519 { public_key: Ed25519PublicKey },
180        Secp256k1 { public_key: Secp256k1PublicKey },
181        Secp256r1 { public_key: Secp256r1PublicKey },
182        ZkLoginDeprecated,
183        Passkey { public_key: PasskeyPublicKey },
184    }
185
186    impl Serialize for PublicKey {
187        fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
188        where
189            S: Serializer,
190        {
191            if serializer.is_human_readable() {
192                let readable = match self {
193                    PublicKey::Ed25519(public_key) => ReadablePublicKey::Ed25519 {
194                        public_key: *public_key,
195                    },
196                    PublicKey::Secp256k1(public_key) => ReadablePublicKey::Secp256k1 {
197                        public_key: *public_key,
198                    },
199                    PublicKey::Secp256r1(public_key) => ReadablePublicKey::Secp256r1 {
200                        public_key: *public_key,
201                    },
202                    PublicKey::Passkey(public_key) => ReadablePublicKey::Passkey {
203                        public_key: public_key.clone(),
204                    },
205                };
206                readable.serialize(serializer)
207            } else {
208                let body = match self {
209                    PublicKey::Ed25519(public_key) => PublicKeyBody::Ed25519(*public_key),
210                    PublicKey::Secp256k1(public_key) => PublicKeyBody::Secp256k1(*public_key),
211                    PublicKey::Secp256r1(public_key) => PublicKeyBody::Secp256r1(*public_key),
212                    PublicKey::Passkey(public_key) => PublicKeyBody::Passkey(public_key.clone()),
213                };
214                body.serialize(serializer)
215            }
216        }
217    }
218
219    impl<'de> Deserialize<'de> for PublicKey {
220        fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
221        where
222            D: Deserializer<'de>,
223        {
224            if deserializer.is_human_readable() {
225                let readable = ReadablePublicKey::deserialize(deserializer)?;
226                Ok(match readable {
227                    ReadablePublicKey::Ed25519 { public_key } => Self::Ed25519(public_key),
228                    ReadablePublicKey::Secp256k1 { public_key } => Self::Secp256k1(public_key),
229                    ReadablePublicKey::Secp256r1 { public_key } => Self::Secp256r1(public_key),
230                    ReadablePublicKey::ZkLoginDeprecated => {
231                        return Err(serde::de::Error::custom(
232                            "zkLoginDeprecated is not supported",
233                        ));
234                    }
235                    ReadablePublicKey::Passkey { public_key } => Self::Passkey(public_key),
236                })
237            } else {
238                let body = PublicKeyBody::deserialize(deserializer)?;
239                Ok(match body {
240                    PublicKeyBody::Ed25519(public_key) => Self::Ed25519(public_key),
241                    PublicKeyBody::Secp256k1(public_key) => Self::Secp256k1(public_key),
242                    PublicKeyBody::Secp256r1(public_key) => Self::Secp256r1(public_key),
243                    PublicKeyBody::ZkLoginDeprecated => {
244                        return Err(serde::de::Error::custom(
245                            "zkLoginDeprecated is not supported",
246                        ));
247                    }
248                    PublicKeyBody::Passkey(public_key) => Self::Passkey(public_key),
249                })
250            }
251        }
252    }
253
254    impl FromStr for PublicKey {
255        type Err = PublicKeyError;
256
257        fn from_str(s: &str) -> Result<Self, Self::Err> {
258            Self::from_base64(s)
259        }
260    }
261}

275            UserSignature::from_base64(passkey_b64).unwrap()
