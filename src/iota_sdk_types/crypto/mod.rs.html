34/// Error returned when decoding a fixed-length value from a base64 string.
35#[derive(Clone, Copy, Debug, Eq, PartialEq, thiserror::Error)]
36#[non_exhaustive]
37pub enum Base64ParseError {
38    /// The input is not valid base64.
39    #[error("invalid Base64 encoding")]
40    InvalidEncoding,
41    /// The decoded bytes do not have the expected length.
42    #[error("invalid Base64 length")]
43    InvalidLength,
44}
45
46impl Base64ParseError {
47    fn from_base64ct(error: base64ct::Error) -> Self {
48        match error {
49            base64ct::Error::InvalidEncoding => Self::InvalidEncoding,
50            base64ct::Error::InvalidLength => Self::InvalidLength,
51        }
52    }
53}

55/// Error returned when decoding a signature or authenticator from its bytes.
56#[cfg(feature = "serde")]
57#[derive(Debug, thiserror::Error)]
58#[non_exhaustive]
59pub enum SignatureFromBytesError {
60    /// The input is not valid base64.
61    #[error("invalid base64")]
62    Base64,
63    /// The bytes are not a valid encoding of the target type.
64    #[error("error deserializing bytes: {0}")]
65    Bytes(String),
66}
67
68#[cfg(feature = "serde")]
69impl SignatureFromBytesError {
70    fn new(msg: impl core::fmt::Display) -> Self {
71        Self::Bytes(msg.to_string())
72    }
73}

78/// Utility for calculating base64 encoding lengths.
79///
80/// In the Base64 encoding each character is used to represent 6 bits (log2(64)
81/// = 6). This means that 4 characters are used to represent 4*6 = 24 bits = 3
82/// bytes. So you need 4*(`n`/3) characters in order to represent `n` bytes, and
83/// this needs to be rounded up to a multiple of 4. The number of unused padding
84/// characters resulting from the rounding will be 0, 1, 2, or 3.
85const fn base64_encoded_length(len: usize) -> usize {
86    ((4 * len / 3) + 3) & !3
87}

89macro_rules! impl_base64_helper {
90    ($base:ident, $display:ident, $fromstr:ident, $test_module:ident, $array_length:literal) => {
91        #[allow(unused)]
92        struct $base;
93
94        impl $base {
95            const LENGTH: usize = $array_length;
96            #[allow(unused)]
97            const ENCODED_LENGTH: usize = base64_encoded_length(Self::LENGTH);
98        }
99
100        #[allow(unused)]
101        struct $display<'a>(&'a [u8; $base::LENGTH]);
102
103        impl<'a> std::fmt::Display for $display<'a> {
104            fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
105                let mut buf = [0; $base::ENCODED_LENGTH];
106                let encoded =
107                    <base64ct::Base64 as base64ct::Encoding>::encode(self.0, &mut buf).unwrap();
108                f.write_str(encoded)
109            }
110        }
111
112        #[allow(unused)]
113        #[derive(Debug, PartialEq)]
114        #[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
115        struct $fromstr([u8; $base::LENGTH]);
116
117        impl std::str::FromStr for $fromstr {
118            type Err = Base64ParseError;
119
120            fn from_str(s: &str) -> Result<Self, Self::Err> {
121                let mut buf = [0; $base::LENGTH];
122                let decoded = <base64ct::Base64 as base64ct::Encoding>::decode(s, &mut buf)
123                    .map_err(Base64ParseError::from_base64ct)?;
124                if decoded.len() != $base::LENGTH {
125                    return Err(Base64ParseError::InvalidLength);
126                }
127                Ok(Self(buf))
128            }
129        }
130
131        #[cfg(feature = "serde")]
132        impl serde_with::SerializeAs<[u8; Self::LENGTH]> for $base {
133            fn serialize_as<S>(
134                source: &[u8; Self::LENGTH],
135                serializer: S,
136            ) -> Result<S::Ok, S::Error>
137            where
138                S: serde::Serializer,
139            {
140                let display = $display(source);
141                serde_with::DisplayFromStr::serialize_as(&display, serializer)
142            }
143        }
144
145        #[cfg(feature = "serde")]
146        impl<'de> serde_with::DeserializeAs<'de, [u8; Self::LENGTH]> for $base {
147            fn deserialize_as<D>(deserializer: D) -> Result<[u8; Self::LENGTH], D::Error>
148            where
149                D: serde::Deserializer<'de>,
150            {
151                let array: $fromstr = serde_with::DisplayFromStr::deserialize_as(deserializer)?;
152                Ok(array.0)
153            }
154        }
155
156        #[cfg(all(test, feature = "proptest"))]
157        mod $test_module {
158            use test_strategy::proptest;
159
160            use super::{Base64ParseError, $display, $fromstr};
161
162            #[proptest]
163            fn roundtrip_display_fromstr(array: $fromstr) {
164                let s = $display(&array.0).to_string();
165                let a = s.parse::<$fromstr>().unwrap();
166                assert_eq!(array, a);
167            }
168
169            // Fuzz the `FromStr` impl to ensure it never panics on
170            // arbitrary input. The strategy is restricted to the base64
171            // alphabet so that decode often succeeds, which exercises the
172            // length-validation path rather than bailing out early on
173            // invalid characters.
174            #[proptest]
175            fn fromstr_does_not_panic(#[strategy("[A-Za-z0-9+/=]{0,128}")] s: String) {
176                let _ = s.parse::<$fromstr>();
177            }
178
179            #[test]
180            fn short_decode_errors() {
181                assert_eq!(
182                    "AAAA".parse::<$fromstr>().unwrap_err(),
183                    Base64ParseError::InvalidLength
184                );
185                assert_eq!(
186                    "!!!!".parse::<$fromstr>().unwrap_err(),
187                    Base64ParseError::InvalidEncoding
188                );
189            }
190        }
191    };
192}

201pub trait PublicKeyExt: Sized {
202    type FromBytesErr;
203
204    /// Returns the public key as bytes.
205    fn as_bytes(&self) -> &[u8];
206
207    /// Tries to create a PublicKey from bytes.
208    fn from_bytes(bytes: impl AsRef<[u8]>) -> Result<Self, Self::FromBytesErr>;
209
210    /// Returns the signature scheme for this public key.
211    fn scheme(&self) -> SignatureScheme;
212
213    /// Returns the bytes with signature scheme flag prepended
214    fn to_flagged_bytes(&self) -> Vec<u8> {
215        let key_bytes = self.as_bytes();
216        let mut bytes = Vec::with_capacity(1 + key_bytes.len());
217        bytes.push(self.scheme().to_u8());
218        bytes.extend_from_slice(key_bytes);
219        bytes
220    }
221}
