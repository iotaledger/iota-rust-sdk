49/// Utility for calculating base64 encoding lengths.
50///
51/// In the Base64 encoding each character is used to represent 6 bits (log2(64)
52/// = 6). This means that 4 characters are used to represent 4*6 = 24 bits = 3
53/// bytes. So you need 4*(`n`/3) characters in order to represent `n` bytes, and
54/// this needs to be rounded up to a multiple of 4. The number of unused padding
55/// characters resulting from the rounding will be 0, 1, 2, or 3.
56const fn base64_encoded_length(len: usize) -> usize {
57    ((4 * len / 3) + 3) & !3
58}
59
60macro_rules! impl_base64_helper {
61    ($base:ident, $display:ident, $fromstr:ident, $test_module:ident, $array_length:literal) => {
62        #[allow(unused)]
63        struct $base;
64
65        impl $base {
66            const LENGTH: usize = $array_length;
67            #[allow(unused)]
68            const ENCODED_LENGTH: usize = base64_encoded_length(Self::LENGTH);
69        }
70
71        #[allow(unused)]
72        struct $display<'a>(&'a [u8; $base::LENGTH]);
73
74        impl<'a> std::fmt::Display for $display<'a> {
75            fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
76                let mut buf = [0; $base::ENCODED_LENGTH];
77                let encoded =
78                    <base64ct::Base64 as base64ct::Encoding>::encode(self.0, &mut buf).unwrap();
79                f.write_str(encoded)
80            }
81        }
82
83        #[allow(unused)]
84        #[derive(Debug, PartialEq)]
85        #[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
86        struct $fromstr([u8; $base::LENGTH]);
87
88        impl std::str::FromStr for $fromstr {
89            type Err = base64ct::Error;
90
91            fn from_str(s: &str) -> Result<Self, Self::Err> {
92                let mut buf = [0; $base::LENGTH];
93                let decoded = <base64ct::Base64 as base64ct::Encoding>::decode(s, &mut buf)?;
94                if decoded.len() != $base::LENGTH {
95                    return Err(base64ct::Error::InvalidLength);
96                }
97                Ok(Self(buf))
98            }
99        }
100
101        #[cfg(feature = "serde")]
102        #[cfg_attr(doc_cfg, doc(cfg(feature = "serde")))]
103        impl serde_with::SerializeAs<[u8; Self::LENGTH]> for $base {
104            fn serialize_as<S>(
105                source: &[u8; Self::LENGTH],
106                serializer: S,
107            ) -> Result<S::Ok, S::Error>
108            where
109                S: serde::Serializer,
110            {
111                let display = $display(source);
112                serde_with::DisplayFromStr::serialize_as(&display, serializer)
113            }
114        }
115
116        #[cfg(feature = "serde")]
117        #[cfg_attr(doc_cfg, doc(cfg(feature = "serde")))]
118        impl<'de> serde_with::DeserializeAs<'de, [u8; Self::LENGTH]> for $base {
119            fn deserialize_as<D>(deserializer: D) -> Result<[u8; Self::LENGTH], D::Error>
120            where
121                D: serde::Deserializer<'de>,
122            {
123                let array: $fromstr = serde_with::DisplayFromStr::deserialize_as(deserializer)?;
124                Ok(array.0)
125            }
126        }
127
128        #[cfg(all(test, feature = "proptest"))]
129        mod $test_module {
130            use test_strategy::proptest;
131
132            use super::{$display, $fromstr};
133
134            #[proptest]
135            fn roundtrip_display_fromstr(array: $fromstr) {
136                let s = $display(&array.0).to_string();
137                let a = s.parse::<$fromstr>().unwrap();
138                assert_eq!(array, a);
139            }
140
141            // Fuzz the `FromStr` impl to ensure it never panics on
142            // arbitrary input. The strategy is restricted to the base64
143            // alphabet so that decode often succeeds, which exercises the
144            // length-validation path rather than bailing out early on
145            // invalid characters.
146            #[proptest]
147            fn fromstr_does_not_panic(#[strategy("[A-Za-z0-9+/=]{0,128}")] s: String) {
148                let _ = s.parse::<$fromstr>();
149            }
150
151            #[test]
152            fn short_decode_errors() {
153                assert_eq!(
154                    "AAAA".parse::<$fromstr>().unwrap_err(),
155                    base64ct::Error::InvalidLength
156                );
157            }
158        }
159    };
160}

169pub trait PublicKeyExt: Sized {
170    type FromBytesErr;
171
172    /// Returns the public key as bytes.
173    fn as_bytes(&self) -> &[u8];
174
175    /// Tries to create a PublicKey from bytes.
176    fn from_bytes(bytes: impl AsRef<[u8]>) -> Result<Self, Self::FromBytesErr>;
177
178    /// Returns the signature scheme for this public key.
179    fn scheme(&self) -> SignatureScheme;
180
181    /// Returns the bytes with signature scheme flag prepended
182    fn to_flagged_bytes(&self) -> Vec<u8> {
183        let key_bytes = self.as_bytes();
184        let mut bytes = Vec::with_capacity(1 + key_bytes.len());
185        bytes.push(self.scheme().to_u8());
186        bytes.extend_from_slice(key_bytes);
187        bytes
188    }
