4use std::str::FromStr;
5
6use crate::{ObjectId, address::Address, iota_names::error::IotaNamesError};
7
8#[derive(Clone, Debug, Eq, PartialEq)]
9#[cfg_attr(
10    feature = "serde",
11    derive(serde::Deserialize, serde::Serialize),
12    serde(rename_all = "kebab-case")
13)]
14pub struct IotaNamesConfig {
15    /// Address of the `iota_names` package.
16    pub package_address: Address,
17    /// ID of the `IotaNames` object.
18    pub object_id: ObjectId,
19    /// Address of the `payments` package.
20    pub payments_package_address: Address,
21    /// ID of the registry table.
22    pub registry_id: ObjectId,
23    /// ID of the reverse registry table.
24    pub reverse_registry_id: ObjectId,
25}
26
27impl crate::TreeDisplay for IotaNamesConfig {
28    fn fmt_tree(&self, w: &mut crate::TreeWriter<'_, '_>) -> std::fmt::Result {
29        w.header("IOTA Names Config")?;
30        w.leaf("Package Address", &self.package_address, false)?;
31        w.leaf("Object ID", &self.object_id, false)?;
32        w.leaf(
33            "Payments Package Address",
34            &self.payments_package_address,
35            false,
36        )?;
37        w.leaf("Registry ID", &self.registry_id, false)?;
38        w.leaf("Reverse Registry ID", &self.reverse_registry_id, true)
39    }
40}
41
42crate::impl_tree_display!(IotaNamesConfig);
43
44impl Default for IotaNamesConfig {
45    fn default() -> Self {
46        Self::mainnet()
47    }
48}
49
50impl IotaNamesConfig {
51    pub fn new(
52        package_address: Address,
53        object_id: ObjectId,
54        payments_package_address: Address,
55        registry_id: ObjectId,
56        reverse_registry_id: ObjectId,
57    ) -> Self {
58        Self {
59            package_address,
60            object_id,
61            payments_package_address,
62            registry_id,
63            reverse_registry_id,
64        }
65    }
66
67    pub fn from_env() -> Result<Self, IotaNamesError> {
68        Ok(Self::new(
69            std::env::var("IOTA_NAMES_PACKAGE_ADDRESS")?.parse()?,
70            std::env::var("IOTA_NAMES_OBJECT_ID")?.parse()?,
71            std::env::var("IOTA_NAMES_PAYMENTS_PACKAGE_ADDRESS")?.parse()?,
72            std::env::var("IOTA_NAMES_REGISTRY_ID")?.parse()?,
73            std::env::var("IOTA_NAMES_REVERSE_REGISTRY_ID")?.parse()?,
74        ))
75    }
