83/// A single record in the registry.
84#[derive(Clone, Debug, Eq, PartialEq)]
85#[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]
86pub struct NameRecord {
87    /// The ID of the registration NFT assigned to this record.
88    ///
89    /// The owner of the corresponding registration NFT has the rights to be
90    /// able to change and adjust the `target_address` of this name.
91    ///
92    /// It is possible that the ID changes if the record expires and is
93    /// purchased by someone else.
94    pub nft_id: ObjectId,
95    /// Timestamp in milliseconds when the record expires.
96    pub expiration_timestamp_ms: u64,
97    /// The target address that this name points to.
98    pub target_address: Option<Address>,
99    /// Additional data which may be stored in a record.
100    #[cfg_attr(feature = "serde", serde(with = "serde_vecmap"))]
101    pub data: HashMap<String, String>,
102}
103
104impl crate::TreeDisplay for NameRecord {
105    fn fmt_tree(&self, w: &mut crate::TreeWriter<'_, '_>) -> std::fmt::Result {
106        w.header("Name Record")?;
107        w.leaf("NFT ID", &self.nft_id, false)?;
108        w.leaf("Expiration (ms)", &self.expiration_timestamp_ms, false)?;
109        w.option_leaf("Target Address", &self.target_address, false)?;
110        w.branch("Data", true, |w| {
111            if self.data.is_empty() {
112                w.leaf("Entries", &"[]", true)
113            } else {
114                let mut entries: Vec<_> = self.data.iter().collect();
115                entries.sort_by_key(|(k, _)| *k);
116                let last_idx = entries.len() - 1;
117                for (i, (k, v)) in entries.iter().enumerate() {
118                    w.leaf(k, v, i == last_idx)?;
119                }
120                Ok(())
121            }
122        })
123    }
124}

134#[cfg(feature = "serde")]
135mod serde_vecmap {
136    use std::collections::HashMap;
137
138    use serde::{Deserialize, Deserializer, Serialize, Serializer};
139
140    #[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize)]
141    pub struct VecMap<K, V> {
142        pub contents: Vec<Entry<K, V>>,
143    }
144
145    #[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize)]
146    pub struct Entry<K, V> {
147        pub key: K,
148        pub value: V,
149    }
150
151    pub fn deserialize<'de, D>(deserializer: D) -> Result<HashMap<String, String>, D::Error>
152    where
153        D: Deserializer<'de>,
154    {
155        let map = VecMap::deserialize(deserializer)?;
156        Ok(map.contents.into_iter().map(|e| (e.key, e.value)).collect())
157    }
158
159    pub fn serialize<S>(value: &HashMap<String, String>, serializer: S) -> Result<S::Ok, S::Error>
160    where
161        S: Serializer,
162    {
163        VecMap {
164            contents: value
165                .iter()
166                .map(|(key, value)| Entry {
167                    key: key.clone(),
168                    value: value.clone(),
169                })
170                .collect(),
171        }
172        .serialize(serializer)
173    }
174}

176#[cfg(feature = "serde")]
177impl TryFrom<crate::Object> for NameRecord {
178    type Error = crate::iota_names::error::IotaNamesError;
179
180    fn try_from(object: crate::Object) -> Result<Self, crate::iota_names::error::IotaNamesError> {
181        #[derive(Clone, Debug, serde::Deserialize, serde::Serialize)]
182        pub struct Field<N, V> {
183            pub id: ObjectId,
184            pub name: N,
185            pub value: V,
186        }
187
188        object
189            .to_rust::<Field<Name, Self>>()
190            .map(|record| record.value)
191            .map_err(|_| crate::iota_names::error::IotaNamesError::MalformedObject(object.id()))
192    }
193}

195impl NameRecord {
196    /// Leaf records expire when their parent expires.
197    /// The `expiration_timestamp_ms` is set to `0` (on-chain) to indicate this.
198    pub fn is_leaf_record(&self) -> bool {
199        self.expiration_timestamp_ms == IOTA_NAMES_LEAF_EXPIRATION_TIMESTAMP
200    }
201
202    /// Validates that a `NameRecord` is a valid parent of a child `NameRecord`.
203    ///
204    /// WARNING: This only applies for `leaf` records.
205    pub fn is_valid_leaf_parent(&self, child: &NameRecord) -> bool {
206        self.nft_id == child.nft_id
207    }
208
209    /// Checks if a `node` name record has expired.
210    /// Expects the latest checkpoint's timestamp.
211    pub fn is_node_expired(&self, checkpoint_timestamp_ms: u64) -> bool {
212        self.expiration_timestamp_ms < checkpoint_timestamp_ms
213    }
214
215    /// Gets the expiration time as a [`SystemTime`].
216    pub fn expiration_time(&self) -> SystemTime {
217        UNIX_EPOCH + Duration::from_millis(self.expiration_timestamp_ms)
218    }
219}

221#[cfg(test)]
222mod tests {
223    use super::*;
224
225    #[test]
226    fn expirations() {
227        let system_time: u64 = 100;
228
229        let mut name = NameRecord {
230            nft_id: ObjectId::ZERO,
231            data: Default::default(),
232            target_address: Some(Address::ZERO),
233            expiration_timestamp_ms: system_time + 10,
234        };
235
236        assert!(!name.is_node_expired(system_time));
237
238        name.expiration_timestamp_ms = system_time - 10;
239
240        assert!(name.is_node_expired(system_time));
241    }
