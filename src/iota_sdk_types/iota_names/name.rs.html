18#[derive(Clone, Debug, Eq, Hash, PartialEq)]
19pub struct Name {
20    // Labels of the name, in reverse order
21    labels: Vec<String>,
22}
23
24impl FromStr for Name {
25    type Err = IotaNamesError;
26
27    fn from_str(s: &str) -> Result<Self, Self::Err> {
28        if s.len() > IOTA_NAMES_MAX_NAME_LENGTH {
29            return Err(IotaNamesError::NameLengthExceeded(
30                s.len(),
31                IOTA_NAMES_MAX_NAME_LENGTH,
32            ));
33        }
34
35        let formatted_string = convert_from_at_format(s, &IOTA_NAMES_SEPARATOR_DOT)?;
36
37        let labels = formatted_string
38            .split(IOTA_NAMES_SEPARATOR_DOT)
39            .rev()
40            .map(validate_label)
41            .collect::<Result<Vec<_>, Self::Err>>()?;
42
43        // A valid name in our system has at least a TLN and an SLN (len == 2).
44        if labels.len() < 2 {
45            return Err(IotaNamesError::NotEnoughLabels);
46        }
47
48        if labels[0] != IOTA_NAMES_TLN {
49            return Err(IotaNamesError::InvalidTln(labels[0].to_string()));
50        }
51
52        let labels = labels.into_iter().map(ToOwned::to_owned).collect();
53
54        Ok(Name { labels })
55    }
56}
57
58impl std::fmt::Display for Name {
59    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
60        // We use to_string() to check on-chain state and parse on-chain data
61        // so we should always default to DOT format.
62        let output = self.format(NameFormat::Dot);
63        f.write_str(&output)?;
64
65        Ok(())
66    }
67}

69impl Name {
70    pub fn type_(package_address: Address) -> StructTag {
71        const IOTA_NAMES_NAME_MODULE: Identifier = Identifier::from_static("name");
72        const IOTA_NAMES_NAME_STRUCT: Identifier = Identifier::from_static("Name");
73
74        StructTag::new(
75            package_address,
76            IOTA_NAMES_NAME_MODULE,
77            IOTA_NAMES_NAME_STRUCT,
78            vec![],
79        )
80    }

82    /// Derive the parent name for a given name. Only subnames have
83    /// parents; second-level names return `None`.
84    ///
85    /// ```
86    /// # use std::str::FromStr;
87    /// # use iota_sdk_types::iota_names::Name;
88    /// assert_eq!(
89    ///     Name::from_str("test.example.iota").unwrap().parent(),
90    ///     Some(Name::from_str("example.iota").unwrap())
91    /// );
92    /// assert_eq!(
93    ///     Name::from_str("sub.test.example.iota").unwrap().parent(),
94    ///     Some(Name::from_str("test.example.iota").unwrap())
95    /// );
96    /// assert_eq!(Name::from_str("example.iota").unwrap().parent(), None);
97    /// ```
98    pub fn parent(&self) -> Option<Self> {
99        if self.is_subname() {
100            Some(Self {
101                labels: self
102                    .labels
103                    .iter()
104                    .take(self.num_labels() - 1)
105                    .cloned()
106                    .collect(),
107            })
108        } else {
109            None
110        }
111    }
112
113    /// Returns whether this name is a second-level name (Ex. `test.iota`)
114    pub fn is_sln(&self) -> bool {
115        self.num_labels() == 2
116    }
117
118    /// Returns whether this name is a subname (Ex. `sub.test.iota`)
119    pub fn is_subname(&self) -> bool {
120        self.num_labels() >= 3
121    }
122
123    /// Returns the number of labels including TLN.
124    ///
125    /// ```
126    /// # use std::str::FromStr;
127    /// # use iota_sdk_types::iota_names::Name;
128    /// assert_eq!(Name::from_str("test.example.iota").unwrap().num_labels(), 3)
129    /// ```
130    pub fn num_labels(&self) -> usize {
131        self.labels.len()
132    }
133
134    /// Get the label at the given index
135    pub fn label(&self, index: usize) -> Option<&String> {
136        self.labels.get(index)
137    }
138
139    /// Get all of the labels. NOTE: These are in reverse order starting with
140    /// the top-level name and proceeding to subnames.
141    pub fn labels(&self) -> &[String] {
142        &self.labels
143    }
144
145    /// Formats a name into a string based on the available output formats.
146    /// The default separator is `.`
147    pub fn format(&self, format: NameFormat) -> String {
148        let mut labels = self.labels.clone();
149        let sep = &IOTA_NAMES_SEPARATOR_DOT.to_string();
150        labels.reverse();
151
152        if format == NameFormat::Dot {
153            // DOT format, all labels joined together with dots, including the TLN.
154            labels.join(sep)
155        } else {
156            // SAFETY: This is a safe operation because we only allow a
157            // name's label vector size to be >= 2 (see `Name::from_str`)
158            let _tln = labels.pop();
159            let sln = labels.pop().unwrap();
160
161            // AT format, labels minus SLN joined together with dots, then joined to SLN
162            // with @, no TLN.
163            format!("{}{IOTA_NAMES_SEPARATOR_AT}{sln}", labels.join(sep))
164        }
165    }
166}

168/// Two different view options for a name.
169/// `At` -> `test@example` | `Dot` -> `test.example.iota`
170#[derive(Clone, Debug, Eq, PartialEq, strum::Display)]
171pub enum NameFormat {
172    At,
173    Dot,
174}

185    /// `Field<Name, _>` object contents are deserialized with.

187    #[serde(rename = "Name")]

190    }

193    #[serde(rename = "Name")]
194    struct BinaryName {
195        labels: Vec<String>,
196    }

198    impl Serialize for Name {
199        fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
200        where
201            S: Serializer,
202        {
203            if serializer.is_human_readable() {
204                serde_with::DisplayFromStr::serialize_as(self, serializer)
205            } else {
206                BinaryNameRef {
207                    labels: &self.labels,
208                }
209                .serialize(serializer)
210            }
211        }
212    }
213
214    impl<'de> Deserialize<'de> for Name {
215        fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
216        where
217            D: Deserializer<'de>,
218        {
219            if deserializer.is_human_readable() {
220                serde_with::DisplayFromStr::deserialize_as(deserializer)
221            } else {
222                BinaryName::deserialize(deserializer).map(|name| Name {
223                    labels: name.labels,
224                })
225            }
226        }
227    }

230/// Converts @label ending to label{separator}iota ending.
231///
232/// E.g. `@example` -> `example.iota` | `test@example` -> `test.example.iota`
233fn convert_from_at_format(s: &str, separator: &char) -> Result<String, IotaNamesError> {
234    let mut splits = s.split(IOTA_NAMES_SEPARATOR_AT);
235
236    let Some(before) = splits.next() else {
237        return Err(IotaNamesError::InvalidSeparator);
238    };
239
240    let Some(after) = splits.next() else {
241        return Ok(before.to_string());
242    };
243
244    if splits.next().is_some() || after.contains(*separator) || after.is_empty() {
245        return Err(IotaNamesError::InvalidSeparator);
246    }
247
248    let mut parts = vec![];
249
250    if !before.is_empty() {
251        parts.push(before);
252    }
253
254    parts.push(after);
255    parts.push(IOTA_NAMES_TLN);
256
257    Ok(parts.join(&separator.to_string()))
258}

260/// Checks the validity of a label according to these rules:
261/// - length must be in
262///   [IOTA_NAMES_MIN_LABEL_LENGTH..IOTA_NAMES_MAX_LABEL_LENGTH]
263/// - must contain only '0'..'9', 'a'..'z' and '-'
264/// - must not start or end with '-'
265pub fn validate_label(label: &str) -> Result<&str, IotaNamesError> {
266    let bytes = label.as_bytes();
267    let len = bytes.len();
268
269    if !(IOTA_NAMES_MIN_LABEL_LENGTH..=IOTA_NAMES_MAX_LABEL_LENGTH).contains(&len) {
270        return Err(IotaNamesError::InvalidLabelLength(
271            len,
272            IOTA_NAMES_MIN_LABEL_LENGTH,
273            IOTA_NAMES_MAX_LABEL_LENGTH,
274        ));
275    }
276
277    for (i, character) in bytes.iter().enumerate() {
278        match character {
279            b'a'..=b'z' | b'0'..=b'9' => continue,
280            b'-' => {
281                if i == 0 || i == len - 1 {
282                    return Err(IotaNamesError::HyphensAsFirstOrLastLabelChar);
283                }
284            }
285            _ => return Err(IotaNamesError::InvalidLabelChar((*character) as char, i)),
286        };
287    }
288
289    Ok(label)
290}

374    #[test]
375    fn outputs() {
376        let mut name = "test.iota".parse::<Name>().unwrap();
377        assert!(name.format(NameFormat::Dot) == "test.iota");
378        assert!(name.format(NameFormat::At) == "@test");
379
380        name = "test.test.iota".parse::<Name>().unwrap();
381        assert!(name.format(NameFormat::Dot) == "test.test.iota");
382        assert!(name.format(NameFormat::At) == "test@test");

384        name = "test.test.test.iota".parse::<Name>().unwrap();
385        assert!(name.format(NameFormat::Dot) == "test.test.test.iota");
386        assert!(name.format(NameFormat::At) == "test.test@test");
387
388        name = "test.test.test.test.iota".parse::<Name>().unwrap();
389        assert!(name.format(NameFormat::Dot) == "test.test.test.test.iota");
390        assert!(name.format(NameFormat::At) == "test.test.test@test");
