9type Blake2b256 = blake2::Blake2b<blake2::digest::consts::U32>;
10
11/// A Blake2b256 Hasher
12#[derive(Debug, Default)]
13pub struct Hasher(Blake2b256);
14
15impl Hasher {
16    /// Initialize a new Blake2b256 Hasher instance.
17    pub fn new() -> Self {
18        Self(Blake2b256::new())
19    }
20
21    /// Process the provided data, updating internal state.
22    pub fn update<T: AsRef<[u8]>>(&mut self, data: T) {
23        self.0.update(data)
24    }
25
26    /// Finalize hashing, consuming the Hasher instance and returning the
27    /// resultant hash or `Digest`.
28    pub fn finalize(self) -> Digest {
29        Digest::new(self.0.finalize().into())
30    }
31
32    /// Convenience function for creating a new Hasher instance, hashing the
33    /// provided data, and returning the resultant `Digest`
34    pub fn digest<T: AsRef<[u8]>>(data: T) -> Digest {
35        let mut hasher = Self::new();
36        hasher.update(data);
37        hasher.finalize()
38    }
39}

41impl std::io::Write for Hasher {
42    fn write(&mut self, buf: &[u8]) -> std::io::Result<usize> {
43        self.0.write(buf)
44    }
45
46    fn flush(&mut self) -> std::io::Result<()> {
47        self.0.flush()
48    }

51/// Hash the bytes written by `write` and turn the digest into an `Address`.
52fn derive_address_from(write: impl FnOnce(&mut Hasher)) -> Address {
53    let mut hasher = Hasher::new();
54    write(&mut hasher);
55    Address::new(hasher.finalize().into_bytes())
56}

58impl crate::Ed25519PublicKey {
59    /// Derive an `Address` from this Public Key
60    ///
61    /// An `Address` can be derived from an `Ed25519PublicKey` by hashing the
62    /// bytes of the public key with no prefix flag.
63    ///
64    /// `hash(32-byte ed25519 public key)`
65    ///
66    /// ```
67    /// use iota_sdk_types::{Address, Ed25519PublicKey, hash::Hasher};
68    ///
69    /// let public_key_bytes = [0; 32];
70    /// let mut hasher = Hasher::new();
71    /// hasher.update(public_key_bytes);
72    /// let address = Address::new(hasher.finalize().into_bytes());
73    /// println!("Address: {}", address);
74    ///
75    /// let public_key = Ed25519PublicKey::new(public_key_bytes);
76    /// assert_eq!(address, public_key.derive_address());
77    /// ```
78    pub fn derive_address(&self) -> Address {
79        derive_address_from(|hasher| self.write_into_hasher(hasher))
80    }
81
82    fn write_into_hasher(&self, hasher: &mut Hasher) {
83        hasher.update(self.bytes());
84    }
85}

291impl crate::MultisigCommittee {
292    /// Derive an `Address` from this MultisigCommittee.
293    ///
294    /// A MultiSig address is defined as the 32-byte Blake2b hash of serializing
295    /// the `SignatureScheme` flag (0x03), the threshold (in little endian), and
296    /// the concatenation of all n flags, public keys and their weights, where
297    /// `flag_i?` is the member's `SignatureScheme` flag — omitted for Ed25519
298    /// keys, matching their plain address derivation.
299    ///
300    /// `hash(0x03 || threshold || flag_1? || pk_1 || weight_1
301    /// || ... || flag_n? || pk_n || weight_n)`.
302    pub fn derive_address(&self) -> Address {
303        derive_address_from(|hasher| {
304            hasher.update([self.scheme().to_u8()]);
305            hasher.update(self.threshold().to_le_bytes());
306
307            for member in self.members() {
308                member.public_key().write_into_hasher(hasher);
309                hasher.update(member.weight().to_le_bytes());
310            }
311        })
312    }
313}

475    impl crate::UserSignature {
476        /// Calculate the auth digest for this signature.
477        ///
478        /// For [`MoveAuthenticator`](crate::MoveAuthenticator) signatures this
479        /// equals
480        /// [`MoveAuthenticator::digest()`](crate::MoveAuthenticator::digest).
481        /// For all other signature types it is the Blake2b256 of the
482        /// serialized (flag-prefixed) signature bytes.
483        pub fn auth_digest(&self) -> Digest {
484            match self {
485                Self::MoveAuthenticator(authenticator) => authenticator.digest(),
486                Self::Simple(_) | Self::Multisig(_) | Self::PasskeyAuthenticator(_) => {
487                    Hasher::digest(self.to_bytes())
488                }
489            }
490        }

493    impl crate::SignedTransaction {
494        /// Computes the auth digest for the sender and, if sponsored, for the
495        /// sponsor. See
496        /// [`UserSignature::auth_digest`](crate::UserSignature::auth_digest)
497        /// for the per-signature logic.
498        ///
499        /// Returns an error if no signature commits to the sender or sponsor
500        /// address.
501        pub fn compute_auth_digests(
502            &self,
503        ) -> Result<(Digest, Option<Digest>), super::MissingSignatureError> {
504            let crate::Transaction::V1(transaction) = &self.transaction;
505
506            let digest_for_address = |address| {
507                self.signatures
508                    .iter()
509                    .find(|signature| signature.derive_address() == address)
510                    .map(crate::UserSignature::auth_digest)
511                    .ok_or(super::MissingSignatureError { address })
512            };
513
514            let sender_auth_digest = digest_for_address(transaction.sender)?;
515            let gas_owner = transaction.gas_payment.owner;
516            let sponsor_auth_digest = if gas_owner != transaction.sender {
517                Some(digest_for_address(gas_owner)?)
518            } else {
519                None
520            };
521
522            Ok((sender_auth_digest, sponsor_auth_digest))
523        }
524    }

526    fn type_digest<T: serde::Serialize>(salt: &str, ty: &T) -> Digest {
527        let mut hasher = Hasher::new();
528        hasher.update(salt);
529        bcs::serialize_into(&mut hasher, ty).expect("bcs serialization failed");
530        hasher.finalize()
531    }

599/// A 1-byte domain separator for hashing Object ID in IOTA. It starts from 0xf0
600/// to ensure no hashing collision for any ObjectId vs Address which is
601/// derived as the hash of `flag || pubkey`.
602#[derive(Clone, Copy, Debug, Eq, Hash, PartialEq)]
603#[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
604#[repr(u8)]
605enum HashingIntent {
606    #[cfg(feature = "serde")]
607    ChildObjectId = 0xf0,
608    RegularObjectId = 0xf1,
609}
610
611impl crate::ObjectId {
612    /// Create an ObjectId from a transaction digest and `count`.
613    ///
614    /// `count` is the number of objects that have been created during a
615    /// transaction.
616    pub fn derive_id(digest: crate::TransactionDigest, count: u64) -> Self {
617        let mut hasher = Hasher::new();
618        hasher.update([HashingIntent::RegularObjectId as u8]);
619        hasher.update(digest);
620        hasher.update(count.to_le_bytes());
621        let digest = hasher.finalize();
622        Self::new(digest.into_bytes())
623    }
624
625    /// Derive an ObjectId for a Dynamic Child Object.
626    ///
627    /// hash(parent || len(key) || key || key_type_tag)
628    #[cfg(feature = "serde")]
629    #[cfg_attr(doc_cfg, doc(cfg(feature = "serde")))]
630    pub fn derive_dynamic_child_id(&self, key_type_tag: &crate::TypeTag, key_bytes: &[u8]) -> Self {
631        let mut hasher = Hasher::new();
632        hasher.update([HashingIntent::ChildObjectId as u8]);
633        hasher.update(self);
634        hasher.update(
635            u64::try_from(key_bytes.len())
636                .expect("key_bytes must fit into a u64")
637                .to_le_bytes(),
638        );
639        hasher.update(key_bytes);
640        bcs::serialize_into(&mut hasher, key_type_tag)
641            .expect("bcs serialization of `TypeTag` cannot fail");
642        let digest = hasher.finalize();
643
644        Self::new(digest.into_bytes())
645    }
646
647    /// Derive the ObjectId of a derived object (`0x2::derived_object`).
648    ///
649    /// hash(parent || len(key) || key || DerivedObjectKey(key_type_tag))
650    #[cfg(feature = "serde")]
651    #[cfg_attr(doc_cfg, doc(cfg(feature = "serde")))]
652    pub fn derive_object_id(&self, key_type_tag: &crate::TypeTag, key_bytes: &[u8]) -> Self {
653        // Wrap the key type into `DerivedObjectKey<K>` to preserve on-chain
654        // namespacing
655        let wrapper_type_tag = crate::TypeTag::Struct(Box::new(crate::StructTag::new(
656            crate::Address::FRAMEWORK,
657            crate::Identifier::DERIVED_OBJECT_MODULE,
658            crate::Identifier::DERIVED_OBJECT_KEY,
659            vec![key_type_tag.clone()],
660        )));
661
662        self.derive_dynamic_child_id(&wrapper_type_tag, key_bytes)
663    }
