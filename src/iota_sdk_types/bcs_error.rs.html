4/// Error returned when BCS serialization or deserialization fails.
5#[derive(Debug, thiserror::Error)]
6#[error(transparent)]
7pub struct BcsError(Box<dyn std::error::Error + Send + Sync + 'static>);
8
9impl BcsError {
10    /// Wraps the error returned by a BCS serializer or deserializer.
11    pub fn new(error: impl std::error::Error + Send + Sync + 'static) -> Self {
12        Self(Box::new(error))
13    }
14}
