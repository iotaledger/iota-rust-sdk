4//! gRPC client for IOTA node operations.
5//!
6//! This crate provides a high-level client for interacting with IOTA nodes
7//! via gRPC. It wraps the low-level proto types and provides ergonomic APIs
8//! using SDK types from `iota_types`.
9//!
10//! # Example
11//!
12//! ```no_run
13//! use iota_sdk_grpc_client::{
14//!     Client,
15//!     read_mask_fields::{ObjectReadMask, TransactionReadMask},
16//! };
17//! use iota_types::{ObjectId, TransactionDigest};
18//!
19//! # async fn example() -> Result<(), Box<dyn std::error::Error>> {
20//! let client = Client::new_localnet()?;
21//!
22//! // Get a transaction with the default field mask.
23//! // The batched reads return one result per request, so a transaction the
24//! // node cannot serve fails only its own slot.
25//! let digest: TransactionDigest = todo!();
26//! let txs = client
27//!     .get_transactions([digest], TransactionReadMask::default())
28//!     .await?;
29//! for tx in txs.body() {
30//!     match tx {
31//!         Ok(tx) => println!("Transaction digest: {:?}", tx.transaction()?.digest()?),
32//!         Err(e) => eprintln!("could not read transaction: {e}"),
33//!     }
34//! }
35//!
36//! // Get an object with the default field mask.
37//! let object_id: ObjectId = "0x2".parse()?;
38//! let objects = client
39//!     .get_objects([object_id], ObjectReadMask::default())
40//!     .await?;
41//! for object in objects.body() {
42//!     match object {
43//!         Ok(object) => println!("Object version: {:?}", object.object_reference()?.version()),
44//!         Err(e) => eprintln!("could not read object: {e}"),
45//!     }
46//! }
47//! # Ok(())
48//! # }
49//! ```
