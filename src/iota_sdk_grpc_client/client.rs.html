27/// gRPC client factory for IOTA gRPC operations.
28#[derive(Clone)]
29pub struct Client {
30    /// Target URI of the gRPC server
31    uri: http::Uri,
32    /// Shared gRPC channel for all service clients
33    channel: tonic::transport::Channel,
34    /// Headers interceptor for adding custom headers to requests
35    headers: HeadersInterceptor,
36    /// Maximum decoding message size for responses
37    max_decoding_message_size: Option<usize>,
38}
39
40impl Client {
41    /// Create a new Client instance for the given gRPC server URI.
42    pub fn new<T>(uri: T) -> Result<Self>
43    where
44        T: TryInto<http::Uri>,
45        T::Error: Into<BoxError>,
46    {
47        let uri = uri
48            .try_into()
49            .map_err(Into::into)
50            .map_err(tonic::Status::from_error)?;
51
52        let endpoint = tonic::transport::Endpoint::from(uri.clone());
53
54        #[cfg(all(
55            feature = "tls-ring",
56            any(feature = "tls-native-roots", feature = "tls-webpki-roots")
57        ))]
58        let endpoint = if uri.scheme() == Some(&http::uri::Scheme::HTTPS) {
59            endpoint
60                .tls_config(tonic::transport::channel::ClientTlsConfig::new().with_enabled_roots())
61                .map_err(Into::into)
62                .map_err(tonic::Status::from_error)?
63        } else {
64            endpoint
65        };
66
67        #[cfg(not(all(
68            feature = "tls-ring",
69            any(feature = "tls-native-roots", feature = "tls-webpki-roots")
70        )))]
71        if uri.scheme() == Some(&http::uri::Scheme::HTTPS) {
72            return Err(tonic::Status::failed_precondition(
73                "HTTPS requires the `tls-ring` feature and either `tls-native-roots` or `tls-webpki-roots` to be enabled",
74            )
75            .into());
76        }
77
78        let channel = endpoint
79            .connect_timeout(Duration::from_secs(5))
80            .http2_keep_alive_interval(Duration::from_secs(5))
81            .connect_lazy();
82
83        Ok(Self {
84            uri,
85            channel,
86            headers: Default::default(),
87            max_decoding_message_size: None,
88        })
89    }
90

179    /// Apply common client configuration (compression, message size limits).
180    fn configure_client<C: GrpcClientConfig>(&self, client: C) -> C {
181        let client = client.accept_compressed(CompressionEncoding::Zstd);
182        if let Some(limit) = self.max_decoding_message_size {
183            client.max_decoding_message_size(limit)
184        } else {
185            client
186        }
187    }
188}
189
190/// Trait for common gRPC client configuration methods.
191///
192/// This trait abstracts over the common configuration methods shared by
193/// tonic-generated service clients, allowing `configure_client` to work
194/// generically.
195trait GrpcClientConfig: Sized {
196    fn accept_compressed(self, encoding: CompressionEncoding) -> Self;
197    fn max_decoding_message_size(self, limit: usize) -> Self;
198}
199
200/// Implement `GrpcClientConfig` for tonic-generated service clients.
201macro_rules! impl_grpc_client_config {
202    ($($client:ty),* $(,)?) => {
203        $(
204            impl GrpcClientConfig for $client {
205                fn accept_compressed(self, encoding: CompressionEncoding) -> Self {
206                    self.accept_compressed(encoding)
207                }
208                fn max_decoding_message_size(self, limit: usize) -> Self {
209                    self.max_decoding_message_size(limit)
210                }
211            }
212        )*
213    };
214}

223#[cfg(test)]
224mod tests {
225    #[cfg(not(feature = "tls-ring"))]
226    #[test]
227    fn https_without_tls_ring_returns_failed_precondition() {
228        use super::Client;
229
230        let status = match Client::new("https://example.com") {
231            Err(crate::api::Error::Grpc(status)) => status,
232            Err(other) => panic!("expected Error::Grpc, got: {other:?}"),
233            Ok(_) => panic!("new should fail without tls-ring"),
234        };
235
236        assert_eq!(
237            status.code(),
238            tonic::Code::FailedPrecondition,
239            "status: {status:?}"
240        );
241        assert!(
242            status.message().contains("tls-ring"),
243            "error should mention `tls-ring` feature, got: {}",
244            status.message()
245        );
246    }
