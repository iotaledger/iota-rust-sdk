5/// Interceptor used to add additional headers to a Request
6#[derive(Clone, Debug, Default)]
7pub struct HeadersInterceptor {
8    headers: tonic::metadata::MetadataMap,
9}
10
11impl HeadersInterceptor {
12    /// Create a new, empty `HeadersInterceptor`.
13    pub fn new() -> Self {
14        Self::default()
15    }
16
17    /// Return reference to the internal `MetadataMap`.
18    pub fn headers(&self) -> &tonic::metadata::MetadataMap {
19        &self.headers
20    }
21
22    /// Get mutable access to the internal `MetadataMap` for modification.
23    pub fn headers_mut(&mut self) -> &mut tonic::metadata::MetadataMap {
24        &mut self.headers
25    }
26
27    /// Enable HTTP basic authentication with a username and optional password.
28    pub fn basic_auth<U, P>(&mut self, username: U, password: Option<P>)
29    where
30        U: std::fmt::Display,
31        P: std::fmt::Display,
32    {
33        use std::io::Write;
34
35        use base64::{prelude::BASE64_STANDARD, write::EncoderWriter};
36
37        let mut buf = b"Basic ".to_vec();
38        {
39            let mut encoder = EncoderWriter::new(&mut buf, &BASE64_STANDARD);
40            let _ = write!(encoder, "{username}:");
41            if let Some(password) = password {
42                let _ = write!(encoder, "{password}");
43            }
44        }
45        let mut header = tonic::metadata::MetadataValue::try_from(buf)
46            .expect("base64 is always valid HeaderValue");
47        header.set_sensitive(true);
48
49        self.headers
50            .insert(http::header::AUTHORIZATION.as_str(), header);
51    }
52
53    /// Enable HTTP bearer authentication.
54    ///
55    /// Returns an error if the token contains characters that are invalid
56    /// in HTTP header values (e.g. control characters).
57    pub fn bearer_auth<T>(
58        &mut self,
59        token: T,
60    ) -> Result<(), tonic::metadata::errors::InvalidMetadataValue>
61    where
62        T: std::fmt::Display,
63    {
64        let header_value = format!("Bearer {token}");
65        let mut header = tonic::metadata::MetadataValue::try_from(header_value)?;
66        header.set_sensitive(true);
67
68        self.headers
69            .insert(http::header::AUTHORIZATION.as_str(), header);
70        Ok(())
71    }
72}
73
74impl tonic::service::Interceptor for &HeadersInterceptor {
75    fn call(
76        &mut self,
77        mut request: tonic::Request<()>,
78    ) -> std::result::Result<tonic::Request<()>, tonic::Status> {
79        if !self.headers.is_empty() {
80            request
81                .metadata_mut()
82                .as_mut()
83                .extend(self.headers.clone().into_headers());
84        }
85        Ok(request)
86    }
87}
88
89impl tonic::service::Interceptor for HeadersInterceptor {
90    fn call(
91        &mut self,
92        request: tonic::Request<()>,
93    ) -> std::result::Result<tonic::Request<()>, tonic::Status> {
94        (&*self).call(request)
95    }
96}

98#[cfg(test)]
99mod tests {
100    use base64::{Engine as _, prelude::BASE64_STANDARD};
101    use tonic::service::Interceptor as _;
102
103    use super::*;
104
105    fn intercepted_authorization(mut interceptor: HeadersInterceptor) -> String {
106        let request = interceptor
107            .call(tonic::Request::new(()))
108            .expect("interceptor should not fail");
109        request
110            .metadata()
111            .get(http::header::AUTHORIZATION.as_str())
112            .expect("authorization header should be set")
113            .to_str()
114            .expect("authorization header should be valid ASCII")
115            .to_string()
116    }
117
118    #[test]
119    fn basic_auth_sets_authorization_header() {
120        let mut interceptor = HeadersInterceptor::new();
121        interceptor.basic_auth("alice", Some("hunter2"));
122
123        let header = intercepted_authorization(interceptor);
124        let encoded = header
125            .strip_prefix("Basic ")
126            .expect("basic auth header should start with 'Basic '");
127        let decoded = BASE64_STANDARD
128            .decode(encoded)
129            .expect("payload should be valid base64");
130        assert_eq!(decoded, b"alice:hunter2");
131    }
132
133    #[test]
134    fn basic_auth_without_password_emits_trailing_colon() {
135        let mut interceptor = HeadersInterceptor::new();
136        interceptor.basic_auth("alice", None::<&str>);
137
138        let header = intercepted_authorization(interceptor);
139        let encoded = header.strip_prefix("Basic ").unwrap();
140        let decoded = BASE64_STANDARD.decode(encoded).unwrap();
141        assert_eq!(decoded, b"alice:");
142    }
143
144    #[test]
145    fn bearer_auth_sets_authorization_header() {
146        let mut interceptor = HeadersInterceptor::new();
147        interceptor
148            .bearer_auth("my-token")
149            .expect("bearer_auth should succeed");
150
151        let header = intercepted_authorization(interceptor);
152        assert_eq!(header, "Bearer my-token");
153    }
154
155    #[test]
156    fn empty_interceptor_does_not_set_authorization_header() {
157        let mut interceptor = HeadersInterceptor::new();
158        let request = interceptor.call(tonic::Request::new(())).unwrap();
159        assert!(
160            request
161                .metadata()
162                .get(http::header::AUTHORIZATION.as_str())
163                .is_none()
164        );
165    }
166}
