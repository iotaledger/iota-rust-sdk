6use iota_grpc_types::{
7    read_mask_fields::{EpochField, EpochReadMask, IntoReadMask},

9        epoch::Epoch,

11    },
12};

14use crate::{
15    GrpcClient, InterceptedChannel,
16    api::{GrpcResult, MetadataEnvelope, TryFromProtoError, define_query},
17};
18
19define_query! {
20    /// Query for [`GrpcClient::epoch`]. Await it to send the request.
21    pub struct GetEpochQuery {
22        service_client: LedgerServiceClient<InterceptedChannel>,
23        epoch: Option<u64>,
24        read_mask: EpochReadMask,
25    }
26    output: GrpcResult<MetadataEnvelope<Epoch>>;
27}
28
29impl GetEpochQuery {
30    /// Set the number of the epoch to query. Without it, queries the current
31    /// epoch.
32    pub fn epoch_number(mut self, epoch_number: u64) -> Self {
33        self.epoch = Some(epoch_number);
34        self
35    }
36
37    /// Set the field mask controlling the returned fields.
38    pub fn read_mask(mut self, read_mask: impl IntoReadMask<EpochReadMask>) -> Self {
39        self.read_mask = read_mask.into_read_mask();
40        self
41    }
42
43    fn into_request(self) -> (LedgerServiceClient<InterceptedChannel>, GetEpochRequest) {
44        let mut request = GetEpochRequest::default().with_read_mask(self.read_mask);
45
46        if let Some(epoch) = self.epoch {
47            request = request.with_epoch(epoch);
48        }
49
50        (self.service_client, request)
51    }
52
53    async fn send(self) -> GrpcResult<MetadataEnvelope<Epoch>> {
54        let (mut service_client, request) = self.into_request();
55        let response = service_client.get_epoch(request).await?;
56
57        MetadataEnvelope::from(response).try_map(|r| {
58            r.epoch
59                .ok_or_else(|| TryFromProtoError::missing("epoch").into())
60        })
61    }
62}

64define_query! {
65    /// Query for [`GrpcClient::reference_gas_price`]. Await it to send the
66    /// request.
67    pub struct GetReferenceGasPriceQuery {
68        service_client: LedgerServiceClient<InterceptedChannel>,
69    }
70    output: GrpcResult<MetadataEnvelope<u64>>;
71}
72
73impl GetReferenceGasPriceQuery {
74    fn into_request(self) -> (LedgerServiceClient<InterceptedChannel>, GetEpochRequest) {
75        let request = GetEpochRequest::default()
76            .with_read_mask(EpochReadMask::from(EpochField::REFERENCE_GAS_PRICE));
77        (self.service_client, request)
78    }
79
80    async fn send(self) -> GrpcResult<MetadataEnvelope<u64>> {
81        let (mut service_client, request) = self.into_request();
82        let response = service_client.get_epoch(request).await?;
83
84        MetadataEnvelope::from(response).try_map(|r| Ok(r.epoch()?.gas_price()?))
85    }
86}

88impl GrpcClient {
89    /// Get epoch information.
90    ///
91    /// Returns the [`Epoch`] proto type of the current epoch, or of the one set
92    /// with [`epoch_number`](GetEpochQuery::epoch_number), with fields
93    /// populated according to the read mask. Without
94    /// [`read_mask`](GetEpochQuery::read_mask), the default mask is used.
95    /// Pass an
96    /// [`EpochReadMask`](iota_grpc_types::read_mask_fields::EpochReadMask)
97    /// built from an
98    /// [`EpochField`](iota_grpc_types::read_mask_fields::EpochField) or any
99    /// slice/array/vec of fields. For individual protocol config map entries,
100    /// use
101    /// [`EpochField::feature_flag`](iota_grpc_types::read_mask_fields::EpochField::feature_flag)
102    /// and
103    /// [`EpochField::attribute`](iota_grpc_types::read_mask_fields::EpochField::attribute).
104    ///
105    /// # Example
106    ///
107    /// ```no_run
108    /// # use iota_sdk_grpc_client::GrpcClient;
109    /// # use iota_sdk_grpc_client::read_mask_fields::{EpochField, EpochReadMask};
110    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
111    /// let client = GrpcClient::new_localnet()?;
112    ///
113    /// // Current epoch with the default mask.
114    /// let epoch = client.epoch().await?;
115    /// println!("Epoch: {:?}", epoch.body().epoch);
116    ///
117    /// // Specific epoch with selected fields.
118    /// let epoch = client
119    ///     .epoch()
120    ///     .epoch_number(0)
121    ///     .read_mask(EpochReadMask::from([
122    ///         EpochField::EPOCH,
123    ///         EpochField::REFERENCE_GAS_PRICE,
124    ///         EpochField::FIRST_CHECKPOINT,
125    ///     ]))
126    ///     .await?;
127    ///
128    /// // All feature flags for the current epoch.
129    /// let epoch = client
130    ///     .epoch()
131    ///     .read_mask(EpochField::PROTOCOL_CONFIG_FEATURE_FLAGS)
132    ///     .await?
133    ///     .into_inner();
134    /// let flags = epoch.protocol_config.unwrap().feature_flags.unwrap().flags;
135    ///
136    /// // A single named feature flag.
137    /// let epoch = client
138    ///     .epoch()
139    ///     .read_mask(EpochField::feature_flag("enable_vdf"))
140    ///     .await?;
141    ///
142    /// // A single named attribute.
143    /// let epoch = client
144    ///     .epoch()
145    ///     .read_mask(EpochField::attribute("max_tx_gas"))
146    ///     .await?;
147    /// # Ok(())
148    /// # }
149    /// ```
150    pub fn epoch(&self) -> GetEpochQuery {
151        GetEpochQuery {
152            service_client: self.ledger_service_client(),
153            epoch: None,
154            read_mask: EpochReadMask::default(),
155        }
156    }
157
158    /// Get the reference gas price for the current epoch.
159    ///
160    /// # Example
161    ///
162    /// ```no_run
163    /// # use iota_sdk_grpc_client::GrpcClient;
164    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
165    /// let client = GrpcClient::new_localnet()?;
166    /// let gas_price = client.reference_gas_price().await?.into_inner();
167    /// println!("Reference gas price: {gas_price} NANOS");
168    /// # Ok(())
169    /// # }
170    /// ```
171    pub fn reference_gas_price(&self) -> GetReferenceGasPriceQuery {
172        GetReferenceGasPriceQuery {
173            service_client: self.ledger_service_client(),
174        }
175    }
176}

178#[cfg(test)]
179mod tests {
180    use iota_grpc_types::read_mask_fields::{EpochField, EpochReadMask};
181
182    use crate::GrpcClient;
183
184    #[tokio::test]
185    async fn epoch_defaults_to_the_current_epoch_and_the_setter_picks_one() {
186        let client = GrpcClient::new("http://localhost").unwrap();
187        let query = client.epoch();
188        assert_eq!(query.epoch, None);
189        assert_eq!(query.read_mask.as_str(), EpochReadMask::default().as_str());
190
191        let query = query
192            .epoch_number(5)
193            .read_mask(EpochField::REFERENCE_GAS_PRICE);
194        assert_eq!(query.epoch, Some(5));
195        assert_eq!(
196            query.read_mask.as_str(),
197            EpochReadMask::from(EpochField::REFERENCE_GAS_PRICE).as_str()
198        );
199    }
200
201    #[tokio::test]
202    async fn the_request_carries_the_epoch_and_the_mask() {
203        let client = GrpcClient::new("http://localhost").unwrap();
204        let (_, request) = client
205            .epoch()
206            .epoch_number(5)
207            .read_mask(EpochField::PROTOCOL_CONFIG_FEATURE_FLAGS)
208            .into_request();
209        assert_eq!(request.epoch, Some(5));
210        assert_eq!(
211            request.read_mask,
212            Some(EpochReadMask::from(EpochField::PROTOCOL_CONFIG_FEATURE_FLAGS).into())
213        );
214
215        let (_, request) = client.epoch().into_request();
216        assert_eq!(request.epoch, None);
217    }
218
219    #[tokio::test]
220    async fn reference_gas_price_asks_the_current_epoch_for_the_price_only() {
221        let client = GrpcClient::new("http://localhost").unwrap();
222        let (_, request) = client.reference_gas_price().into_request();
223        assert_eq!(request.epoch, None);
224        assert_eq!(
225            request.read_mask.map(|mask| mask.paths),
226            Some(vec!["reference_gas_price".to_owned()])
227        );
228    }
