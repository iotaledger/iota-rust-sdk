27define_query! {
28    /// Query for [`GrpcClient::objects`] and
29    /// [`GrpcClient::objects_with_versions`]. Await it to send the request.
30    pub struct GetObjectsQuery {
31        service_client: LedgerServiceClient<InterceptedChannel>,
32        max_message_size: Option<usize>,
33        refs: Vec<(ObjectId, Option<Version>)>,
34    }
35    output: GrpcResult<MetadataEnvelope<Vec<GrpcResult<Object>>>>;
36    read_mask: ObjectReadMask;
37}
38
39impl GetObjectsQuery {
40    fn request(&self) -> GetObjectsRequest {
41        let requests = ObjectRequests::default().with_requests(
42            self.refs
43                .iter()
44                .map(|(id, version)| {
45                    let mut object_ref =
46                        ObjectReference::default().with_object_id(proto_object_id(*id));
47
48                    if let Some(v) = version {
49                        object_ref = object_ref.with_version(v.as_u64());
50                    }
51
52                    ObjectRequest::default().with_object_ref(object_ref)
53                })
54                .collect(),
55        );
56
57        let mut request = GetObjectsRequest::default()
58            .with_requests(requests)
59            .with_read_mask(self.read_mask.clone());
60
61        if let Some(max_size) = self.max_message_size {
62            request = request.with_max_message_size_bytes(saturating_usize_to_u32(max_size));
63        }
64
65        request
66    }
67
68    async fn send(mut self) -> GrpcResult<MetadataEnvelope<Vec<GrpcResult<Object>>>> {
69        if self.refs.is_empty() {
70            return Err(GrpcError::EmptyRequest);
71        }
72
73        let request = self.request();
74        let response = self.service_client.get_objects(request).await?;
75        let (stream, metadata) = MetadataEnvelope::from(response).into_parts();
76
77        // Server guarantees results are returned in request order
78        let response = collect_stream(stream, metadata, |msg| {
79            Ok((msg.has_next, into_item_results(msg.objects)))
80        })
81        .await?;
82        check_result_count(response.body(), self.refs.len())?;
83        check_object_identity(response.body(), &self.refs)?;
84
85        Ok(response)
86    }
87}

89define_query! {
90    /// Query for [`GrpcClient::object_references`]. Await it to send the
91    /// request.
92    pub struct GetObjectReferencesQuery {
93        objects: GetObjectsQuery,
94    }
95    output: GrpcResult<MetadataEnvelope<Vec<GrpcResult<iota_types::ObjectReference>>>>;
96}
97
98impl GetObjectReferencesQuery {
99    async fn send(
100        self,
101    ) -> GrpcResult<MetadataEnvelope<Vec<GrpcResult<iota_types::ObjectReference>>>> {
102        Ok(self.objects.send().await?.map(|objects| {
103            objects
104                .into_iter()
105                .map(|object| Ok(object?.object_reference()?))
106                .collect()
107        }))
108    }
109}

111impl GrpcClient {
112    /// Get objects by their IDs.
113    ///
114    /// Returns proto `Object` types. Use `obj.object()` to convert to SDK
115    /// type, or use `obj.object_reference()` to get the object reference.
116    ///
117    /// Results are returned in the same order as the input IDs, one per ID.
118    ///
119    /// # Errors
120    ///
121    /// Returns [`GrpcError::EmptyRequest`] if `refs` is empty.
122    ///
123    /// Each ID gets its own result: an object that is not found (never
124    /// existed, was deleted, or has been pruned by the serving node) yields
125    /// [`GrpcError::Server`] with code `NOT_FOUND` in that slot only, leaving
126    /// the other objects intact. The outer `GrpcResult` is reserved for
127    /// failures of the call itself, such as a transport error, and for a
128    /// server that answered with a different number of results than IDs
129    /// requested ([`UnexpectedResultCount`]), which leaves no way to tell
130    /// which ID each result belongs to, or answered a position with a
131    /// different object than the one requested there
132    /// ([`UnexpectedObject`]). The answered id is read from the object
133    /// reference or its BCS, so a read mask that includes neither leaves
134    /// nothing to check.
135    ///
136    /// [`UnexpectedResultCount`]: crate::ProtocolError::UnexpectedResultCount
137    /// [`UnexpectedObject`]: crate::ProtocolError::UnexpectedObject
138    ///
139    /// # Read Mask
140    ///
141    /// Without [`read_mask`](GetObjectsQuery::read_mask), the default mask is
142    /// used. Pass an
143    /// [`ObjectReadMask`](iota_grpc_types::read_mask_fields::ObjectReadMask)
144    /// built from an
145    /// [`ObjectField`](iota_grpc_types::read_mask_fields::ObjectField) or any
146    /// slice/array/vec of fields to choose the returned fields.
147    ///
148    /// # Example
149    ///
150    /// ```no_run
151    /// # use iota_sdk_grpc_client::GrpcClient;
152    /// # use iota_sdk_grpc_client::read_mask_fields::{ObjectField, ObjectReadMask};
153    /// # use iota_types::ObjectId;
154    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
155    /// let client = GrpcClient::new_localnet()?;
156    /// let object_id: ObjectId = "0x2".parse()?;
157    /// let ids = [object_id];
158    ///
159    /// // Default mask
160    /// let objs = client.objects(ids).await?;
161    ///
162    /// // Selected fields
163    /// let objs = client
164    ///     .objects(ids)
165    ///     .read_mask(ObjectReadMask::from([
166    ///         ObjectField::REFERENCE,
167    ///         ObjectField::BCS,
168    ///     ]))
169    ///     .await?;
170    ///
171    /// for obj in objs.body() {
172    ///     let obj = match obj {
173    ///         Ok(obj) => obj,
174    ///         // Only this ID failed; the remaining objects are still usable
175    ///         Err(e) => {
176    ///             eprintln!("could not read object: {e}");
177    ///             continue;
178    ///         }
179    ///     };
180    ///
181    ///     // Convert proto object to SDK type
182    ///     let sdk_obj = obj.object()?;
183    ///     println!("Got object ID: {:?}", sdk_obj.id());
184    ///     let obj_ref = obj.object_reference()?;
185    ///     println!("Object version: {:?}", obj_ref.version());
186    /// }
187    ///
188    /// // Results line up with the requested IDs, so pair them to find out
189    /// // which objects the node does not have
190    /// let missing: Vec<ObjectId> = ids
191    ///     .iter()
192    ///     .zip(objs.body())
193    ///     .filter_map(|(id, result)| match result {
194    ///         Err(e) if e.is_not_found() => Some(*id),
195    ///         _ => None,
196    ///     })
197    ///     .collect();
198    /// println!("Missing objects: {missing:?}");
199    /// # Ok(())
200    /// # }
201    /// ```
202    pub fn objects(&self, refs: impl IntoIterator<Item = ObjectId>) -> GetObjectsQuery {
203        self.objects_query(refs.into_iter().map(|id| (id, None)).collect())
204    }
205
206    /// Get objects by their IDs and optional versions.
207    ///
208    /// Returns proto `Object` types. Use `obj.object()` to convert to SDK
209    /// type, or use `obj.object_reference()` to get the object reference.
210    ///
211    /// Results are returned in the same order as the input refs, one per ref.
212    ///
213    /// # Errors
214    ///
215    /// Returns [`GrpcError::EmptyRequest`] if `refs` is empty.
216    ///
217    /// Each ref gets its own result, with the same meaning as in
218    /// [`objects`](GrpcClient::objects): a requested version the serving
219    /// node does not have fails only its own slot.
220    ///
221    /// # Read Mask
222    ///
223    /// As for [`objects`](GrpcClient::objects), set with
224    /// [`read_mask`](GetObjectsQuery::read_mask).
225    ///
226    /// # Example
227    ///
228    /// ```no_run
229    /// # use iota_sdk_grpc_client::GrpcClient;
230    /// # use iota_sdk_grpc_client::read_mask_fields::{ObjectField, ObjectReadMask};
231    /// # use iota_types::ObjectId;
232    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
233    /// let client = GrpcClient::new_localnet()?;
234    /// let object_id: ObjectId = "0x2".parse()?;
235    ///
236    /// // Default mask
237    /// let objs = client.objects_with_versions([(object_id, None)]).await?;
238    ///
239    /// // Selected fields
240    /// let objs = client
241    ///     .objects_with_versions([(object_id, None)])
242    ///     .read_mask(ObjectReadMask::from(ObjectField::REFERENCE_OBJECT_ID))
243    ///     .await?;
244    ///
245    /// for obj in objs.body() {
246    ///     let obj = match obj {
247    ///         Ok(obj) => obj,
248    ///         // Only this ref failed; the remaining objects are still usable
249    ///         Err(e) => {
250    ///             eprintln!("could not read object: {e}");
251    ///             continue;
252    ///         }
253    ///     };
254    ///
255    ///     // Convert proto object to SDK type
256    ///     let sdk_obj = obj.object()?;
257    ///     println!("Got object ID: {:?}", sdk_obj.id());
258    ///     let obj_ref = obj.object_reference()?;
259    ///     println!("Object version: {:?}", obj_ref.version());
260    /// }
261    /// # Ok(())
262    /// # }
263    /// ```
264    pub fn objects_with_versions(
265        &self,
266        refs: impl IntoIterator<Item = (ObjectId, Option<Version>)>,
267    ) -> GetObjectsQuery {
268        self.objects_query(refs.into_iter().collect())
269    }
270
271    /// Get the current references of objects by their IDs.
272    ///
273    /// Requests the `reference` field only, so no object contents travel.
274    /// Results are returned in the same order as the input IDs, one per ID.
275    ///
276    /// # Errors
277    ///
278    /// Returns [`GrpcError::EmptyRequest`] if `ids` is empty. As for
279    /// [`objects`](Self::objects), an ID that is not found yields
280    /// [`GrpcError::Server`] with code `NOT_FOUND` in its slot only; the outer
281    /// `GrpcResult` is reserved for failures of the call itself.
282    ///
283    /// # Example
284    ///
285    /// ```no_run
286    /// # use iota_sdk_grpc_client::GrpcClient;
287    /// # use iota_types::ObjectId;
288    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
289    /// let client = GrpcClient::new_localnet()?;
290    /// let gas: ObjectId = "0x2".parse()?;
291    /// let mut refs = client.object_references([gas]).await?.into_inner();
292    /// let gas_ref = refs.remove(0)?;
293    /// println!("gas version: {:?}", gas_ref.version());
294    /// # Ok(())
295    /// # }
296    /// ```
297    pub fn object_references(
298        &self,
299        ids: impl IntoIterator<Item = ObjectId>,
300    ) -> GetObjectReferencesQuery {
301        GetObjectReferencesQuery {
302            objects: self.objects(ids).read_mask([ObjectField::REFERENCE]),
303        }
304    }
305
306    fn objects_query(&self, refs: Vec<(ObjectId, Option<Version>)>) -> GetObjectsQuery {
307        GetObjectsQuery {
308            service_client: self.ledger_service_client(),
309            max_message_size: self.max_decoding_message_size(),
310            refs,
311            read_mask: ObjectReadMask::default(),
312        }
313    }

316#[cfg(test)]
317mod tests {
318    use iota_grpc_types::read_mask_fields::{ObjectField, ObjectReadMask};
319    use iota_types::{ObjectId, Version};
320
321    use crate::{GrpcClient, GrpcError, api::proto_object_id};
322
323    #[tokio::test]
324    async fn objects_asks_for_the_latest_version_with_the_default_mask() {
325        let client = GrpcClient::new("http://localhost").unwrap();
326        let query = client.objects([ObjectId::ZERO]);
327        assert_eq!(query.refs, vec![(ObjectId::ZERO, None)]);
328        assert_eq!(query.read_mask.as_str(), ObjectReadMask::default().as_str());
329    }
330
331    #[tokio::test]
332    async fn objects_with_versions_keeps_the_versions() {
333        let client = GrpcClient::new("http://localhost").unwrap();
334        let version = Version::from_u64(7);
335        let query = client.objects_with_versions([(ObjectId::ZERO, Some(version))]);
336        assert_eq!(query.refs, vec![(ObjectId::ZERO, Some(version))]);
337    }
338
339    #[tokio::test]
340    async fn read_mask_replaces_the_default_mask() {
341        let client = GrpcClient::new("http://localhost").unwrap();
342        let query = client
343            .objects([ObjectId::ZERO])
344            .read_mask(ObjectField::REFERENCE);
345        assert_eq!(
346            query.read_mask.as_str(),
347            ObjectReadMask::from(ObjectField::REFERENCE).as_str()
348        );
349    }
350
351    #[tokio::test]
352    async fn object_references_asks_only_for_the_reference() {
353        let client = GrpcClient::new("http://localhost").unwrap();
354        let query = client.object_references([ObjectId::ZERO]);
355        assert_eq!(query.objects.refs, vec![(ObjectId::ZERO, None)]);
356        assert_eq!(
357            query.objects.read_mask.as_str(),
358            ObjectReadMask::from(ObjectField::REFERENCE).as_str()
359        );
360    }
361
362    #[tokio::test]
363    async fn awaiting_no_ids_is_an_empty_request() {
364        let client = GrpcClient::new("http://localhost").unwrap();
365        let result = client.objects(Vec::new()).await;
366        assert!(matches!(result, Err(GrpcError::EmptyRequest)));
367    }
368
369    #[tokio::test]
370    async fn the_request_carries_every_ref_the_mask_and_the_message_size() {
371        let client = GrpcClient::new("http://localhost")
372            .unwrap()
373            .with_max_decoding_message_size(1024);
374        let pinned: ObjectId = "0x5".parse().unwrap();
375        let query = client
376            .objects_with_versions([(ObjectId::ZERO, None), (pinned, Some(Version::from_u64(7)))])
377            .read_mask(ObjectField::REFERENCE);
378        let request = query.request();
379
380        let refs: Vec<_> = request
381            .requests
382            .unwrap()
383            .requests
384            .into_iter()
385            .map(|r| {
386                let object_ref = r.object_ref.unwrap();
387                (object_ref.object_id, object_ref.version)
388            })
389            .collect();
390        assert_eq!(
391            refs,
392            vec![
393                (Some(proto_object_id(ObjectId::ZERO)), None),
394                (Some(proto_object_id(pinned)), Some(7)),
395            ]
396        );
397        assert_eq!(
398            request.read_mask,
399            Some(ObjectReadMask::from(ObjectField::REFERENCE).into())
400        );
401        assert_eq!(request.max_message_size_bytes, Some(1024));
402    }
403
404    #[tokio::test]
405    async fn object_references_sends_the_ids_with_the_reference_mask() {
406        let client = GrpcClient::new("http://localhost").unwrap();
407        let request = client.object_references([ObjectId::ZERO]).objects.request();
408        assert_eq!(
409            request.read_mask,
410            Some(ObjectReadMask::from(ObjectField::REFERENCE).into())
411        );
412        assert_eq!(request.requests.map(|r| r.requests.len()), Some(1));
413    }
