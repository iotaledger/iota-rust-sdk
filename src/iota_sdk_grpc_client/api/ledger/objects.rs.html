24impl Client {
25    /// Get objects by their IDs.
26    ///
27    /// Returns proto `Object` types. Use `obj.object()` to convert to SDK
28    /// type, or use `obj.object_reference()` to get the object reference.
29    ///
30    /// Results are returned in the same order as the input IDs, one per ID.
31    ///
32    /// # Errors
33    ///
34    /// Returns [`GrpcError::EmptyRequest`] if `refs` is empty.
35    ///
36    /// Each ID gets its own result: an object that is not found (never
37    /// existed, was deleted, or has been pruned by the serving node) yields
38    /// [`GrpcError::Server`] with code `NOT_FOUND` in that slot only, leaving
39    /// the other objects intact. The outer `GrpcResult` is reserved for
40    /// failures of the call itself, such as a transport error, and for a
41    /// server that answered with a different number of results than IDs
42    /// requested ([`UnexpectedResultCount`]), which leaves no way to tell
43    /// which ID each result belongs to, or answered a position with a
44    /// different object than the one requested there
45    /// ([`UnexpectedObject`]). The answered id is read from the object
46    /// reference or its BCS, so a read mask that includes neither leaves
47    /// nothing to check.
48    ///
49    /// [`UnexpectedResultCount`]: crate::ProtocolError::UnexpectedResultCount
50    /// [`UnexpectedObject`]: crate::ProtocolError::UnexpectedObject
51    ///
52    /// # Read Mask
53    ///
54    /// The `read_mask` parameter controls which fields the server returns; use
55    /// `ObjectReadMask::default()` for the default mask, or pass an
56    /// [`ObjectReadMask`](iota_grpc_types::read_mask_fields::ObjectReadMask)
57    /// built from an
58    /// [`ObjectField`](iota_grpc_types::read_mask_fields::ObjectField) or any
59    /// slice/array/vec of fields.
60    ///
61    /// # Example
62    ///
63    /// ```no_run
64    /// # use iota_sdk_grpc_client::Client;
65    /// # use iota_sdk_grpc_client::read_mask_fields::{ObjectField, ObjectReadMask};
66    /// # use iota_types::ObjectId;
67    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
68    /// let client = Client::new_localnet()?;
69    /// let object_id: ObjectId = "0x2".parse()?;
70    /// let ids = [object_id];
71    ///
72    /// // Default mask
73    /// let objs = client.objects(ids, ObjectReadMask::default()).await?;
74    ///
75    /// // Selected fields
76    /// let objs = client
77    ///     .objects(
78    ///         ids,
79    ///         ObjectReadMask::from([ObjectField::REFERENCE, ObjectField::BCS]),
80    ///     )
81    ///     .await?;
82    ///
83    /// for obj in objs.body() {
84    ///     let obj = match obj {
85    ///         Ok(obj) => obj,
86    ///         // Only this ID failed; the remaining objects are still usable
87    ///         Err(e) => {
88    ///             eprintln!("could not read object: {e}");
89    ///             continue;
90    ///         }
91    ///     };
92    ///
93    ///     // Convert proto object to SDK type
94    ///     let sdk_obj = obj.object()?;
95    ///     println!("Got object ID: {:?}", sdk_obj.id());
96    ///     let obj_ref = obj.object_reference()?;
97    ///     println!("Object version: {:?}", obj_ref.version());
98    /// }
99    ///
100    /// // Results line up with the requested IDs, so pair them to find out
101    /// // which objects the node does not have
102    /// let missing: Vec<ObjectId> = ids
103    ///     .iter()
104    ///     .zip(objs.body())
105    ///     .filter_map(|(id, result)| match result {
106    ///         Err(e) if e.is_not_found() => Some(*id),
107    ///         _ => None,
108    ///     })
109    ///     .collect();
110    /// println!("Missing objects: {missing:?}");
111    /// # Ok(())
112    /// # }
113    /// ```
114    pub async fn objects(
115        &self,
116        refs: impl IntoIterator<Item = ObjectId>,
117        read_mask: impl IntoReadMask<ObjectReadMask>,
118    ) -> GrpcResult<MetadataEnvelope<Vec<GrpcResult<Object>>>> {
119        let refs = refs.into_iter().map(|id| (id, None)).collect::<Vec<_>>();
120
121        self.objects_internal(refs, read_mask.into_read_mask())
122            .await
123    }

125    /// Get objects by their IDs and optional versions.
126    ///
127    /// Returns proto `Object` types. Use `obj.object()` to convert to SDK
128    /// type, or use `obj.object_reference()` to get the object reference.
129    ///
130    /// Results are returned in the same order as the input refs, one per ref.
131    ///
132    /// # Errors
133    ///
134    /// Returns [`GrpcError::EmptyRequest`] if `refs` is empty.
135    ///
136    /// Each ref gets its own result, with the same meaning as in
137    /// [`objects`](Client::objects): a requested version the serving
138    /// node does not have fails only its own slot.
139    ///
140    /// # Read Mask
141    ///
142    /// The `read_mask` parameter controls which fields the server returns; use
143    /// `ObjectReadMask::default()` for the default mask, or pass an
144    /// [`ObjectReadMask`](iota_grpc_types::read_mask_fields::ObjectReadMask)
145    /// built from an
146    /// [`ObjectField`](iota_grpc_types::read_mask_fields::ObjectField) or any
147    /// slice/array/vec of fields.
148    ///
149    /// # Example
150    ///
151    /// ```no_run
152    /// # use iota_sdk_grpc_client::Client;
153    /// # use iota_sdk_grpc_client::read_mask_fields::{ObjectField, ObjectReadMask};
154    /// # use iota_types::ObjectId;
155    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
156    /// let client = Client::new_localnet()?;
157    /// let object_id: ObjectId = "0x2".parse()?;
158    ///
159    /// // Default mask
160    /// let objs = client
161    ///     .objects_with_versions([(object_id, None)], ObjectReadMask::default())
162    ///     .await?;
163    ///
164    /// // Selected fields
165    /// let objs = client
166    ///     .objects_with_versions(
167    ///         [(object_id, None)],
168    ///         ObjectReadMask::from(ObjectField::REFERENCE_OBJECT_ID),
169    ///     )
170    ///     .await?;
171    ///
172    /// for obj in objs.body() {
173    ///     let obj = match obj {
174    ///         Ok(obj) => obj,
175    ///         // Only this ref failed; the remaining objects are still usable
176    ///         Err(e) => {
177    ///             eprintln!("could not read object: {e}");
178    ///             continue;
179    ///         }
180    ///     };
181    ///
182    ///     // Convert proto object to SDK type
183    ///     let sdk_obj = obj.object()?;
184    ///     println!("Got object ID: {:?}", sdk_obj.id());
185    ///     let obj_ref = obj.object_reference()?;
186    ///     println!("Object version: {:?}", obj_ref.version());
187    /// }
188    /// # Ok(())
189    /// # }
190    /// ```
191    pub async fn objects_with_versions(
192        &self,
193        refs: impl IntoIterator<Item = (ObjectId, Option<Version>)>,
194        read_mask: impl IntoReadMask<ObjectReadMask>,
195    ) -> GrpcResult<MetadataEnvelope<Vec<GrpcResult<Object>>>> {
196        self.objects_internal(refs.into_iter().collect(), read_mask.into_read_mask())
197            .await
198    }
199
200    async fn objects_internal(
201        &self,
202        refs: Vec<(ObjectId, Option<Version>)>,
203        read_mask: ObjectReadMask,
204    ) -> GrpcResult<MetadataEnvelope<Vec<GrpcResult<Object>>>> {
205        if refs.is_empty() {
206            return Err(GrpcError::EmptyRequest);
207        }
208
209        let requests = ObjectRequests::default().with_requests(
210            refs.iter()
211                .map(|(id, version)| {
212                    let mut object_ref =
213                        ObjectReference::default().with_object_id(proto_object_id(*id));
214
215                    if let Some(v) = version {
216                        object_ref = object_ref.with_version(v.as_u64());
217                    }
218
219                    ObjectRequest::default().with_object_ref(object_ref)
220                })
221                .collect(),
222        );
223
224        let mut request = GetObjectsRequest::default()
225            .with_requests(requests)
226            .with_read_mask(read_mask);
227
228        if let Some(max_size) = self.max_decoding_message_size() {
229            request = request.with_max_message_size_bytes(saturating_usize_to_u32(max_size));
230        }
231
232        let mut client = self.ledger_service_client();
233
234        let response = client.get_objects(request).await?;
235        let (stream, metadata) = MetadataEnvelope::from(response).into_parts();
236
237        // Server guarantees results are returned in request order
238        let response = collect_stream(stream, metadata, |msg| {
239            Ok((msg.has_next, into_item_results(msg.objects)))
240        })
241        .await?;
242        check_result_count(response.body(), refs.len())?;
243        check_object_identity(response.body(), &refs)?;
244
245        Ok(response)
246    }
