24impl Client {
25    /// Get objects by their IDs.
26    ///
27    /// Returns proto `Object` types. Use `obj.object()` to convert to SDK
28    /// type, or use `obj.object_reference()` to get the object reference.
29    ///
30    /// Results are returned in the same order as the input IDs, one per ID.
31    ///
32    /// # Errors
33    ///
34    /// Returns [`Error::EmptyRequest`] if `refs` is empty.
35    ///
36    /// Each ID gets its own result: an object that is not found (never
37    /// existed, was deleted, or has been pruned by the serving node) yields
38    /// [`Error::Server`] with code `NOT_FOUND` in that slot only, leaving the
39    /// other objects intact. The outer `Result` is reserved for failures of the
40    /// call itself, such as a transport error, and for a server that answered
41    /// with a different number of results than IDs requested
42    /// ([`UnexpectedResultCount`]), which leaves no way to tell which ID each
43    /// result belongs to, or answered a position with a different object than
44    /// the one requested there ([`UnexpectedObject`]). The answered id is read
45    /// from the object reference or its BCS, so a read mask that includes
46    /// neither leaves nothing to check.
47    ///
48    /// [`UnexpectedResultCount`]: crate::ProtocolError::UnexpectedResultCount
49    /// [`UnexpectedObject`]: crate::ProtocolError::UnexpectedObject
50    ///
51    /// # Read Mask
52    ///
53    /// The `read_mask` parameter controls which fields the server returns; use
54    /// `ObjectReadMask::default()` for the default mask, or pass an
55    /// [`ObjectReadMask`](iota_grpc_types::read_mask_fields::ObjectReadMask)
56    /// built from an
57    /// [`ObjectField`](iota_grpc_types::read_mask_fields::ObjectField) or any
58    /// slice/array/vec of fields.
59    ///
60    /// # Example
61    ///
62    /// ```no_run
63    /// # use iota_sdk_grpc_client::Client;
64    /// # use iota_sdk_grpc_client::read_mask_fields::{ObjectField, ObjectReadMask};
65    /// # use iota_types::ObjectId;
66    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
67    /// let client = Client::new_localnet()?;
68    /// let object_id: ObjectId = "0x2".parse()?;
69    /// let ids = [object_id];
70    ///
71    /// // Default mask
72    /// let objs = client.get_objects(ids, ObjectReadMask::default()).await?;
73    ///
74    /// // Selected fields
75    /// let objs = client
76    ///     .get_objects(
77    ///         ids,
78    ///         ObjectReadMask::from([ObjectField::REFERENCE, ObjectField::BCS]),
79    ///     )
80    ///     .await?;
81    ///
82    /// for obj in objs.body() {
83    ///     let obj = match obj {
84    ///         Ok(obj) => obj,
85    ///         // Only this ID failed; the remaining objects are still usable
86    ///         Err(e) => {
87    ///             eprintln!("could not read object: {e}");
88    ///             continue;
89    ///         }
90    ///     };
91    ///
92    ///     // Convert proto object to SDK type
93    ///     let sdk_obj = obj.object()?;
94    ///     println!("Got object ID: {:?}", sdk_obj.id());
95    ///     let obj_ref = obj.object_reference()?;
96    ///     println!("Object version: {:?}", obj_ref.version());
97    /// }
98    ///
99    /// // Results line up with the requested IDs, so pair them to find out
100    /// // which objects the node does not have
101    /// let missing: Vec<ObjectId> = ids
102    ///     .iter()
103    ///     .zip(objs.body())
104    ///     .filter_map(|(id, result)| match result {
105    ///         Err(e) if e.is_not_found() => Some(*id),
106    ///         _ => None,
107    ///     })
108    ///     .collect();
109    /// println!("Missing objects: {missing:?}");
110    /// # Ok(())
111    /// # }
112    /// ```
113    pub async fn get_objects(
114        &self,
115        refs: impl IntoIterator<Item = ObjectId>,
116        read_mask: impl IntoReadMask<ObjectReadMask>,
117    ) -> Result<MetadataEnvelope<Vec<Result<Object>>>> {
118        let refs = refs.into_iter().map(|id| (id, None)).collect::<Vec<_>>();
119
120        self.get_objects_internal(refs, read_mask.into_read_mask())
121            .await
122    }

124    /// Get objects by their IDs and optional versions.
125    ///
126    /// Returns proto `Object` types. Use `obj.object()` to convert to SDK
127    /// type, or use `obj.object_reference()` to get the object reference.
128    ///
129    /// Results are returned in the same order as the input refs, one per ref.
130    ///
131    /// # Errors
132    ///
133    /// Returns [`Error::EmptyRequest`] if `refs` is empty.
134    ///
135    /// Each ref gets its own result, with the same meaning as in
136    /// [`get_objects`](Client::get_objects): a requested version the serving
137    /// node does not have fails only its own slot.
138    ///
139    /// # Read Mask
140    ///
141    /// The `read_mask` parameter controls which fields the server returns; use
142    /// `ObjectReadMask::default()` for the default mask, or pass an
143    /// [`ObjectReadMask`](iota_grpc_types::read_mask_fields::ObjectReadMask)
144    /// built from an
145    /// [`ObjectField`](iota_grpc_types::read_mask_fields::ObjectField) or any
146    /// slice/array/vec of fields.
147    ///
148    /// # Example
149    ///
150    /// ```no_run
151    /// # use iota_sdk_grpc_client::Client;
152    /// # use iota_sdk_grpc_client::read_mask_fields::{ObjectField, ObjectReadMask};
153    /// # use iota_types::ObjectId;
154    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
155    /// let client = Client::new_localnet()?;
156    /// let object_id: ObjectId = "0x2".parse()?;
157    ///
158    /// // Default mask
159    /// let objs = client
160    ///     .get_objects_with_versions([(object_id, None)], ObjectReadMask::default())
161    ///     .await?;
162    ///
163    /// // Selected fields
164    /// let objs = client
165    ///     .get_objects_with_versions(
166    ///         [(object_id, None)],
167    ///         ObjectReadMask::from(ObjectField::REFERENCE_OBJECT_ID),
168    ///     )
169    ///     .await?;
170    ///
171    /// for obj in objs.body() {
172    ///     let obj = match obj {
173    ///         Ok(obj) => obj,
174    ///         // Only this ref failed; the remaining objects are still usable
175    ///         Err(e) => {
176    ///             eprintln!("could not read object: {e}");
177    ///             continue;
178    ///         }
179    ///     };
180    ///
181    ///     // Convert proto object to SDK type
182    ///     let sdk_obj = obj.object()?;
183    ///     println!("Got object ID: {:?}", sdk_obj.id());
184    ///     let obj_ref = obj.object_reference()?;
185    ///     println!("Object version: {:?}", obj_ref.version());
186    /// }
187    /// # Ok(())
188    /// # }
189    /// ```
190    pub async fn get_objects_with_versions(
191        &self,
192        refs: impl IntoIterator<Item = (ObjectId, Option<Version>)>,
193        read_mask: impl IntoReadMask<ObjectReadMask>,
194    ) -> Result<MetadataEnvelope<Vec<Result<Object>>>> {
195        self.get_objects_internal(refs.into_iter().collect(), read_mask.into_read_mask())
196            .await
197    }
198
199    async fn get_objects_internal(
200        &self,
201        refs: Vec<(ObjectId, Option<Version>)>,
202        read_mask: ObjectReadMask,
203    ) -> Result<MetadataEnvelope<Vec<Result<Object>>>> {
204        if refs.is_empty() {
205            return Err(Error::EmptyRequest);
206        }
207
208        let requests = ObjectRequests::default().with_requests(
209            refs.iter()
210                .map(|(id, version)| {
211                    let mut object_ref =
212                        ObjectReference::default().with_object_id(proto_object_id(*id));
213
214                    if let Some(v) = version {
215                        object_ref = object_ref.with_version(v.as_u64());
216                    }
217
218                    ObjectRequest::default().with_object_ref(object_ref)
219                })
220                .collect(),
221        );
222
223        let mut request = GetObjectsRequest::default()
224            .with_requests(requests)
225            .with_read_mask(read_mask);
226
227        if let Some(max_size) = self.max_decoding_message_size() {
228            request = request.with_max_message_size_bytes(saturating_usize_to_u32(max_size));
229        }
230
231        let mut client = self.ledger_service_client();
232
233        let response = client.get_objects(request).await?;
234        let (stream, metadata) = MetadataEnvelope::from(response).into_parts();
235
236        // Server guarantees results are returned in request order
237        let response = collect_stream(stream, metadata, |msg| {
238            Ok((msg.has_next, into_item_results(msg.objects)))
239        })
240        .await?;
241        check_result_count(response.body(), refs.len())?;
242        check_object_identity(response.body(), &refs)?;
243
244        Ok(response)
245    }
