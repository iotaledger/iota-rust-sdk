23impl Client {
24    /// Get transactions by their digests.
25    ///
26    /// Returns proto `ExecutedTransaction` for each transaction. Use the lazy
27    /// conversion methods to extract data:
28    /// - `tx.digest()` - Get transaction digest
29    /// - `tx.transaction()` - Deserialize transaction
30    /// - `tx.signatures()` - Deserialize signatures
31    /// - `tx.effects()` - Deserialize effects
32    /// - `tx.events()` - Deserialize events (if available)
33    /// - `tx.checkpoint_sequence_number()` - Get checkpoint number
34    /// - `tx.timestamp_ms()` - Get timestamp
35    ///
36    /// Results are returned in the same order as the input digests, one per
37    /// digest.
38    ///
39    /// # Errors
40    ///
41    /// Returns [`Error::EmptyRequest`] if `digests` is empty.
42    ///
43    /// Each digest gets its own result: a transaction the node does not have
44    /// (never executed, or pruned) yields [`Error::Server`] with code
45    /// `NOT_FOUND` in that slot only, leaving the other transactions intact. A
46    /// slot can also carry `FAILED_PRECONDITION` when the transaction itself is
47    /// present but an object a requested field needs is gone, as described
48    /// under Read Mask below. The outer `Result` is reserved for failures
49    /// of the call itself, such as a transport error, and for a server that
50    /// answered with a different number of results than digests requested
51    /// ([`UnexpectedResultCount`]), which leaves no way to tell which digest
52    /// each result belongs to, or answered a position with a different
53    /// transaction than the one requested there ([`UnexpectedTransaction`]).
54    /// The answered digest is read from the response or computed from the
55    /// transaction's BCS, so a read mask that includes neither leaves nothing
56    /// to check.
57    ///
58    /// [`UnexpectedResultCount`]: crate::ProtocolError::UnexpectedResultCount
59    /// [`UnexpectedTransaction`]: crate::ProtocolError::UnexpectedTransaction
60    ///
61    /// # Read Mask
62    ///
63    /// The `read_mask` controls which fields the server returns; use
64    /// `TransactionReadMask::default()` for the default field mask, or pass a
65    /// [`TransactionReadMask`](iota_grpc_types::read_mask_fields::TransactionReadMask)
66    /// built from a
67    /// [`TransactionField`](iota_grpc_types::read_mask_fields::TransactionField)
68    /// or any slice/array/vec of fields.
69    ///
70    /// The `input_objects`, `output_objects`, `balance_changes` and
71    /// `object_changes` fields (also included by wildcard masks) require the
72    /// serving node to still have the transaction's objects. If one has been
73    /// pruned, the transaction's result is a `FAILED_PRECONDITION` error
74    /// instead of a silently incomplete answer — narrow the read mask, or
75    /// fetch objects individually via
76    /// [`get_objects`](Client::get_objects) for best-effort retrieval.
77    ///
78    /// # Example
79    ///
80    /// ```no_run
81    /// # use iota_sdk_grpc_client::Client;
82    /// # use iota_sdk_grpc_client::read_mask_fields::{TransactionField, TransactionReadMask};
83    /// # use iota_types::TransactionDigest;
84    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
85    /// let client = Client::new_localnet()?;
86    /// let digest: TransactionDigest = TransactionDigest::ZERO;
87    ///
88    /// // Default mask
89    /// let txs = client
90    ///     .get_transactions([digest], TransactionReadMask::default())
91    ///     .await?;
92    /// for tx in txs.body() {
93    ///     let tx = match tx {
94    ///         Ok(tx) => tx,
95    ///         // Only this digest failed; the remaining transactions are still
96    ///         // usable
97    ///         Err(e) => {
98    ///             eprintln!("could not read transaction: {e}");
99    ///             continue;
100    ///         }
101    ///     };
102    ///
103    ///     // Lazy conversion - only deserialize what you need
104    ///     let effects = tx.effects()?.effects()?;
105    ///     println!("Status: {:?}", effects.as_v1().status);
106    ///
107    ///     // Access checkpoint number
108    ///     let checkpoint = tx.checkpoint_sequence_number()?;
109    ///     println!("Checkpoint: {}", checkpoint);
110    /// }
111    ///
112    /// // Selected fields
113    /// let txs = client
114    ///     .get_transactions(
115    ///         [digest],
116    ///         TransactionReadMask::from([TransactionField::EFFECTS, TransactionField::CHECKPOINT]),
117    ///     )
118    ///     .await?;
119    /// # Ok(())
120    /// # }
121    /// ```
122    pub async fn get_transactions(
123        &self,
124        digests: impl IntoIterator<Item = TransactionDigest>,
125        read_mask: impl IntoReadMask<TransactionReadMask>,
126    ) -> Result<MetadataEnvelope<Vec<Result<ExecutedTransaction>>>> {
127        let digests = digests.into_iter().collect::<Vec<_>>();
128        if digests.is_empty() {
129            return Err(Error::EmptyRequest);
130        }
131
132        let requests = TransactionRequests::default().with_requests(
133            digests
134                .iter()
135                .map(|d| TransactionRequest::default().with_digest(*d))
136                .collect(),
137        );
138
139        let mut request = GetTransactionsRequest::default()
140            .with_requests(requests)
141            .with_read_mask(read_mask.into_read_mask());
142
143        if let Some(max_size) = self.max_decoding_message_size() {
144            request = request.with_max_message_size_bytes(saturating_usize_to_u32(max_size));
145        }
146
147        let mut client = self.ledger_service_client();
148
149        let response = client.get_transactions(request).await?;
150        let (stream, metadata) = MetadataEnvelope::from(response).into_parts();
151
152        // Server guarantees results are returned in request order
153        let response = collect_stream(stream, metadata, |msg| {
154            Ok((msg.has_next, into_item_results(msg.transaction_results)))
155        })
156        .await?;
157        check_result_count(response.body(), digests.len())?;
158        check_transaction_identity(response.body(), &digests)?;
159
160        Ok(response)
161    }
162}
