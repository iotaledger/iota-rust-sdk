26define_query! {
27    /// Query for [`GrpcClient::transactions`]. Await it to send the request.
28    pub struct GetTransactionsQuery {
29        service_client: LedgerServiceClient<InterceptedChannel>,
30        max_message_size: Option<usize>,
31        digests: Vec<TransactionDigest>,
32        read_mask: TransactionReadMask,
33    }
34    output: GrpcResult<MetadataEnvelope<Vec<GrpcResult<ExecutedTransaction>>>>;
35}
36
37impl GetTransactionsQuery {
38    /// Set the field mask controlling the returned fields.
39    pub fn read_mask(mut self, read_mask: impl IntoReadMask<TransactionReadMask>) -> Self {
40        self.read_mask = read_mask.into_read_mask();
41        self
42    }
43
44    fn request(&self) -> GetTransactionsRequest {
45        let requests = TransactionRequests::default().with_requests(
46            self.digests
47                .iter()
48                .map(|d| TransactionRequest::default().with_digest(*d))
49                .collect(),
50        );
51
52        let mut request = GetTransactionsRequest::default()
53            .with_requests(requests)
54            .with_read_mask(self.read_mask.clone());
55
56        if let Some(max_size) = self.max_message_size {
57            request = request.with_max_message_size_bytes(saturating_usize_to_u32(max_size));
58        }
59
60        request
61    }
62
63    async fn send(mut self) -> GrpcResult<MetadataEnvelope<Vec<GrpcResult<ExecutedTransaction>>>> {
64        if self.digests.is_empty() {
65            return Err(GrpcError::EmptyRequest);
66        }
67
68        let request = self.request();
69        let response = self.service_client.get_transactions(request).await?;
70        let (stream, metadata) = MetadataEnvelope::from(response).into_parts();
71
72        // Server guarantees results are returned in request order
73        let response = collect_stream(stream, metadata, |msg| {
74            Ok((msg.has_next, into_item_results(msg.transaction_results)))
75        })
76        .await?;
77        check_result_count(response.body(), self.digests.len())?;
78        check_transaction_identity(response.body(), &self.digests)?;
79
80        Ok(response)
81    }
82}

84impl GrpcClient {
85    /// Get transactions by their digests.
86    ///
87    /// Returns proto `ExecutedTransaction` for each transaction. Use the lazy
88    /// conversion methods to extract data:
89    /// - `tx.digest()` - Get transaction digest
90    /// - `tx.transaction()` - Deserialize transaction
91    /// - `tx.signatures()` - Deserialize signatures
92    /// - `tx.effects()` - Deserialize effects
93    /// - `tx.events()` - Deserialize events (if available)
94    /// - `tx.checkpoint_sequence_number()` - Get checkpoint number
95    /// - `tx.timestamp_ms()` - Get timestamp
96    ///
97    /// Results are returned in the same order as the input digests, one per
98    /// digest.
99    ///
100    /// # Errors
101    ///
102    /// Returns [`GrpcError::EmptyRequest`] if `digests` is empty.
103    ///
104    /// Each digest gets its own result: a transaction the node does not have
105    /// (never executed, or pruned) yields [`GrpcError::Server`] with code
106    /// `NOT_FOUND` in that slot only, leaving the other transactions intact. A
107    /// slot can also carry `FAILED_PRECONDITION` when the transaction itself is
108    /// present but an object a requested field needs is gone, as described
109    /// under Read Mask below. The outer `GrpcResult` is reserved for failures
110    /// of the call itself, such as a transport error, and for a server that
111    /// answered with a different number of results than digests requested
112    /// ([`UnexpectedResultCount`]), which leaves no way to tell which digest
113    /// each result belongs to, or answered a position with a different
114    /// transaction than the one requested there ([`UnexpectedTransaction`]).
115    /// The answered digest is read from the response or computed from the
116    /// transaction's BCS, so a read mask that includes neither leaves nothing
117    /// to check.
118    ///
119    /// [`UnexpectedResultCount`]: crate::ProtocolError::UnexpectedResultCount
120    /// [`UnexpectedTransaction`]: crate::ProtocolError::UnexpectedTransaction
121    ///
122    /// # Read Mask
123    ///
124    /// Without [`read_mask`](GetTransactionsQuery::read_mask), the default
125    /// mask is used. Pass a
126    /// [`TransactionReadMask`](iota_grpc_types::read_mask_fields::TransactionReadMask)
127    /// built from a
128    /// [`TransactionField`](iota_grpc_types::read_mask_fields::TransactionField)
129    /// or any slice/array/vec of fields to choose the returned fields.
130    ///
131    /// The `input_objects`, `output_objects`, `balance_changes` and
132    /// `object_changes` fields (also included by wildcard masks) require the
133    /// serving node to still have the transaction's objects. If one has been
134    /// pruned, the transaction's result is a `FAILED_PRECONDITION` error
135    /// instead of a silently incomplete answer — narrow the read mask, or
136    /// fetch objects individually via
137    /// [`objects`](GrpcClient::objects) for best-effort retrieval.
138    ///
139    /// # Example
140    ///
141    /// ```no_run
142    /// # use iota_sdk_grpc_client::GrpcClient;
143    /// # use iota_sdk_grpc_client::read_mask_fields::{TransactionField, TransactionReadMask};
144    /// # use iota_types::TransactionDigest;
145    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
146    /// let client = GrpcClient::new_localnet()?;
147    /// let digest: TransactionDigest = TransactionDigest::ZERO;
148    ///
149    /// // Default mask
150    /// let txs = client.transactions([digest]).await?;
151    /// for tx in txs.body() {
152    ///     let tx = match tx {
153    ///         Ok(tx) => tx,
154    ///         // Only this digest failed; the remaining transactions are still
155    ///         // usable
156    ///         Err(e) => {
157    ///             eprintln!("could not read transaction: {e}");
158    ///             continue;
159    ///         }
160    ///     };
161    ///
162    ///     // Lazy conversion - only deserialize what you need
163    ///     let effects = tx.effects()?.effects()?;
164    ///     println!("Status: {:?}", effects.as_v1().status);
165    ///
166    ///     // Access checkpoint number
167    ///     let checkpoint = tx.checkpoint_sequence_number()?;
168    ///     println!("Checkpoint: {}", checkpoint);
169    /// }
170    ///
171    /// // Selected fields
172    /// let txs = client
173    ///     .transactions([digest])
174    ///     .read_mask(TransactionReadMask::from([
175    ///         TransactionField::EFFECTS,
176    ///         TransactionField::CHECKPOINT,
177    ///     ]))
178    ///     .await?;
179    /// # Ok(())
180    /// # }
181    /// ```
182    pub fn transactions(
183        &self,
184        digests: impl IntoIterator<Item = TransactionDigest>,
185    ) -> GetTransactionsQuery {
186        GetTransactionsQuery {
187            service_client: self.ledger_service_client(),
188            max_message_size: self.max_decoding_message_size(),
189            digests: digests.into_iter().collect(),
190            read_mask: TransactionReadMask::default(),
191        }
192    }
193}

195#[cfg(test)]
196mod tests {
197    use iota_grpc_types::read_mask_fields::{TransactionField, TransactionReadMask};
198    use iota_types::TransactionDigest;
199
200    use crate::{GrpcClient, GrpcError};
201
202    #[tokio::test]
203    async fn read_mask_replaces_the_default_mask() {
204        let client = GrpcClient::new("http://localhost").unwrap();
205        let query = client.transactions([TransactionDigest::ZERO]);
206        assert_eq!(
207            query.read_mask.as_str(),
208            TransactionReadMask::default().as_str()
209        );
210
211        let query = query.read_mask(TransactionField::EFFECTS);
212        assert_eq!(
213            query.read_mask.as_str(),
214            TransactionReadMask::from(TransactionField::EFFECTS).as_str()
215        );
216    }
217
218    #[tokio::test]
219    async fn awaiting_no_digests_is_an_empty_request() {
220        let client = GrpcClient::new("http://localhost").unwrap();
221        let result = client.transactions(Vec::new()).await;
222        assert!(matches!(result, Err(GrpcError::EmptyRequest)));
223    }
224
225    #[tokio::test]
226    async fn the_request_carries_every_digest_the_mask_and_the_message_size() {
227        let client = GrpcClient::new("http://localhost")
228            .unwrap()
229            .with_max_decoding_message_size(1024);
230        let query = client
231            .transactions([TransactionDigest::ZERO])
232            .read_mask(TransactionField::EFFECTS);
233        let request = query.request();
234
235        let digests: Vec<_> = request
236            .requests
237            .unwrap()
238            .requests
239            .into_iter()
240            .map(|r| r.digest)
241            .collect();
242        assert_eq!(digests, vec![Some(TransactionDigest::ZERO.into())]);
243        assert_eq!(
244            request.read_mask,
245            Some(TransactionReadMask::from(TransactionField::EFFECTS).into())
246        );
247        assert_eq!(request.max_message_size_bytes, Some(1024));
248    }
