4//! High-level API for checkpoint queries.
5//!
6//! # Read Mask
7//!
8//! The checkpoint queries and streams take a mask through their `read_mask`
9//! setter to control which data is included in the response. Without the
10//! setter the default mask is used. Set a
11//! [`CheckpointResponseField`](iota_grpc_types::read_mask_fields::CheckpointResponseField)
12//! (or any slice/array/vec of fields) to choose the returned fields.

39define_query! {
40    /// Query for [`GrpcClient::checkpoint_latest`],
41    /// [`GrpcClient::checkpoint_by_sequence_number`] and
42    /// [`GrpcClient::checkpoint_by_digest`]. Await it to send the request.
43    pub struct GetCheckpointQuery {
44        service_client: LedgerServiceClient<InterceptedChannel>,
45        max_message_size: Option<usize>,
46        checkpoint_id: get_checkpoint_request::CheckpointId,
47        transactions_filter: Option<grpc_filter::TransactionFilter>,
48        events_filter: Option<grpc_filter::EventFilter>,
49        read_mask: CheckpointResponseReadMask,
50    }
51    output: GrpcResult<MetadataEnvelope<CheckpointResponse>>;
52}
53
54impl GetCheckpointQuery {
55    /// Set the filter to apply to transactions.
56    pub fn transactions_filter(
57        mut self,
58        transactions_filter: grpc_filter::TransactionFilter,
59    ) -> Self {
60        self.transactions_filter = Some(transactions_filter);
61        self
62    }
63
64    /// Set the filter to apply to events.
65    pub fn events_filter(mut self, events_filter: grpc_filter::EventFilter) -> Self {
66        self.events_filter = Some(events_filter);
67        self
68    }
69
70    /// Set the field mask controlling the returned fields.
71    pub fn read_mask(mut self, read_mask: impl IntoReadMask<CheckpointResponseReadMask>) -> Self {
72        self.read_mask = read_mask.into_read_mask();
73        self
74    }
75
76    fn into_request(
77        self,
78    ) -> GrpcResult<(
79        LedgerServiceClient<InterceptedChannel>,
80        GetCheckpointRequest,
81    )> {
82        let mut request = match self.checkpoint_id {
83            get_checkpoint_request::CheckpointId::Latest(val) => {
84                GetCheckpointRequest::default().with_latest(val)
85            }
86            get_checkpoint_request::CheckpointId::SequenceNumber(val) => {
87                GetCheckpointRequest::default().with_sequence_number(val)
88            }
89            get_checkpoint_request::CheckpointId::Digest(val) => {
90                GetCheckpointRequest::default().with_digest(val)
91            }
92            _ => {
93                return Err(GrpcError::Protocol(ProtocolError::UnknownVariant(
94                    "checkpoint ID",
95                )));
96            }
97        }
98        .with_read_mask(self.read_mask);
99
100        if let Some(tf) = self.transactions_filter {
101            request = request.with_transactions_filter(tf);
102        }
103        if let Some(ef) = self.events_filter {
104            request = request.with_events_filter(ef);
105        }
106        if let Some(max_size) = self.max_message_size.map(saturating_usize_to_u32) {
107            request = request.with_max_message_size_bytes(max_size);
108        }
109
110        Ok((self.service_client, request))
111    }
112
113    async fn send(self) -> GrpcResult<MetadataEnvelope<CheckpointResponse>> {
114        let (mut service_client, request) = self.into_request()?;
115        let response = service_client.get_checkpoint(request).await?;
116        let (stream, metadata) = MetadataEnvelope::from(response).into_parts();
117
118        let reassembled = GrpcClient::reassemble_checkpoint_data_stream(stream);
119        futures::pin_mut!(reassembled);
120
121        // Skip any progress messages and find the first checkpoint
122        let checkpoint = loop {
123            match reassembled.next().await {
124                Some(Ok(CheckpointStreamItem::Checkpoint(cp))) => break *cp,
125                Some(Ok(CheckpointStreamItem::Progress { .. })) => continue,
126                Some(Err(e)) => return Err(e),
127                None => {
128                    return Err(TryFromProtoError::missing("checkpoint data").into());
129                }
130            }
131        };
132
133        Ok(MetadataEnvelope::new(checkpoint, metadata))
134    }
135}

144struct CheckpointStreamOptions {
145    service_client: LedgerServiceClient<InterceptedChannel>,
146    max_message_size: Option<usize>,
147    start_sequence_number: Option<CheckpointSequenceNumber>,
148    end_sequence_number: Option<CheckpointSequenceNumber>,
149    transactions_filter: Option<grpc_filter::TransactionFilter>,
150    events_filter: Option<grpc_filter::EventFilter>,
151    read_mask: CheckpointResponseReadMask,
152}
153
154impl CheckpointStreamOptions {
155    async fn open(
156        (mut service_client, request): StreamCheckpointsCall,
157    ) -> GrpcResult<MetadataEnvelope<CheckpointItemStream>> {
158        let response = service_client.stream_checkpoints(request).await?;
159        let (stream, metadata) = MetadataEnvelope::from(response).into_parts();
160
161        Ok(MetadataEnvelope::new(
162            Box::pin(GrpcClient::reassemble_checkpoint_data_stream(stream)),
163            metadata,
164        ))
165    }
166
167    fn into_request(
168        self,
169        filter_checkpoints: bool,
170        progress_interval_ms: Option<u32>,
171    ) -> StreamCheckpointsCall {
172        let mut request = StreamCheckpointsRequest::default().with_read_mask(self.read_mask);
173
174        if let Some(start) = self.start_sequence_number {
175            request = request.with_start_sequence_number(start);
176        }
177        if let Some(end) = self.end_sequence_number {
178            request = request.with_end_sequence_number(end);
179        }
180        if let Some(tf) = self.transactions_filter {
181            request = request.with_transactions_filter(tf);
182        }
183        if let Some(ef) = self.events_filter {
184            request = request.with_events_filter(ef);
185        }
186        if filter_checkpoints {
187            request = request.with_filter_checkpoints(true);
188        }
189        if let Some(ms) = progress_interval_ms {
190            request = request.with_progress_interval_ms(ms);
191        }
192        if let Some(max_size) = self.max_message_size.map(saturating_usize_to_u32) {
193            request = request.with_max_message_size_bytes(max_size);
194        }
195
196        (self.service_client, request)
197    }
198}

250
251    async fn send(self) -> GrpcResult<MetadataEnvelope<CheckpointStream>> {
252        let (stream, metadata) = CheckpointStreamOptions::open(self.into_request())
253            .await?
254            .into_parts();
255
256        // remove the wrapping CheckpointStreamItem layer since we know
257        // filter_checkpoints is false and thus only Checkpoint items will be
258        // produced
259        let filtered = stream.filter_map(|item| async {
260            match item {
261                Ok(CheckpointStreamItem::Checkpoint(cp)) => Some(Ok(*cp)),
262                Ok(CheckpointStreamItem::Progress { .. }) => None,
263                Err(e) => Some(Err(e)),
264            }
265        });
266
267        Ok(MetadataEnvelope::new(Box::pin(filtered), metadata))
268    }

443    /// Stream checkpoints across a range of checkpoints.
444    ///
445    /// Returns a stream of [`CheckpointResponse`] objects, each representing
446    /// a complete checkpoint with its transactions and events. Every checkpoint
447    /// in the range is yielded, even if the filters produce no matching
448    /// transactions or events within it.
449    ///
450    /// To skip non-matching checkpoints entirely, use
451    /// [`checkpoints_stream_filtered`](Self::checkpoints_stream_filtered).
452    ///
453    /// Without [`start_sequence_number`](CheckpointsStreamQuery::start_sequence_number)
454    /// the stream starts from the latest checkpoint, and without
455    /// [`end_sequence_number`](CheckpointsStreamQuery::end_sequence_number) it
456    /// runs indefinitely. Filter with
457    /// [`transactions_filter`](CheckpointsStreamQuery::transactions_filter) and
458    /// [`events_filter`](CheckpointsStreamQuery::events_filter), and choose
459    /// the returned fields with
460    /// [`read_mask`](CheckpointsStreamQuery::read_mask).
461    ///
462    /// **Note:** The metadata in the returned [`MetadataEnvelope`] is captured
463    /// from the initial gRPC response headers when the stream is opened. It is
464    /// **not** updated as subsequent checkpoint data arrives.
465    ///
466    /// # Example
467    ///
468    /// ```no_run
469    /// # use iota_sdk_grpc_client::GrpcClient;
470    /// # use futures::StreamExt;
471    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
472    /// let client = GrpcClient::new_localnet()?;
473    /// let mut stream = client
474    ///     .checkpoints_stream()
475    ///     .start_sequence_number(0)
476    ///     .end_sequence_number(10)
477    ///     .await?;
478    ///
479    /// while let Some(checkpoint) = stream.body_mut().next().await {
480    ///     let checkpoint = checkpoint?;
481    ///     println!("Received checkpoint {}", checkpoint.sequence_number());
482    /// }
483    /// # Ok(())
484    /// # }
485    /// ```
486    pub fn checkpoints_stream(&self) -> CheckpointsStreamQuery {
487        CheckpointsStreamQuery {
488            options: self.checkpoint_stream_options(),
489        }
490    }

492    /// Stream checkpoints, skipping those with no matching data.
493    ///
494    /// Unlike [`checkpoints_stream`](Self::checkpoints_stream), this method
495    /// sets `filter_checkpoints = true` on the server, which means checkpoints
496    /// without any matching transactions or events are skipped entirely.
497    ///
498    /// The returned stream yields [`CheckpointStreamItem`], which is either a
499    /// [`CheckpointStreamItem::Checkpoint`] or a
500    /// [`CheckpointStreamItem::Progress`]. Progress messages are sent
501    /// periodically during scanning to indicate liveness and the current scan
502    /// position (default every 2 seconds, configurable via
503    /// [`progress_interval_ms`](CheckpointsStreamFilteredQuery::progress_interval_ms)).
504    ///
505    /// For liveness detection, wrap `stream.next()` in
506    /// `tokio::time::timeout()`: if neither a `Checkpoint` nor a `Progress`
507    /// arrives within your chosen duration plus some buffer for connection
508    /// latency, the connection is likely dead.
509    ///
510    /// At least one of
511    /// [`transactions_filter`](CheckpointsStreamFilteredQuery::transactions_filter)
512    /// or [`events_filter`](CheckpointsStreamFilteredQuery::events_filter) must
513    /// be set. The range and the read mask work as for
514    /// [`checkpoints_stream`](Self::checkpoints_stream).
515    ///
516    /// # Example
517    ///
518    /// ```no_run
519    /// # use iota_sdk_grpc_client::{GrpcClient, CheckpointStreamItem};
520    /// # use iota_grpc_types::v1::filter as grpc_filter;
521    /// # use futures::StreamExt;
522    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
523    /// let client = GrpcClient::new_localnet()?;
524    /// // At least one filter is required
525    /// let tx_filter = grpc_filter::TransactionFilter::default();
526    /// let mut stream = client
527    ///     .checkpoints_stream_filtered()
528    ///     .start_sequence_number(0)
529    ///     .transactions_filter(tx_filter)
530    ///     .await?;
531    ///
532    /// while let Some(item) = stream.body_mut().next().await {
533    ///     match item? {
534    ///         CheckpointStreamItem::Checkpoint(cp) => {
535    ///             println!("Matched checkpoint {}", cp.sequence_number());
536    ///         }
537    ///         CheckpointStreamItem::Progress {
538    ///             latest_scanned_sequence_number,
539    ///         } => {
540    ///             println!("Scanned up to {latest_scanned_sequence_number}");
541    ///         }
542    ///         _ => {}
543    ///     }
544    /// }
545    /// # Ok(())
546    /// # }
547    /// ```
548    pub fn checkpoints_stream_filtered(&self) -> CheckpointsStreamFilteredQuery {
549        CheckpointsStreamFilteredQuery {
550            options: self.checkpoint_stream_options(),
551            progress_interval_ms: None,
552        }
553    }

567    /// Reassemble a stream of checkpoint data chunks into complete checkpoints.
568    ///
569    /// The server sends checkpoint data in multiple messages:
570    /// - `Checkpoint` - Contains the checkpoint summary and contents
571    /// - `Transactions` - Contains executed transactions
572    /// - `Events` - Contains events from transactions
573    /// - `Progress` - Liveness indicator during filtered scanning
574    /// - `EndMarker` - Signals the end of one checkpoint's data
575    ///
576    /// This function buffers the chunks and yields [`CheckpointStreamItem`]
577    /// values: either complete [`CheckpointResponse`] objects when an
578    /// `EndMarker` is received, or [`CheckpointStreamItem::Progress`] when
579    /// a progress message arrives.
580    fn reassemble_checkpoint_data_stream<S, E>(
581        stream: S,
582    ) -> impl Stream<Item = GrpcResult<CheckpointStreamItem>>
583    where
584        S: Stream<
585            Item = std::result::Result<iota_grpc_types::v1::ledger_service::CheckpointData, E>,
586        >,
587        E: Into<GrpcError>,
588    {
589        async_stream::try_stream! {
590            futures::pin_mut!(stream);
591
592            // State for accumulating checkpoint data
593            let mut current_sequence_number: Option<CheckpointSequenceNumber> = None;
594            let mut current_summary: Option<checkpoint::CheckpointSummary> = None;
595            let mut current_signature: Option<ProtoValidatorAggregatedSignature> = None;
596            let mut current_contents: Option<checkpoint::CheckpointContents> = None;
597            let mut current_transactions: Vec<ExecutedTransaction> = Vec::new();
598            let mut current_events: Vec<event::Event> = Vec::new();
599
600            while let Some(data) = stream.next().await {
601                let data = data.map_err(|e| e.into())?;
602
603                match data.payload {
604                    Some(checkpoint_data::Payload::Checkpoint(checkpoint)) => {
605                        if checkpoint.sequence_number.is_none() {
606                            Err(TryFromProtoError::missing("checkpoint.sequence_number"))?;
607                        }
608
609                        // Start of new checkpoint - throw error if previous checkpoint was incomplete
610                        if current_sequence_number.is_some() {
611                            Err(GrpcError::Protocol(CheckpointStreamError::IncompleteCheckpoint.into()))?;
612                        }
613                        current_sequence_number = checkpoint.sequence_number;
614
615                        // Store proto summary (optional, no deserialization)
616                        current_summary = checkpoint.summary;
617
618                        // Store proto signature (optional, no deserialization)
619                        current_signature = checkpoint.signature;
620
621                        // Store proto contents (optional, no deserialization)
622                        current_contents = checkpoint.contents;
623
624                        // Reset accumulators for new checkpoint (in case Transactions or Events
625                        // arrived between endmarker and Checkpoint)
626                        current_transactions.clear();
627                        current_events.clear();
628                    }
629
630                    Some(checkpoint_data::Payload::ExecutedTransactions(txs)) => {
631                        if current_sequence_number.is_none() {
632                            Err(GrpcError::Protocol(CheckpointStreamError::DataBeforeHeader { data_kind: "transactions" }.into()))?;
633                        }
634
635                        // Accumulate proto transactions (no deserialization)
636                        current_transactions.extend(txs.executed_transactions.into_iter());
637                    }
638
639                    Some(checkpoint_data::Payload::Events(events)) => {
640                        if current_sequence_number.is_none() {
641                            Err(GrpcError::Protocol(CheckpointStreamError::DataBeforeHeader { data_kind: "events" }.into()))?;
642                        }
643
644                        // Accumulate proto events (no deserialization)
645                        current_events.extend(events.events);
646                    }
647
648                    Some(checkpoint_data::Payload::EndMarker(marker)) => {
649                        // End of current checkpoint - assemble the result and yield it
650                         let sequence_number = current_sequence_number
651                        .take()
652                        .ok_or_else(|| -> GrpcError { GrpcError::Protocol(CheckpointStreamError::DataBeforeHeader { data_kind: "end marker" }.into()) })?;
653
654                        let marker_sequence_number = marker.sequence_number
655                        .ok_or_else(|| -> GrpcError { TryFromProtoError::missing("end_marker.sequence_number").into() })?;
656
657                        if marker_sequence_number != sequence_number {
658                            Err(GrpcError::Protocol(CheckpointStreamError::SequenceNumberMismatch {
659                                expected: sequence_number,
660                                actual: marker_sequence_number,
661                            }.into()))?;
662                        }
663
664                        let response = CheckpointResponse {
665                            sequence_number,
666                            summary: current_summary.take(),
667                            signature: current_signature.take(),
668                            contents: current_contents.take(),
669                            executed_transactions: std::mem::take(&mut current_transactions),
670                            events: std::mem::take(&mut current_events),
671                        };
672
673                        yield CheckpointStreamItem::Checkpoint(Box::new(response));
674                    }
675
676                    Some(checkpoint_data::Payload::Progress(progress)) => {
677                        yield CheckpointStreamItem::Progress {
678                            latest_scanned_sequence_number: progress.latest_scanned_sequence_number,
679                        };
680                    }
681
682                    None => {
683                        // Empty payload - skip
684                        continue;
685                    }
686
687                    Some(_) => {
688                        // Unknown payload type
689                        Err(GrpcError::Protocol(CheckpointStreamError::UnknownPayload.into()))?;
690                    }
691                }
692            }
693
694            // Check if stream ended with incomplete checkpoint data
695            if let Some(sequence_number) = current_sequence_number {
696                Err(GrpcError::Protocol(CheckpointStreamError::IncompleteStream { sequence_number }.into()))?;
697            }
698        }
699    }
700}
