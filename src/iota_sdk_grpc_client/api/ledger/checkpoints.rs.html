6//! # Read Mask
7//!
8//! All checkpoint query methods accept a `read_mask` to control which data is
9//! included in the response. Pass `CheckpointResponseReadMask::default()` for
10//! the default mask, or a
11//! [`CheckpointResponseField`](iota_grpc_types::read_mask_fields::CheckpointResponseField)
12//! (or any slice/array/vec of fields) — conversion is automatic.

174    /// Internal helper to fetch checkpoint by any ID type.
175    async fn get_checkpoint_internal(
176        &self,
177        checkpoint_id: get_checkpoint_request::CheckpointId,
178        transactions_filter: Option<grpc_filter::TransactionFilter>,
179        events_filter: Option<grpc_filter::EventFilter>,
180        read_mask: CheckpointResponseReadMask,
181    ) -> Result<MetadataEnvelope<CheckpointResponse>> {
182        let mut request = match checkpoint_id {
183            get_checkpoint_request::CheckpointId::Latest(val) => {
184                GetCheckpointRequest::default().with_latest(val)
185            }
186            get_checkpoint_request::CheckpointId::SequenceNumber(val) => {
187                GetCheckpointRequest::default().with_sequence_number(val)
188            }
189            get_checkpoint_request::CheckpointId::Digest(val) => {
190                GetCheckpointRequest::default().with_digest(val)
191            }
192            _ => {
193                return Err(Error::Protocol(ProtocolError::UnknownVariant(
194                    "checkpoint ID",
195                )));
196            }
197        }
198        .with_read_mask(read_mask);
199
200        if let Some(tf) = transactions_filter {
201            request = request.with_transactions_filter(tf);
202        }
203        if let Some(ef) = events_filter {
204            request = request.with_events_filter(ef);
205        }
206        if let Some(max_size) = self
207            .max_decoding_message_size()
208            .map(saturating_usize_to_u32)
209        {
210            request = request.with_max_message_size_bytes(max_size);
211        }
212
213        let mut client = self.ledger_service_client();
214        let response = client.get_checkpoint(request).await?;
215        let (stream, metadata) = MetadataEnvelope::from(response).into_parts();
216
217        let reassembled = Self::reassemble_checkpoint_data_stream(stream);
218        futures::pin_mut!(reassembled);
219
220        // Skip any progress messages and find the first checkpoint
221        let checkpoint = loop {
222            match reassembled.next().await {
223                Some(Ok(CheckpointStreamItem::Checkpoint(cp))) => break *cp,
224                Some(Ok(CheckpointStreamItem::Progress { .. })) => continue,
225                Some(Err(e)) => return Err(e),
226                None => {
227                    return Err(TryFromProtoError::missing("checkpoint data").into());
228                }
229            }
230        };
231
232        Ok(MetadataEnvelope::new(checkpoint, metadata))
233    }

235    /// Stream checkpoints across a range of checkpoints.
236    ///
237    /// Returns a stream of [`CheckpointResponse`] objects, each representing
238    /// a complete checkpoint with its transactions and events. Every checkpoint
239    /// in the range is yielded, even if the filters produce no matching
240    /// transactions or events within it.
241    ///
242    /// To skip non-matching checkpoints entirely, use
243    /// [`stream_checkpoints_filtered`](Self::stream_checkpoints_filtered).
244    ///
245    /// The `read_mask` controls which fields the server returns; use
246    /// `CheckpointResponseReadMask::default()` for the default field mask.
247    /// Pass a
248    /// [`CheckpointResponseField`](iota_grpc_types::read_mask_fields::CheckpointResponseField)
249    /// or any slice/array/vec of fields — conversion is automatic.
250    ///
251    /// **Note:** The metadata in the returned [`MetadataEnvelope`] is captured
252    /// from the initial gRPC response headers when the stream is opened. It is
253    /// **not** updated as subsequent checkpoint data arrives.
254    ///
255    /// # Parameters
256    ///
257    /// * `start_sequence_number` - Optional starting checkpoint. If `None`,
258    ///   starts from the latest checkpoint.
259    /// * `end_sequence_number` - Optional ending checkpoint. If `None`, streams
260    ///   indefinitely.
261    /// * `transactions_filter` - Optional filter to apply to transactions
262    /// * `events_filter` - Optional filter to apply to events
263    /// * `read_mask` - Field mask controlling the returned fields
264    ///
265    /// # Example
266    ///
267    /// ```no_run
268    /// # use iota_sdk_grpc_client::Client;
269    /// # use iota_sdk_grpc_client::read_mask_fields::CheckpointResponseReadMask;
270    /// # use futures::StreamExt;
271    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
272    /// let client = Client::new_localnet()?;
273    /// let mut stream = client
274    ///     .stream_checkpoints(
275    ///         Some(0),
276    ///         Some(10),
277    ///         None,
278    ///         None,
279    ///         CheckpointResponseReadMask::default(),
280    ///     )
281    ///     .await?;
282    ///
283    /// while let Some(checkpoint) = stream.body_mut().next().await {
284    ///     let checkpoint = checkpoint?;
285    ///     println!("Received checkpoint {}", checkpoint.sequence_number);
286    /// }
287    /// # Ok(())
288    /// # }
289    /// ```
290    pub async fn stream_checkpoints(
291        &self,
292        start_sequence_number: impl Into<Option<CheckpointSequenceNumber>>,
293        end_sequence_number: impl Into<Option<CheckpointSequenceNumber>>,
294        transactions_filter: impl Into<Option<grpc_filter::TransactionFilter>>,
295        events_filter: impl Into<Option<grpc_filter::EventFilter>>,
296        read_mask: impl IntoReadMask<CheckpointResponseReadMask>,
297    ) -> Result<MetadataEnvelope<Pin<Box<dyn Stream<Item = Result<CheckpointResponse>> + Send>>>>
298    {
299        self.stream_checkpoints_internal(
300            start_sequence_number.into(),
301            end_sequence_number.into(),
302            transactions_filter.into(),
303            events_filter.into(),
304            read_mask.into_read_mask(),
305        )
306        .await
307    }

309    async fn stream_checkpoints_internal(
310        &self,
311        start_sequence_number: Option<CheckpointSequenceNumber>,
312        end_sequence_number: Option<CheckpointSequenceNumber>,
313        transactions_filter: Option<grpc_filter::TransactionFilter>,
314        events_filter: Option<grpc_filter::EventFilter>,
315        read_mask: CheckpointResponseReadMask,
316    ) -> Result<MetadataEnvelope<Pin<Box<dyn Stream<Item = Result<CheckpointResponse>> + Send>>>>
317    {
318        let envelope = self
319            .stream_checkpoints_raw(
320                start_sequence_number,
321                end_sequence_number,
322                transactions_filter,
323                events_filter,
324                false,
325                None,
326                read_mask,
327            )
328            .await?;
329
330        let (stream, metadata) = envelope.into_parts();
331
332        // remove the wrapping CheckpointStreamItem layer since we know
333        // filter_checkpoints is false and thus only Checkpoint items will be produced
334        let filtered = stream.filter_map(|item| async {
335            match item {
336                Ok(CheckpointStreamItem::Checkpoint(cp)) => Some(Ok(*cp)),
337                Ok(CheckpointStreamItem::Progress { .. }) => None,
338                Err(e) => Some(Err(e)),
339            }
340        });
341
342        Ok(MetadataEnvelope::new(Box::pin(filtered), metadata))
343    }

345    /// Stream checkpoints, skipping those with no matching data.
346    ///
347    /// Unlike [`stream_checkpoints`](Self::stream_checkpoints), this method
348    /// sets `filter_checkpoints = true` on the server, which means checkpoints
349    /// without any matching transactions or events are skipped entirely.
350    ///
351    /// The returned stream yields [`CheckpointStreamItem`], which is either a
352    /// [`CheckpointStreamItem::Checkpoint`] or a
353    /// [`CheckpointStreamItem::Progress`]. Progress messages are sent
354    /// periodically during scanning to indicate liveness and the current scan
355    /// position (default every 2 seconds, configurable via
356    /// `progress_interval_ms`).
357    ///
358    /// For liveness detection, wrap `stream.next()` in
359    /// `tokio::time::timeout()` — if neither a `Checkpoint` nor a `Progress`
360    /// arrives within your chosen duration plus some buffer for connection
361    /// latency, the connection is likely dead.
362    ///
363    /// At least one of `transactions_filter` or `events_filter` must be set.
364    ///
365    /// The `read_mask` controls which fields the server returns; use
366    /// `CheckpointResponseReadMask::default()` for the default field mask.
367    /// Pass a
368    /// [`CheckpointResponseField`](iota_grpc_types::read_mask_fields::CheckpointResponseField)
369    /// or any slice/array/vec of fields — conversion is automatic.
370    ///
371    /// # Parameters
372    ///
373    /// * `start_sequence_number` - Optional starting checkpoint. If `None`,
374    ///   starts from the latest checkpoint.
375    /// * `end_sequence_number` - Optional ending checkpoint. If `None`, streams
376    ///   indefinitely.
377    /// * `transactions_filter` - Optional filter to apply to transactions
378    /// * `events_filter` - Optional filter to apply to events
379    /// * `progress_interval_ms` - Optional progress message interval in
380    ///   milliseconds. Defaults to 2000ms. Minimum 500ms.
381    /// * `read_mask` - Field mask controlling the returned fields
382    ///
383    /// # Example
384    ///
385    /// ```no_run
386    /// # use iota_sdk_grpc_client::{Client, CheckpointStreamItem};
387    /// # use iota_sdk_grpc_client::read_mask_fields::CheckpointResponseReadMask;
388    /// # use iota_grpc_types::v1::filter as grpc_filter;
389    /// # use futures::StreamExt;
390    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
391    /// let client = Client::new_localnet()?;
392    /// // At least one filter is required
393    /// let tx_filter = grpc_filter::TransactionFilter::default();
394    /// let mut stream = client
395    ///     .stream_checkpoints_filtered(
396    ///         Some(0),
397    ///         None,
398    ///         Some(tx_filter),
399    ///         None,
400    ///         None,
401    ///         CheckpointResponseReadMask::default(),
402    ///     )
403    ///     .await?;
404    ///
405    /// while let Some(item) = stream.body_mut().next().await {
406    ///     match item? {
407    ///         CheckpointStreamItem::Checkpoint(cp) => {
408    ///             println!("Matched checkpoint {}", cp.sequence_number);
409    ///         }
410    ///         CheckpointStreamItem::Progress {
411    ///             latest_scanned_sequence_number,
412    ///         } => {
413    ///             println!("Scanned up to {latest_scanned_sequence_number}");
414    ///         }
415    ///         _ => {}
416    ///     }
417    /// }
418    /// # Ok(())
419    /// # }
420    /// ```
421    pub async fn stream_checkpoints_filtered(
422        &self,
423        start_sequence_number: impl Into<Option<CheckpointSequenceNumber>>,
424        end_sequence_number: impl Into<Option<CheckpointSequenceNumber>>,
425        transactions_filter: impl Into<Option<grpc_filter::TransactionFilter>>,
426        events_filter: impl Into<Option<grpc_filter::EventFilter>>,
427        progress_interval_ms: impl Into<Option<u32>>,
428        read_mask: impl IntoReadMask<CheckpointResponseReadMask>,
429    ) -> Result<MetadataEnvelope<Pin<Box<dyn Stream<Item = Result<CheckpointStreamItem>> + Send>>>>
430    {
431        self.stream_checkpoints_raw(
432            start_sequence_number.into(),
433            end_sequence_number.into(),
434            transactions_filter.into(),
435            events_filter.into(),
436            true,
437            progress_interval_ms.into(),
438            read_mask.into_read_mask(),
439        )
440        .await
441    }

443    /// Internal helper that builds the stream request and returns the raw
444    /// [`CheckpointStreamItem`] stream.
445    #[expect(clippy::too_many_arguments)]
446    async fn stream_checkpoints_raw(
447        &self,
448        start_sequence_number: Option<CheckpointSequenceNumber>,
449        end_sequence_number: Option<CheckpointSequenceNumber>,
450        transactions_filter: Option<grpc_filter::TransactionFilter>,
451        events_filter: Option<grpc_filter::EventFilter>,
452        filter_checkpoints: bool,
453        progress_interval_ms: Option<u32>,
454        read_mask: CheckpointResponseReadMask,
455    ) -> Result<MetadataEnvelope<Pin<Box<dyn Stream<Item = Result<CheckpointStreamItem>> + Send>>>>
456    {
457        let mut request = StreamCheckpointsRequest::default().with_read_mask(read_mask);
458
459        if let Some(start) = start_sequence_number {
460            request = request.with_start_sequence_number(start);
461        }
462        if let Some(end) = end_sequence_number {
463            request = request.with_end_sequence_number(end);
464        }
465        if let Some(tf) = transactions_filter {
466            request = request.with_transactions_filter(tf);
467        }
468        if let Some(ef) = events_filter {
469            request = request.with_events_filter(ef);
470        }
471        if filter_checkpoints {
472            request = request.with_filter_checkpoints(true);
473        }
474        if let Some(ms) = progress_interval_ms {
475            request = request.with_progress_interval_ms(ms);
476        }
477        if let Some(max_size) = self
478            .max_decoding_message_size()
479            .map(saturating_usize_to_u32)
480        {
481            request = request.with_max_message_size_bytes(max_size);
482        }
483
484        let mut client = self.ledger_service_client();
485        let response = client.stream_checkpoints(request).await?;
486        let (stream, metadata) = MetadataEnvelope::from(response).into_parts();
487
488        Ok(MetadataEnvelope::new(
489            Box::pin(Self::reassemble_checkpoint_data_stream(stream)),
490            metadata,
491        ))
492    }

494    /// Reassemble a stream of checkpoint data chunks into complete checkpoints.
495    ///
496    /// The server sends checkpoint data in multiple messages:
497    /// - `Checkpoint` - Contains the checkpoint summary and contents
498    /// - `Transactions` - Contains executed transactions
499    /// - `Events` - Contains events from transactions
500    /// - `Progress` - Liveness indicator during filtered scanning
501    /// - `EndMarker` - Signals the end of one checkpoint's data
502    ///
503    /// This function buffers the chunks and yields [`CheckpointStreamItem`]
504    /// values: either complete [`CheckpointResponse`] objects when an
505    /// `EndMarker` is received, or [`CheckpointStreamItem::Progress`] when
506    /// a progress message arrives.
507    fn reassemble_checkpoint_data_stream<S, E>(
508        stream: S,
509    ) -> impl Stream<Item = Result<CheckpointStreamItem>>
510    where
511        S: Stream<
512            Item = std::result::Result<iota_grpc_types::v1::ledger_service::CheckpointData, E>,
513        >,
514        E: Into<Error>,
515    {
516        async_stream::try_stream! {
517            futures::pin_mut!(stream);
518
519            // State for accumulating checkpoint data
520            let mut current_sequence_number: Option<CheckpointSequenceNumber> = None;
521            let mut current_summary: Option<checkpoint::CheckpointSummary> = None;
522            let mut current_signature: Option<ProtoValidatorAggregatedSignature> = None;
523            let mut current_contents: Option<checkpoint::CheckpointContents> = None;
524            let mut current_transactions: Vec<ExecutedTransaction> = Vec::new();
525            let mut current_events: Vec<event::Event> = Vec::new();
526
527            while let Some(data) = stream.next().await {
528                let data = data.map_err(|e| e.into())?;
529
530                match data.payload {
531                    Some(checkpoint_data::Payload::Checkpoint(checkpoint)) => {
532                        if checkpoint.sequence_number.is_none() {
533                            Err(TryFromProtoError::missing("checkpoint.sequence_number"))?;
534                        }
535
536                        // Start of new checkpoint - throw error if previous checkpoint was incomplete
537                        if current_sequence_number.is_some() {
538                            Err(Error::Protocol(CheckpointStreamError::IncompleteCheckpoint.into()))?;
539                        }
540                        current_sequence_number = checkpoint.sequence_number;
541
542                        // Store proto summary (optional, no deserialization)
543                        current_summary = checkpoint.summary;
544
545                        // Store proto signature (optional, no deserialization)
546                        current_signature = checkpoint.signature;
547
548                        // Store proto contents (optional, no deserialization)
549                        current_contents = checkpoint.contents;
550
551                        // Reset accumulators for new checkpoint (in case Transactions or Events
552                        // arrived between endmarker and Checkpoint)
553                        current_transactions.clear();
554                        current_events.clear();
555                    }
556
557                    Some(checkpoint_data::Payload::ExecutedTransactions(txs)) => {
558                        if current_sequence_number.is_none() {
559                            Err(Error::Protocol(CheckpointStreamError::DataBeforeHeader { data_kind: "transactions" }.into()))?;
560                        }
561
562                        // Accumulate proto transactions (no deserialization)
563                        current_transactions.extend(txs.executed_transactions.into_iter());
564                    }
565
566                    Some(checkpoint_data::Payload::Events(events)) => {
567                        if current_sequence_number.is_none() {
568                            Err(Error::Protocol(CheckpointStreamError::DataBeforeHeader { data_kind: "events" }.into()))?;
569                        }
570
571                        // Accumulate proto events (no deserialization)
572                        current_events.extend(events.events);
573                    }
574
575                    Some(checkpoint_data::Payload::EndMarker(marker)) => {
576                        // End of current checkpoint - assemble the result and yield it
577                         let sequence_number = current_sequence_number
578                        .take()
579                        .ok_or_else(|| -> Error { Error::Protocol(CheckpointStreamError::DataBeforeHeader { data_kind: "end marker" }.into()) })?;
580
581                        let marker_sequence_number = marker.sequence_number
582                        .ok_or_else(|| -> Error { TryFromProtoError::missing("end_marker.sequence_number").into() })?;
583
584                        if marker_sequence_number != sequence_number {
585                            Err(Error::Protocol(CheckpointStreamError::SequenceNumberMismatch {
586                                expected: sequence_number,
587                                actual: marker_sequence_number,
588                            }.into()))?;
589                        }
590
591                        let response = CheckpointResponse {
592                            sequence_number,
593                            summary: current_summary.take(),
594                            signature: current_signature.take(),
595                            contents: current_contents.take(),
596                            executed_transactions: std::mem::take(&mut current_transactions),
597                            events: std::mem::take(&mut current_events),
598                        };
599
600                        yield CheckpointStreamItem::Checkpoint(Box::new(response));
601                    }
602
603                    Some(checkpoint_data::Payload::Progress(progress)) => {
604                        yield CheckpointStreamItem::Progress {
605                            latest_scanned_sequence_number: progress.latest_scanned_sequence_number,
606                        };
607                    }
608
609                    None => {
610                        // Empty payload - skip
611                        continue;
612                    }
613
614                    Some(_) => {
615                        // Unknown payload type
616                        Err(Error::Protocol(CheckpointStreamError::UnknownPayload.into()))?;
617                    }
618                }
619            }
620
621            // Check if stream ended with incomplete checkpoint data
622            if let Some(sequence_number) = current_sequence_number {
623                Err(Error::Protocol(CheckpointStreamError::IncompleteStream { sequence_number }.into()))?;
