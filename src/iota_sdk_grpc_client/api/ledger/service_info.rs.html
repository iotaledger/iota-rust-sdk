4//! High-level API for service info queries.
5
6use iota_grpc_types::{
7    read_mask_fields::ServiceInfoReadMask,
8    v1::ledger_service::{
9        GetServiceInfoRequest, GetServiceInfoResponse, ledger_service_client::LedgerServiceClient,
10    },
11};
12
13use crate::{
14    GrpcClient, InterceptedChannel,
15    api::{GrpcResult, MetadataEnvelope, define_query},
16};
17
18define_query! {
19    /// Query for [`GrpcClient::service_info`]. Await it to send the request.
20    pub struct GetServiceInfoQuery {
21        service_client: LedgerServiceClient<InterceptedChannel>,
22    }
23    output: GrpcResult<MetadataEnvelope<GetServiceInfoResponse>>;
24    read_mask: ServiceInfoReadMask;
25}
26
27impl GetServiceInfoQuery {
28    fn into_request(
29        self,
30    ) -> (
31        LedgerServiceClient<InterceptedChannel>,
32        GetServiceInfoRequest,
33    ) {
34        let request = GetServiceInfoRequest::default().with_read_mask(self.read_mask);
35        (self.service_client, request)
36    }
37
38    async fn send(self) -> GrpcResult<MetadataEnvelope<GetServiceInfoResponse>> {
39        let (mut service_client, request) = self.into_request();
40        let response = service_client.get_service_info(request).await?;
41
42        Ok(MetadataEnvelope::from(response))
43    }
44}
45
46impl GrpcClient {
47    /// Get service info from the node.
48    ///
49    /// Returns the [`GetServiceInfoResponse`] proto type with fields populated
50    /// according to the read mask. Without
51    /// [`read_mask`](GetServiceInfoQuery::read_mask), the default mask is
52    /// used. Pass a
53    /// [`ServiceInfoReadMask`](iota_grpc_types::read_mask_fields::ServiceInfoReadMask)
54    /// built from a
55    /// [`ServiceInfoField`](iota_grpc_types::read_mask_fields::ServiceInfoField)
56    /// or any slice/array/vec of fields to choose the returned fields.
57    ///
58    /// # Example
59    ///
60    /// ```no_run
61    /// # use iota_sdk_grpc_client::GrpcClient;
62    /// # use iota_sdk_grpc_client::read_mask_fields::{ServiceInfoField, ServiceInfoReadMask};
63    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
64    /// let client = GrpcClient::new_localnet()?;
65    ///
66    /// let info = client.service_info().await?;
67    /// println!("Chain ID: {:?}", info.body().chain_id);
68    /// println!("Epoch: {:?}", info.body().epoch);
69    ///
70    /// // With a custom mask.
71    /// let info = client
72    ///     .service_info()
73    ///     .read_mask(ServiceInfoReadMask::from([
74    ///         ServiceInfoField::CHAIN_ID,
75    ///         ServiceInfoField::EPOCH,
76    ///     ]))
77    ///     .await?;
78    /// # Ok(())
79    /// # }
80    /// ```
81    pub fn service_info(&self) -> GetServiceInfoQuery {
82        GetServiceInfoQuery {
83            service_client: self.ledger_service_client(),
84            read_mask: ServiceInfoReadMask::default(),
85        }
86    }
87}

89#[cfg(test)]
90mod tests {
91    use iota_grpc_types::read_mask_fields::{ServiceInfoField, ServiceInfoReadMask};
92
93    use crate::GrpcClient;
94
95    #[tokio::test]
96    async fn read_mask_replaces_the_default_mask() {
97        let client = GrpcClient::new("http://localhost").unwrap();
98        let query = client.service_info();
99        assert_eq!(
100            query.read_mask.as_str(),
101            ServiceInfoReadMask::default().as_str()
102        );
103
104        let query = query.read_mask(ServiceInfoField::CHAIN_ID);
105        assert_eq!(
106            query.read_mask.as_str(),
107            ServiceInfoReadMask::from(ServiceInfoField::CHAIN_ID).as_str()
108        );
109    }
110
111    #[tokio::test]
112    async fn the_request_carries_the_mask() {
113        let client = GrpcClient::new("http://localhost").unwrap();
114        let (_, request) = client
115            .service_info()
116            .read_mask(ServiceInfoField::CHAIN_ID)
117            .into_request();
118        assert_eq!(
119            request.read_mask,
120            Some(ServiceInfoReadMask::from(ServiceInfoField::CHAIN_ID).into())
121        );
122    }
123}
