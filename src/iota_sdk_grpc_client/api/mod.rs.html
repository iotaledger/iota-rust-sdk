27/// An item from a checkpoint data stream.
28///
29/// When `filter_checkpoints` is enabled, the server may skip checkpoints that
30/// don't match the provided filters. In that case, `Progress` items are sent
31/// periodically to indicate liveness and the current scan position. When
32/// `filter_checkpoints` is disabled, only `Checkpoint` items are produced.
33///
34/// For liveness detection with `filter_checkpoints`, wrap `stream.next()` in
35/// `tokio::time::timeout()` — if neither a `Checkpoint` nor a `Progress`
36/// arrives within your chosen duration plus some buffer for connection latency,
37/// the connection is likely dead.
38#[derive(Clone, Debug)]
39#[non_exhaustive]
40pub enum CheckpointStreamItem {
41    /// A complete checkpoint with its transactions and events.
42    Checkpoint(Box<CheckpointResponse>),
43    /// A progress indicator sent during filtered scanning.
44    /// Contains the sequence number of the latest scanned checkpoint.
45    Progress {
46        latest_scanned_sequence_number: CheckpointSequenceNumber,
47    },
48}
49
50impl CheckpointStreamItem {
51    /// Returns the contained checkpoint, or `None` if this is a progress
52    /// message.
53    pub fn into_checkpoint(self) -> Option<CheckpointResponse> {
54        match self {
55            Self::Checkpoint(c) => Some(*c),
56            Self::Progress { .. } => None,
57        }
58    }
59
60    /// Returns the progress sequence number, or `None` if this is a
61    /// checkpoint.
62    pub fn into_progress(self) -> Option<CheckpointSequenceNumber> {
63        match self {
64            Self::Checkpoint(_) => None,
65            Self::Progress {
66                latest_scanned_sequence_number,
67            } => Some(latest_scanned_sequence_number),
68        }
69    }
70
71    /// Returns `true` if this is a checkpoint item.
72    pub fn is_checkpoint(&self) -> bool {
73        matches!(self, Self::Checkpoint(_))
74    }
75
76    /// Returns `true` if this is a progress item.
77    pub fn is_progress(&self) -> bool {
78        matches!(self, Self::Progress { .. })
79    }
80}

82/// Response for a checkpoint query.
83///
84/// Contains checkpoint summary, signature, contents, transactions, and events.
85/// Fields are proto types that can be accessed directly or converted to SDK
86/// types using their conversion methods (e.g.,
87/// `response.summary()?.summary()?`, `response.contents()?.contents()?`).
88#[derive(Clone, Debug)]
89pub struct CheckpointResponse {
90    /// The checkpoint sequence number.
91    pub sequence_number: CheckpointSequenceNumber,
92    /// Proto checkpoint summary. Use `response.summary()?.summary()` to convert
93    /// to SDK type.
94    pub summary: Option<iota_grpc_types::v1::checkpoint::CheckpointSummary>,
95    /// Proto validator signature. Use `response.signature()?.signature()` to
96    /// convert to SDK type.
97    pub signature: Option<iota_grpc_types::v1::signatures::ValidatorAggregatedSignature>,
98    /// Proto checkpoint contents. Use `response.contents()?.contents()` to
99    /// convert to SDK type.
100    pub contents: Option<iota_grpc_types::v1::checkpoint::CheckpointContents>,
101    /// Proto executed transactions. Use methods like `tx.effects()?`,
102    /// `tx.transaction()?`, etc.
103    pub executed_transactions: Vec<iota_grpc_types::v1::transaction::ExecutedTransaction>,
104    /// Proto events. Use `event.try_into()` or `event.events()` to convert to
105    /// SDK types.
106    pub events: Vec<iota_grpc_types::v1::event::Event>,
107}
108
109impl CheckpointResponse {
110    /// Get the checkpoint sequence number.
111    ///
112    /// Always available regardless of the read mask.
113    pub fn sequence_number(&self) -> CheckpointSequenceNumber {
114        self.sequence_number
115    }
116
117    /// Get the checkpoint summary.
118    ///
119    /// Returns the proto
120    /// [`CheckpointSummary`](iota_grpc_types::v1::checkpoint::CheckpointSummary)
121    /// which provides:
122    /// - [`digest()`](iota_grpc_types::v1::checkpoint::CheckpointSummary::digest) — the summary digest
123    /// - [`summary()`](iota_grpc_types::v1::checkpoint::CheckpointSummary::summary) — the deserialized SDK `CheckpointSummary`
124    ///
125    /// **Read mask:** `"checkpoint.summary"` (see
126    /// [`CHECKPOINT_RESPONSE_SUMMARY`])
127    pub fn summary(&self) -> Result<&iota_grpc_types::v1::checkpoint::CheckpointSummary> {
128        self.summary
129            .as_ref()
130            .ok_or_else(|| TryFromProtoError::missing("summary").into())
131    }
132
133    /// Build a [`SignedCheckpointSummary`](iota_types::SignedCheckpointSummary)
134    /// from the response.
135    ///
136    /// Requires the checkpoint summary and signature to be present.
137    ///
138    /// **Read mask:** see [`CHECKPOINT_RESPONSE_SIGNED_SUMMARY`]
139    pub fn signed_summary(&self) -> Result<iota_types::SignedCheckpointSummary> {
140        Ok(iota_types::SignedCheckpointSummary {
141            checkpoint: self.summary()?.summary()?,
142            signature: self.signature()?.signature()?,
143        })
144    }
145
146    /// Get the validator aggregated signature for the checkpoint.
147    ///
148    /// **Read mask:** `"checkpoint.signature"` (see
149    /// [`CHECKPOINT_RESPONSE_SIGNATURE`])
150    pub fn signature(
151        &self,
152    ) -> Result<&iota_grpc_types::v1::signatures::ValidatorAggregatedSignature> {
153        self.signature
154            .as_ref()
155            .ok_or_else(|| TryFromProtoError::missing("signature").into())
156    }
157
158    /// Get the checkpoint contents.
159    ///
160    /// Returns the proto
161    /// [`CheckpointContents`](iota_grpc_types::v1::checkpoint::CheckpointContents)
162    /// which provides:
163    /// - [`digest()`](iota_grpc_types::v1::checkpoint::CheckpointContents::digest) — the contents digest
164    /// - [`contents()`](iota_grpc_types::v1::checkpoint::CheckpointContents::contents) — the deserialized SDK `CheckpointContents`
165    ///
166    /// **Read mask:** `"checkpoint.contents"` (see
167    /// [`CHECKPOINT_RESPONSE_CONTENTS`])
168    pub fn contents(&self) -> Result<&iota_grpc_types::v1::checkpoint::CheckpointContents> {
169        self.contents
170            .as_ref()
171            .ok_or_else(|| TryFromProtoError::missing("contents").into())
172    }
173
174    /// Get the executed transactions in this checkpoint.
175    ///
176    /// Returns proto
177    /// [`ExecutedTransaction`](iota_grpc_types::v1::transaction::ExecutedTransaction)
178    /// values. Which sub-fields are populated depends on the read mask; use
179    /// paths like `"transactions.effects"` or `"transactions.transaction"`.
180    ///
181    /// **Read mask:** `"transactions"` for all sub-fields (see
182    /// [`CHECKPOINT_RESPONSE_EXECUTED_TRANSACTIONS`])
183    pub fn executed_transactions(
184        &self,
185    ) -> &Vec<iota_grpc_types::v1::transaction::ExecutedTransaction> {
186        &self.executed_transactions
187    }
188
189    /// Get the top-level events for this checkpoint.
190    ///
191    /// Returns proto [`Event`](iota_grpc_types::v1::event::Event) values.
192    /// Which sub-fields are populated depends on the read mask; use paths like
193    /// `"events.bcs"` or `"events.event_type"`.
194    ///
195    /// **Read mask:** `"events"` for all sub-fields (see
196    /// [`CHECKPOINT_RESPONSE_EVENTS`])
197    pub fn events(&self) -> &Vec<iota_grpc_types::v1::event::Event> {
198        &self.events
199    }
200
201    /// Build a full
202    /// [`CheckpointData`](iota_types::checkpoint::CheckpointData)
203    /// from the response.
204    ///
205    /// Requires the checkpoint summary, signature, contents, and all
206    /// transaction data (transaction, signatures, effects, events,
207    /// input_objects, output_objects) to be present in the response.
208    ///
209    /// **Read mask:** see [`CHECKPOINT_RESPONSE_CHECKPOINT_DATA`]
210    ///
211    /// # Example
212    ///
213    /// ```no_run
214    /// # use iota_sdk_grpc_client::Client;
215    /// # use iota_sdk_grpc_client::read_mask_fields::CheckpointResponseReadMask;
216    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
217    /// let client = Client::new_localnet()?;
218    /// let cp = client
219    ///     .get_checkpoint_latest(None, None, CheckpointResponseReadMask::default())
220    ///     .await?;
221    /// let data = cp.body().checkpoint_data()?;
222    /// # Ok(())
223    /// # }
224    /// ```
225    pub fn checkpoint_data(&self) -> Result<iota_types::checkpoint::CheckpointData> {
226        Ok(iota_types::checkpoint::CheckpointData {
227            checkpoint_contents: self.contents()?.contents()?,
228            checkpoint_summary: iota_types::SignedCheckpointSummary {
229                checkpoint: self.summary()?.summary()?,
230                signature: self.signature()?.signature()?,
231            },
232            transactions: self
233                .executed_transactions()
234                .iter()
235                .map(TryInto::try_into)
236                .collect::<std::result::Result<Vec<_>, _>>()?,
237        })
238    }
239}
