4//! High-level API for listing coins owned by an address.
5//!
6//! Wraps [`GrpcClient::owned_objects`](crate::GrpcClient::owned_objects) with a coin type filter
7//! and converts each returned proto `Object` into an [`iota_types::framework::Coin`].
8
9use iota_grpc_types::{
10    read_mask_fields::OwnedObjectReadMask,
11    v1::{
12        state_service::{ListOwnedObjectsRequest, state_service_client::StateServiceClient},
13        types::Address as ProtoAddress,
14    },
15};
16use iota_types::{Address, Identifier, StructTag, framework::Coin};
17
18use crate::{
19    GrpcClient, InterceptedChannel,
20    api::{GrpcError, GrpcResult, TryFromProtoError, define_list_query},
21};
22
23define_list_query! {
24    /// Builder for listing coins owned by an address.
25    ///
26    /// Created by [`GrpcClient::coins`]. Await directly for a single page (with access to
27    /// `next_page_token`), or call [`.collect(limit)`](Self::collect) to auto-paginate.
28    pub struct GetCoinsQuery {
29        service_client: StateServiceClient<InterceptedChannel>,
30        request: ListOwnedObjectsRequest,
31        item: Coin,
32        rpc_method: list_owned_objects,
33        items_field: objects,
34        map_item: object_to_coin,
35    }
36}
37
38fn object_to_coin(obj: &iota_grpc_types::v1::object::Object) -> GrpcResult<Coin> {
39    let sdk_obj = obj.object()?;
40    Coin::try_from_object(&sdk_obj)
41        .map_err(|e| GrpcError::from(TryFromProtoError::invalid("coin", e)))
42}
43
44impl GetCoinsQuery {
45    /// Filter by coin type, the inner type `T` of `Coin<T>`. Without it, lists all coin types (with
46    /// type `0x2::coin::Coin`).
47    pub fn coin_type(mut self, coin_type: StructTag) -> Self {
48        self.base_request.object_type = Some(coin_object_type(Some(coin_type)));
49        self
50    }
51}
52
53fn coin_object_type(coin_type: Option<StructTag>) -> String {
54    // Without a coin type, query for the generic `0x2::coin::Coin` (no type params); the server
55    // returns all `Coin<T>` objects regardless of `T`.
56    coin_type
57        .map(StructTag::new_coin)
58        .unwrap_or_else(|| {
59            StructTag::new(
60                Address::FRAMEWORK,
61                Identifier::COIN_MODULE,
62                Identifier::COIN,
63                Vec::new(),
64            )
65        })
66        .to_string()
67}
68
69impl GrpcClient {
70    /// List coins owned by an address.
71    ///
72    /// Returns a query builder. Await it directly for a single page (with access to
73    /// `next_page_token`), or call `.collect(limit)` to auto-paginate through all results. Filter
74    /// by coin type with [`coin_type`](GetCoinsQuery::coin_type) and page with
75    /// [`page_size`](GetCoinsQuery::page_size) and [`page_token`](GetCoinsQuery::page_token).
76    ///
77    /// Each returned [`Coin`] is converted from the underlying proto `Object`.
78    ///
79    /// # Parameters
80    ///
81    /// - `owner` - The address that owns the coins.
82    ///
83    /// # Examples
84    ///
85    /// Single page:
86    /// ```no_run
87    /// # use iota_sdk_grpc_client::GrpcClient;
88    /// # use iota_types::Address;
89    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
90    /// let client = GrpcClient::new_localnet()?;
91    /// let owner: Address = "0x1".parse()?;
92    ///
93    /// let page = client.coins(owner).await?;
94    /// for coin in &page.body().items {
95    ///     println!("Coin {}: {}", coin.id(), coin.balance());
96    /// }
97    /// # Ok(())
98    /// # }
99    /// ```
100    ///
101    /// Auto-paginate:
102    /// ```no_run
103    /// # use iota_sdk_grpc_client::GrpcClient;
104    /// # use iota_types::Address;
105    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
106    /// let client = GrpcClient::new_localnet()?;
107    /// let owner: Address = "0x1".parse()?;
108    ///
109    /// let all = client.coins(owner).page_size(50).collect(500).await?;
110    /// for coin in all.body() {
111    ///     println!("Coin {}: {}", coin.id(), coin.balance());
112    /// }
113    /// # Ok(())
114    /// # }
115    /// ```
116    pub fn coins(&self, owner: Address) -> GetCoinsQuery {
117        let base_request = ListOwnedObjectsRequest::default()
118            .with_owner(ProtoAddress::default().with_address(Vec::from(owner)))
119            .with_object_type(coin_object_type(None))
120            .with_read_mask(OwnedObjectReadMask::default());
121
122        GetCoinsQuery::new(
123            self.state_service_client(),
124            base_request,
125            self.max_decoding_message_size(),
126        )
127    }
128}

130#[cfg(test)]
131mod tests {
132    use iota_types::{Address, StructTag};
133
134    use crate::GrpcClient;
135
136    #[tokio::test]
137    async fn coins_without_a_coin_type_lists_every_coin_type() {
138        let client = GrpcClient::new("http://localhost").unwrap();
139        let query = client.coins(Address::ZERO);
140        assert_eq!(
141            query.base_request.object_type.as_deref(),
142            Some("0x2::coin::Coin")
143        );
144    }
145
146    #[tokio::test]
147    async fn coin_type_sets_the_coin_filter() {
148        let client = GrpcClient::new("http://localhost").unwrap();
149        let query = client
150            .coins(Address::ZERO)
151            .coin_type("0x2::iota::IOTA".parse::<StructTag>().unwrap());
152        assert_eq!(
153            query.base_request.object_type.as_deref(),
154            Some("0x2::coin::Coin<0x2::iota::IOTA>")
155        );
156    }
157
158    #[tokio::test]
159    async fn the_request_carries_the_owner_and_the_coin_type() {
160        let client = GrpcClient::new("http://localhost").unwrap();
161        let owner: Address = "0x5".parse().unwrap();
162        let (_, request) = client
163            .coins(owner)
164            .coin_type("0x2::iota::IOTA".parse::<StructTag>().unwrap())
165            .into_request();
166        assert_eq!(
167            request.owner.map(|o| o.address).as_deref(),
168            Some(Vec::from(owner).as_slice())
169        );
170        assert_eq!(
171            request.object_type.as_deref(),
172            Some("0x2::coin::Coin<0x2::iota::IOTA>")
173        );
174    }
175}
