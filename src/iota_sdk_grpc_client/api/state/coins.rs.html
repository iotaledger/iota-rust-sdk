4//! High-level API for listing coins owned by an address.
5//!
6//! Wraps [`Client::list_owned_objects`](crate::Client::list_owned_objects)
7//! with a coin type filter and converts each returned proto `Object` into an
8//! [`iota_types::framework::Coin`].
9
10use iota_grpc_types::{
11    read_mask_fields::OwnedObjectReadMask,
12    v1::{
13        state_service::{ListOwnedObjectsRequest, state_service_client::StateServiceClient},
14        types::Address as ProtoAddress,
15    },
16};
17use iota_types::{Address, Identifier, StructTag, framework::Coin};
18
19use crate::{
20    Client, InterceptedChannel,
21    api::{Error, Result, TryFromProtoError, define_list_query},
22};
23
24define_list_query! {
25    /// Builder for listing coins owned by an address.
26    ///
27    /// Created by [`Client::get_coins`]. Await directly for a single page
28    /// (with access to `next_page_token`), or call
29    /// [`.collect(limit)`](Self::collect) to auto-paginate.
30    pub struct GetCoinsQuery {
31        service_client: StateServiceClient<InterceptedChannel>,
32        request: ListOwnedObjectsRequest,
33        item: Coin,
34        rpc_method: list_owned_objects,
35        items_field: objects,
36        map_item: object_to_coin,
37    }
38}
39
40fn object_to_coin(obj: &iota_grpc_types::v1::object::Object) -> Result<Coin> {
41    let sdk_obj = obj.object()?;
42    Coin::try_from_object(&sdk_obj).map_err(|e| Error::from(TryFromProtoError::invalid("coin", e)))
43}
44
45impl Client {
46    /// List coins owned by an address.
47    ///
48    /// Returns a query builder. Await it directly for a single page (with
49    /// access to `next_page_token`), or call `.collect(limit)` to
50    /// auto-paginate through all results.
51    ///
52    /// Each returned [`Coin`] is converted from the underlying proto
53    /// `Object`.
54    ///
55    /// # Parameters
56    ///
57    /// - `owner` - The address that owns the coins.
58    /// - `coin_type` - Optional coin type filter as a [`StructTag`]. The value
59    ///   must be the inner type `T` of `Coin<T>`. If `None`, lists all coin
60    ///   types (with type `0x2::coin::Coin`).
61    /// - `page_size` - Optional maximum number of coins per page.
62    /// - `page_token` - Optional continuation token from a previous page.
63    ///
64    /// # Examples
65    ///
66    /// Single page:
67    /// ```no_run
68    /// # use iota_sdk_grpc_client::Client;
69    /// # use iota_types::Address;
70    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
71    /// let client = Client::new_localnet()?;
72    /// let owner: Address = "0x1".parse()?;
73    ///
74    /// let page = client.get_coins(owner, None, None, None).await?;
75    /// for coin in &page.body().items {
76    ///     println!("Coin {}: {}", coin.id(), coin.balance());
77    /// }
78    /// # Ok(())
79    /// # }
80    /// ```
81    ///
82    /// Auto-paginate:
83    /// ```no_run
84    /// # use iota_sdk_grpc_client::Client;
85    /// # use iota_types::Address;
86    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
87    /// let client = Client::new_localnet()?;
88    /// let owner: Address = "0x1".parse()?;
89    ///
90    /// let all = client
91    ///     .get_coins(owner, None, Some(50), None)
92    ///     .collect(Some(500))
93    ///     .await?;
94    /// for coin in all.body() {
95    ///     println!("Coin {}: {}", coin.id(), coin.balance());
96    /// }
97    /// # Ok(())
98    /// # }
99    /// ```
100    pub fn get_coins(
101        &self,
102        owner: Address,
103        coin_type: impl Into<Option<StructTag>>,
104        page_size: impl Into<Option<u32>>,
105        page_token: impl Into<Option<prost::bytes::Bytes>>,
106    ) -> GetCoinsQuery {
107        self.get_coins_internal(owner, coin_type.into(), page_size.into(), page_token.into())
108    }
109
110    fn get_coins_internal(
111        &self,
112        owner: Address,
113        coin_type: Option<StructTag>,
114        page_size: Option<u32>,
115        page_token: Option<prost::bytes::Bytes>,
116    ) -> GetCoinsQuery {
117        // When no specific coin type is provided, query for the generic
118        // `0x2::coin::Coin` (no type params); the server returns all
119        // `Coin<T>` objects regardless of `T`.
120        let coin_type = coin_type.map(StructTag::new_coin).unwrap_or_else(|| {
121            StructTag::new(
122                Address::FRAMEWORK,
123                Identifier::COIN_MODULE,
124                Identifier::COIN,
125                Vec::new(),
126            )
127        });
128
129        let base_request = ListOwnedObjectsRequest::default()
130            .with_owner(ProtoAddress::default().with_address(Vec::from(owner)))
131            .with_object_type(coin_type.to_string())
132            .with_read_mask(OwnedObjectReadMask::default());
133
134        GetCoinsQuery::new(
135            self.state_service_client(),
136            base_request,
137            self.max_decoding_message_size(),
138            page_size,
139            page_token,
140        )
141    }
142}
