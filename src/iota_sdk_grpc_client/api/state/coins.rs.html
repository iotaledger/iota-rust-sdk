4//! High-level API for listing coins owned by an address.
5//!
6//! Wraps [`GrpcClient::owned_objects`](crate::GrpcClient::owned_objects)
7//! with a coin type filter and converts each returned proto `Object` into an
8//! [`iota_types::framework::Coin`].
9
10use iota_grpc_types::{
11    read_mask_fields::OwnedObjectReadMask,
12    v1::{
13        state_service::{ListOwnedObjectsRequest, state_service_client::StateServiceClient},
14        types::Address as ProtoAddress,
15    },
16};
17use iota_types::{Address, Identifier, StructTag, framework::Coin};
18
19use crate::{
20    GrpcClient, InterceptedChannel,
21    api::{GrpcError, GrpcResult, TryFromProtoError, define_list_query},
22};
23
24define_list_query! {
25    /// Builder for listing coins owned by an address.
26    ///
27    /// Created by [`GrpcClient::coins`]. Await directly for a single page
28    /// (with access to `next_page_token`), or call
29    /// [`.collect(limit)`](Self::collect) to auto-paginate.
30    pub struct GetCoinsQuery {
31        service_client: StateServiceClient<InterceptedChannel>,
32        request: ListOwnedObjectsRequest,
33        item: Coin,
34        rpc_method: list_owned_objects,
35        items_field: objects,
36        map_item: object_to_coin,
37    }
38}
39
40fn object_to_coin(obj: &iota_grpc_types::v1::object::Object) -> GrpcResult<Coin> {
41    let sdk_obj = obj.object()?;
42    Coin::try_from_object(&sdk_obj)
43        .map_err(|e| GrpcError::from(TryFromProtoError::invalid("coin", e)))
44}
45
46impl GetCoinsQuery {
47    /// Filter by coin type, the inner type `T` of `Coin<T>`. Without it, lists
48    /// all coin types (with type `0x2::coin::Coin`).
49    pub fn coin_type(mut self, coin_type: StructTag) -> Self {
50        self.base_request.object_type = Some(coin_object_type(Some(coin_type)));
51        self
52    }
53}
54
55fn coin_object_type(coin_type: Option<StructTag>) -> String {
56    // Without a coin type, query for the generic `0x2::coin::Coin` (no type
57    // params); the server returns all `Coin<T>` objects regardless of `T`.
58    coin_type
59        .map(StructTag::new_coin)
60        .unwrap_or_else(|| {
61            StructTag::new(
62                Address::FRAMEWORK,
63                Identifier::COIN_MODULE,
64                Identifier::COIN,
65                Vec::new(),
66            )
67        })
68        .to_string()
69}
70
71impl GrpcClient {
72    /// List coins owned by an address.
73    ///
74    /// Returns a query builder. Await it directly for a single page (with
75    /// access to `next_page_token`), or call `.collect(limit)` to
76    /// auto-paginate through all results. Filter by coin type with
77    /// [`coin_type`](GetCoinsQuery::coin_type) and page with
78    /// [`page_size`](GetCoinsQuery::page_size) and
79    /// [`page_token`](GetCoinsQuery::page_token).
80    ///
81    /// Each returned [`Coin`] is converted from the underlying proto
82    /// `Object`.
83    ///
84    /// # Parameters
85    ///
86    /// - `owner` - The address that owns the coins.
87    ///
88    /// # Examples
89    ///
90    /// Single page:
91    /// ```no_run
92    /// # use iota_sdk_grpc_client::GrpcClient;
93    /// # use iota_types::Address;
94    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
95    /// let client = GrpcClient::new_localnet()?;
96    /// let owner: Address = "0x1".parse()?;
97    ///
98    /// let page = client.coins(owner).await?;
99    /// for coin in &page.body().items {
100    ///     println!("Coin {}: {}", coin.id(), coin.balance());
101    /// }
102    /// # Ok(())
103    /// # }
104    /// ```
105    ///
106    /// Auto-paginate:
107    /// ```no_run
108    /// # use iota_sdk_grpc_client::GrpcClient;
109    /// # use iota_types::Address;
110    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
111    /// let client = GrpcClient::new_localnet()?;
112    /// let owner: Address = "0x1".parse()?;
113    ///
114    /// let all = client.coins(owner).page_size(50).collect(500).await?;
115    /// for coin in all.body() {
116    ///     println!("Coin {}: {}", coin.id(), coin.balance());
117    /// }
118    /// # Ok(())
119    /// # }
120    /// ```
121    pub fn coins(&self, owner: Address) -> GetCoinsQuery {
122        let base_request = ListOwnedObjectsRequest::default()
123            .with_owner(ProtoAddress::default().with_address(Vec::from(owner)))
124            .with_object_type(coin_object_type(None))
125            .with_read_mask(OwnedObjectReadMask::default());
126
127        GetCoinsQuery::new(
128            self.state_service_client(),
129            base_request,
130            self.max_decoding_message_size(),
131        )
132    }
133}

135#[cfg(test)]
136mod tests {
137    use iota_types::{Address, StructTag};
138
139    use crate::GrpcClient;
140
141    #[tokio::test]
142    async fn coins_without_a_coin_type_lists_every_coin_type() {
143        let client = GrpcClient::new("http://localhost").unwrap();
144        let query = client.coins(Address::ZERO);
145        assert_eq!(
146            query.base_request.object_type.as_deref(),
147            Some("0x2::coin::Coin")
148        );
149    }
150
151    #[tokio::test]
152    async fn coin_type_sets_the_coin_filter() {
153        let client = GrpcClient::new("http://localhost").unwrap();
154        let query = client
155            .coins(Address::ZERO)
156            .coin_type("0x2::iota::IOTA".parse::<StructTag>().unwrap());
157        assert_eq!(
158            query.base_request.object_type.as_deref(),
159            Some("0x2::coin::Coin<0x2::iota::IOTA>")
160        );
161    }
162
163    #[tokio::test]
164    async fn the_request_carries_the_owner_and_the_coin_type() {
165        let client = GrpcClient::new("http://localhost").unwrap();
166        let owner: Address = "0x5".parse().unwrap();
167        let (_, request) = client
168            .coins(owner)
169            .coin_type("0x2::iota::IOTA".parse::<StructTag>().unwrap())
170            .into_request();
171        assert_eq!(
172            request.owner.map(|o| o.address).as_deref(),
173            Some(Vec::from(owner).as_slice())
174        );
175        assert_eq!(
176            request.object_type.as_deref(),
177            Some("0x2::coin::Coin<0x2::iota::IOTA>")
178        );
179    }
180}
