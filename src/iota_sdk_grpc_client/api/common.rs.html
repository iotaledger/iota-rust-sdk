27/// Errors that can occur during gRPC client API operations.
28#[derive(Debug, thiserror::Error)]
29#[non_exhaustive]
30pub enum GrpcError {
31    /// Error converting proto types to SDK types.
32    #[error("proto conversion error: {0}")]
33    ProtoConversion(#[from] Box<TryFromProtoError>),
34
35    /// Per-item error returned by the server (preserves code, message,
36    /// details).
37    #[error("server error (code {code}): {msg}", code = .0.code, msg = .0.message)]
38    Server(RpcStatus),
39
40    /// GrpcClient-side protocol error (e.g. checkpoint stream reassembly).
41    #[error("protocol error: {0}")]
42    Protocol(ProtocolError),
43
44    /// Error converting signatures to proto format.
45    #[error("signature conversion error: {0}")]
46    Signature(GrpcConversionError),
47
48    /// The caller passed an empty request (e.g. no object IDs or digests).
49    #[error("empty request: at least one item must be provided")]
50    EmptyRequest,
51
52    /// The server stream ended unexpectedly while `has_next` was still true.
53    #[error("stream ended unexpectedly: server indicated more results with has_next=true")]
54    UnexpectedEndOfStream,
55
56    /// gRPC transport or protocol error.
57    #[error("grpc error: {0}")]
58    Grpc(Box<tonic::Status>),
59}
60
61impl GrpcError {
62    /// Returns `true` if the error carries a `NOT_FOUND` status, whether the
63    /// server reported it for the call or for a single item of a batched
64    /// request.
65    ///
66    /// **Warning:** do not test the outer error of a batched read to decide
67    /// that an item is absent. Those calls report an absent item against the
68    /// request that asked for it, so a `NOT_FOUND` for the call means the
69    /// endpoint is wrong; treating it as a missing item turns a misconfigured
70    /// endpoint into a normal "not there" answer. Test the per-item result
71    /// instead. Calls returning a single response, such as fetching a
72    /// checkpoint, do report absence at the call level.
73    pub fn is_not_found(&self) -> bool {
74        match self {
75            GrpcError::Server(status) => status.code == i32::from(tonic::Code::NotFound),
76            GrpcError::Grpc(status) => status.code() == tonic::Code::NotFound,
77            _ => false,
78        }
79    }
80}

82impl From<TryFromProtoError> for GrpcError {
83    fn from(err: TryFromProtoError) -> Self {
84        GrpcError::ProtoConversion(Box::new(err))
85    }

88impl From<tonic::Status> for GrpcError {
89    fn from(status: tonic::Status) -> Self {
90        GrpcError::Grpc(Box::new(status))
91    }

94impl From<GrpcError> for tonic::Status {
95    fn from(err: GrpcError) -> Self {
96        match err {
97            GrpcError::ProtoConversion(e) => {
98                tonic::Status::internal(format!("proto conversion error: {e}"))
99            }
100            GrpcError::Server(status) => status.to_tonic_status(),
101            GrpcError::Protocol(err) => tonic::Status::internal(format!("protocol error: {err}")),
102            GrpcError::Signature(err) => {
103                tonic::Status::internal(format!("signature conversion error: {err}"))
104            }
105            GrpcError::EmptyRequest => {
106                tonic::Status::invalid_argument("empty request: at least one item must be provided")
107            }
108            GrpcError::UnexpectedEndOfStream => {
109                tonic::Status::internal("stream ended unexpectedly: has_next was true")
110            }
111            GrpcError::Grpc(status) => *status,
112        }
113    }
114}

116/// Protocol-level errors encountered while processing gRPC responses.
117#[derive(Debug, thiserror::Error)]
118#[non_exhaustive]
119pub enum ProtocolError {
120    /// Server returned an unrecognized proto oneof variant.
121    #[error("unknown {0} variant")]
122    UnknownVariant(&'static str),
123
124    /// A required response field was unexpectedly empty.
125    #[error("empty response field: {0}")]
126    EmptyResponseField(&'static str),
127
128    /// Error during checkpoint data stream reassembly.
129    #[error("checkpoint stream error: {0}")]
130    CheckpointStream(#[from] CheckpointStreamError),
131
132    /// A batched read returned a number of results that does not match the
133    /// number of requested items.
134    #[error("expected {expected} results, got {actual}")]
135    UnexpectedResultCount { expected: usize, actual: usize },
136
137    /// `objects()` answered a position with a different object than the one
138    /// requested there.
139    #[error("requested object {expected} at position {position}, but got {actual}")]
140    UnexpectedObject {
141        position: usize,
142        expected: ObjectId,
143        actual: ObjectId,
144    },
145
146    /// `transactions()` answered a position with a different transaction than
147    /// the one requested there.
148    #[error("requested transaction {expected} at position {position}, but got {actual}")]
149    UnexpectedTransaction {
150        position: usize,
151        expected: TransactionDigest,
152        actual: TransactionDigest,
153    },
154}

184/// Safely convert a `usize` to `u32`, saturating at `u32::MAX` instead of
185/// silently truncating on 64-bit platforms.
186pub fn saturating_usize_to_u32(value: usize) -> u32 {
187    u32::try_from(value).unwrap_or(u32::MAX)
188}

190/// A trait for proto result types that follow the pattern of having
191/// `Some(Result::Value)`, `Some(Result::Error)`, or `None`.
192///
193/// This allows generic handling of gRPC response results that can be either
194/// a success value, a server error, or missing.
195pub trait ProtoResult {
196    /// The success value type.
197    type Value;
198
199    /// Extract the result, converting to our error types.
200    fn into_result(self) -> GrpcResult<Self::Value>;
201}
202
203/// Convert a batch of proto results into one result per requested item,
204/// preserving request order.
205///
206/// The batched RPCs report a failure for a single item as a `google.rpc.Status`
207/// in that item's slot, so the outcome for one item is independent of the
208/// others: a caller that needs every item can
209/// `collect::<GrpcResult<Vec<_>>>()`, while one that tolerates gaps can inspect
210/// each slot.
211pub fn into_item_results<T: ProtoResult>(batch: Vec<T>) -> Vec<GrpcResult<T::Value>> {
212    batch.into_iter().map(ProtoResult::into_result).collect()
213}

215/// Check that a batched read answered every requested item.
216///
217/// Callers pair results with requests by position, so a count that does not
218/// match the request leaves no way to tell which item each result belongs to.
219pub fn check_result_count<T>(results: &[T], expected: usize) -> GrpcResult<()> {
220    if results.len() == expected {
221        Ok(())
222    } else {
223        Err(GrpcError::Protocol(ProtocolError::UnexpectedResultCount {
224            expected,
225            actual: results.len(),
226        }))
227    }
228}

230/// Check that each answered object is the one requested in that position.
231///
232/// A matching count only says how many results came back, not that position `i`
233/// holds object `i`, so the pairing callers rely on is checked rather than
234/// trusted.
235pub fn check_object_identity(
236    results: &[GrpcResult<ProtoObject>],
237    requested: &[(ObjectId, Option<Version>)],
238) -> GrpcResult<()> {
239    for (position, (result, (expected, _))) in results.iter().zip(requested).enumerate() {
240        let Ok(object) = result else { continue };
241        let Some(actual) = answered_object_id(object)? else {
242            continue;
243        };
244        if actual != *expected {
245            return Err(GrpcError::Protocol(ProtocolError::UnexpectedObject {
246                position,
247                expected: *expected,
248                actual,
249            }));
250        }
251    }
252    Ok(())
253}

255/// Check that each answered transaction is the one requested in that position.
256///
257/// See [`check_object_identity`] for why the pairing is checked rather than
258/// trusted.
259pub fn check_transaction_identity(
260    results: &[GrpcResult<ExecutedTransaction>],
261    requested: &[TransactionDigest],
262) -> GrpcResult<()> {
263    for (position, (result, expected)) in results.iter().zip(requested).enumerate() {
264        let Ok(transaction) = result else { continue };
265        let Some(actual) = answered_transaction_digest(transaction)? else {
266            continue;
267        };
268        if actual != *expected {
269            return Err(GrpcError::Protocol(ProtocolError::UnexpectedTransaction {
270                position,
271                expected: *expected,
272                actual,
273            }));
274        }
275    }
276    Ok(())
277}

279/// The id of an answered object, taken from its reference or, when the read
280/// mask left that out, from its BCS. `None` when it carries neither, leaving
281/// nothing to compare.
282fn answered_object_id(object: &ProtoObject) -> GrpcResult<Option<ObjectId>> {
283    if let Some(id) = object
284        .reference
285        .as_ref()
286        .and_then(|reference| reference.object_id.as_ref())
287    {
288        return Ok(Some(id.try_into()?));
289    }
290    if object.bcs.is_some() {
291        return Ok(Some(object.object()?.id()));
292    }
293    Ok(None)
294}
295
296/// The digest of an answered transaction, taken from the response or, when the
297/// read mask left it out, computed from the transaction's BCS. `None` when it
298/// carries neither, leaving nothing to compare.
299fn answered_transaction_digest(
300    transaction: &ExecutedTransaction,
301) -> GrpcResult<Option<TransactionDigest>> {
302    let Some(transaction) = transaction.transaction.as_ref() else {
303        return Ok(None);
304    };
305    if let Some(digest) = transaction.digest.as_ref() {
306        return Ok(Some(digest.try_into()?));
307    }
308    if transaction.bcs.is_some() {
309        return Ok(Some(transaction.transaction()?.digest()));
310    }
311    Ok(None)
312}

389/// Collect all items from a paginated gRPC stream into a single `Vec`.
390///
391/// This handles the common pattern of iterating over a `tonic::Streaming<T>`,
392/// extracting items from each message via the `extract` closure, and checking
393/// that the stream was not truncated (i.e. `has_next` is `false` on the last
394/// message).
395///
396/// The `extract` closure receives each stream message and must return
397/// `(has_next, items)`.  Because some streams require fallible per-item
398/// conversion (e.g. via [`ProtoResult`]), the closure itself returns
399/// `GrpcResult<…>`.
400pub async fn collect_stream<T, I, F>(
401    mut stream: tonic::Streaming<T>,
402    metadata: tonic::metadata::MetadataMap,
403    extract: F,
404) -> GrpcResult<MetadataEnvelope<Vec<I>>>
405where
406    F: Fn(T) -> GrpcResult<(bool, Vec<I>)>,
407{
408    let mut results = Vec::new();
409    let mut has_next = false;
410
411    while let Some(response) = stream.message().await? {
412        let (next, items) = extract(response)?;
413        has_next = next;
414        results.extend(items);
415    }
416
417    if has_next {
418        return Err(GrpcError::UnexpectedEndOfStream);
419    }
420
421    Ok(MetadataEnvelope::new(results, metadata))
422}

424/// A single page of results from a paginated list endpoint.
425///
426/// Returned when awaiting a list query builder directly (single-page mode).
427/// Contains the items from this page plus an optional continuation token.
428#[derive(Clone, Debug)]
429pub struct Page<T> {
430    /// The items returned in this page.
431    pub items: Vec<T>,
432    /// Token to retrieve the next page. `None` when this is the last page.
433    pub next_page_token: Option<::prost::bytes::Bytes>,
434}

436/// Generate a query object: a struct that runs its query when awaited.
437///
438/// The struct's [`IntoFuture`](std::future::IntoFuture) boxes the future of
439/// `send(self) -> $output`, which each invocation writes by hand in an
440/// inherent impl.
441macro_rules! define_query {
442    (
443        $(#[$meta:meta])*
444        pub struct $name:ident $(<$generic:ident: $bound:path>)? {
445            $($field:ident: $field_ty:ty),* $(,)?
446        }
447        output: $output:ty;
448    ) => {
449        $(#[$meta])*
450        #[must_use]
451        pub struct $name $(<$generic>)? {
452            $($field: $field_ty,)*
453        }
454
455        impl $(<$generic: $bound + 'static>)? ::std::future::IntoFuture for $name $(<$generic>)? {
456            type Output = $output;
457            type IntoFuture = ::std::pin::Pin<
458                Box<dyn ::std::future::Future<Output = Self::Output> + Send>,
459            >;
460
461            fn into_future(self) -> Self::IntoFuture {
462                Box::pin(self.send())
463            }
464        }
465    };
466}

470/// Generate a paginated query builder for a list endpoint.
471///
472/// The generated struct implements [`IntoFuture`](std::future::IntoFuture) for
473/// single-page retrieval and provides a [`collect`] method for auto-pagination.
474///
475/// # Parameters
476///
477/// - `$query_name` — name of the generated builder struct, optionally with a
478///   single bounded type parameter (e.g. `ListOwnedMoveObjectsQuery<T:
479///   MoveObject>`) usable in `item` and `map_item`
480/// - `$service_client_type` — the tonic service client type
481/// - `$item_type` — the item type exposed by the builder
482/// - `$rpc_method` — the RPC method name on the service client
483/// - `$items_field` — the field name on the response containing the items vec
484/// - `map_item` (optional) — a fallible `fn(&ProtoItem) ->
485///   GrpcResult<$item_type>` applied to each response element. When omitted,
486///   items are passed through unchanged (so `$item_type` must be the response
487///   field's element type).
488///
489/// # Example
490///
491/// ```ignore
492/// define_list_query! {
493///     pub struct ListOwnedObjectsQuery {
494///         service_client: StateServiceClient<InterceptedChannel>,
495///         request: ListOwnedObjectsRequest,
496///         item: Object,
497///         rpc_method: list_owned_objects,
498///         items_field: objects,
499///     }
500/// }
501/// ```
502///
503/// With a per-item conversion:
504///
505/// ```ignore
506/// define_list_query! {
507///     pub struct GetCoinsQuery {
508///         service_client: StateServiceClient<InterceptedChannel>,
509///         request: ListOwnedObjectsRequest,
510///         item: Coin,
511///         rpc_method: list_owned_objects,
512///         items_field: objects,
513///         map_item: object_to_coin, // fn(&Object) -> GrpcResult<Coin>
514///     }
515/// }
516/// ```
517macro_rules! define_list_query {
518    // Pass-through variant: `$item_type` is the response element type.
519    (
520        $(#[$meta:meta])*
521        pub struct $query_name:ident $(<$generic:ident: $bound:path>)? {
522            service_client: $service_client_type:ty,
523            request: $request_type:ty,
524            item: $item_type:ty,
525            rpc_method: $rpc_method:ident,
526            items_field: $items_field:ident,
527        }
528    ) => {
529        $crate::api::define_list_query! {
530            @impl
531            $(#[$meta])*
532            pub struct $query_name $(<$generic: $bound>)? {
533                service_client: $service_client_type,
534                request: $request_type,
535                item: $item_type,
536                rpc_method: $rpc_method,
537                items_field: $items_field,
538                map_item: |item| $crate::api::GrpcResult::Ok(item),
539            }
540        }
541    };
542
543    // Conversion variant: each response element is mapped through `$map_item`,
544    // a fallible `fn(&ProtoItem) -> GrpcResult<$item_type>`.
545    (
546        $(#[$meta:meta])*
547        pub struct $query_name:ident $(<$generic:ident: $bound:path>)? {
548            service_client: $service_client_type:ty,
549            request: $request_type:ty,
550            item: $item_type:ty,
551            rpc_method: $rpc_method:ident,
552            items_field: $items_field:ident,
553            map_item: $map_item:expr,
554        }
555    ) => {
556        $crate::api::define_list_query! {
557            @impl
558            $(#[$meta])*
559            pub struct $query_name $(<$generic: $bound>)? {
560                service_client: $service_client_type,
561                request: $request_type,
562                item: $item_type,
563                rpc_method: $rpc_method,
564                items_field: $items_field,
565                map_item: |item| $map_item(&item),
566            }
567        }
568    };
569
570    (
571        @impl
572        $(#[$meta:meta])*
573        pub struct $query_name:ident $(<$generic:ident: $bound:path>)? {
574            service_client: $service_client_type:ty,
575            request: $request_type:ty,
576            item: $item_type:ty,
577            rpc_method: $rpc_method:ident,
578            items_field: $items_field:ident,
579            map_item: $map_item:expr,
580        }
581    ) => {
582        $crate::api::define_query! {
583            $(#[$meta])*
584            pub struct $query_name $(<$generic: $bound>)? {
585                service_client: $service_client_type,
586                base_request: $request_type,
587                max_message_size: Option<usize>,
588                page_size: Option<u32>,
589                page_token: Option<::prost::bytes::Bytes>,
590                _marker: ::std::marker::PhantomData<fn() -> ($($generic,)?)>,
591            }
592            output: $crate::api::GrpcResult<
593                $crate::api::MetadataEnvelope<$crate::api::Page<$item_type>>,
594            >;
595        }
596
597        impl $(<$generic: $bound>)? $query_name $(<$generic>)? {
598            pub(crate) fn new(
599                service_client: $service_client_type,
600                base_request: $request_type,
601                max_message_size: Option<usize>,
602            ) -> Self {
603                Self {
604                    service_client,
605                    base_request,
606                    max_message_size,
607                    page_size: None,
608                    page_token: None,
609                    _marker: ::std::marker::PhantomData,
610                }
611            }
612
613            /// Set the maximum number of items per page.
614            pub fn page_size(mut self, page_size: u32) -> Self {
615                self.page_size = Some(page_size);
616                self
617            }
618
619            /// Set the continuation token from a previous page.
620            pub fn page_token(
621                mut self,
622                page_token: ::prost::bytes::Bytes,
623            ) -> Self {
624                self.page_token = Some(page_token);
625                self
626            }
627
628            /// Auto-paginate through all pages, collecting up to `limit` items.
629            ///
630            /// If `limit` is `None`, collects all items across all pages.
631            pub async fn collect(
632                self,
633                limit: impl Into<Option<u32>>,
634            ) -> $crate::api::GrpcResult<$crate::api::MetadataEnvelope<Vec<$item_type>>> {
635                let limit = limit.into();
636                let mut all_items: Vec<$item_type> = Vec::new();
637                let mut next_page_token = self.page_token;
638                let mut result_metadata = None;
639                let mut service_client = self.service_client;
640
641                loop {
642                    let mut request = self.base_request.clone();
643
644                    // Cap page_size to the remaining items needed when a
645                    // limit is set, so we don't over-fetch from the server.
646                    let effective_page_size = match (self.page_size, limit) {
647                        (Some(ps), Some(l)) => {
648                            let remaining = (l as usize).saturating_sub(all_items.len());
649                            Some(ps.min(remaining as u32))
650                        }
651                        (Some(ps), None) => Some(ps),
652                        (None, Some(l)) => {
653                            let remaining = (l as usize).saturating_sub(all_items.len());
654                            Some(remaining as u32)
655                        }
656                        (None, None) => None,
657                    };
658                    if let Some(ps) = effective_page_size {
659                        request = request.with_page_size(ps);
660                    }
661                    if let Some(token) = next_page_token.take() {
662                        request = request.with_page_token(token);
663                    }
664                    if let Some(max_size) = self.max_message_size {
665                        request = request.with_max_message_size_bytes(
666                            $crate::api::saturating_usize_to_u32(max_size),
667                        );
668                    }
669
670                    let response = service_client.$rpc_method(request).await?;
671                    let (body, metadata) =
672                        $crate::api::MetadataEnvelope::from(response).into_parts();
673                    if result_metadata.is_none() {
674                        result_metadata = Some(metadata);
675                    }
676
677                    let map_item = $map_item;
678                    for item in body.$items_field {
679                        all_items.push(map_item(item)?);
680                    }
681
682                    match body.next_page_token {
683                        Some(token) => next_page_token = Some(token),
684                        None => break,
685                    }
686
687                    if limit.is_some_and(|l| all_items.len() >= l as usize) {
688                        break;
689                    }
690                }
691
692                Ok($crate::api::MetadataEnvelope::new(
693                    all_items,
694                    result_metadata.unwrap_or_default(),
695                ))
696            }
697
698            fn into_request(self) -> ($service_client_type, $request_type) {
699                let mut request = self.base_request;
700
701                if let Some(ps) = self.page_size {
702                    request = request.with_page_size(ps);
703                }
704                if let Some(token) = self.page_token {
705                    request = request.with_page_token(token);
706                }
707                if let Some(max_size) = self.max_message_size {
708                    request = request.with_max_message_size_bytes(
709                        $crate::api::saturating_usize_to_u32(max_size),
710                    );
711                }
712
713                (self.service_client, request)
714            }
715
716            async fn send(
717                self,
718            ) -> $crate::api::GrpcResult<
719                $crate::api::MetadataEnvelope<$crate::api::Page<$item_type>>,
720            > {
721                let (mut service_client, request) = self.into_request();
722                let response = service_client.$rpc_method(request).await?;
723                let (body, metadata) =
724                    $crate::api::MetadataEnvelope::from(response).into_parts();
725
726                let map_item = $map_item;
727                let items = body
728                    .$items_field
729                    .into_iter()
730                    .map(map_item)
731                    .collect::<$crate::api::GrpcResult<Vec<$item_type>>>()?;
732
733                Ok($crate::api::MetadataEnvelope::new(
734                    $crate::api::Page {
735                        items,
736                        next_page_token: body.next_page_token,
737                    },
738                    metadata,
739                ))
740            }
741        }
742    };
743}

747/// Convert an `ObjectId` to the gRPC proto `ObjectId` type.
748pub fn proto_object_id(id: ObjectId) -> ProtoObjectId {
749    ProtoObjectId::default().with_object_id(Vec::from(id))

752/// Build a proto Transaction from serializable transaction data and digest.
753pub fn build_proto_transaction<T: Serialize>(
754    data: &T,
755    digest: TransactionDigest,
756) -> GrpcResult<ProtoTransaction> {
757    let bcs = BcsData::serialize(data)
758        .map_err(|e| GrpcError::from(TryFromProtoError::invalid("transaction", e)))?;
759
760    let proto_transaction = ProtoTransaction::default()
761        .with_digest(digest)
762        .with_bcs(bcs);
763
764    Ok(proto_transaction)
765}
