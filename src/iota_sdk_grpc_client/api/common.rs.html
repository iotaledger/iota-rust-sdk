32/// Errors that can occur during gRPC client API operations.
33#[derive(Debug, thiserror::Error)]
34#[non_exhaustive]
35pub enum Error {
36    /// Error converting proto types to SDK types.
37    #[error("proto conversion error: {0}")]
38    ProtoConversion(#[from] Box<TryFromProtoError>),
39
40    /// Per-item error returned by the server (preserves code, message,
41    /// details).
42    #[error("server error (code {code}): {msg}", code = .0.code, msg = .0.message)]
43    Server(RpcStatus),
44
45    /// Client-side protocol error (e.g. checkpoint stream reassembly).
46    #[error("protocol error: {0}")]
47    Protocol(ProtocolError),
48
49    /// Error converting signatures to proto format.
50    #[error("signature conversion error: {0}")]
51    Signature(GrpcConversionError),
52
53    /// The caller passed an empty request (e.g. no object IDs or digests).
54    #[error("empty request: at least one item must be provided")]
55    EmptyRequest,
56
57    /// The server stream ended unexpectedly while `has_next` was still true.
58    #[error("stream ended unexpectedly: server indicated more results with has_next=true")]
59    UnexpectedEndOfStream,
60
61    /// gRPC transport or protocol error.
62    #[error("grpc error: {0}")]
63    Grpc(Box<tonic::Status>),
64}
65
66impl Error {
67    /// Returns `true` if the error carries a `NOT_FOUND` status, whether the
68    /// server reported it for the call or for a single item of a batched
69    /// request.
70    ///
71    /// **Warning:** do not test the outer error of a batched read to decide
72    /// that an item is absent. Those calls report an absent item against the
73    /// request that asked for it, so a `NOT_FOUND` for the call means the
74    /// endpoint is wrong; treating it as a missing item turns a misconfigured
75    /// endpoint into a normal "not there" answer. Test the per-item result
76    /// instead. Calls returning a single response, such as fetching a
77    /// checkpoint, do report absence at the call level.
78    pub fn is_not_found(&self) -> bool {
79        match self {
80            Error::Server(status) => status.code == i32::from(tonic::Code::NotFound),
81            Error::Grpc(status) => status.code() == tonic::Code::NotFound,
82            _ => false,
83        }
84    }
85}
86
87impl From<TryFromProtoError> for Error {
88    fn from(err: TryFromProtoError) -> Self {
89        Error::ProtoConversion(Box::new(err))
90    }
91}
92
93impl From<tonic::Status> for Error {
94    fn from(status: tonic::Status) -> Self {
95        Error::Grpc(Box::new(status))
96    }
97}
98
99impl From<Error> for tonic::Status {
100    fn from(err: Error) -> Self {
101        match err {
102            Error::ProtoConversion(e) => {
103                tonic::Status::internal(format!("proto conversion error: {e}"))
104            }
105            Error::Server(status) => status.to_tonic_status(),
106            Error::Protocol(err) => tonic::Status::internal(format!("protocol error: {err}")),
107            Error::Signature(err) => {
108                tonic::Status::internal(format!("signature conversion error: {err}"))
109            }
110            Error::EmptyRequest => {
111                tonic::Status::invalid_argument("empty request: at least one item must be provided")
112            }
113            Error::UnexpectedEndOfStream => {
114                tonic::Status::internal("stream ended unexpectedly: has_next was true")
115            }
116            Error::Grpc(status) => *status,
117        }
118    }
119}

121/// Protocol-level errors encountered while processing gRPC responses.
122#[derive(Debug, thiserror::Error)]
123#[non_exhaustive]
124pub enum ProtocolError {
125    /// Server returned an unrecognized proto oneof variant.
126    #[error("unknown {0} variant")]
127    UnknownVariant(&'static str),
128
129    /// A required response field was unexpectedly empty.
130    #[error("empty response field: {0}")]
131    EmptyResponseField(&'static str),
132
133    /// Error during checkpoint data stream reassembly.
134    #[error("checkpoint stream error: {0}")]
135    CheckpointStream(#[from] CheckpointStreamError),
136
137    /// A batched read returned a number of results that does not match the
138    /// number of requested items.
139    #[error("expected {expected} results, got {actual}")]
140    UnexpectedResultCount { expected: usize, actual: usize },
141
142    /// `get_objects` answered a position with a different object than the one
143    /// requested there.
144    #[error("requested object {expected} at position {position}, but got {actual}")]
145    UnexpectedObject {
146        position: usize,
147        expected: ObjectId,
148        actual: ObjectId,
149    },
150
151    /// `get_transactions` answered a position with a different transaction than
152    /// the one requested there.
153    #[error("requested transaction {expected} at position {position}, but got {actual}")]
154    UnexpectedTransaction {
155        position: usize,
156        expected: TransactionDigest,
157        actual: TransactionDigest,
158    },
159}

193/// A low-level read mask string.
194///
195/// Most callers should use the scoped per-endpoint mask types in
196/// [`read_mask_fields`](crate::read_mask_fields)
197/// (e.g. [`ObjectReadMask`](crate::read_mask_fields::ObjectReadMask)) which
198/// are passed directly to the client methods. This type is the underlying
199/// string holder, useful when composing masks by hand:
200///
201/// ```
202/// use iota_sdk_grpc_client::ReadMask;
203///
204/// let mask = ReadMask::from("effects,checkpoint");
205/// assert_eq!(mask.as_str(), "effects,checkpoint");
206/// ```
207#[derive(Clone, Debug)]
208pub struct ReadMask<'a>(Cow<'a, str>);
209
210impl<'a> ReadMask<'a> {
211    /// Returns the comma-separated field mask string.
212    pub fn as_str(&self) -> &str {
213        &self.0
214    }
215}
216
217impl<'a> From<&'a str> for ReadMask<'a> {
218    fn from(s: &'a str) -> Self {
219        Self(Cow::Borrowed(s))
220    }
221}
222
223impl From<String> for ReadMask<'_> {
224    fn from(s: String) -> Self {
225        Self(Cow::Owned(s))
226    }
227}

229impl From<&[&str]> for ReadMask<'_> {
230    /// Paths are normalized: broader paths subsume their sub-paths.
231    fn from(paths: &[&str]) -> Self {
232        Self(Cow::Owned(field_mask_normalize(&paths.join(","))))
233    }
234}

236impl<const N: usize> From<&[&str; N]> for ReadMask<'_> {
237    /// Paths are normalized: broader paths subsume their sub-paths.
238    fn from(paths: &[&str; N]) -> Self {
239        Self::from(paths.as_slice())
240    }
241}

245    fn from(mask: FieldMask) -> Self {
246        Self(Cow::Owned(field_mask_normalize(&mask.paths.join(","))))
247    }

250/// Safely convert a `usize` to `u32`, saturating at `u32::MAX` instead of
251/// silently truncating on 64-bit platforms.
252pub fn saturating_usize_to_u32(value: usize) -> u32 {
253    u32::try_from(value).unwrap_or(u32::MAX)
254}

256/// A trait for proto result types that follow the pattern of having
257/// `Some(Result::Value)`, `Some(Result::Error)`, or `None`.
258///
259/// This allows generic handling of gRPC response results that can be either
260/// a success value, a server error, or missing.
261pub trait ProtoResult {
262    /// The success value type.
263    type Value;
264
265    /// Extract the result, converting to our error types.
266    fn into_result(self) -> Result<Self::Value>;
267}

269/// Convert a batch of proto results into one result per requested item,
270/// preserving request order.
271///
272/// The batched RPCs report a failure for a single item as a `google.rpc.Status`
273/// in that item's slot, so the outcome for one item is independent of the
274/// others: a caller that needs every item can `collect::<Result<Vec<_>>>()`,
275/// while one that tolerates gaps can inspect each slot.
276pub fn into_item_results<T: ProtoResult>(batch: Vec<T>) -> Vec<Result<T::Value>> {
277    batch.into_iter().map(ProtoResult::into_result).collect()
278}

280/// Check that a batched read answered every requested item.
281///
282/// Callers pair results with requests by position, so a count that does not
283/// match the request leaves no way to tell which item each result belongs to.
284pub fn check_result_count<T>(results: &[T], expected: usize) -> Result<()> {
285    if results.len() == expected {
286        Ok(())
287    } else {
288        Err(Error::Protocol(ProtocolError::UnexpectedResultCount {
289            expected,
290            actual: results.len(),
291        }))
292    }
293}

295/// Check that each answered object is the one requested in that position.
296///
297/// A matching count only says how many results came back, not that position `i`
298/// holds object `i`, so the pairing callers rely on is checked rather than
299/// trusted.
300pub fn check_object_identity(
301    results: &[Result<ProtoObject>],
302    requested: &[(ObjectId, Option<Version>)],
303) -> Result<()> {
304    for (position, (result, (expected, _))) in results.iter().zip(requested).enumerate() {
305        let Ok(object) = result else { continue };
306        let Some(actual) = answered_object_id(object)? else {
307            continue;
308        };
309        if actual != *expected {
310            return Err(Error::Protocol(ProtocolError::UnexpectedObject {
311                position,
312                expected: *expected,
313                actual,
314            }));
315        }
316    }
317    Ok(())
318}

320/// Check that each answered transaction is the one requested in that position.
321///
322/// See [`check_object_identity`] for why the pairing is checked rather than
323/// trusted.
324pub fn check_transaction_identity(
325    results: &[Result<ExecutedTransaction>],
326    requested: &[TransactionDigest],
327) -> Result<()> {
328    for (position, (result, expected)) in results.iter().zip(requested).enumerate() {
329        let Ok(transaction) = result else { continue };
330        let Some(actual) = answered_transaction_digest(transaction)? else {
331            continue;
332        };
333        if actual != *expected {
334            return Err(Error::Protocol(ProtocolError::UnexpectedTransaction {
335                position,
336                expected: *expected,
337                actual,
338            }));
339        }
340    }
341    Ok(())
342}

344/// The id of an answered object, taken from its reference or, when the read
345/// mask left that out, from its BCS. `None` when it carries neither, leaving
346/// nothing to compare.
347fn answered_object_id(object: &ProtoObject) -> Result<Option<ObjectId>> {
348    if let Some(id) = object
349        .reference
350        .as_ref()
351        .and_then(|reference| reference.object_id.as_ref())
352    {
353        return Ok(Some(id.try_into()?));
354    }
355    if object.bcs.is_some() {
356        return Ok(Some(object.object()?.id()));
357    }
358    Ok(None)
359}
360
361/// The digest of an answered transaction, taken from the response or, when the
362/// read mask left it out, computed from the transaction's BCS. `None` when it
363/// carries neither, leaving nothing to compare.
364fn answered_transaction_digest(
365    transaction: &ExecutedTransaction,
366) -> Result<Option<TransactionDigest>> {
367    let Some(transaction) = transaction.transaction.as_ref() else {
368        return Ok(None);
369    };
370    if let Some(digest) = transaction.digest.as_ref() {
371        return Ok(Some(digest.try_into()?));
372    }
373    if transaction.bcs.is_some() {
374        return Ok(Some(transaction.transaction()?.digest()));
375    }
376    Ok(None)
377}

454/// Collect all items from a paginated gRPC stream into a single `Vec`.
455///
456/// This handles the common pattern of iterating over a `tonic::Streaming<T>`,
457/// extracting items from each message via the `extract` closure, and checking
458/// that the stream was not truncated (i.e. `has_next` is `false` on the last
459/// message).
460///
461/// The `extract` closure receives each stream message and must return
462/// `(has_next, items)`.  Because some streams require fallible per-item
463/// conversion (e.g. via [`ProtoResult`]), the closure itself returns
464/// `Result<…>`.
465pub async fn collect_stream<T, I, F>(
466    mut stream: tonic::Streaming<T>,
467    metadata: tonic::metadata::MetadataMap,
468    extract: F,
469) -> Result<MetadataEnvelope<Vec<I>>>
470where
471    F: Fn(T) -> Result<(bool, Vec<I>)>,
472{
473    let mut results = Vec::new();
474    let mut has_next = false;
475
476    while let Some(response) = stream.message().await? {
477        let (next, items) = extract(response)?;
478        has_next = next;
479        results.extend(items);
480    }
481
482    if has_next {
483        return Err(Error::UnexpectedEndOfStream);
484    }
485
486    Ok(MetadataEnvelope::new(results, metadata))
487}

489/// A single page of results from a paginated list endpoint.
490///
491/// Returned when awaiting a list query builder directly (single-page mode).
492/// Contains the items from this page plus an optional continuation token.
493#[derive(Clone, Debug)]
494pub struct Page<T> {
495    /// The items returned in this page.
496    pub items: Vec<T>,
497    /// Token to retrieve the next page. `None` when this is the last page.
498    pub next_page_token: Option<::prost::bytes::Bytes>,
499}

501/// Generate a paginated query builder for a list endpoint.
502///
503/// The generated struct implements [`IntoFuture`](std::future::IntoFuture) for
504/// single-page retrieval and provides a [`collect`] method for auto-pagination.
505///
506/// # Parameters
507///
508/// - `$query_name` — name of the generated builder struct
509/// - `$service_client_type` — the tonic service client type
510/// - `$item_type` — the item type exposed by the builder
511/// - `$rpc_method` — the RPC method name on the service client
512/// - `$items_field` — the field name on the response containing the items vec
513/// - `map_item` (optional) — a fallible `fn(&ProtoItem) -> Result<$item_type>`
514///   applied to each response element. When omitted, items are passed through
515///   unchanged (so `$item_type` must be the response field's element type).
516///
517/// # Example
518///
519/// ```ignore
520/// define_list_query! {
521///     pub struct ListOwnedObjectsQuery {
522///         service_client: StateServiceClient<InterceptedChannel>,
523///         request: ListOwnedObjectsRequest,
524///         item: Object,
525///         rpc_method: list_owned_objects,
526///         items_field: objects,
527///     }
528/// }
529/// ```
530///
531/// With a per-item conversion:
532///
533/// ```ignore
534/// define_list_query! {
535///     pub struct GetCoinsQuery {
536///         service_client: StateServiceClient<InterceptedChannel>,
537///         request: ListOwnedObjectsRequest,
538///         item: Coin,
539///         rpc_method: list_owned_objects,
540///         items_field: objects,
541///         map_item: object_to_coin, // fn(&Object) -> Result<Coin>
542///     }
543/// }
544/// ```
545macro_rules! define_list_query {
546    // Pass-through variant: `$item_type` is the response element type.
547    (
548        $(#[$meta:meta])*
549        pub struct $query_name:ident {
550            service_client: $service_client_type:ty,
551            request: $request_type:ty,
552            item: $item_type:ty,
553            rpc_method: $rpc_method:ident,
554            items_field: $items_field:ident,
555        }
556    ) => {
557        $crate::api::define_list_query! {
558            @impl
559            $(#[$meta])*
560            pub struct $query_name {
561                service_client: $service_client_type,
562                request: $request_type,
563                item: $item_type,
564                rpc_method: $rpc_method,
565                items_field: $items_field,
566                map_item: |item| $crate::api::Result::Ok(item),
567            }
568        }
569    };
570
571    // Conversion variant: each response element is mapped through `$map_item`,
572    // a fallible `fn(&ProtoItem) -> Result<$item_type>`.
573    (
574        $(#[$meta:meta])*
575        pub struct $query_name:ident {
576            service_client: $service_client_type:ty,
577            request: $request_type:ty,
578            item: $item_type:ty,
579            rpc_method: $rpc_method:ident,
580            items_field: $items_field:ident,
581            map_item: $map_item:expr,
582        }
583    ) => {
584        $crate::api::define_list_query! {
585            @impl
586            $(#[$meta])*
587            pub struct $query_name {
588                service_client: $service_client_type,
589                request: $request_type,
590                item: $item_type,
591                rpc_method: $rpc_method,
592                items_field: $items_field,
593                map_item: |item| $map_item(&item),
594            }
595        }
596    };
597
598    (
599        @impl
600        $(#[$meta:meta])*
601        pub struct $query_name:ident {
602            service_client: $service_client_type:ty,
603            request: $request_type:ty,
604            item: $item_type:ty,
605            rpc_method: $rpc_method:ident,
606            items_field: $items_field:ident,
607            map_item: $map_item:expr,
608        }
609    ) => {
610        $(#[$meta])*
611        pub struct $query_name {
612            service_client: $service_client_type,
613            base_request: $request_type,
614            max_message_size: Option<usize>,
615            page_size: Option<u32>,
616            page_token: Option<::prost::bytes::Bytes>,
617        }
618
619        impl $query_name {
620            pub(crate) fn new(
621                service_client: $service_client_type,
622                base_request: $request_type,
623                max_message_size: Option<usize>,
624                page_size: Option<u32>,
625                page_token: Option<::prost::bytes::Bytes>,
626            ) -> Self {
627                Self {
628                    service_client,
629                    base_request,
630                    max_message_size,
631                    page_size,
632                    page_token,
633                }
634            }
635
636            /// Auto-paginate through all pages, collecting up to `limit` items.
637            ///
638            /// If `limit` is `None`, collects all items across all pages.
639            pub async fn collect(
640                self,
641                limit: impl Into<Option<u32>>,
642            ) -> $crate::api::Result<$crate::api::MetadataEnvelope<Vec<$item_type>>> {
643                let limit = limit.into();
644                let mut all_items: Vec<$item_type> = Vec::new();
645                let mut next_page_token = self.page_token;
646                let mut result_metadata = None;
647                let mut service_client = self.service_client;
648
649                loop {
650                    let mut request = self.base_request.clone();
651
652                    // Cap page_size to the remaining items needed when a
653                    // limit is set, so we don't over-fetch from the server.
654                    let effective_page_size = match (self.page_size, limit) {
655                        (Some(ps), Some(l)) => {
656                            let remaining = (l as usize).saturating_sub(all_items.len());
657                            Some(ps.min(remaining as u32))
658                        }
659                        (Some(ps), None) => Some(ps),
660                        (None, Some(l)) => {
661                            let remaining = (l as usize).saturating_sub(all_items.len());
662                            Some(remaining as u32)
663                        }
664                        (None, None) => None,
665                    };
666                    if let Some(ps) = effective_page_size {
667                        request = request.with_page_size(ps);
668                    }
669                    if let Some(token) = next_page_token.take() {
670                        request = request.with_page_token(token);
671                    }
672                    if let Some(max_size) = self.max_message_size {
673                        request = request.with_max_message_size_bytes(
674                            $crate::api::saturating_usize_to_u32(max_size),
675                        );
676                    }
677
678                    let response = service_client.$rpc_method(request).await?;
679                    let (body, metadata) =
680                        $crate::api::MetadataEnvelope::from(response).into_parts();
681                    if result_metadata.is_none() {
682                        result_metadata = Some(metadata);
683                    }
684
685                    let map_item = $map_item;
686                    for item in body.$items_field {
687                        all_items.push(map_item(item)?);
688                    }
689
690                    match body.next_page_token {
691                        Some(token) => next_page_token = Some(token),
692                        None => break,
693                    }
694
695                    if limit.is_some_and(|l| all_items.len() >= l as usize) {
696                        break;
697                    }
698                }
699
700                Ok($crate::api::MetadataEnvelope::new(
701                    all_items,
702                    result_metadata.unwrap_or_default(),
703                ))
704            }
705        }
706
707        impl ::std::future::IntoFuture for $query_name {
708            type Output = $crate::api::Result<
709                $crate::api::MetadataEnvelope<$crate::api::Page<$item_type>>,
710            >;
711            type IntoFuture = ::std::pin::Pin<
712                Box<dyn ::std::future::Future<Output = Self::Output> + Send>,
713            >;
714
715            fn into_future(self) -> Self::IntoFuture {
716                Box::pin(async move {
717                    let mut service_client = self.service_client;
718                    let mut request = self.base_request;
719
720                    if let Some(ps) = self.page_size {
721                        request = request.with_page_size(ps);
722                    }
723                    if let Some(token) = self.page_token {
724                        request = request.with_page_token(token);
725                    }
726                    if let Some(max_size) = self.max_message_size {
727                        request = request.with_max_message_size_bytes(
728                            $crate::api::saturating_usize_to_u32(max_size),
729                        );
730                    }
731
732                    let response = service_client.$rpc_method(request).await?;
733                    let (body, metadata) =
734                        $crate::api::MetadataEnvelope::from(response).into_parts();
735
736                    let map_item = $map_item;
737                    let items = body
738                        .$items_field
739                        .into_iter()
740                        .map(map_item)
741                        .collect::<$crate::api::Result<Vec<$item_type>>>()?;
742
743                    Ok($crate::api::MetadataEnvelope::new(
744                        $crate::api::Page {
745                            items,
746                            next_page_token: body.next_page_token,
747                        },
748                        metadata,
749                    ))
750                })
751            }
752        }
753    };
754}

758/// Convert an `ObjectId` to the gRPC proto `ObjectId` type.
759pub fn proto_object_id(id: ObjectId) -> ProtoObjectId {
760    ProtoObjectId::default().with_object_id(Vec::from(id))
761}

763/// Build a proto Transaction from serializable transaction data and digest.
764pub fn build_proto_transaction<T: Serialize>(
765    data: &T,
766    digest: TransactionDigest,
767) -> Result<ProtoTransaction> {
768    let bcs = BcsData::serialize(data)
769        .map_err(|e| Error::from(TryFromProtoError::invalid("transaction", e)))?;
770
771    let proto_transaction = ProtoTransaction::default()
772        .with_digest(digest)
773        .with_bcs(bcs);
774
775    Ok(proto_transaction)
776}
