4//! High-level API for calling Move view functions.
5
6use iota_grpc_types::{
7    proto::json_to_prost_stringify_numbers,
8    read_mask_fields::{IntoReadMask, ViewFunctionCallReadMask},
9    v1::{
10        command::InputArgument,
11        transaction_execution_service::{
12            ViewFunctionCallItem, ViewFunctionCallOutputs, ViewFunctionCallsRequest,
13        },
14    },
15};
16use iota_types::TypeTag;
17
18use crate::{
19    Client,
20    api::{Error, MetadataEnvelope, ProtocolError, Result, check_result_count, into_item_results},
21};
22
23impl Client {
24    /// Call a Move view function and read back what it returns, without
25    /// submitting a transaction.
26    ///
27    /// Arguments are passed as JSON and encoded by the node against the
28    /// parameter's Move type. Numbers go over the wire as strings. To
29    /// pass BCS-encoded arguments instead, build the [`InputArgument`]s
30    /// yourself and use [`view_function_calls`](Self::view_function_calls).
31    ///
32    /// # Parameters
33    ///
34    /// - `fq_function_name`: Fully qualified Move view function name
35    /// - `type_args`: Type arguments
36    /// - `call_args`: Value arguments
37    ///
38    /// Returns [`ViewFunctionCallOutputs`] which contains:
39    /// - `return_values()` - View function return values in case of success
40    /// - `execution_error()` - View function execution error in case of failure
41    ///
42    /// # Example
43    ///
44    /// ```no_run
45    /// # use iota_sdk_grpc_client::Client;
46    /// # use iota_sdk_grpc_client::read_mask_fields::ViewFunctionCallReadMask;
47    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
48    /// let client = Client::new_localnet()?;
49    ///
50    /// // `discounted_price` has to be declared `#[view]` in the package.
51    /// let outputs = client
52    ///     .view_function_call(
53    ///         "0x1234::shop::discounted_price",
54    ///         &[],
55    ///         &[serde_json::json!(100), serde_json::json!(25)],
56    ///         ViewFunctionCallReadMask::default(),
57    ///     )
58    ///     .await?;
59    ///
60    /// // The call ran either way; `execution_error` says whether it aborted.
61    /// match outputs.body().return_values() {
62    ///     Some(values) => println!("returned {} value(s)", values.outputs.len()),
63    ///     None => println!("aborted: {:?}", outputs.body().execution_error()),
64    /// }
65    /// # Ok(())
66    /// # }
67    /// ```
68    ///
69    /// The `read_mask` controls which fields the server returns; use
70    /// `ViewFunctionCallReadMask::default()` for the default mask. Pass a
71    /// [`ViewFunctionCallField`](iota_grpc_types::read_mask_fields::ViewFunctionCallField)
72    /// or any slice/array/vec of fields — conversion is automatic.
73    ///
74    /// # Errors
75    ///
76    /// Returns [`Error::EmptyRequest`] if `fq_function_name` is empty.
77    /// Returns [`Error::Server`] if the node rejected the call (an unknown
78    /// function, a wrong argument count, a non-view function). A call that ran
79    /// and *aborted* is not an error here — it comes back as
80    /// [`ViewFunctionCallOutputs::execution_error`].
81    pub async fn view_function_call(
82        &self,
83        fq_function_name: &str,
84        type_args: &[TypeTag],
85        call_args: &[serde_json::Value],
86        read_mask: impl IntoReadMask<ViewFunctionCallReadMask>,
87    ) -> Result<MetadataEnvelope<ViewFunctionCallOutputs>> {
88        if fq_function_name.is_empty() {
89            return Err(Error::EmptyRequest);
90        }
91
92        self.view_function_calls(
93            vec![
94                ViewFunctionCallItem::default()
95                    .with_fq_function_name(fq_function_name)
96                    .with_type_args(type_args.iter().map(|t| t.into()).collect())
97                    .with_inputs(
98                        call_args
99                            .iter()
100                            .map(|arg| {
101                                InputArgument::default()
102                                    .with_json(json_to_prost_stringify_numbers(arg))
103                            })
104                            .collect(),
105                    ),
106            ],
107            read_mask,
108        )
109        .await?
110        .try_map(|results| {
111            results
112                .into_iter()
113                .next()
114                .ok_or_else(|| Error::Protocol(ProtocolError::EmptyResponseField("call_results")))?
115        })
116    }
117
118    /// Call a batch of Move view functions.
119    ///
120    /// Each call runs in its own transaction on the server, so one call being
121    /// rejected leaves the rest untouched.
122    ///
123    /// Returns a `Vec<Result<ViewFunctionCallOutputs>>` in the same order as
124    /// the input. Each element is either the outputs of that call or the
125    /// per-item error the server returned for it. Note that a call which ran
126    /// and aborted lands in the `Ok` slot — the abort is reported by
127    /// [`ViewFunctionCallOutputs::execution_error`]; only a call the server
128    /// refused to run yields `Err`.
129    ///
130    /// The `read_mask` controls which fields the server returns for each
131    /// `ViewFunctionCallOutputs`; use `ViewFunctionCallReadMask::default()`
132    /// for the default mask. Pass a
133    /// [`ViewFunctionCallField`](iota_grpc_types::read_mask_fields::ViewFunctionCallField)
134    /// or any slice/array/vec of fields — conversion is automatic.
135    ///
136    /// # Errors
137    ///
138    /// Returns [`Error::EmptyRequest`] if `function_calls` is empty.
139    /// Returns a transport-level [`Error::Grpc`] if the entire RPC fails
140    /// (e.g. batch size exceeded).
141    /// Returns [`ProtocolError::UnexpectedResultCount`] if the server did not
142    /// answer every call, since results are paired with calls by position.
143    pub async fn view_function_calls(
144        &self,
145        function_calls: Vec<ViewFunctionCallItem>,
146        read_mask: impl IntoReadMask<ViewFunctionCallReadMask>,
147    ) -> Result<MetadataEnvelope<Vec<Result<ViewFunctionCallOutputs>>>> {
148        if function_calls.is_empty() {
149            return Err(Error::EmptyRequest);
150        }
151
152        let expected_results = function_calls.len();
153        let read_mask = read_mask.into_read_mask();
154        let request = ViewFunctionCallsRequest::default()
155            .with_view_function_calls(function_calls)
156            .with_read_mask(read_mask);
157
158        let response = self
159            .execution_service_client()
160            .view_function_calls(request)
161            .await?;
162
163        let response = MetadataEnvelope::from(response).map(|r| into_item_results(r.call_results));
164        check_result_count(response.body(), expected_results)?;
165
166        Ok(response)
167    }
168}
