27define_query! {
28    /// Query for [`GrpcClient::execute_transaction`]. Await it to send the
29    /// request.
30    pub struct ExecuteTransactionQuery {
31        batch: ExecuteTransactionsQuery,
32    }
33    output: GrpcResult<MetadataEnvelope<ExecutedTransaction>>;
34}
35
36impl ExecuteTransactionQuery {
37    /// Wait up to `checkpoint_inclusion_timeout_ms` milliseconds for the
38    /// transaction to be included in a checkpoint. Without it, or with `0`, the
39    /// server returns without waiting; it may clamp the timeout to a configured
40    /// maximum.
41    pub fn checkpoint_inclusion_timeout_ms(mut self, checkpoint_inclusion_timeout_ms: u64) -> Self {
42        self.batch = self
43            .batch
44            .checkpoint_inclusion_timeout_ms(checkpoint_inclusion_timeout_ms);
45        self
46    }
47
48    /// Set the field mask controlling the returned fields.
49    pub fn read_mask(mut self, read_mask: impl IntoReadMask<ExecuteTransactionReadMask>) -> Self {
50        self.batch = self.batch.read_mask(read_mask);
51        self
52    }
53
54    async fn send(self) -> GrpcResult<MetadataEnvelope<ExecutedTransaction>> {
55        self.batch
56            .send()
57            .await?
58            .try_map(extract_single_execution_result)
59    }
60}

62define_query! {
63    /// Query for [`GrpcClient::execute_transactions`]. Await it to send the
64    /// request.
65    pub struct ExecuteTransactionsQuery {
66        service_client: TransactionExecutionServiceClient<InterceptedChannel>,
67        transactions: Vec<SignedTransaction>,
68        checkpoint_inclusion_timeout_ms: Option<u64>,
69        read_mask: ExecuteTransactionReadMask,
70    }
71    output: GrpcResult<MetadataEnvelope<Vec<GrpcResult<ExecutedTransaction>>>>;
72}
73
74impl ExecuteTransactionsQuery {
75    /// Wait up to `checkpoint_inclusion_timeout_ms` milliseconds for all
76    /// executed transactions to be included in a checkpoint. Without it, or
77    /// with `0`, the server returns without waiting; it may clamp the timeout
78    /// to a configured maximum.
79    pub fn checkpoint_inclusion_timeout_ms(mut self, checkpoint_inclusion_timeout_ms: u64) -> Self {
80        self.checkpoint_inclusion_timeout_ms = Some(checkpoint_inclusion_timeout_ms);
81        self
82    }
83
84    /// Set the field mask controlling the returned fields.
85    pub fn read_mask(mut self, read_mask: impl IntoReadMask<ExecuteTransactionReadMask>) -> Self {
86        self.read_mask = read_mask.into_read_mask();
87        self
88    }
89
90    fn into_request(
91        self,
92    ) -> GrpcResult<(
93        TransactionExecutionServiceClient<InterceptedChannel>,
94        ExecuteTransactionsRequest,
95    )> {
96        let items = self
97            .transactions
98            .into_iter()
99            .map(build_execute_item)
100            .collect::<GrpcResult<Vec<_>>>()?;
101
102        let mut request = ExecuteTransactionsRequest::default()
103            .with_transactions(items)
104            .with_read_mask(self.read_mask);
105
106        if let Some(timeout_ms) = self.checkpoint_inclusion_timeout_ms {
107            request = request.with_checkpoint_inclusion_timeout_ms(timeout_ms);
108        }
109
110        Ok((self.service_client, request))
111    }
112
113    async fn send(self) -> GrpcResult<MetadataEnvelope<Vec<GrpcResult<ExecutedTransaction>>>> {
114        if self.transactions.is_empty() {
115            return Err(GrpcError::EmptyRequest);
116        }
117
118        let expected_results = self.transactions.len();
119        let (mut service_client, request) = self.into_request()?;
120        let response = service_client.execute_transactions(request).await?;
121
122        let response =
123            MetadataEnvelope::from(response).map(|r| into_item_results(r.transaction_results));
124        check_result_count(response.body(), expected_results)?;
125
126        Ok(response)
127    }
128}

130impl GrpcClient {
131    /// Execute a signed transaction.
132    ///
133    /// This submits the transaction to the network for execution and waits for
134    /// the result. The transaction must be signed with valid signatures.
135    ///
136    /// Returns proto `ExecutedTransaction`. Use lazy conversion methods to
137    /// extract data:
138    /// - `result.effects()` - Get transaction effects
139    /// - `result.events()` - Get transaction events (if available)
140    /// - `result.input_objects()` - Get input objects (if requested)
141    /// - `result.output_objects()` - Get output objects (if requested)
142    /// - `result.balance_changes()` - Get balance changes (if requested)
143    /// - `result.object_changes()` - Get object changes (if requested)
144    ///
145    /// Without [`read_mask`](ExecuteTransactionQuery::read_mask), the default
146    /// mask is used. Pass a
147    /// [`TransactionField`](iota_grpc_types::read_mask_fields::TransactionField)
148    /// or any slice/array/vec of fields to choose the returned fields.
149    ///
150    /// # Checkpoint Inclusion
151    ///
152    /// If
153    /// [`checkpoint_inclusion_timeout_ms`](ExecuteTransactionQuery::checkpoint_inclusion_timeout_ms)
154    /// is set, the server will wait up to the specified duration (in
155    /// milliseconds) for the transaction to be included in a checkpoint before
156    /// returning. When set, include `checkpoint` and `timestamp` in the read
157    /// mask to receive the data.
158    ///
159    /// # Example
160    ///
161    /// ```no_run
162    /// # use iota_sdk_grpc_client::GrpcClient;
163    /// # use iota_types::SignedTransaction;
164    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
165    /// let client = GrpcClient::new_localnet()?;
166    ///
167    /// let signed_tx: SignedTransaction = todo!();
168    /// let result = client.execute_transaction(signed_tx).await?;
169    ///
170    /// let effects = result.body().effects()?.effects()?;
171    /// println!("Status: {:?}", effects.as_v1().status);
172    ///
173    /// let events = result.body().events()?.events()?;
174    /// if !events.0.is_empty() {
175    ///     println!("Events: {}", events.0.len());
176    /// }
177    /// # Ok(())
178    /// # }
179    /// ```
180    pub fn execute_transaction(
181        &self,
182        signed_transaction: SignedTransaction,
183    ) -> ExecuteTransactionQuery {
184        ExecuteTransactionQuery {
185            batch: self.execute_transactions(vec![signed_transaction]),
186        }
187    }
188
189    /// Execute a batch of signed transactions.
190    ///
191    /// Transactions are executed concurrently on the server, so their relative
192    /// execution order is not guaranteed. Send transactions that must run in a
193    /// given order (for example, ones that use the same shared object) in
194    /// separate batches. Failure of one transaction does not abort the rest.
195    ///
196    /// Returns a `Vec<GrpcResult<ExecutedTransaction>>` in the same order as
197    /// the input. Each element is either the successfully executed
198    /// transaction or the per-item error returned by the server.
199    ///
200    /// Without [`read_mask`](ExecuteTransactionsQuery::read_mask), the default
201    /// mask is used for each `ExecutedTransaction`. Pass a
202    /// [`TransactionField`](iota_grpc_types::read_mask_fields::TransactionField)
203    /// or any slice/array/vec of fields to choose the returned fields.
204    ///
205    /// # Checkpoint Inclusion
206    ///
207    /// If
208    /// [`checkpoint_inclusion_timeout_ms`](ExecuteTransactionsQuery::checkpoint_inclusion_timeout_ms)
209    /// is set, the server will wait up to the specified duration (in
210    /// milliseconds) for all executed transactions to be included in a
211    /// checkpoint before returning. When set, include `checkpoint` and
212    /// `timestamp` in the read mask to receive the data.
213    ///
214    /// # Errors
215    ///
216    /// Returns [`GrpcError::EmptyRequest`] if `transactions` is empty.
217    /// Returns a transport-level [`GrpcError::Grpc`] if the entire RPC fails
218    /// (e.g. batch size exceeded).
219    pub fn execute_transactions(
220        &self,
221        transactions: Vec<SignedTransaction>,
222    ) -> ExecuteTransactionsQuery {
223        ExecuteTransactionsQuery {
224            service_client: self.execution_service_client(),
225            transactions,
226            checkpoint_inclusion_timeout_ms: None,
227            read_mask: ExecuteTransactionReadMask::default(),
228        }
229    }
230}

232fn extract_single_execution_result(
233    results: Vec<GrpcResult<ExecutedTransaction>>,
234) -> GrpcResult<ExecutedTransaction> {
235    results.into_iter().next().ok_or_else(|| {
236        GrpcError::Protocol(ProtocolError::EmptyResponseField("transaction_results"))
237    })?
238}

240/// Convert a `SignedTransaction` into a proto `ExecuteTransactionItem`.
241fn build_execute_item(signed_transaction: SignedTransaction) -> GrpcResult<ExecuteTransactionItem> {
242    let tx_digest = signed_transaction.transaction.digest();
243    let proto_transaction = build_proto_transaction(&signed_transaction.transaction, tx_digest)?;
244
245    let proto_signatures = UserSignatures::default().with_signatures(
246        signed_transaction
247            .signatures
248            .into_iter()
249            .map(|sig| ProtoUserSignature::try_from(sig).map_err(GrpcError::Signature))
250            .collect::<GrpcResult<Vec<_>>>()?,
251    );
252
253    Ok(ExecuteTransactionItem::default()
254        .with_transaction(proto_transaction)
255        .with_signatures(proto_signatures))
256}

268    fn signed_transaction() -> SignedTransaction {
269        SignedTransaction {
270            transaction: Transaction::V1(TransactionV1 {
271                kind: TransactionKind::Programmable(ProgrammableTransaction {
272                    inputs: Vec::new(),
273                    commands: Vec::new(),
274                }),
275                sender: Address::ZERO,
276                gas_payment: GasPayment {
277                    objects: Vec::new(),
278                    owner: Address::ZERO,

280                    budget: 1,
281                },
282                expiration: TransactionExpiration::None,
283            }),
284            signatures: Vec::new(),

288    #[tokio::test]
289    async fn execute_transaction_setters_reach_the_one_item_batch() {
290        let client = GrpcClient::new("http://localhost").unwrap();
291        let query = client.execute_transaction(signed_transaction());
292        assert_eq!(query.batch.transactions, vec![signed_transaction()]);
293        assert_eq!(query.batch.checkpoint_inclusion_timeout_ms, None);
294        assert_eq!(
295            query.batch.read_mask.as_str(),
296            ExecuteTransactionReadMask::default().as_str()
297        );
298
299        let query = query
300            .checkpoint_inclusion_timeout_ms(5_000)
301            .read_mask(TransactionField::EFFECTS);
302        assert_eq!(query.batch.checkpoint_inclusion_timeout_ms, Some(5_000));
303        assert_eq!(
304            query.batch.read_mask.as_str(),
305            ExecuteTransactionReadMask::from(TransactionField::EFFECTS).as_str()
306        );
307    }

316    #[tokio::test]
317    async fn the_request_carries_the_timeout_and_the_mask() {
318        let client = GrpcClient::new("http://localhost").unwrap();
319        let (_, request) = client
320            .execute_transaction(signed_transaction())
321            .checkpoint_inclusion_timeout_ms(5_000)
322            .read_mask(TransactionField::CHECKPOINT)
323            .batch
324            .into_request()
325            .unwrap();
326        assert_eq!(request.transactions.len(), 1);
327        assert_eq!(request.checkpoint_inclusion_timeout_ms, Some(5_000));
328        assert_eq!(
329            request.read_mask,
330            Some(ExecuteTransactionReadMask::from(TransactionField::CHECKPOINT).into())
331        );
332
333        let (_, request) = client
334            .execute_transactions(vec![signed_transaction(), signed_transaction()])
335            .into_request()
336            .unwrap();
337        assert_eq!(request.transactions.len(), 2);
338        assert_eq!(request.checkpoint_inclusion_timeout_ms, None);
339    }
