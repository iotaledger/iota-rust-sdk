4//! Capture real on-chain BCS fixtures for the Move types mirrored by
5//! `iota-sdk-move-types`.
6//!
7//! For each entry in [`FIXTURES`], the script queries a live IOTA
8//! GraphQL endpoint, fetches the pinned object, extracts the raw BCS
9//! bytes of the inner Move struct, and writes them to
10//! `crates/iota-sdk-move-types/tests/fixtures/<name>.bcs`.
11//!
12//! This is a standalone package (rather than an example of
13//! `iota-sdk-move-types`) so its GraphQL/`tokio` dependencies stay out of
14//! `iota-sdk-move-types`' own dependency graph — that crate runs its tests
15//! under wasm, where `tokio`/`reqwest` don't build.
16//!
17//! Invocation:
18//!
19//! ```bash
20//! cargo run -p capture-move-type-fixtures
21//! # optional: pick a different network (defaults to mainnet)
22//! IOTA_NETWORK=testnet cargo run -p capture-move-type-fixtures
23//! IOTA_NETWORK=devnet cargo run -p capture-move-type-fixtures
24//! ```
25//!
26//! # Capture strategies
27//!
28//! - [`Source::ObjectId`]: fetch a specific object by its known address. The
29//!   default for almost all fixtures — every entry carries a hard-pinned
30//!   mainnet object ID so re-runs produce the same bytes byte-for-byte. When a
31//!   pinned object is spent / deleted, capture for that fixture fails loudly;
32//!   the operator picks a fresh ID (often via [`Source::TypeFilter`] locally)
33//!   and commits the new pin alongside the new fixture.
34//! - [`Source::TypeFilter`]: ask the GraphQL endpoint for any object matching
35//!   the given fully-qualified type tag and take the first one returned. Useful
36//!   for *discovery* (finding a candidate to pin), but inherently
37//!   non-deterministic across runs so we don't leave this in `FIXTURES`
38//!   permanently.
39//! - [`Source::DynamicField`]: walk a dynamic field of a parent object. Used
40//!   for the versioned inner state (`IotaSystemStateInnerV2` sits as a dynamic
41//!   field of the 0x5 wrapper, keyed by version number).
42//! - [`Source::Event`]: fetch the BCS contents of an event, pinned by the
43//!   digest of the transaction that emitted it plus its fully-qualified type.
44//!   Events are immutable history, so unlike object pins these can never be
45//!   spent or deleted — re-runs always produce the same bytes.
46//! - [`Source::DynamicFieldName`]: capture the BCS of a dynamic field's *name*
47//!   struct (not its value). Used for the small key types that only ever exist
48//!   as dynamic-field names — e.g. `kiosk::Item`/`Listing`/`Lock`, which key
49//!   the items, listings and locks stored on a `Kiosk`. Takes the first field
50//!   on the parent whose name type matches, so the bytes are stable as long as
51//!   the parent's dynamic fields are unchanged.
52//!
53//! Run this manually whenever you want to refresh the fixtures
54//! against current chain state — there is no automated CI job that
55//! re-runs it.
56

64/// What to capture and where to put it.
65struct Fixture {
66    /// Output file name (without extension). Written to
67    /// `crates/iota-sdk-move-types/tests/fixtures/<file>.bcs`.
68    file: &'static str,
69    /// How to find an instance.
70    source: Source,
71}
72
73enum Source {
74    /// Fetch the singleton object at the given address.
75    ObjectId(&'static str),
76    /// Find any object whose top-level type tag matches the given
77    /// fully-qualified Move type string. *Not used in committed
78    /// fixtures* — non-deterministic, only useful locally to find a
79    /// candidate to pin.
80    TypeFilter(&'static str),
81    /// Walk a dynamic field off the given parent.
82    ///
83    /// `name_bcs` is the BCS-encoded value of the field name (e.g. a
84    /// `u64` epoch number, BCS-encoded as little-endian).
85    DynamicField {
86        parent: &'static str,
87        name_type: &'static str,
88        name_bcs: &'static [u8],
89    },
90    /// Fetch the first event of the given fully-qualified type emitted by
91    /// the given transaction. Events are immutable once the transaction is
92    /// checkpointed, so this pin is deterministic forever.
93    Event {
94        event_type: &'static str,
95        tx_digest: &'static str,
96    },
97    /// Capture the BCS of the *name* of the first dynamic field on `parent`
98    /// whose name type matches `name_type`. Used for key structs that only
99    /// live as dynamic-field names (e.g. `kiosk::Item`).
100    DynamicFieldName {
101        parent: &'static str,
102        name_type: &'static str,
103    },
104}

515        // `TokenPolicy` enforcing an allowlist rule.
516        source: Source::DynamicFieldName {
517            parent: "0xa10b50fb6e9b582eebb2b7b156a68c48489525f26022d6e877f63732292812af",
518            name_type: "0x2::token::RuleKey<0x229b368f5086b9030778f54dd123bb2e8debb8da559658f21f4a6fd11083e7d7::allowlist_rule::Allowlist>",
519        },
520    },
521];
522
523#[tokio::main]
524async fn main() -> Result<(), Box<dyn std::error::Error + Send + Sync + 'static>> {
525    let network = std::env::var("IOTA_NETWORK").unwrap_or_else(|_| "mainnet".to_string());
526    let client = match network.as_str() {
527        "testnet" => Client::new_testnet(),
528        "devnet" => Client::new_devnet(),
529        "mainnet" => Client::new_mainnet(),
530        other => {
531            return Err(format!(
532                "unknown IOTA_NETWORK={other}; expected one of: testnet, devnet, mainnet"
533            )
534            .into());
535        }
536    };
537
538    // Discovery mode: when `IOTA_DISCOVER_TYPE` is set to a fully-qualified
539    // Move type tag, find one live object of that type and print its ID so it
540    // can be pinned in FIXTURES as a `Source::ObjectId`. This is a local,
541    // ad-hoc workflow that writes no fixture file.
542    if let Ok(type_str) = std::env::var("IOTA_DISCOVER_TYPE") {
543        // The process exits right after, so leaking the single string that the
544        // `Source::TypeFilter` borrow needs is harmless.
545        let type_str: &'static str = Box::leak(type_str.into_boxed_str());
546        let bytes = capture(&client, &Source::TypeFilter(type_str)).await?;
547        eprintln!(
548            "  ({} bytes; not written — pin the discovered ID above)",
549            bytes.len()
550        );
551        return Ok(());
552    }
553
554    // The fixtures belong to `iota-sdk-move-types`; this generator is a
555    // standalone package under that crate's `examples/` to keep its
556    // GraphQL/`tokio` dependencies out of the crate's own graph.
557    let out_dir = PathBuf::from(env!("CARGO_MANIFEST_DIR")).join("../../tests/fixtures");
558    std::fs::create_dir_all(&out_dir)?;
559
560    eprintln!("network: {network}");
561    eprintln!("writing fixtures to {}", out_dir.display());
562    eprintln!();
563
564    let mut failures = 0;
565    for fixture in FIXTURES {
566        let path = out_dir.join(format!("{}.bcs", fixture.file));
567        match capture(&client, &fixture.source).await {
568            Ok(bytes) => {
569                std::fs::write(&path, &bytes)?;
570                eprintln!(
571                    "  {:32} {:>6} bytes  →  {}",
572                    fixture.file,
573                    bytes.len(),
574                    path.display()
575                );
576            }
577            Err(e) => {
578                failures += 1;
579                eprintln!("  {:32} FAILED: {e}", fixture.file);
580            }
581        }
582    }
583
584    eprintln!();
585    if failures > 0 {
586        eprintln!(
587            "{failures} of {} fixtures failed to capture; a pinned object may have been spent/deleted",
588            FIXTURES.len()
589        );
590        std::process::exit(1);
591    }
592
593    eprintln!("captured {} fixtures", FIXTURES.len());
594    Ok(())
595}

597/// Resolve a [`Source`] into the raw BCS bytes of the Move struct
598/// contents.
599async fn capture(
600    client: &Client,
601    source: &Source,
602) -> Result<Vec<u8>, Box<dyn std::error::Error + Send + Sync + 'static>> {
603    match source {
604        Source::TypeFilter(type_str) => {
605            let filter = iota_sdk::graphql_client::query_types::ObjectFilter::default()
606                .with_type((*type_str).to_string());
607            let page = client
608                .objects(
609                    filter,
610                    iota_sdk::graphql_client::PaginationFilter::default(),
611                )
612                .await?;
613            let object = page
614                .data()
615                .first()
616                .ok_or_else(|| format!("no objects of type `{type_str}` on this network"))?;
617            // Print the discovered ID so an operator can copy-paste it
618            // into FIXTURES as a `Source::ObjectId` pin.
619            eprintln!("    └─ discovered: {}", object.id());
620            let move_struct = object
621                .as_opt_struct()
622                .ok_or("object is not a Move struct")?;
623            Ok(move_struct.contents().to_vec())
624        }
625        Source::ObjectId(id_str) => {
626            let id: ObjectId = id_str.parse()?;
627            let object = client
628                .object(id, None)
629                .await?
630                .ok_or_else(|| format!("object `{id_str}` not found on this network"))?;
631            let move_struct = object
632                .as_opt_struct()
633                .ok_or("object is not a Move struct")?;
634            Ok(move_struct.contents().to_vec())
635        }
636        Source::Event {
637            event_type,
638            tx_digest,
639        } => {
640            use base64ct::Encoding as _;
641            let filter = iota_sdk::graphql_client::query_types::EventFilter::default()
642                .with_event_type((*event_type).to_string())
643                .with_transaction_digest((*tx_digest).to_string());
644            let page = client
645                .events(
646                    filter,
647                    iota_sdk::graphql_client::PaginationFilter::default(),
648                )
649                .await?;
650            let event = page
651                .data()
652                .first()
653                .ok_or_else(|| format!("no `{event_type}` event in transaction `{tx_digest}`"))?;
654            Ok(base64ct::Base64::decode_vec(&event.bcs.0)?)
655        }
656        Source::DynamicField {
657            parent,
658            name_type,
659            name_bcs,
660        } => {
661            let parent_addr: Address = parent.parse::<ObjectId>()?.into();
662            let type_tag: TypeTag = name_type.parse()?;
663            let df = client
664                .dynamic_field(
665                    parent_addr,
666                    type_tag,
667                    iota_sdk::graphql_client::BcsName(name_bcs.to_vec()),
668                )
669                .await?
670                .ok_or_else(|| format!("dynamic field `{name_type}` on `{parent}` not found"))?;
671            let value = df
672                .value
673                .ok_or("dynamic field has no value (was it removed?)")?;
674            Ok(value.bcs)
675        }
676        Source::DynamicFieldName { parent, name_type } => {
677            let parent_addr: Address = parent.parse::<ObjectId>()?.into();
678            let want: TypeTag = name_type.parse()?;
679            let page = client
680                .dynamic_fields(
681                    parent_addr,
682                    iota_sdk::graphql_client::PaginationFilter::default(),
683                )
684                .await?;
685            let df = page
686                .data()
687                .iter()
688                .find(|df| df.name.type_tag == want)
689                .ok_or_else(|| {
690                    format!("no dynamic field of name type `{name_type}` on `{parent}`")
691                })?;
692            Ok(df.name.bcs.clone())
693        }
694    }
695}
