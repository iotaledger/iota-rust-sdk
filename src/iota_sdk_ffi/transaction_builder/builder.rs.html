29/// A builder for creating transactions. Use `finish` to finalize the
30/// transaction data.
31#[derive(Debug, derive_more::From, uniffi::Object)]
32#[uniffi::export(Debug)]
33pub struct TransactionBuilder(RwLock<iota_sdk::transaction_builder::TransactionBuilder<()>>);
34
35impl TransactionBuilder {
36    fn read<F, T>(&self, f: F) -> T
37    where
38        F: FnOnce(&iota_sdk::transaction_builder::TransactionBuilder<()>) -> T,
39    {
40        let lock = self.0.read().expect("error reading from builder");
41        f(&lock)
42    }
43
44    fn write<F, T>(&self, f: F) -> T
45    where
46        F: FnOnce(&mut iota_sdk::transaction_builder::TransactionBuilder<()>) -> T,
47    {
48        let mut lock = self.0.write().expect("error writing to builder");
49        f(&mut lock)
50    }
51}

62    /// Reconstruct a transaction builder from a finalized transaction.
63    /// Calling `finish` on the returned builder produces a transaction equal
64    /// to the input.
65    ///
66    /// Only programmable transactions are supported; other transaction kinds
67    /// will return an error.
68    #[uniffi::constructor]
69    pub fn from_transaction(transaction: &Transaction) -> Result<Self> {
70        Ok(Self(
71            iota_sdk::transaction_builder::TransactionBuilder::try_from(transaction.0.clone())?
72                .into(),
73        ))
74    }

76    /// Create a transaction builder from a programmable transaction.
77    ///
78    /// The returned builder has the original inputs and commands but no
79    /// sender, gas payment, sponsor, or expiration; the sender defaults to
80    /// the zero address and must be set via `sender` before `finish`
81    /// is called.
82    #[uniffi::constructor]
83    pub fn from_programmable_transaction(ptb: &ProgrammableTransaction) -> Self {
84        Self(iota_sdk::transaction_builder::TransactionBuilder::from(ptb.0.clone()).into())
85    }

135    /// Set the gas station sponsor.
136    #[uniffi::method(default(duration = None, headers = None))]
137    pub fn gas_station_sponsor(
138        self: Arc<Self>,
139        url: String,
140        duration: Option<Duration>,
141        headers: Option<HashMap<String, Vec<String>>>,
142    ) -> Arc<Self> {
143        self.write(|builder| {
144            let b = builder.gas_station_sponsor(url.parse().expect("invalid URL"));
145            if let Some(duration) = duration {
146                b.gas_reservation_duration(duration);
147            }
148            if let Some(headers) = headers {
149                for (name, values) in headers {
150                    for value in values {
151                        b.add_gas_station_header(
152                            name.parse().expect("invalid header name"),
153                            value.parse().expect("invalid header value"),
154                        );
155                    }
156                }
157            }
158        });
159        self
160    }

172    /// Call a Move function with the given arguments.
173    #[uniffi::method(default(type_args = [], arguments = [], names = []))]
174    pub fn move_call(
175        self: Arc<Self>,
176        package: &Address,
177        module: &Identifier,
178        function: &Identifier,
179        arguments: Vec<Arc<PTBArgument>>,
180        type_args: Vec<Arc<TypeTag>>,
181        names: Vec<String>,
182    ) -> Arc<Self> {
183        self.write(|builder| {
184            builder
185                .move_call(**package, &module.as_str(), &function.as_str())
186                .arguments(arguments)
187                .type_tags(type_args.into_iter().map(|v| v.0.clone()))
188                .assign(names);
189        });
190        self
191    }

205    /// Transfer some coins to a recipient address. If multiple coins are
206    /// provided then they will be merged.
207    ///
208    /// The `amount` parameter specifies the quantity in NANOS, where 1 IOTA
209    /// equals 1_000_000_000 NANOS.
210    /// If `amount` is provided, that amount is split from the provided coins
211    /// and sent.
212    /// If `amount` is `None`, the entire coins are transferred.
213    ///
214    /// All provided coins must have the same coin type. Mixing coins of
215    /// different types will result in an error.
216    ///
217    /// If you intend to transfer all provided coins to another address in a
218    /// single transaction, consider using
219    /// `TransactionBuilder::transfer_objects()` instead.
220    #[uniffi::method(default(amount = None))]
221    pub fn send_coins(
222        self: Arc<Self>,
223        coins: Vec<Arc<PTBArgument>>,
224        recipient: &Address,
225        amount: Option<Arc<PTBArgument>>,
226    ) -> Arc<Self> {
227        self.write(|builder| {
228            builder.send_coins::<_, &PTBArgument>(coins, **recipient, amount.as_deref());
229        });
230        self
231    }

246    /// Send coins to multiple recipients, each paired with the amount to
247    /// send.
248    ///
249    /// The amounts specify quantities in the coins' smallest unit (NANOS for
250    /// IOTA coins, where 1 IOTA equals 1_000_000_000 NANOS).
251    ///
252    /// The coins are merged into the first one, the amounts are split off it
253    /// in a single command, and each split coin is transferred to its
254    /// corresponding recipient, with one transfer command per unique
255    /// recipient. The remainder stays in the first coin.
256    ///
257    /// All provided coins must have the same coin type. Mixing coins of
258    /// different types will result in an error.
259    ///
260    /// To pay IOTA directly from the gas coin, use
261    /// `TransactionBuilder::pay_iota()` instead.
262    ///
263    /// For a single recipient, consider using
264    /// `TransactionBuilder::send_coins()` or `TransactionBuilder::send_iota()`
265    /// instead.
266    pub fn pay(self: Arc<Self>, coins: Vec<Arc<PTBArgument>>, payments: Vec<Payment>) -> Arc<Self> {
267        self.write(|builder| {
268            builder.pay(
269                coins,
270                payments.into_iter().map(|p| (**p.recipient, p.amount)),
271            );
272        });
273        self
274    }

276    /// Send IOTA to multiple recipients, each paired with the amount to
277    /// send.
278    ///
279    /// The amounts specify quantities in NANOS, where 1 IOTA equals
280    /// 1_000_000_000 NANOS. They are split off the gas coin in a single
281    /// command, and each split coin is transferred to its corresponding
282    /// recipient, with one transfer command per unique recipient.
283    ///
284    /// To pay with specific coins, or with a coin type other than IOTA, use
285    /// `TransactionBuilder::pay()`. For a single recipient, consider using
286    /// `TransactionBuilder::send_iota()` instead.
287    pub fn pay_iota(self: Arc<Self>, payments: Vec<Payment>) -> Arc<Self> {
288        self.write(|builder| {
289            builder.pay_iota(payments.into_iter().map(|p| (**p.recipient, p.amount)));
290        });
291        self
292    }

308    /// Merge multiple coins into one.
309    ///
310    /// This method combines the balances of multiple coins of the same coin
311    /// type into a single coin. The `primary_coin` will receive the balances
312    /// from all `consumed_coins`. After merging, the `consumed_coins` will
313    /// be consumed and no longer exist.
314    pub fn merge_coins(
315        self: Arc<Self>,
316        primary_coin: &PTBArgument,
317        consumed_coins: Vec<Arc<PTBArgument>>,
318    ) -> Arc<Self> {
319        self.write(|builder| {
320            builder.merge_coins(primary_coin, consumed_coins);
321        });
322        self
323    }

325    /// Divide a coin into `count` coins of equal value, all kept by the
326    /// sender.
327    ///
328    /// Unlike `split_coins`, the new coins are transferred to the sender by
329    /// `0x2::pay::divide_and_keep` itself, so no transfer command is needed
330    /// for them. In exchange they are not available as command results and
331    /// cannot be used by later commands in the same transaction.
332    ///
333    /// The coin defaults an IOTA coin. For any other coin type, set it
334    /// with `coin_type`, which is the `T` of `0x2::coin::Coin<T>`.
335    ///
336    /// `count - 1` new coins are created, each holding `value / count`, and
337    /// the divided coin keeps its own share plus the remainder of the
338    /// division. The transaction aborts if `count` is zero or larger than the
339    /// coin's value.
340    ///
341    /// The coin is passed by reference, so the gas coin
342    /// (`PTBArgument::Gas`) can be divided as well,
343    /// as long as it retains enough balance to pay for the transaction.
344    #[uniffi::method(default(coin_type = None))]
345    pub fn divide_coin(
346        self: Arc<Self>,
347        coin: &PTBArgument,
348        count: u64,
349        coin_type: Option<Arc<TypeTag>>,
350    ) -> Arc<Self> {
351        self.write(|builder| {
352            let builder = builder.divide_coin(coin, count);
353            if let Some(coin_type) = coin_type {
354                builder.coin_type_tag(coin_type.0.clone());
355            }
356        });
357        self
358    }

360    /// Make a move vector from a list of elements. The elements must all be of
361    /// the type indicated by `type_tag`.
362    pub fn make_move_vec(
363        self: Arc<Self>,
364        elements: Vec<Arc<MoveArg>>,
365        type_tag: &TypeTag,
366        name: String,
367    ) -> Arc<Self> {
368        use iota_sdk::transaction_builder::unresolved::{Command, MakeMoveVector};
369        self.write(|builder| {
370            let cmd = Command::MakeMoveVector(MakeMoveVector::new(
371                Some(type_tag.0.clone()),
372                elements
373                    .iter()
374                    .map(|e| builder.apply_argument(e.as_ref()))
375                    .collect(),
376            ));
377            builder.assigned_command(cmd, name);
378        });
379        self
380    }

382    /// Publish a list of modules with the given dependencies. The result
383    /// assigned to `upgrade_cap_name` is the `0x2::package::UpgradeCap`
384    /// Move type. Note that the upgrade capability needs to be handled
385    /// after this call:
386    ///  - transfer it to the transaction sender or another address
387    ///  - burn it
388    ///  - wrap it for access control
389    ///  - discard the it to make a package immutable
390    ///
391    /// The arguments required for this command are:
392    ///  - `modules`: is the modules' bytecode to be published
393    ///  - `dependencies`: is the list of IDs of the transitive dependencies of
394    ///    the package
395    pub fn publish_package(
396        self: Arc<Self>,
397        package_data: &MovePackageData,
398        upgrade_cap_name: String,
399    ) -> Arc<Self> {
400        self.write(|builder| {
401            builder
402                .publish_package(package_data.0.clone())
403                .upgrade_cap(upgrade_cap_name);
404        });
405        self
406    }

408    /// Authorize a package upgrade by calling
409    /// `0x2::package::authorize_upgrade`.
410    ///
411    /// The result assigned to `upgrade_ticket_name` is the `UpgradeTicket`
412    /// expected by `TransactionBuilder::upgrade()`. The digest is the digest
413    /// of the package being upgraded to (`MovePackageData::digest()`).
414    ///
415    /// For the standard upgrade flow, use
416    /// `TransactionBuilder::upgrade_package()` instead.
417    pub fn authorize_upgrade(
418        self: Arc<Self>,
419        upgrade_capability: &PTBArgument,
420        upgrade_policy: &UpgradePolicy,
421        digest: &Digest,
422        upgrade_ticket_name: String,
423    ) -> Arc<Self> {
424        self.write(|builder| {
425            builder
426                .authorize_upgrade(upgrade_capability, upgrade_policy.as_u8(), **digest)
427                .assign(upgrade_ticket_name);
428        });
429        self
430    }

432    /// Upgrade a Move package.
433    ///
434    ///  - `modules`: is the modules' bytecode for the modules to be published
435    ///  - `dependencies`: is the list of IDs of the transitive dependencies of
436    ///    the package to be upgraded
437    ///  - `package`: is the ID of the current package being upgraded
438    ///  - `ticket`: is the upgrade ticket
439    ///
440    /// To get the ticket, use `TransactionBuilder::authorize_upgrade()`. The
441    /// result assigned to `name` is the `UpgradeReceipt` expected by
442    /// `TransactionBuilder::commit_upgrade()`.
443    #[uniffi::method(default(name = None))]
444    pub fn upgrade(
445        self: Arc<Self>,
446        package_id: &ObjectId,
447        package_data: &MovePackageData,
448        upgrade_ticket: &PTBArgument,
449        name: Option<String>,
450    ) -> Arc<Self> {
451        self.write(|builder| {
452            builder
453                .upgrade(**package_id, package_data.0.clone(), upgrade_ticket)
454                .assign(name);
455        });
456        self
457    }

464        upgrade_capability: &PTBArgument,
465        upgrade_receipt: &PTBArgument,

467        self.write(|builder| {

473    /// Upgrade a Move package.
474    ///
475    /// This is a high-level function which chains
476    /// `TransactionBuilder::authorize_upgrade()`,
477    /// `TransactionBuilder::upgrade()` and
478    /// `TransactionBuilder::commit_upgrade()` using the digest from the
479    /// provided `MovePackageData`. Use the individual functions for custom
480    /// upgrade flows.
481    pub fn upgrade_package(
482        self: Arc<Self>,
483        package_id: &ObjectId,
484        package_data: &MovePackageData,
485        upgrade_capability: &PTBArgument,
486        upgrade_policy: &UpgradePolicy,
487    ) -> Arc<Self> {
488        self.write(|builder| {
489            builder.upgrade_package(
490                **package_id,
491                package_data.0.clone(),
492                upgrade_capability,
493                upgrade_policy.as_u8(),
494            );
495        });
496        self
497    }

518    /// Convert this builder into a transaction.
519    pub fn finish(&self) -> Result<Transaction> {
520        Ok(Transaction(self.read(|builder| builder.clone().finish())?))
521    }

523    /// Execute the transaction using the gas station and return the JSON
524    /// transaction effects. This will fail unless data is set with the
525    /// `gas_station_sponsor` function.
526    ///
527    /// NOTE: These effects are not necessarily compatible with
528    /// `TransactionEffects`
529    pub async fn execute_with_gas_station(
530        &self,
531        signer: &TransactionSigner,
532    ) -> Result<serde_json::Value> {
533        Ok(self
534            .read(|builder| builder.clone().execute_with_gas_station(signer))
535            .await?)
536    }
