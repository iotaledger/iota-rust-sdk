29/// A builder for creating transactions. Use `finish` to finalize the
30/// transaction data.
31#[derive(Debug, derive_more::From, uniffi::Object)]
32#[uniffi::export(Debug)]
33pub struct TransactionBuilder(RwLock<iota_sdk::transaction_builder::TransactionBuilder<()>>);
34
35impl TransactionBuilder {
36    fn read<F, T>(&self, f: F) -> T
37    where
38        F: FnOnce(&iota_sdk::transaction_builder::TransactionBuilder<()>) -> T,
39    {
40        let lock = self.0.read().expect("error reading from builder");
41        f(&lock)
42    }
43
44    fn write<F, T>(&self, f: F) -> T
45    where
46        F: FnOnce(&mut iota_sdk::transaction_builder::TransactionBuilder<()>) -> T,
47    {
48        let mut lock = self.0.write().expect("error writing to builder");
49        f(&mut lock)
50    }
51}

59        Self(iota_sdk::transaction_builder::TransactionBuilder::new(**sender).into())

62    /// Reconstruct a transaction builder from a finalized transaction.
63    /// Calling `finish` on the returned builder produces a transaction equal
64    /// to the input.
65    ///
66    /// Only programmable transactions are supported; other transaction kinds
67    /// will return an error.
68    #[uniffi::constructor]
69    pub fn from_transaction(transaction: &Transaction) -> Result<Self> {
70        Ok(Self(
71            iota_sdk::transaction_builder::TransactionBuilder::try_from(transaction.0.clone())?
72                .into(),
73        ))
74    }
75
76    /// Create a transaction builder from a programmable transaction.
77    ///
78    /// The returned builder has the original inputs and commands but no
79    /// sender, gas payment, sponsor, or expiration; the sender defaults to
80    /// the zero address and must be set via `sender` before `finish`
81    /// is called.
82    #[uniffi::constructor]
83    pub fn from_programmable_transaction(ptb: &ProgrammableTransaction) -> Self {
84        Self(iota_sdk::transaction_builder::TransactionBuilder::from(ptb.0.clone()).into())
85    }

134    /// Set the gas station sponsor.
135    #[uniffi::method(default(duration = None, headers = None))]
136    pub fn gas_station_sponsor(
137        self: Arc<Self>,
138        url: String,
139        duration: Option<Duration>,
140        headers: Option<HashMap<String, Vec<String>>>,
141    ) -> Arc<Self> {
142        self.write(|builder| {
143            let b = builder.gas_station_sponsor(url.parse().expect("invalid URL"));
144            if let Some(duration) = duration {
145                b.gas_reservation_duration(duration);
146            }
147            if let Some(headers) = headers {
148                for (name, values) in headers {
149                    for value in values {
150                        b.add_gas_station_header(
151                            name.parse().expect("invalid header name"),
152                            value.parse().expect("invalid header value"),
153                        );
154                    }
155                }
156            }
157        });
158        self
159    }

171    /// Call a Move function with the given arguments.
172    #[uniffi::method(default(type_args = [], arguments = [], names = []))]
173    pub fn move_call(
174        self: Arc<Self>,
175        package: &Address,
176        module: &Identifier,
177        function: &Identifier,
178        arguments: Vec<Arc<PTBArgument>>,
179        type_args: Vec<Arc<TypeTag>>,
180        names: Vec<String>,
181    ) -> Arc<Self> {
182        self.write(|builder| {
183            builder
184                .move_call(**package, &module.as_str(), &function.as_str())
185                .arguments(arguments)
186                .type_tags(type_args.into_iter().map(|v| v.0.clone()))
187                .assign(names);
188        });
189        self
190    }

204    /// Transfer some coins to a recipient address. If multiple coins are
205    /// provided then they will be merged.
206    ///
207    /// The `amount` parameter specifies the quantity in NANOS, where 1 IOTA
208    /// equals 1_000_000_000 NANOS.
209    /// If `amount` is provided, that amount is split from the provided coins
210    /// and sent.
211    /// If `amount` is `None`, the entire coins are transferred.
212    ///
213    /// All provided coins must have the same coin type. Mixing coins of
214    /// different types will result in an error.
215    ///
216    /// If you intend to transfer all provided coins to another address in a
217    /// single transaction, consider using
218    /// `TransactionBuilder::transfer_objects()` instead.
219    #[uniffi::method(default(amount = None))]
220    pub fn send_coins(
221        self: Arc<Self>,
222        coins: Vec<Arc<PTBArgument>>,
223        recipient: &Address,
224        amount: Option<Arc<PTBArgument>>,
225    ) -> Arc<Self> {
226        self.write(|builder| {
227            builder.send_coins::<_, &PTBArgument>(coins, **recipient, amount.as_deref());
228        });
229        self
230    }

245    /// Send coins to multiple recipients, each paired with the amount to
246    /// send.
247    ///
248    /// The amounts specify quantities in the coins' smallest unit (NANOS for
249    /// IOTA coins, where 1 IOTA equals 1_000_000_000 NANOS).
250    ///
251    /// The coins are merged into the first one, the amounts are split off it
252    /// in a single command, and each split coin is transferred to its
253    /// corresponding recipient, with one transfer command per unique
254    /// recipient. The remainder stays in the first coin.
255    ///
256    /// All provided coins must have the same coin type. Mixing coins of
257    /// different types will result in an error.
258    ///
259    /// To pay IOTA directly from the gas coin, use
260    /// `TransactionBuilder::pay_iota()` instead.
261    ///
262    /// For a single recipient, consider using
263    /// `TransactionBuilder::send_coins()` or `TransactionBuilder::send_iota()`
264    /// instead.
265    pub fn pay(self: Arc<Self>, coins: Vec<Arc<PTBArgument>>, payments: Vec<Payment>) -> Arc<Self> {
266        self.write(|builder| {
267            builder.pay(
268                coins,
269                payments.into_iter().map(|p| (**p.recipient, p.amount)),
270            );
271        });
272        self
273    }

275    /// Send IOTA to multiple recipients, each paired with the amount to
276    /// send.
277    ///
278    /// The amounts specify quantities in NANOS, where 1 IOTA equals
279    /// 1_000_000_000 NANOS. They are split off the gas coin in a single
280    /// command, and each split coin is transferred to its corresponding
281    /// recipient, with one transfer command per unique recipient.
282    ///
283    /// To pay with specific coins, or with a coin type other than IOTA, use
284    /// `TransactionBuilder::pay()`. For a single recipient, consider using

286    pub fn pay_iota(self: Arc<Self>, payments: Vec<Payment>) -> Arc<Self> {
287        self.write(|builder| {
288            builder.pay_iota(payments.into_iter().map(|p| (**p.recipient, p.amount)));
289        });
290        self
291    }

324    /// Make a move vector from a list of elements. The elements must all be of
325    /// the type indicated by `type_tag`.
326    pub fn make_move_vec(
327        self: Arc<Self>,
328        elements: Vec<Arc<MoveArg>>,
329        type_tag: &TypeTag,
330        name: String,
331    ) -> Arc<Self> {
332        use iota_sdk::transaction_builder::unresolved::{Command, MakeMoveVector};
333        self.write(|builder| {
334            let cmd = Command::MakeMoveVector(MakeMoveVector {
335                type_tag: Some(type_tag.0.clone()),
336                elements: elements
337                    .iter()
338                    .map(|e| builder.apply_argument(e.as_ref()))
339                    .collect(),
340            });
341            builder.assigned_command(cmd, name);
342        });
343        self
344    }

346    /// Publish a list of modules with the given dependencies. The result
347    /// assigned to `upgrade_cap_name` is the `0x2::package::UpgradeCap`
348    /// Move type. Note that the upgrade capability needs to be handled
349    /// after this call:
350    ///  - transfer it to the transaction sender or another address
351    ///  - burn it
352    ///  - wrap it for access control
353    ///  - discard the it to make a package immutable
354    ///
355    /// The arguments required for this command are:
356    ///  - `modules`: is the modules' bytecode to be published
357    ///  - `dependencies`: is the list of IDs of the transitive dependencies of
358    ///    the package
359    pub fn publish_package(
360        self: Arc<Self>,
361        package_data: &MovePackageData,
362        upgrade_cap_name: String,
363    ) -> Arc<Self> {
364        self.write(|builder| {
365            builder
366                .publish_package(package_data.0.clone())
367                .upgrade_cap(upgrade_cap_name);
368        });
369        self
370    }

372    /// Authorize a package upgrade by calling
373    /// `0x2::package::authorize_upgrade`.
374    ///
375    /// The result assigned to `upgrade_ticket_name` is the `UpgradeTicket`
376    /// expected by `TransactionBuilder::upgrade()`. The digest is the digest
377    /// of the package being upgraded to (`MovePackageData::digest()`).
378    ///
379    /// For the standard upgrade flow, use
380    /// `TransactionBuilder::upgrade_package()` instead.
381    pub fn authorize_upgrade(
382        self: Arc<Self>,
383        upgrade_capability: &PTBArgument,
384        upgrade_policy: &UpgradePolicy,
385        digest: &Digest,
386        upgrade_ticket_name: String,
387    ) -> Arc<Self> {
388        self.write(|builder| {
389            builder
390                .authorize_upgrade(upgrade_capability, upgrade_policy.as_u8(), **digest)
391                .assign(upgrade_ticket_name);
392        });
393        self
394    }

396    /// Upgrade a Move package.
397    ///
398    ///  - `modules`: is the modules' bytecode for the modules to be published
399    ///  - `dependencies`: is the list of IDs of the transitive dependencies of
400    ///    the package to be upgraded
401    ///  - `package`: is the ID of the current package being upgraded
402    ///  - `ticket`: is the upgrade ticket
403    ///
404    /// To get the ticket, use `TransactionBuilder::authorize_upgrade()`. The
405    /// result assigned to `name` is the `UpgradeReceipt` expected by
406    /// `TransactionBuilder::commit_upgrade()`.
407    #[uniffi::method(default(name = None))]
408    pub fn upgrade(
409        self: Arc<Self>,
410        package_id: &ObjectId,
411        package_data: &MovePackageData,
412        upgrade_ticket: &PTBArgument,
413        name: Option<String>,
414    ) -> Arc<Self> {
415        self.write(|builder| {
416            builder
417                .upgrade(**package_id, package_data.0.clone(), upgrade_ticket)
418                .assign(name);
419        });
420        self
421    }

423    /// Commit a package upgrade by calling `0x2::package::commit_upgrade`,
424    /// consuming the `UpgradeReceipt` returned by
425    /// `TransactionBuilder::upgrade()`.
426    pub fn commit_upgrade(
427        self: Arc<Self>,
428        upgrade_capability: &PTBArgument,
429        upgrade_receipt: &PTBArgument,
430    ) -> Arc<Self> {
431        self.write(|builder| {
432            builder.commit_upgrade(upgrade_capability, upgrade_receipt);

434        self

437    /// Upgrade a Move package.
438    ///
439    /// This is a high-level function which chains
440    /// `TransactionBuilder::authorize_upgrade()`,
441    /// `TransactionBuilder::upgrade()` and
442    /// `TransactionBuilder::commit_upgrade()` using the digest from the
443    /// provided `MovePackageData`. Use the individual functions for custom
444    /// upgrade flows.
445    pub fn upgrade_package(
446        self: Arc<Self>,
447        package_id: &ObjectId,
448        package_data: &MovePackageData,
449        upgrade_capability: &PTBArgument,
450        upgrade_policy: &UpgradePolicy,
451    ) -> Arc<Self> {
452        self.write(|builder| {
453            builder.upgrade_package(
454                **package_id,
455                package_data.0.clone(),
456                upgrade_capability,
457                upgrade_policy.as_u8(),
458            );
459        });
460        self
461    }

482    /// Convert this builder into a transaction.
483    pub fn finish(&self) -> Result<Transaction> {
484        Ok(Transaction(self.read(|builder| builder.clone().finish())?))
485    }

488    /// transaction effects. This will fail unless data is set with the
