4use std::{collections::HashMap, sync::Arc, time::Duration};
5
6use crate::{
7    error::{Result, SdkFfiError},
8    http::HttpClientOptions,
9};
10
11/// The IOTA gas station, sponsoring transactions over its HTTP API.
12///
13/// Build one and pass it to a transaction builder's `execute_with_gas_station`.
14#[derive(Debug, derive_more::From, uniffi::Object)]
15#[uniffi::export(Debug)]
16pub struct GasStation(pub(crate) iota_sdk::transaction_builder::GasStation);
17
18#[uniffi::export]
19impl GasStation {
20    /// Create a gas station reachable at `url`.
21    ///
22    /// `reservation_duration` is how long the station holds the gas it
23    /// reserves, defaulting to 60 seconds. `headers` are sent with every
24    /// request, typically to carry an authorization token. `http_options`
25    /// describes the HTTP client the requests go out on.
26    #[uniffi::constructor(default(reservation_duration = None, headers = None, http_options = None))]
27    pub fn new(
28        url: String,
29        reservation_duration: Option<Duration>,
30        headers: Option<HashMap<String, Vec<String>>>,
31        http_options: Option<HttpClientOptions>,
32    ) -> Result<Arc<Self>> {
33        let mut builder = iota_sdk::transaction_builder::GasStation::builder(
34            url.parse().map_err(SdkFfiError::new)?,
35        );
36
37        if let Some(duration) = reservation_duration {
38            builder = builder.reservation_duration(duration);
39        }
40        if let Some(options) = http_options {
41            builder = builder.http_client(options.build()?);
42        }
43        for (name, values) in headers.into_iter().flatten() {
44            let name: iota_sdk::transaction_builder::HeaderName =
45                name.parse().map_err(SdkFfiError::new)?;
46            for value in values {
47                builder = builder.header(name.clone(), value.parse().map_err(SdkFfiError::new)?);
48            }
49        }
50
51        Ok(Arc::new(Self(builder.build())))
52    }
53
54    /// The URL this station is reached at.
55    pub fn url(&self) -> String {
56        self.0.url().to_string()
57    }
58}
