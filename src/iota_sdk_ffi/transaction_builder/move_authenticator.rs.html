4use std::sync::Arc;
5
6use crate::{
7    error::Result,
8    graphql::client::GraphQLClient,
9    transaction_builder::ptb_arg::PTBArgument,
10    types::{crypto::move_authenticator::MoveAuthenticator, move_core::TypeTag, object::ObjectId},
11};
12
13#[derive(uniffi::Object)]
14pub struct MoveAuthenticatorBuilder(pub iota_sdk::transaction_builder::MoveAuthenticatorBuilder);
15
16#[cfg_attr(not(target_arch = "wasm32"), uniffi::export(async_runtime = "tokio"))]
17#[cfg_attr(target_arch = "wasm32", uniffi::export)]
18impl MoveAuthenticatorBuilder {
19    /// Create a new move authenticator call with the account ID, function
20    /// inputs, and generic types.
21    #[uniffi::constructor]
22    pub fn new(
23        account_id: &ObjectId,
24        call_args: Vec<Arc<PTBArgument>>,
25        type_args: Vec<Arc<TypeTag>>,
26    ) -> MoveAuthenticatorBuilder {
27        Self(
28            iota_sdk::transaction_builder::MoveAuthenticatorBuilder::new(account_id.0)
29                .call_args(call_args)
30                .type_args(type_args.into_iter().map(|v| v.0.clone())),
31        )
32    }
33
34    /// Resolve this move authenticator builder into a `MoveAuthenticator` which
35    /// can be used to execute a transaction.
36    pub async fn finish(&self, client: &GraphQLClient) -> Result<MoveAuthenticator> {
37        Ok(MoveAuthenticator(self.0.clone().finish(client).await?))
38    }
39}
