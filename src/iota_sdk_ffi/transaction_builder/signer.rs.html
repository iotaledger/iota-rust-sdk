26/// The result of an async sign call containing the `UserSignature`.
27#[derive(uniffi::Record)]
28pub struct TransactionSignerFnOutput {
29    signature: Arc<UserSignature>,
30}
31
32/// Defines a type which can sign a transaction asynchronously.
33///
34/// This trait can be implemented downstream to enable signing when using the
35/// `TransactionBuilder::execute` function.
36#[uniffi::export(with_foreign)]
37#[cfg_attr(not(target_arch = "wasm32"), async_trait::async_trait)]
38#[cfg_attr(target_arch = "wasm32", async_trait::async_trait(?Send))]
39pub trait TransactionSignerFn: ThreadSafety + std::fmt::Debug {
40    /// Sign a transaction and return a `UserSignature`.
41    async fn sign(&self, transaction: Arc<Transaction>) -> Result<TransactionSignerFnOutput>;
42}

115/// An async signer implementation which wraps a `TransactionSignerFn`
116/// definition, which can be used to sign a transaction with a callback.
117#[derive(uniffi::Object)]
118pub struct TransactionSigner(Arc<dyn TransactionSignerFn>);
119
120#[cfg_attr(not(target_arch = "wasm32"), uniffi::export(async_runtime = "tokio"))]
121#[cfg_attr(target_arch = "wasm32", uniffi::export)]
122impl TransactionSigner {
123    #[uniffi::constructor]
124    pub fn new(signer_fn: Arc<dyn TransactionSignerFn>) -> Self {
125        Self(signer_fn)
126    }
127
128    #[uniffi::constructor]
129    pub fn from_move_authenticator(auth: Arc<MoveAuthenticator>) -> Self {
130        Self(auth as Arc<_>)
131    }
132
133    pub async fn sign(&self, txn: Arc<Transaction>) -> Result<Arc<UserSignature>> {
134        Ok(self.0.sign(txn).await?.signature)
135    }
136}

162#[derive(Debug)]
163pub struct TransactionSignerError(crate::error::SdkFfiError);
164
165impl std::error::Error for TransactionSignerError {}
166
167impl std::fmt::Display for TransactionSignerError {
168    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
169        let crate::error::SdkFfiError::Generic(s) = &self.0;
170        write!(f, "{s}")
171    }
172}

174impl iota_sdk::transaction_builder::TransactionSigner for TransactionSigner {
175    type Error = TransactionSignerError;
176
177    async fn sign(
178        &self,
179        transaction: &iota_sdk::types::Transaction,
180    ) -> std::result::Result<iota_sdk::types::UserSignature, Self::Error> {
181        Ok(self
182            .sign(Arc::new(transaction.clone().into()))
183            .await
184            .map_err(TransactionSignerError)?
185            .0
186            .clone())
187    }
