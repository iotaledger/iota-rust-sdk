8/// Unique identifier for an Account on the IOTA blockchain.
9///
10/// An `Address` is a 32-byte pseudonymous identifier used to uniquely identify
11/// an account and asset-ownership on the IOTA blockchain. Often, human-readable
12/// addresses are encoded in hexadecimal with a `0x` prefix. For example, this
13/// is a valid IOTA address:
14/// `0x02a212de6a9dfa3a69e22387acfbafbb1a9e591bd9d636e7895dcfc8de05f331`.
15///
16/// # Deriving an Address
17///
18/// Addresses are cryptographically derived from a number of user account
19/// authenticators, the simplest of which is an
20/// `Ed25519PublicKey`.
21///
22/// Deriving an address consists of the Blake2b256 hash of the sequence of bytes
23/// of its corresponding authenticator, prefixed with a domain-separator (except
24/// ed25519, for compatibility reasons). For each other authenticator, this
25/// domain-separator is the single byte-value of its
26/// `SignatureScheme` flag. E.g. `hash(signature schema flag || authenticator
27/// bytes)`.
28///
29/// Each authenticator has a method for deriving its `Address` as well as
30/// documentation for the specifics of how the derivation is done. See
31/// `Ed25519PublicKey::derive_address` for an example.
32///
33/// ## Relationship to ObjectIds
34///
35/// `ObjectId`s and `Address`es share the same 32-byte addressable space but
36/// are derived leveraging different domain-separator values to ensure that,
37/// cryptographically, there won't be any overlap, e.g. there can't be a
38/// valid `Object` who's `ObjectId` is equal to that of the `Address` of a user
39/// account.
40///
41/// # BCS
42///
43/// An `Address`'s BCS serialized form is defined by the following:
44///
45/// ```text
46/// address = 32OCTET
47/// ```
48#[derive(
49    Debug,
50    derive_more::Deref,
51    derive_more::Display,
52    derive_more::From,
53    Eq,
54    Hash,
55    Ord,
56    PartialEq,
57    PartialOrd,
58    uniffi::Object,
59)]
60#[uniffi::export(Debug, Display, Eq, Hash)]
61pub struct Address(pub iota_sdk::types::Address);
62
63#[uniffi::export]
64impl Address {
65    #[uniffi::constructor]
66    pub fn from_bytes(bytes: Vec<u8>) -> Result<Self> {
67        Ok(Self(iota_sdk::types::Address::from_bytes(bytes)?))
68    }
69
70    /// Parses an Address from a full-length hex string (64 hex characters),
71    /// with or without a `0x` prefix. Will return an error if the string is not
72    /// exactly 64 hex characters long (excluding the `0x` prefix).
73    #[uniffi::constructor]
74    pub fn from_hex(hex: &str) -> Result<Self> {
75        Ok(Self(iota_sdk::types::Address::from_hex(hex)?))
76    }
77
78    /// Parses an Address from a full-length hex string (64 hex characters),
79    /// with a mandatory `0x` prefix. Will return an error if the string is not
80    /// exactly 64 hex characters long (excluding the `0x` prefix).
81    #[uniffi::constructor]
82    pub fn from_prefixed_hex(hex: &str) -> Result<Self> {
83        Ok(Self(iota_sdk::types::Address::from_prefixed_hex(hex)?))
84    }
85
86    /// Parses an Address from a full-length hex string (64 hex characters),
87    /// without a `0x` prefix. Will return an error if the string has a `0x`
88    /// prefix or is not exactly 64 hex characters long.
89    #[uniffi::constructor]
90    pub fn from_raw_hex(hex: &str) -> Result<Self> {
91        Ok(Self(iota_sdk::types::Address::from_raw_hex(hex)?))
92    }
93
94    /// Parses an Address from a hex string, with or without a `0x` prefix.
95    /// The string can be of variable length; if it's shorter than 64 hex
96    /// characters, it will be left-padded with `0`s.
97    #[uniffi::constructor]
98    pub fn from_short_hex(hex: &str) -> Result<Self> {
99        Ok(Self(iota_sdk::types::Address::from_short_hex(hex)?))
100    }
101
102    /// Parses an Address from a hex string with a mandatory `0x` prefix.
103    /// The string can be of variable length; if it's shorter than 64 hex
104    /// characters, it will be left-padded with `0`s.
105    #[uniffi::constructor]
106    pub fn from_prefixed_short_hex(hex: &str) -> Result<Self> {
107        Ok(Self(iota_sdk::types::Address::from_prefixed_short_hex(
108            hex,
109        )?))
110    }
111
112    /// Parses an Address from a hex string without a `0x` prefix.
113    /// The string can be of variable length; if it's shorter than 64 hex
114    /// characters, it will be left-padded with `0`s. Will return an error if
115    /// the string has a `0x` prefix.
116    #[uniffi::constructor]
117    pub fn from_raw_short_hex(hex: &str) -> Result<Self> {
118        Ok(Self(iota_sdk::types::Address::from_raw_short_hex(hex)?))
119    }
120
121    #[uniffi::constructor]
122    pub fn random() -> Self {
123        Self(iota_sdk::types::Address::random())
124    }
125
126    pub fn to_bytes(&self) -> Vec<u8> {
127        self.0.as_bytes().to_vec()
128    }
129
130    /// Returns the string representation of this address in hex format with
131    /// `0x` prefix.
132    pub fn to_hex(&self) -> String {
133        self.0.to_hex()
134    }
135
136    /// Returns the string representation of this address in hex format without
137    /// `0x` prefix.
138    pub fn to_raw_hex(&self) -> String {
139        self.0.to_raw_hex()
140    }
141
142    /// Returns the string representation of this address using the
143    /// canonical display, with or without a `0x` prefix.
144    pub fn to_canonical_string(&self, with_prefix: bool) -> String {
145        self.0.to_canonical_string(with_prefix)
146    }
147
148    /// Returns the shortest possible string representation of the address (i.e.
149    /// with leading zeroes trimmed).
150    pub fn to_short_hex(&self) -> String {
151        self.0.to_short_hex()
152    }
153
154    /// Returns the shortest possible string representation of the address (i.e.
155    /// with leading zeroes trimmed), without `0x` prefix.
156    pub fn to_raw_short_hex(&self) -> String {
157        self.0.to_raw_short_hex()
158    }
159
160    /// Returns the next digest in byte-increasing order.
161    pub fn next_lexicographical(&self) -> Self {
162        Self(self.0.next_lexicographical())
163    }
164
165    /// Returns the next digest in byte-increasing order, or `None` if the
166    /// result would overflow.
167    pub fn next_lexicographical_opt(&self) -> Option<Arc<Self>> {
168        self.0.next_lexicographical_opt().map(Self).map(Arc::new)
169    }
170}

172macro_rules! named_address {
173    ($($constant:ident),+ $(,)?) => {
174        paste::paste! {
175            #[uniffi::export]
176            impl Address {$(
177                #[uniffi::constructor]
178                pub const fn [< $constant:lower >]() -> Self {
179                    Self(iota_sdk::types::Address::$constant)
180                }
181            )+}
182        }
183    }
184}
