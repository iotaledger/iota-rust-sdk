4use std::sync::Arc;
5
6use crate::error::Result;
7
8#[derive(derive_more::Deref, derive_more::From, uniffi::Object)]
9pub struct Version(iota_sdk::types::Version);
10
11#[uniffi::export]
12impl Version {
13    /// Create a new Version from a u64 value
14    #[uniffi::constructor]
15    pub const fn from_u64(value: u64) -> Self {
16        Self(iota_sdk::types::Version::from_u64(value))
17    }
18
19    /// Get the underlying u64 value of this version
20    pub const fn as_u64(&self) -> u64 {
21        self.0.as_u64()
22    }
23
24    /// Returns a special version used for congested shared objects:
25    /// `Version::MIN_CONGESTED + suggested_gas_price`,
26    /// where `suggested_gas_price` is embedded into a congested version
27    /// to facilitate a gas price feedback mechanism for transactions
28    /// canceled due to shared object congestion.
29    #[uniffi::constructor]
30    pub fn new_congested_with_suggested_gas_price(suggested_gas_price: u64) -> Result<Self> {
31        Ok(Self(
32            iota_sdk::types::Version::new_congested_with_suggested_gas_price(suggested_gas_price)?,
33        ))
34    }
35
36    /// Check if this version is congested, i.e., the corresponding
37    /// object is the reason for transaction cancellation.
38    pub fn is_congested(&self) -> bool {
39        self.0.is_congested()
40    }
41
42    /// Returns the `suggested_gas_price` embedded in this congested shared
43    /// object version. The `suggested_gas_price` here is used for a
44    /// gas price feedback mechanism for transactions canceled due to
45    /// shared object congestion.
46    pub fn get_congested_version_suggested_gas_price(&self) -> Result<u64> {
47        Ok(self.0.get_congested_version_suggested_gas_price()?)
48    }
49
50    /// Returns a new version that is greater than all versions
51    /// in `inputs`, assuming this operation will not overflow.
52    #[uniffi::constructor]
53    pub fn lamport_increment(inputs: Vec<Arc<Self>>) -> Result<Self> {
54        Ok(Self(iota_sdk::types::Version::lamport_increment(
55            inputs.into_iter().map(|v| v.0),
56        )?))
57    }
58
59    /// Checks if this version is canceled, i.e., the corresponding
60    /// object appears in a canceled transaction.
61    pub fn is_canceled(&self) -> bool {
62        self.0.is_canceled()
63    }
64
65    /// Checks if this version is valid, i.e., the corresponding
66    /// object does not appear in a canceled transaction.
67    pub fn is_valid(&self) -> bool {
68        self.0.is_valid()
69    }
70
71    /// An inclusive lower limit on a valid version.
72    ///
73    /// A valid version means an object, which this version
74    /// is assigned to, does not appear in a canceled transaction.
75    #[uniffi::constructor]
76    pub fn min_valid_incl() -> Self {
77        Self(iota_sdk::types::Version::MIN_VALID_INCL)
78    }
79
80    /// An exclusive upper limit on a valid version: versions
81    /// strictly smaller than this limit are valid versions.
82    ///
83    /// A valid version means an object, which this version
84    /// is assigned to, does not appear in a canceled transaction.
85    /// Versions larger than this value are "special" and
86    /// assigned to objects that appear in canceled transactions.
87    #[uniffi::constructor]
88    pub fn max_valid_excl() -> Self {
89        Self(iota_sdk::types::Version::MAX_VALID_EXCL)
90    }
91
92    /// Special version that is assigned to objects which are accessed
93    /// immutably in a canceled transaction.
94    #[uniffi::constructor]
95    pub fn canceled_read() -> Self {
96        Self(iota_sdk::types::Version::CANCELED_READ)
97    }
98
99    /// Special version that was assigned to congested objects which
100    /// cause transaction cancellations. Note that this special version
101    /// was only used prior to the introduction of a gas price feedback
102    /// mechanism, but it is kept for backward compatibility.
103    #[uniffi::constructor]
104    pub fn congested_prior_to_gas_price_feedback() -> Self {
105        Self(iota_sdk::types::Version::CONGESTED_PRIOR_TO_GAS_PRICE_FEEDBACK)
106    }
107
108    #[uniffi::constructor]
109    pub fn randomness_unavailable() -> Self {
110        Self(iota_sdk::types::Version::RANDOMNESS_UNAVAILABLE)
111    }
112
113    /// Returns the next version, or an error if overflow occurs.
114    pub fn next(&self) -> Result<Self> {
115        Ok(Self(self.0.next()?))
116    }
117
118    /// Returns the previous version, or an error if underflow occurs.
119    pub fn previous(&self) -> Result<Self> {
120        Ok(Self(self.0.previous()?))
121    }
122}
