19/// Flag use to disambiguate the signature schemes supported by IOTA.
20///
21/// # BCS
22///
23/// The BCS serialized form for this type is defined by the following ABNF:
24///
25/// ```text
26/// signature-scheme = ed25519-flag / secp256k1-flag / secp256r1-flag /
27///                    multisig-flag / bls-flag / passkey-auth-flag /
28///                    move-auth-flag
29/// ed25519-flag                    = %d00
30/// secp256k1-flag                  = %d01
31/// secp256r1-flag                  = %d02
32/// multisig-flag                   = %d03
33/// bls-flag                        = %d04
34/// passkey-auth-flag               = %d06
35/// move-auth-flag                  = %d07
36/// ```
37///
38/// Flag `%d05` is reserved: it was formerly used for the now-removed zklogin
39/// authenticator (which was never enabled on chain) and is intentionally
40/// skipped.
41#[derive(Clone, uniffi::Enum)]
42#[repr(u8)]
43pub enum SignatureScheme {
44    Ed25519 = 0x00,
45    Secp256k1 = 0x01,
46    Secp256r1 = 0x02,
47    Multisig = 0x03,
48    Bls12381 = 0x04,
49    PasskeyAuthenticator = 0x06,
50    MoveAuthenticator = 0x07,
51}
52
53impl From<iota_sdk::types::SignatureScheme> for SignatureScheme {
54    fn from(value: iota_sdk::types::SignatureScheme) -> Self {
55        match value {
56            iota_sdk::types::SignatureScheme::Ed25519 => Self::Ed25519,
57            iota_sdk::types::SignatureScheme::Secp256k1 => Self::Secp256k1,
58            iota_sdk::types::SignatureScheme::Secp256r1 => Self::Secp256r1,
59            iota_sdk::types::SignatureScheme::Multisig => Self::Multisig,
60            iota_sdk::types::SignatureScheme::Bls12381 => Self::Bls12381,
61            iota_sdk::types::SignatureScheme::PasskeyAuthenticator => Self::PasskeyAuthenticator,
62            iota_sdk::types::SignatureScheme::MoveAuthenticator => Self::MoveAuthenticator,
63            _ => unimplemented!("a new SignatureScheme variant was added and needs to be handled"),
64        }
65    }
66}
67
68impl From<SignatureScheme> for iota_sdk::types::SignatureScheme {
69    fn from(value: SignatureScheme) -> Self {
70        match value {
71            SignatureScheme::Ed25519 => Self::Ed25519,
72            SignatureScheme::Secp256k1 => Self::Secp256k1,
73            SignatureScheme::Secp256r1 => Self::Secp256r1,
74            SignatureScheme::Multisig => Self::Multisig,
75            SignatureScheme::Bls12381 => Self::Bls12381,
76            SignatureScheme::PasskeyAuthenticator => Self::PasskeyAuthenticator,
77            SignatureScheme::MoveAuthenticator => Self::MoveAuthenticator,
78        }
79    }
80}

82/// A signature from a user
83///
84/// A `UserSignature` is most commonly used to authorize the execution and
85/// inclusion of a transaction to the blockchain.
86///
87/// # BCS
88///
89/// The BCS serialized form for this type is defined by the following ABNF:
90///
91/// ```text
92/// user-signature = bytes ; where the contents of the bytes are defined by
93///                        ; <user-signature-body>
94/// user-signature-body = (%d00 ed25519-signature ed25519-public-key) /
95///                       (%d01 secp256k1-signature secp256k1-public-key) /
96///                       (%d02 secp256r1-signature secp256r1-public-key) /
97///                       (%d03 multisig-aggregated-signature) /
98///                       (%d06 passkey-authenticator) /
99///                       (%d07 move-authenticator)
100/// ```
101///
102/// Note: Due to historical reasons, signatures are serialized slightly
103/// different from the majority of the types in IOTA. In particular if a
104/// signature is ever embedded in another structure it generally is serialized
105/// as `bytes` meaning it has a length prefix that defines the length of
106/// the completely serialized signature.
107#[derive(Debug, derive_more::From, Eq, PartialEq, uniffi::Object)]
108#[uniffi::export(Debug, Eq)]
109pub struct UserSignature(pub iota_sdk::types::UserSignature);
110
111#[uniffi::export]
112impl UserSignature {
113    #[uniffi::constructor]
114    pub fn new_simple(signature: &SimpleSignature) -> Self {
115        Self(iota_sdk::types::UserSignature::Simple(signature.0.clone()))
116    }
117
118    #[uniffi::constructor]
119    pub fn new_multisig(signature: &MultisigAggregatedSignature) -> Self {
120        Self(iota_sdk::types::UserSignature::Multisig(
121            signature.0.clone(),
122        ))
123    }
124
125    #[uniffi::constructor]
126    pub fn new_passkey_authenticator(authenticator: &PasskeyAuthenticator) -> Self {
127        Self(iota_sdk::types::UserSignature::PasskeyAuthenticator(
128            authenticator.0.clone(),

130    }
131
132    #[uniffi::constructor]
133    pub fn new_move_authenticator(authenticator: &MoveAuthenticator) -> Self {
134        Self(iota_sdk::types::UserSignature::MoveAuthenticator(
135            authenticator.0.clone(),
136        ))
137    }
138

151    }

163    pub fn from_base64(base64: String) -> Result<Self> {

175        self.0
176            .as_opt_simple()
177            .cloned()
178            .map(Into::into)

252/// A basic signature
253///
254/// This enumeration defines the set of simple or basic signature schemes
255/// supported by IOTA. Most signature schemes supported by IOTA end up
256/// comprising of a at least one simple signature scheme.
257///
258/// # BCS
259///
260/// The BCS serialized form for this type is defined by the following ABNF:
261///
262/// ```text
263/// simple-signature = bytes ; where the contents of the bytes are defined by
264///                          ; <simple-signature-body>
265/// simple-signature-body = (ed25519-flag ed25519-signature ed25519-public-key) /
266///                         (secp256k1-flag secp256k1-signature secp256k1-public-key) /
267///                         (secp256r1-flag secp256r1-signature secp256r1-public-key)
268/// ```
269///
270/// Note: Due to historical reasons, signatures are serialized slightly
271/// different from the majority of the types in IOTA. In particular if a
272/// signature is ever embedded in another structure it generally is serialized
273/// as `bytes` meaning it has a length prefix that defines the length of
274/// the completely serialized signature.
275#[derive(Debug, derive_more::From, Eq, Hash, PartialEq, uniffi::Object)]
276#[uniffi::export(Debug, Eq, Hash)]
277pub struct SimpleSignature(pub iota_sdk::types::SimpleSignature);
278
