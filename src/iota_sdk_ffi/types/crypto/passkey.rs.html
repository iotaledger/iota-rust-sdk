4use crate::types::{address::Address, crypto::Secp256r1PublicKey, signature::SimpleSignature};

6/// A passkey authenticator.
7///
8/// # BCS
9///
10/// The BCS serialized form for this type is defined by the following ABNF:
11///
12/// ```text
13/// passkey-bcs = bytes               ; where the contents of the bytes are
14///                                   ; defined by <passkey>
15/// passkey     = passkey-flag
16///               bytes               ; passkey authenticator data
17///               client-data-json    ; valid json
18///               simple-signature    ; required to be a secp256r1 signature
19///
20/// client-data-json = string ; valid json
21/// ```
22///
23/// See <https://www.w3.org/TR/webauthn-2/#dictdef-collectedclientdata> for
24/// the required json-schema for the `client-data-json` rule. In addition, IOTA
25/// currently requires that the `CollectedClientData.type` field is required to
26/// be `webauthn.get`.
27///
28/// Note: Due to historical reasons, signatures are serialized slightly
29/// different from the majority of the types in IOTA. In particular if a
30/// signature is ever embedded in another structure it generally is serialized
31/// as `bytes` meaning it has a length prefix that defines the length of
32/// the completely serialized signature.
33#[derive(Debug, derive_more::From, Eq, PartialEq, uniffi::Object)]
34#[uniffi::export(Debug, Eq)]
35pub struct PasskeyAuthenticator(pub iota_sdk::types::PasskeyAuthenticator);
36
37#[uniffi::export]
38impl PasskeyAuthenticator {
39    /// Opaque authenticator data for this passkey signature.
40    ///
41    /// See <https://www.w3.org/TR/webauthn-2/#sctn-authenticator-data>
42    /// for more information on this field.
43    pub fn authenticator_data(&self) -> Vec<u8> {
44        self.0.authenticator_data().to_vec()
45    }
46
47    /// Structured, unparsed, JSON for this passkey signature.
48    ///
49    /// See <https://www.w3.org/TR/webauthn-2/#dictdef-collectedclientdata>
50    /// for more information on this field.
51    pub fn client_data_json(&self) -> String {
52        self.0.client_data_json().to_owned()
53    }
54
55    /// The parsed challenge message for this passkey signature.
56    ///
57    /// This is parsed by decoding the base64url data from the
58    /// `client_data_json.challenge` field.
59    pub fn challenge(&self) -> Vec<u8> {
60        self.0.challenge().to_vec()
61    }
62
63    /// The passkey signature.
64    pub fn signature(&self) -> SimpleSignature {
65        self.0.signature().into()
66    }
67
68    /// The passkey public key.
69    pub fn public_key(&self) -> PasskeyPublicKey {
70        self.0.public_key().into()
71    }
72
73    /// Derive the `Address` of the passkey that produced this authenticator.
74    ///
75    /// See `PasskeyPublicKey::derive_address`.
76    pub fn derive_address(&self) -> Address {
77        self.0.derive_address().into()
78    }
79}

81/// Public key of a `PasskeyAuthenticator`.
82///
83/// This is used to derive the onchain `Address` for a `PasskeyAuthenticator`.
84///
85/// # BCS
86///
87/// The BCS serialized form for this type is defined by the following ABNF:
88///
89/// ```text
90/// passkey-public-key = passkey-flag secp256r1-public-key
91/// ```
92#[derive(Debug, derive_more::From, Eq, PartialEq, uniffi::Object)]
93#[uniffi::export(Debug, Eq)]
94pub struct PasskeyPublicKey(iota_sdk::types::PasskeyPublicKey);
95
96#[uniffi::export]
97impl PasskeyPublicKey {
98    #[uniffi::constructor]
99    pub fn new(public_key: &Secp256r1PublicKey) -> Self {
100        Self(iota_sdk::types::PasskeyPublicKey::new(**public_key))
101    }
102
103    pub fn inner(&self) -> Secp256r1PublicKey {
104        (*self.0.inner()).into()
105    }
106
107    /// Derive an `Address` from this Passkey Public Key
108    ///
109    /// An `Address` can be derived from a `PasskeyPublicKey` by hashing the
110    /// bytes of the `Secp256r1PublicKey` that corresponds to this passkey
111    /// prefixed with the Passkey `SignatureScheme` flag (`0x06`).
112    ///
113    /// `hash( 0x06 || 33-byte secp256r1 public key)`
114    pub fn derive_address(&self) -> Address {
115        self.0.derive_address().into()
116    }
117}
