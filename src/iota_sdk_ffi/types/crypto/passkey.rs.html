6/// A passkey authenticator.
7///
8/// # BCS
9///
10/// The BCS serialized form of this type is specified in
11/// [`bcs-schema.abnf`](https://github.com/iotaledger/iota-rust-sdk/blob/develop/crates/iota-sdk-types/bcs-schema.abnf).
12///
13/// See <https://www.w3.org/TR/webauthn-2/#dictdef-collectedclientdata> for
14/// the required json-schema for the `client-data-json` rule. In addition, IOTA
15/// currently requires that the `CollectedClientData.type` field is required to
16/// be `webauthn.get`.
17///
18/// Note: Due to historical reasons, signatures are serialized slightly
19/// different from the majority of the types in IOTA. In particular if a
20/// signature is ever embedded in another structure it generally is serialized
21/// as `bytes` meaning it has a length prefix that defines the length of
22/// the completely serialized signature.
23#[derive(Debug, derive_more::From, Eq, PartialEq, uniffi::Object)]
24#[uniffi::export(Debug, Eq)]
25pub struct PasskeyAuthenticator(pub iota_sdk::types::PasskeyAuthenticator);
26
27#[uniffi::export]
28impl PasskeyAuthenticator {
29    /// Opaque authenticator data for this passkey signature.
30    ///
31    /// See <https://www.w3.org/TR/webauthn-2/#sctn-authenticator-data>
32    /// for more information on this field.
33    pub fn authenticator_data(&self) -> Vec<u8> {
34        self.0.authenticator_data().to_vec()
35    }
36
37    /// Structured, unparsed, JSON for this passkey signature.
38    ///
39    /// See <https://www.w3.org/TR/webauthn-2/#dictdef-collectedclientdata>
40    /// for more information on this field.
41    pub fn client_data_json(&self) -> String {
42        self.0.client_data_json().to_owned()
43    }
44
45    /// The parsed challenge message for this passkey signature.
46    ///
47    /// This is parsed by decoding the base64url data from the
48    /// `client_data_json.challenge` field.
49    pub fn challenge(&self) -> Vec<u8> {
50        self.0.challenge().to_vec()
51    }
52
53    /// The passkey signature.
54    pub fn signature(&self) -> SimpleSignature {
55        self.0.signature().into()
56    }
57
58    /// The passkey public key.
59    pub fn public_key(&self) -> PasskeyPublicKey {
60        self.0.public_key().into()
61    }
62
63    /// Derive the `Address` of the passkey that produced this authenticator.
64    ///
65    /// See `PasskeyPublicKey::derive_address`.
66    pub fn derive_address(&self) -> Address {
67        self.0.derive_address().into()
68    }
69}
70
71/// Public key of a `PasskeyAuthenticator`.
72///
73/// This is used to derive the onchain `Address` for a `PasskeyAuthenticator`.
74///
75/// # BCS
76///
77/// The BCS serialized form of this type is specified in
78/// [`bcs-schema.abnf`](https://github.com/iotaledger/iota-rust-sdk/blob/develop/crates/iota-sdk-types/bcs-schema.abnf).
79#[derive(Debug, derive_more::From, Eq, PartialEq, uniffi::Object)]
80#[uniffi::export(Debug, Eq)]
81pub struct PasskeyPublicKey(iota_sdk::types::PasskeyPublicKey);
82
83#[uniffi::export]
84impl PasskeyPublicKey {
85    #[uniffi::constructor]
86    pub fn new(public_key: &Secp256r1PublicKey) -> Self {
87        Self(iota_sdk::types::PasskeyPublicKey::new(**public_key))
88    }
89
90    pub fn inner(&self) -> Secp256r1PublicKey {
91        (*self.0.inner()).into()
92    }
93
94    /// Derive an `Address` from this Passkey Public Key
95    ///
96    /// An `Address` can be derived from a `PasskeyPublicKey` by hashing the
97    /// bytes of the `Secp256r1PublicKey` that corresponds to this passkey
98    /// prefixed with the Passkey `SignatureScheme` flag (`0x06`).
99    ///
100    /// `hash( 0x06 || 33-byte secp256r1 public key)`
101    pub fn derive_address(&self) -> Address {
102        self.0.derive_address().into()
103    }
104}
