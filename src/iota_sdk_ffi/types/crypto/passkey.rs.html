4use crate::types::{address::Address, crypto::Secp256r1PublicKey, signature::SimpleSignature};

6/// A passkey authenticator.
7///
8/// # BCS
9///
10/// The BCS serialized form of this type is specified in
11/// [`bcs-schema.abnf`](https://github.com/iotaledger/iota-rust-sdk/blob/develop/crates/iota-sdk-types/bcs-schema.abnf).
12///
13/// See <https://www.w3.org/TR/webauthn-2/#dictdef-collectedclientdata> for the required json-schema
14/// for the `client-data-json` rule. In addition, IOTA currently requires that the
15/// `CollectedClientData.type` field is required to be `webauthn.get`.
16///
17/// Note: Due to historical reasons, signatures are serialized slightly different from the majority
18/// of the types in IOTA. In particular if a signature is ever embedded in another structure it
19/// generally is serialized as `bytes` meaning it has a length prefix that defines the length of the
20/// completely serialized signature.
21#[derive(Debug, derive_more::From, Eq, PartialEq, uniffi::Object)]
22#[uniffi::export(Debug, Eq)]
23pub struct PasskeyAuthenticator(pub iota_sdk::types::PasskeyAuthenticator);
24
25#[uniffi::export]
26impl PasskeyAuthenticator {
27    /// Opaque authenticator data for this passkey signature.
28    ///
29    /// See <https://www.w3.org/TR/webauthn-2/#sctn-authenticator-data> for more information on this
30    /// field.
31    pub fn authenticator_data(&self) -> Vec<u8> {
32        self.0.authenticator_data().to_vec()
33    }
34
35    /// Structured, unparsed, JSON for this passkey signature.
36    ///
37    /// See <https://www.w3.org/TR/webauthn-2/#dictdef-collectedclientdata> for more information on
38    /// this field.
39    pub fn client_data_json(&self) -> String {
40        self.0.client_data_json().to_owned()
41    }
42
43    /// The parsed challenge message for this passkey signature.
44    ///
45    /// This is parsed by decoding the base64url data from the `client_data_json.challenge` field.
46    pub fn challenge(&self) -> Vec<u8> {
47        self.0.challenge().to_vec()
48    }
49
50    /// The passkey signature.
51    pub fn signature(&self) -> SimpleSignature {
52        self.0.signature().into()
53    }
54
55    /// The passkey public key.
56    pub fn public_key(&self) -> PasskeyPublicKey {
57        self.0.public_key().into()
58    }
59
60    /// Derive the `Address` of the passkey that produced this authenticator.
61    ///
62    /// See `PasskeyPublicKey::derive_address`.
63    pub fn derive_address(&self) -> Address {
64        self.0.derive_address().into()
65    }
66}
67
68/// Public key of a `PasskeyAuthenticator`.
69///
70/// This is used to derive the onchain `Address` for a `PasskeyAuthenticator`.
71///
72/// # BCS
73///
74/// The BCS serialized form of this type is specified in
75/// [`bcs-schema.abnf`](https://github.com/iotaledger/iota-rust-sdk/blob/develop/crates/iota-sdk-types/bcs-schema.abnf).
76#[derive(Debug, derive_more::From, Eq, PartialEq, uniffi::Object)]
77#[uniffi::export(Debug, Eq)]
78pub struct PasskeyPublicKey(iota_sdk::types::PasskeyPublicKey);
79
80#[uniffi::export]
81impl PasskeyPublicKey {
82    #[uniffi::constructor]
83    pub fn new(public_key: &Secp256r1PublicKey) -> Self {
84        Self(iota_sdk::types::PasskeyPublicKey::new(**public_key))
85    }
86
87    pub fn inner(&self) -> Secp256r1PublicKey {
88        (*self.0.inner()).into()
89    }
90
91    /// Derive an `Address` from this Passkey Public Key
92    ///
93    /// An `Address` can be derived from a `PasskeyPublicKey` by hashing the bytes of the
94    /// `Secp256r1PublicKey` that corresponds to this passkey prefixed with the Passkey
95    /// `SignatureScheme` flag (`0x06`).
96    ///
97    /// `hash( 0x06 || 33-byte secp256r1 public key)`
98    pub fn derive_address(&self) -> Address {
99        self.0.derive_address().into()
100    }
101}
