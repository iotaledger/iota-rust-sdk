6/// A move identifier
7///
8/// # BCS
9///
10/// The BCS serialized form for this type is defined by the following ABNF:
11///
12/// ```text
13/// identifier = %d1-128    ; length of the identifier
14///              (ALPHA *127(ALPHA / DIGIT / UNDERSCORE)) /
15///              (UNDERSCORE 1*127(ALPHA / DIGIT / UNDERSCORE))
16///
17/// UNDERSCORE = %x95
18/// ```
19#[derive(Debug, derive_more::Display, derive_more::From, Eq, Hash, PartialEq, uniffi::Object)]
20#[uniffi::export(Debug, Display, Eq, Hash)]
21pub struct Identifier(pub iota_sdk::types::Identifier);
22
23#[uniffi::export]
24impl Identifier {
25    #[uniffi::constructor]
26    pub fn new(identifier: String) -> Result<Self> {
27        Ok(Self(iota_sdk::types::Identifier::new(identifier)?))
28    }
29
30    pub fn as_str(&self) -> String {
31        self.0.as_str().to_owned()
32    }
33}
