8/// Type of a move value
9///
10/// # BCS
11///
12/// The BCS serialized form for this type is defined by the following ABNF:
13///
14/// ```text
15/// type-tag = %d00            ; Bool
16///          / %d01            ; U8
17///          / %d02            ; U64
18///          / %d03            ; U128
19///          / %d04            ; Address
20///          / %d05            ; Signer
21///          / %d06 type-tag   ; Vector
22///          / %d07 struct-tag ; Struct
23///          / %d08            ; U16
24///          / %d09            ; U32
25///          / %d10            ; U256
26/// ```
27#[derive(
28    Debug,
29    derive_more::Display,
30    derive_more::From,
31    Eq,
32    Hash,
33    Ord,
34    PartialEq,
35    PartialOrd,
36    uniffi::Object,
37)]
38#[uniffi::export(Debug, Display, Eq, Hash)]
39pub struct TypeTag(pub iota_sdk::types::TypeTag);
40
41#[uniffi::export]
42impl TypeTag {
43    #[inline]
44    /// Checks if this type tag is a u8.
45    pub fn is_u8(&self) -> bool {
46        self.0.is_u8()
47    }
48
49    #[inline]
50    /// Checks if this type tag is a u16.

52        self.0.is_u16()
53    }
54

58        self.0.is_u32()
59    }
60
61    #[inline]
62    /// Checks if this type tag is a u64.

64        self.0.is_u64()
65    }
66
67    #[inline]
68    /// Checks if this type tag is a u128.

71    }
72
73    #[inline]
74    /// Checks if this type tag is a u256.
75    pub fn is_u256(&self) -> bool {
76        self.0.is_u256()
77    }
78
79    #[inline]
80    /// Checks if this type tag is a boolean.
81    pub fn is_bool(&self) -> bool {
82        self.0.is_bool()
83    }
84
85    #[inline]
86    /// Checks if this type tag is an address.

91    #[inline]

95    }
96
97    #[inline]
98    /// Checks if this type tag is a vector.
99    pub fn is_vector(&self) -> bool {
100        self.0.is_vector()
101    }
102
103    #[inline]
104    /// Converts this type tag into the inner type tag of a vector, if it is
105    /// one, or returns `None` otherwise.
106    pub fn as_opt_vector_type_tag(&self) -> Option<Arc<TypeTag>> {
107        self.0
108            .as_opt_vector_type_tag()
109            .cloned()
110            .map(Into::into)
111            .map(Arc::new)
112    }
113
114    #[inline]
115    /// Converts this type tag into the inner type tag of a vector, if it is
116    /// one, or panics otherwise.
117    pub fn as_vector_type_tag(&self) -> TypeTag {
118        self.0.as_vector_type_tag().clone().into()
119    }
120
121    #[inline]
122    /// Checks if this type tag is a struct.
123    pub fn is_struct(&self) -> bool {
124        self.0.is_struct()
125    }
126
127    #[inline]
128    /// Converts this type tag into a struct tag, if it is one, or returns
129    /// `None` otherwise.
130    pub fn as_opt_struct_tag(&self) -> Option<Arc<StructTag>> {
131        self.0
132            .as_opt_struct_tag()
133            .cloned()
134            .map(Into::into)
135            .map(Arc::new)
136    }
137
138    #[inline]
139    /// Converts this type tag into a struct tag, if it is one, or panics

141    pub fn as_struct_tag(&self) -> StructTag {
142        self.0.as_struct_tag().clone().into()
143    }
