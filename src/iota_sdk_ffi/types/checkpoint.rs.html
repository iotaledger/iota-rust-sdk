22/// A header for a Checkpoint on the IOTA blockchain.
23///
24/// On the IOTA network, checkpoints define the history of the blockchain. They
25/// are quite similar to the concept of blocks used by other blockchains like
26/// Bitcoin or Ethereum. The IOTA blockchain, however, forms checkpoints after
27/// transaction execution has already happened to provide a certified history of
28/// the chain, instead of being formed before execution.
29///
30/// Checkpoints commit to a variety of state including but not limited to:
31/// - The hash of the previous checkpoint.
32/// - The set of transaction digests, their corresponding effects digests, as
33///   well as the set of user signatures which authorized its execution.
34/// - The object's produced by a transaction.
35/// - The set of live objects that make up the current state of the chain.
36/// - On epoch transitions, the next validator committee.
37///
38/// `CheckpointSummary`s themselves don't directly include all of the above
39/// information but they are the top-level type by which all the above are
40/// committed to transitively via cryptographic hashes included in the summary.
41/// `CheckpointSummary`s are signed and certified by a quorum of the validator
42/// committee in a given epoch in order to allow verification of the chain's
43/// state.
44///
45/// # BCS
46///
47/// The BCS serialized form for this type is defined by the following ABNF:
48///
49/// ```text
50/// checkpoint-summary = u64                            ; epoch
51///                      u64                            ; sequence_number
52///                      u64                            ; network_total_transactions
53///                      digest                         ; contents_digest
54///                      (option digest)                ; previous_digest
55///                      gas-cost-summary               ; epoch_rolling_gas_cost_summary
56///                      u64                            ; timestamp_ms
57///                      (vector checkpoint-commitment) ; checkpoint_commitments
58///                      (option end-of-epoch-data)     ; end_of_epoch_data
59///                      bytes                          ; version_specific_data
60/// ```
61#[derive(derive_more::From, uniffi::Object)]
62pub struct CheckpointSummary(pub iota_sdk::types::CheckpointSummary);
63
64#[uniffi::export]
65impl CheckpointSummary {
66    #[uniffi::constructor]
67    #[expect(clippy::too_many_arguments)]
68    pub fn new(
69        epoch: EpochId,
70        sequence_number: CheckpointSequenceNumber,
71        network_total_transactions: u64,
72        contents_digest: &CheckpointContentsDigest,
73        previous_digest: Option<Arc<CheckpointDigest>>,
74        epoch_rolling_gas_cost_summary: GasCostSummary,
75        timestamp_ms: CheckpointTimestamp,
76        checkpoint_commitments: Vec<Arc<CheckpointCommitment>>,
77        end_of_epoch_data: Option<EndOfEpochData>,
78        version_specific_data: Vec<u8>,
79    ) -> Self {
80        Self(iota_sdk::types::CheckpointSummary::new(
81            epoch,
82            sequence_number,
83            network_total_transactions,
84            **contents_digest,
85            previous_digest.map(|v| **v),
86            epoch_rolling_gas_cost_summary.into(),
87            timestamp_ms,
88            checkpoint_commitments
89                .into_iter()
90                .map(|v| v.0.clone())
91                .collect(),
92            end_of_epoch_data.map(Into::into),
93            version_specific_data,
94        ))
95    }
96
97    /// Epoch that this checkpoint belongs to.
98    pub fn epoch(&self) -> u64 {
99        self.0.epoch
100    }
101
102    /// The height of this checkpoint.
103    pub fn sequence_number(&self) -> u64 {
104        self.0.sequence_number
105    }
106
107    /// Total number of transactions committed since genesis, including those in
108    /// this checkpoint.
109    pub fn network_total_transactions(&self) -> u64 {
110        self.0.network_total_transactions
111    }
112
113    /// The hash of the `CheckpointContents` for this checkpoint.
114    pub fn contents_digest(&self) -> CheckpointContentsDigest {
115        self.0.contents_digest.into()
116    }
117
118    /// The hash of the previous `CheckpointSummary`.
119    ///
120    /// This will be only be `None` for the first, or genesis checkpoint.
121    pub fn previous_digest(&self) -> Option<Arc<CheckpointDigest>> {
122        self.0.previous_digest.map(Into::into).map(Arc::new)
123    }
124
125    /// The running total gas costs of all transactions included in the current
126    /// epoch so far until this checkpoint.
127    pub fn epoch_rolling_gas_cost_summary(&self) -> GasCostSummary {
128        self.0.epoch_rolling_gas_cost_summary.clone().into()
129    }
130
131    /// Timestamp of the checkpoint - number of milliseconds from the Unix epoch
132    /// Checkpoint timestamps are monotonic, but not strongly monotonic -
133    /// subsequent checkpoints can have same timestamp if they originate
134    /// from the same underlining consensus commit
135    pub fn timestamp_ms(&self) -> u64 {
136        self.0.timestamp_ms
137    }
138
139    /// Commitments to checkpoint-specific state.
140    pub fn checkpoint_commitments(&self) -> Vec<Arc<CheckpointCommitment>> {
141        self.0
142            .checkpoint_commitments
143            .iter()
144            .cloned()
145            .map(Into::into)
146            .map(Arc::new)
147            .collect()
148    }
149
150    /// Extra data only present in the final checkpoint of an epoch.
151    pub fn end_of_epoch_data(&self) -> Option<EndOfEpochData> {
152        self.0.end_of_epoch_data.clone().map(Into::into)
153    }
154
155    /// CheckpointSummary is not an evolvable structure - it must be readable by
156    /// any version of the code. Therefore, in order to allow extensions to
157    /// be added to CheckpointSummary, we allow opaque data to be added to
158    /// checkpoints which can be deserialized based on the current
159    /// protocol version.
160    pub fn version_specific_data(&self) -> Vec<u8> {
161        self.0.version_specific_data.clone()
162    }
163
164    pub fn digest(&self) -> CheckpointDigest {
165        self.0.digest().into()
166    }
167
168    pub fn signing_message(&self) -> Vec<u8> {
169        self.0.signing_message()
170    }
171
172    pub fn signing_message_hex(&self) -> String {
173        self.0.signing_message_hex()
174    }
175}

201/// The committed to contents of a checkpoint.
202///
203/// `CheckpointContents` contains a list of digests of Transactions, their
204/// effects, and the user signatures that authorized their execution included in
205/// a checkpoint.
206///
207/// # BCS
208///
209/// The BCS serialized form for this type is defined by the following ABNF:
210///
211/// ```text
212/// checkpoint-contents = %d00 checkpoint-contents-v1 ; variant 0
213/// ```
214#[derive(derive_more::From, uniffi::Object)]
215pub struct CheckpointContents(pub iota_sdk::types::CheckpointContents);
216
217#[uniffi::export]
218impl CheckpointContents {
219    #[uniffi::constructor]
220    pub fn new_v1(contents: &CheckpointContentsV1) -> Self {
221        Self(iota_sdk::types::CheckpointContents::new_v1(
222            contents.0.clone(),
223        ))
224    }
225
226    pub fn is_v1(&self) -> bool {
227        self.0.is_v1()
228    }
229
230    pub fn as_v1(&self) -> CheckpointContentsV1 {
231        self.0.as_v1().clone().into()
232    }
233
234    pub fn transactions(&self) -> Vec<Arc<CheckpointTransactionInfo>> {
235        self.0
236            .transactions()
237            .iter()
238            .cloned()
239            .map(Into::into)
240            .map(Arc::new)
241            .collect()
242    }
243
244    pub fn digest(&self) -> CheckpointContentsDigest {
245        self.0.digest().into()
246    }
247
248    /// The number of transactions in this checkpoint.
249    pub fn len(&self) -> u64 {

251    }

254    pub fn is_empty(&self) -> bool {
255        self.0.is_empty()
256    }

259/// CheckpointContents are the transactions included in an upcoming checkpoint.
260/// They must have already been causally ordered. Since the causal order
261/// algorithm is the same among validators, we expect all honest validators to
262/// come up with the same order for each checkpoint content.
263///
264/// # BCS
265///
266/// The BCS serialized form for this type is defined by the following ABNF:
267///
268/// ```text
269/// checkpoint-contents-v1 = (vector execution-digests)      ; transaction and effect digests
270///                          (vector (vector user-signature)) ; set of user signatures for each
271///                                                           ; transaction. MUST be the same
272///                                                           ; length as the vector of digests
273///
274/// execution-digests = transaction-digest transaction-effects-digest   ; transaction, effects
275/// ```
276#[derive(derive_more::From, uniffi::Object)]
277pub struct CheckpointContentsV1(pub iota_sdk::types::CheckpointContentsV1);
278
279#[uniffi::export]
280impl CheckpointContentsV1 {
281    #[uniffi::constructor]
282    pub fn new(transaction_info: Vec<Arc<CheckpointTransactionInfo>>) -> Self {
283        Self(iota_sdk::types::CheckpointContentsV1::new(
284            transaction_info.into_iter().map(|v| v.0.clone()).collect(),
285        ))
286    }
287
288    pub fn transactions(&self) -> Vec<Arc<CheckpointTransactionInfo>> {
289        self.0
290            .transactions()
291            .iter()
292            .cloned()
293            .map(Into::into)
294            .map(Arc::new)
295            .collect()
296    }
297
298    /// The number of transactions in this checkpoint.
299    pub fn len(&self) -> u64 {
300        self.0.len() as _
301    }
302
303    /// Whether this checkpoint has no transactions.
304    pub fn is_empty(&self) -> bool {
305        self.0.is_empty()
306    }
307}

372/// Data which, when included in a [`CheckpointSummary`], signals the end of an
373/// `Epoch`.
374///
375/// # BCS
376///
377/// The BCS serialized form for this type is defined by the following ABNF:
378///
379/// ```text
380/// end-of-epoch-data = (vector validator-committee-member) ; next_epoch_committee
381///                     u64                                 ; next_epoch_protocol_version
382///                     (vector checkpoint-commitment)      ; epoch_commitments
383///                     i64                                 ; epoch_supply_change
384/// ```
385#[derive(Clone, uniffi::Record)]
386pub struct EndOfEpochData {
387    pub next_epoch_committee: Vec<ValidatorCommitteeMember>,
388    pub next_epoch_protocol_version: u64,
389    pub epoch_commitments: Vec<Arc<CheckpointCommitment>>,
390    pub epoch_supply_change: i64,
391}
392
393impl From<iota_sdk::types::EndOfEpochData> for EndOfEpochData {
394    fn from(value: iota_sdk::types::EndOfEpochData) -> Self {
395        Self {
396            next_epoch_committee: value
397                .next_epoch_committee
398                .into_iter()
399                .map(Into::into)
400                .collect(),
401            next_epoch_protocol_version: value.next_epoch_protocol_version,
402            epoch_commitments: value
403                .epoch_commitments
404                .into_iter()
405                .map(Into::into)
406                .map(Arc::new)
407                .collect(),
408            epoch_supply_change: value.epoch_supply_change,
409        }
410    }
411}

430}
