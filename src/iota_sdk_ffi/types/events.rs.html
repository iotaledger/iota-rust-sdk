13/// An event emitted during the successful execution of a transaction.
14///
15/// This mirrors the core chain [`iota_sdk::types::Event`] one-to-one: every
16/// field is required and the type round-trips through BCS/JSON. For events
17/// returned by the GraphQL `events` query — which may originate from system
18/// transactions and therefore lack a sender or emitting module — see
19/// [`GraphQLEvent`](crate::graphql::query_types::GraphQLEvent).
20///
21/// # BCS
22///
23/// The BCS serialized form for this type is defined by the following ABNF:
24///
25/// ```text
26/// event = object-id identifier address struct-tag bytes
27/// ```
28#[derive(Clone, uniffi::Record)]
29pub struct Event {
30    /// Package id of the top-level function invoked by a MoveCall command which
31    /// triggered this event to be emitted.
32    pub package_id: Arc<ObjectId>,
33    /// Module name of the top-level function invoked by a MoveCall command
34    /// which triggered this event to be emitted.
35    pub module: String,
36    /// Address of the account that sent the transaction where this event was
37    /// emitted.
38    pub sender: Arc<Address>,
39    /// The type of the event emitted
40    pub struct_tag: String,
41    /// BCS serialized bytes of the event
42    pub contents: Vec<u8>,
43}
44
45impl TryFrom<Event> for iota_sdk::types::Event {
46    type Error = crate::error::SdkFfiError;
47
48    fn try_from(value: Event) -> Result<Self> {
49        Ok(Self {
50            package_id: (**value.package_id),
51            module: Identifier::from_str(&value.module)?,
52            sender: (**value.sender),
53            struct_tag: StructTag::from_str(&value.struct_tag)?,
54            contents: value.contents,
55        })
56    }
57}
58
59impl From<iota_sdk::types::Event> for Event {
60    fn from(value: iota_sdk::types::Event) -> Self {
61        Self {
62            package_id: Arc::new(value.package_id.into()),
63            module: value.module.to_string(),
64            sender: Arc::new(value.sender.into()),
65            struct_tag: value.struct_tag.to_string(),
66            contents: value.contents,
67        }
68    }
69}

71/// Events emitted during the successful execution of a transaction
72///
73/// # BCS
74///
75/// The BCS serialized form for this type is defined by the following ABNF:
76///
77/// ```text
78/// transaction-events = vector event
79/// ```
80#[derive(derive_more::From, uniffi::Object)]
81pub struct TransactionEvents(pub iota_sdk::types::TransactionEvents);
82
83#[uniffi::export]
84impl TransactionEvents {
85    #[uniffi::constructor]
86    pub fn new(events: Vec<Event>) -> Result<Self> {
87        Ok(Self(iota_sdk::types::TransactionEvents(
88            events
89                .into_iter()
90                .map(TryInto::try_into)
91                .collect::<Result<_>>()?,
92        )))
93    }
94
95    pub fn events(&self) -> Vec<Event> {
96        self.0.0.iter().cloned().map(Into::into).collect()
97    }
98
99    pub fn digest(&self) -> TransactionEventsDigest {
100        self.0.digest().into()
101    }
102}

108impl Event {
109    /// Render this type as human-readable text.
110    ///
111    /// The layout is meant for reading and can change between releases. Use the
112    /// JSON or BCS conversions for output that gets parsed.
113    pub fn to_display_string(&self) -> Result<String> {
114        let data: iota_sdk::types::Event = self.clone().try_into()?;
115        Ok(data.to_string())
116    }
