10/// An object to manage a second-level name (SLN).
11#[derive(Debug, derive_more::From, Eq, PartialEq, uniffi::Object)]
12#[uniffi::export(Debug, Eq)]
13pub struct NameRegistration(iota_sdk::types::iota_names::NameRegistration);
14
15#[uniffi::export]
16impl NameRegistration {
17    #[uniffi::constructor]
18    pub fn new(id: &ObjectId, name: &Name, name_str: String, expiration_timestamp_ms: u64) -> Self {
19        Self(iota_sdk::types::iota_names::NameRegistration::new(
20            **id,
21            name.0.clone(),
22            name_str,
23            expiration_timestamp_ms,
24        ))
25    }
26
27    pub fn id(&self) -> ObjectId {
28        self.0.id().into()
29    }
30
31    pub fn name(&self) -> Name {
32        self.0.name().clone().into()
33    }
34
35    pub fn name_str(&self) -> String {
36        self.0.name_str().to_owned()
37    }
38
39    pub fn expiration_timestamp_ms(&self) -> u64 {
40        self.0.expiration_timestamp_ms()
41    }
42}
43
44#[derive(Debug, derive_more::Display, derive_more::From, Eq, Hash, PartialEq, uniffi::Object)]
45#[uniffi::export(Debug, Display, Eq, Hash)]
46pub struct Name(iota_sdk::types::iota_names::Name);
47
48#[uniffi::export]
49impl Name {
50    #[uniffi::constructor]
51    pub fn from_str(s: &str) -> Result<Self> {
52        Ok(iota_sdk::types::iota_names::Name::from_str(s)?.into())
53    }
54
55    // Derive the parent name for a given name. Only subnames have
56    /// parents; second-level names return `None`.
57    pub fn parent(&self) -> Option<Arc<Self>> {
58        self.0.parent().map(Into::into).map(Arc::new)
59    }
60
61    /// Returns whether this name is a second-level name (Ex. `test.iota`)
62    pub fn is_sln(&self) -> bool {
63        self.num_labels() == 2
64    }
65
66    /// Returns whether this name is a subname (Ex. `sub.test.iota`)
67    pub fn is_subname(&self) -> bool {
68        self.num_labels() >= 3
69    }
70
71    /// Returns the number of labels including TLN.
72    pub fn num_labels(&self) -> u32 {
73        self.0.num_labels() as _
74    }
75
76    /// Get the label at the given index
77    pub fn label(&self, index: u32) -> Option<String> {
78        self.0.label(index as _).cloned()
79    }
80
81    /// Get all of the labels. NOTE: These are in reverse order starting with
82    /// the top-level name and proceeding to subnames.
83    pub fn labels(&self) -> Vec<String> {
84        self.0.labels().to_vec()
85    }
86
87    /// Formats a name into a string based on the available output formats.
88    /// The default separator is `.`
89    pub fn format(&self, format: NameFormat) -> String {
90        self.0.format(format.into())
91    }
92}
93
94/// Two different view options for a name.
95/// `At` -> `test@example` | `Dot` -> `test.example.iota`
96#[derive(Clone, Copy, Debug, uniffi::Enum)]
97pub enum NameFormat {
98    At,
99    Dot,
100}
101
102impl From<NameFormat> for iota_sdk::types::iota_names::NameFormat {
103    fn from(value: NameFormat) -> Self {
104        match value {
105            NameFormat::At => Self::At,
106            NameFormat::Dot => Self::Dot,
107        }
108    }
109}
110
111impl From<iota_sdk::types::iota_names::NameFormat> for NameFormat {
112    fn from(value: iota_sdk::types::iota_names::NameFormat) -> Self {
113        match value {
114            iota_sdk::types::iota_names::NameFormat::At => Self::At,
115            iota_sdk::types::iota_names::NameFormat::Dot => Self::Dot,
116        }
117    }
118}
