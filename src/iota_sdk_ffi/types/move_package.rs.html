11/// Representation of upgrade policy constants in `iota::package`.
12#[derive(Debug, derive_more::Display, derive_more::From, Eq, PartialEq, uniffi::Object)]
13#[uniffi::export(Debug, Display, Eq)]
14pub struct UpgradePolicy(pub iota_sdk::types::UpgradePolicy);
15
16#[uniffi::export]
17impl UpgradePolicy {
18    /// The least restrictive policy. Permits changes to all function
19    /// implementations, the removal of ability constraints on generic type
20    /// parameters in function signatures, and modifications to private,
21    /// public(friend), and entry function signatures. However, public function
22    /// signatures and existing types cannot be changed.
23    #[uniffi::constructor]
24    pub fn compatible() -> Self {
25        Self(iota_sdk::types::UpgradePolicy::Compatible)
26    }
27
28    /// Allows adding new functionalities (e.g., new public functions or
29    /// structs) but restricts changes to existing functionalities.
30    #[uniffi::constructor]
31    pub fn additive() -> Self {
32        Self(iota_sdk::types::UpgradePolicy::Additive)
33    }
34
35    /// Limits modifications to the package’s dependencies only.
36    #[uniffi::constructor]
37    pub fn dep_only() -> Self {
38        Self(iota_sdk::types::UpgradePolicy::DepOnly)
39    }
40
41    /// Returns the internal representation.
42    pub fn as_u8(&self) -> u8 {
43        self.0 as u8
44    }
45}
46
47/// Type corresponding to the output of `iota move build
48/// --dump-bytecode-as-base64`
49#[derive(Debug, derive_more::From, uniffi::Object)]
50#[uniffi::export(Debug)]
51pub struct MovePackageData(pub iota_sdk::types::MovePackageData);
52
53#[uniffi::export]
54impl MovePackageData {
55    #[uniffi::constructor]
56    pub fn new(modules: Vec<Vec<u8>>, dependencies: Vec<Arc<ObjectId>>) -> Self {
57        Self(iota_sdk::types::MovePackageData::new(
58            modules,
59            dependencies.into_iter().map(|o| **o).collect(),
60        ))
61    }
62
63    pub fn to_base64(&self) -> String {
64        self.0.to_base64()
65    }
66
67    #[uniffi::constructor]
68    pub fn from_base64(base64: &str) -> Result<Self> {
69        Ok(Self(iota_sdk::types::MovePackageData::from_base64(base64)?))
70    }
71
72    pub fn to_json(&self) -> String {
73        serde_json::to_string(&self.0).expect("failed to serialize move package data")
74    }
75
76    #[uniffi::constructor]
77    pub fn from_json(json: &str) -> Result<Self> {
78        Ok(Self(serde_json::from_str(json)?))
79    }
80
81    /// Returns the package modules.
82    pub fn modules(&self) -> Vec<Vec<u8>> {
83        self.0.modules.clone()
84    }
85
86    /// Returns the package dependencies.
87    pub fn dependencies(&self) -> Vec<Arc<ObjectId>> {
88        self.0
89            .dependencies
90            .iter()
91            .copied()
92            .map(Into::into)
93            .map(Arc::new)
94            .collect()
95    }
96
97    /// Returns the package digest.
98    pub fn digest(&self) -> Digest {
99        self.0.digest.into()
100    }
101}
