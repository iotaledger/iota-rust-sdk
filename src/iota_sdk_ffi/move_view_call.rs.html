4//! Arguments for Move view function calls, shared by the GraphQL and gRPC
5//! clients.
6
7use std::sync::Arc;
8
9use crate::types::{address::Address, object::ObjectId};
10
11/// An argument for a Move View Function call.
12///
13/// This enum represents the different types of values that can be passed
14/// as arguments to a Move View Function.
15#[derive(Clone, uniffi::Object)]
16pub enum MoveViewArg {
17    /// A boolean value.
18    Bool(bool),
19    /// An unsigned 8-bit integer.
20    U8(u8),
21    /// An unsigned 16-bit integer.
22    U16(u16),
23    /// An unsigned 32-bit integer.
24    U32(u32),
25    /// An unsigned 64-bit integer.
26    U64(u64),
27    /// An unsigned 128-bit integer (as string to avoid precision loss).
28    U128(String),
29    /// A string value.
30    String(String),
31    /// An object ID.
32    ObjectId(iota_sdk::types::ObjectId),
33    /// An address.
34    Address(iota_sdk::types::Address),
35    /// A vector/array of arguments.
36    Vector(Vec<MoveViewArg>),
37    /// An optional value (for Option types).
38    Option(Option<Arc<MoveViewArg>>),
39    /// A raw JSON value (as string, will be parsed).
40    Json(String),
41}
42
43#[uniffi::export]
44impl MoveViewArg {
45    #[uniffi::constructor]
46    pub fn bool(value: bool) -> Self {
47        Self::Bool(value)
48    }
49
50    #[uniffi::constructor]
51    pub fn u8(value: u8) -> Self {
52        Self::U8(value)
53    }
54
55    #[uniffi::constructor]
56    pub fn u16(value: u16) -> Self {
57        Self::U16(value)
58    }
59
60    #[uniffi::constructor]
61    pub fn u32(value: u32) -> Self {
62        Self::U32(value)
63    }
64
65    #[uniffi::constructor]
66    pub fn u64(value: u64) -> Self {
67        Self::U64(value)
68    }
69
70    #[uniffi::constructor]
71    pub fn u128(value: String) -> Self {
72        Self::U128(value)
73    }
74
75    #[uniffi::constructor]
76    pub fn string(value: String) -> Self {
77        Self::String(value)
78    }
79
80    #[uniffi::constructor]
81    pub fn object_id(value: &ObjectId) -> Self {
82        Self::ObjectId(**value)
83    }
84
85    #[uniffi::constructor]
86    pub fn address(value: &Address) -> Self {
87        Self::Address(**value)
88    }
89
90    #[uniffi::constructor]
91    pub fn option(value: Option<Arc<MoveViewArg>>) -> Self {
92        Self::Option(value)
93    }
94
95    #[uniffi::constructor]
96    pub fn null() -> Self {
97        Self::Option(None)
98    }
99
100    #[uniffi::constructor]
101    pub fn json(value: String) -> Self {
102        Self::Json(value)
103    }
104
105    #[uniffi::constructor]
106    pub fn u8_vec(values: Vec<u8>) -> Self {
107        Self::Vector(values.into_iter().map(MoveViewArg::u8).collect())
108    }
109
110    #[uniffi::constructor]
111    pub fn string_vec(values: Vec<String>) -> Self {
112        Self::Vector(values.into_iter().map(MoveViewArg::string).collect())
113    }
114}
115
116impl MoveViewArg {
117    /// Convert this argument to a JSON value.
118    pub(crate) fn to_json(&self) -> serde_json::Value {
119        match self {
120            MoveViewArg::Bool(value) => serde_json::Value::Bool(*value),
121            MoveViewArg::U8(value) => serde_json::Value::Number((*value).into()),
122            MoveViewArg::U16(value) => serde_json::Value::Number((*value).into()),
123            MoveViewArg::U32(value) => serde_json::Value::Number((*value).into()),
124            MoveViewArg::U64(value) => serde_json::Value::String(value.to_string()),
125            MoveViewArg::U128(value) => serde_json::Value::String(value.clone()),
126            MoveViewArg::String(value) => serde_json::Value::String(value.clone()),
127            MoveViewArg::ObjectId(value) => serde_json::Value::String(value.to_string()),
128            MoveViewArg::Address(value) => serde_json::Value::String(value.to_string()),
129            MoveViewArg::Vector(value) => {
130                serde_json::Value::Array(value.iter().map(|v| v.to_json()).collect())
131            }
132            MoveViewArg::Option(value) => match value {
133                Some(v) => v.to_json(),
134                None => serde_json::Value::Null,
135            },
136            MoveViewArg::Json(value) => {
137                serde_json::from_str(value).unwrap_or(serde_json::Value::String(value.clone()))
138            }
139        }
140    }
141}
