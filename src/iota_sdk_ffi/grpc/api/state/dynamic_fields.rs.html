16/// The kind of a dynamic field.
17#[derive(uniffi::Enum)]
18pub enum DynamicFieldKind {
19    /// The kind of the dynamic field is unknown.
20    Unknown,
21    /// A dynamic field.
22    Field,
23    /// A dynamic object field.
24    Object,
25}
26
27impl From<proto::dynamic_field::dynamic_field::DynamicFieldKind> for DynamicFieldKind {
28    fn from(value: proto::dynamic_field::dynamic_field::DynamicFieldKind) -> Self {
29        match value {
30            proto::dynamic_field::dynamic_field::DynamicFieldKind::Field => Self::Field,
31            proto::dynamic_field::dynamic_field::DynamicFieldKind::Object => Self::Object,
32            _ => Self::Unknown,
33        }
34    }
35}

37/// A dynamic field of an object.
38#[derive(uniffi::Record)]
39pub struct DynamicField {
40    /// The kind of the dynamic field.
41    pub kind: Option<DynamicFieldKind>,
42    /// The id of the dynamic field's parent object.
43    pub parent: Option<Arc<ObjectId>>,
44    /// The id of the dynamic field object.
45    pub field_id: Option<Arc<ObjectId>>,
46    /// The dynamic field object itself.
47    pub field_object: Option<Arc<Object>>,
48    /// The BCS representation of the dynamic field's name.
49    pub name_bcs: Option<Vec<u8>>,
50    /// The BCS representation of the dynamic field's value.
51    ///
52    /// For regular dynamic fields this contains the BCS-encoded value whose
53    /// type is given by `value_type`. For dynamic *object* fields this
54    /// contains the BCS-encoded id of the child object; use `child_object`
55    /// to access the full object.
56    pub value_bcs: Option<Vec<u8>>,
57    /// The type of the dynamic field's value.
58    pub value_type: Option<String>,
59    /// The id of the child object when a child is a dynamic object field.
60    pub child_id: Option<Arc<ObjectId>>,
61    /// The object itself when a child is a dynamic object field.
62    pub child_object: Option<Arc<Object>>,
63}
64
65impl TryFrom<&proto::dynamic_field::DynamicField> for DynamicField {
66    type Error = SdkFfiError;
67
68    fn try_from(value: &proto::dynamic_field::DynamicField) -> Result<Self> {
69        Ok(Self {
70            kind: value.kind.map(|kind| {
71                proto::dynamic_field::dynamic_field::DynamicFieldKind::try_from(kind)
72                    .map_or(DynamicFieldKind::Unknown, Into::into)
73            }),
74            parent: value
75                .parent
76                .as_ref()
77                .map(iota_sdk::types::ObjectId::try_from)
78                .transpose()?
79                .map(Into::into)
80                .map(Arc::new),
81            field_id: value
82                .field_id
83                .as_ref()
84                .map(iota_sdk::types::ObjectId::try_from)
85                .transpose()?
86                .map(Into::into)
87                .map(Arc::new),
88            field_object: value
89                .field_object
90                .as_ref()
91                .filter(|object| object.bcs.is_some())
92                .map(|object| object.object().map_err(SdkFfiError::new))
93                .transpose()?
94                .map(Into::into)
95                .map(Arc::new),
96            name_bcs: value.name.as_ref().map(Vec::from),
97            value_bcs: value.value.as_ref().map(Vec::from),
98            value_type: value.value_type.clone(),
99            child_id: value
100                .child_id
101                .as_ref()
102                .map(iota_sdk::types::ObjectId::try_from)
103                .transpose()?
104                .map(Into::into)
105                .map(Arc::new),
106            child_object: value
107                .child_object
108                .as_ref()
109                .filter(|object| object.bcs.is_some())
110                .map(|object| object.object().map_err(SdkFfiError::new))
111                .transpose()?
112                .map(Into::into)
113                .map(Arc::new),
114        })
115    }
116}

118/// A page of dynamic fields returned by the gRPC server.
119#[derive(uniffi::Record)]
120pub struct DynamicFieldPage {
121    /// The dynamic fields returned in the page.
122    pub dynamic_fields: Vec<DynamicField>,
123    /// Token to retrieve the next page. `None` when this is the last page.
124    pub next_page_token: Option<Vec<u8>>,
125}
126
127#[uniffi::export(async_runtime = "tokio")]
128impl GrpcClient {
129    /// List a single page of dynamic fields of an object.
130    ///
131    /// Pass the returned `next_page_token` back in to retrieve the next page.
132    ///
133    /// The optional `read_mask` controls which fields the server returns.
134    /// If `None`, the parent and field id are returned.
135    #[uniffi::method(default(page_size = None, page_token = None, read_mask = None))]
136    pub async fn dynamic_fields(
137        &self,
138        parent: &ObjectId,
139        page_size: Option<u32>,
140        page_token: Option<Vec<u8>>,
141        read_mask: Option<Vec<String>>,
142    ) -> Result<DynamicFieldPage> {
143        let query = self.client().dynamic_fields(
144            **parent,
145            page_size,
146            page_token.map(Into::into),
147            crate::grpc::api::read_mask::<DynamicFieldReadMask>(&read_mask),
148        );
149        let page = query.await?.into_inner();
150        Ok(DynamicFieldPage {
151            dynamic_fields: page
152                .items
153                .iter()
154                .map(TryInto::try_into)
155                .collect::<Result<_>>()?,
156            next_page_token: page.next_page_token.map(|token| token.to_vec()),
157        })
158    }
159
160    /// List all dynamic fields of an object, auto-paginating up to `limit`
161    /// fields. If `limit` is `None`, all fields are returned.
162    ///
163    /// The optional `read_mask` controls which fields the server returns.
164    /// If `None`, the parent and field id are returned.
165    #[uniffi::method(default(limit = None, read_mask = None))]
166    pub async fn all_dynamic_fields(
167        &self,
168        parent: &ObjectId,
169        limit: Option<u32>,
170        read_mask: Option<Vec<String>>,
171    ) -> Result<Vec<DynamicField>> {
172        let query = self.client().dynamic_fields(
173            **parent,
174            None,
175            None,
176            crate::grpc::api::read_mask::<DynamicFieldReadMask>(&read_mask),
177        );
178        query
179            .collect(limit)
180            .await?
181            .into_inner()
182            .iter()
183            .map(TryInto::try_into)
184            .collect()
185    }
186}
