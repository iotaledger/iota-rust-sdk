4//! Owned objects API implementation.
5
6use std::sync::Arc;
7
8use iota_sdk::grpc_client::{ListOwnedObjectsQuery, read_mask_fields::OwnedObjectReadMask};
9
10use crate::{
11    error::Result,
12    grpc::{
13        api::ledger::objects::GrpcObject, client::GrpcClient,
14        read_mask_fields::GrpcOwnedObjectField,
15    },
16    helpers::SetIfSome,
17    types::{address::Address, move_core::StructTag},
18};
19
20/// A page of objects returned by the gRPC server.
21#[derive(uniffi::Record)]
22pub struct GrpcOwnedObjectPage {
23    /// The objects returned in the page.
24    pub objects: Vec<GrpcObject>,
25    /// Token to retrieve the next page. `None` when this is the last page.
26    pub next_page_token: Option<Vec<u8>>,
27}
28
29#[uniffi::export(async_runtime = "tokio")]
30impl GrpcClient {
31    /// List a single page of objects owned by an address, optionally filtered
32    /// by object type.
33    ///
34    /// Pass the returned `next_page_token` back in to retrieve the next page.
35    ///
36    /// The optional `read_mask` controls which fields the server returns.
37    /// If `None`, the object reference and the BCS-encoded object are
38    /// returned.
39    #[uniffi::method(default(
40        object_type = None,
41        page_size = None,
42        page_token = None,
43        read_mask = None
44    ))]
45    pub async fn owned_objects(
46        &self,
47        owner: &Address,
48        object_type: Option<Arc<StructTag>>,
49        page_size: Option<u32>,
50        page_token: Option<Vec<u8>>,
51        read_mask: Option<Vec<GrpcOwnedObjectField>>,
52    ) -> Result<GrpcOwnedObjectPage> {
53        let query = self
54            .client()
55            .owned_objects(**owner)
56            .set_if_some(
57                object_type.map(|object_type| object_type.0.clone()),
58                ListOwnedObjectsQuery::object_type,
59            )
60            .set_if_some(page_size, ListOwnedObjectsQuery::page_size)
61            .set_if_some(
62                page_token.map(Into::into),
63                ListOwnedObjectsQuery::page_token,
64            )
65            .read_mask(crate::grpc::api::read_mask::<OwnedObjectReadMask, _>(
66                read_mask,
67            ));
68        let page = query.await?.into_inner();
69        Ok(GrpcOwnedObjectPage {
70            objects: page
71                .items
72                .iter()
73                .map(TryInto::try_into)
74                .collect::<Result<_>>()?,
75            next_page_token: page.next_page_token.map(|token| token.to_vec()),
76        })
77    }
78
79    /// List all objects owned by an address, optionally filtered by object
80    /// type, auto-paginating up to `limit` objects. If `limit` is `None`,
81    /// all objects are returned.
82    ///
83    /// The optional `read_mask` controls which fields the server returns.
84    /// If `None`, the object reference and the BCS-encoded object are
85    /// returned.
86    #[uniffi::method(default(object_type = None, limit = None, read_mask = None))]
87    pub async fn all_owned_objects(
88        &self,
89        owner: &Address,
90        object_type: Option<Arc<StructTag>>,
91        limit: Option<u32>,
92        read_mask: Option<Vec<GrpcOwnedObjectField>>,
93    ) -> Result<Vec<GrpcObject>> {
94        let query = self
95            .client()
96            .owned_objects(**owner)
97            .set_if_some(
98                object_type.map(|object_type| object_type.0.clone()),
99                ListOwnedObjectsQuery::object_type,
100            )
101            .read_mask(crate::grpc::api::read_mask::<OwnedObjectReadMask, _>(
102                read_mask,
103            ));
104        query
105            .collect(limit)
106            .await?
107            .into_inner()
108            .iter()
109            .map(TryInto::try_into)
110            .collect()
111    }
112}
