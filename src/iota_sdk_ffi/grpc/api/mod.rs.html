4//! High-level API for gRPC client operations.
5//!
6//! The gRPC API lets callers control which fields the server returns via read
7//! masks. Fields that were not requested (or that the server did not populate)
8//! are `None` in the corresponding record.
9//!
10//! Complex types (transactions, effects, events, objects, ...) are eagerly
11//! deserialized from their BCS representation, so the read mask must include
12//! the corresponding `bcs` sub-fields for those record fields to be
13//! populated.
14
15use iota_sdk::grpc_types::field_mask_normalize;
16
17pub mod ledger;
18
19/// Convert an optional list of field paths into an endpoint read mask.
20///
21/// The paths are normalized like the typed field lists of the Rust client
22/// (duplicates and subsumed paths dropped). When nothing is left, or no paths
23/// were given at all, the endpoint's default mask is used.
24pub(crate) fn read_mask<M: Default + From<String>>(paths: &Option<Vec<String>>) -> M {
25    match paths {
26        Some(paths) => {
27            let mask = field_mask_normalize(&paths.join(","));
28            if mask.is_empty() {
29                M::default()
30            } else {
31                M::from(mask)
32            }
33        }
34        None => M::default(),
35    }
36}

38#[cfg(test)]
39mod tests {
40    use iota_sdk::grpc_client::read_mask_fields::EpochReadMask;
41
42    use super::read_mask;
43
44    #[test]
45    fn read_mask_falls_back_to_the_default_without_paths() {
46        let default = EpochReadMask::default();
47        assert_eq!(read_mask::<EpochReadMask>(&None).as_str(), default.as_str());
48        assert_eq!(
49            read_mask::<EpochReadMask>(&Some(vec![])).as_str(),
50            default.as_str()
51        );
52        assert_eq!(
53            read_mask::<EpochReadMask>(&Some(vec![String::new()])).as_str(),
54            default.as_str()
55        );
56    }
57
58    #[test]
59    fn read_mask_normalizes_the_given_paths() {
60        let mask = read_mask::<EpochReadMask>(&Some(vec![
61            "protocol_config.feature_flags".to_owned(),
62            "protocol_config".to_owned(),
63            "epoch".to_owned(),
64            "epoch".to_owned(),
65        ]));
66        assert_eq!(mask.as_str(), "epoch,protocol_config");
67    }
68
69    #[test]
70    fn read_mask_joins_the_given_paths() {
71        let mask = read_mask::<EpochReadMask>(&Some(vec![
72            "epoch".to_owned(),
73            "reference_gas_price".to_owned(),
74        ]));
75        assert_eq!(mask.as_str(), "epoch,reference_gas_price");
76    }
77}
