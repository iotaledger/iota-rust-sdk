30/// Response for a checkpoint query.
31///
32/// Which fields are populated depends on the read mask used for the query;
33/// the default read mask only includes the checkpoint summary, and every
34/// other field is `None`.
35///
36/// The `summary`, `signature`, `contents`, and `events` fields are
37/// deserialized from BCS, so the read mask must include the corresponding
38/// `bcs` sub-fields for them to be populated; digest-only read masks populate
39/// only the digest fields.
40#[derive(uniffi::Record)]
41pub struct CheckpointResponse {
42    /// The checkpoint sequence number. Always available regardless of the
43    /// read mask.
44    pub sequence_number: u64,
45    /// The digest of the checkpoint summary.
46    pub summary_digest: Option<Arc<CheckpointDigest>>,
47    /// The checkpoint summary.
48    pub summary: Option<Arc<CheckpointSummary>>,
49    /// The aggregated validator signature of the checkpoint.
50    pub signature: Option<Arc<ValidatorAggregatedSignature>>,
51    /// The digest of the checkpoint contents.
52    pub contents_digest: Option<Arc<CheckpointContentsDigest>>,
53    /// The checkpoint contents.
54    pub contents: Option<Arc<CheckpointContents>>,
55    /// The transactions executed in the checkpoint. `None` unless the read
56    /// mask requests `transactions` or one of its sub-fields; with a
57    /// transactions filter that matches nothing, an empty list.
58    pub transactions: Option<Vec<ExecutedTransaction>>,
59    /// The events emitted in the checkpoint. `None` unless the read mask
60    /// requests `events.bcs`; a checkpoint with no events, or an events
61    /// filter that matches nothing, yields an empty list.
62    pub events: Option<Vec<Event>>,
63}
64
65impl CheckpointResponse {
66    /// Convert a client response, using the read mask it was requested with
67    /// to tell fields that were not requested from fields that are empty.
68    fn from_response(
69        value: &iota_sdk::grpc_client::CheckpointResponse,
70        read_mask: &CheckpointResponseReadMask,
71    ) -> Result<Self> {
72        let summary = value.summary().ok();
73        let contents = value.contents().ok();
74        Ok(Self {
75            sequence_number: value.sequence_number(),
76            summary_digest: summary
77                .and_then(|summary| summary.digest.as_ref())
78                .map(iota_sdk::types::CheckpointDigest::try_from)
79                .transpose()?
80                .map(Into::into)
81                .map(Arc::new),
82            summary: summary
83                .filter(|summary| summary.bcs.is_some())
84                .map(|summary| summary.summary().map_err(SdkFfiError::new))
85                .transpose()?
86                .map(Into::into)
87                .map(Arc::new),
88            signature: value
89                .signature()
90                .ok()
91                .filter(|signature| signature.bcs.is_some())
92                .map(|signature| signature.signature().map_err(SdkFfiError::new))
93                .transpose()?
94                .map(Into::into)
95                .map(Arc::new),
96            contents_digest: contents
97                .and_then(|contents| contents.digest.as_ref())
98                .map(iota_sdk::types::CheckpointContentsDigest::try_from)
99                .transpose()?
100                .map(Into::into)
101                .map(Arc::new),
102            contents: contents
103                .filter(|contents| contents.bcs.is_some())
104                .map(|contents| contents.contents().map_err(SdkFfiError::new))
105                .transpose()?
106                .map(Into::into)
107                .map(Arc::new),
108            transactions: read_mask_requests(read_mask, CheckpointResponseField::TRANSACTIONS)
109                .then(|| {
110                    value
111                        .executed_transactions()
112                        .iter()
113                        .map(TryInto::try_into)
114                        .collect::<Result<Vec<_>>>()
115                })
116                .transpose()?,
117            events: read_mask_requests(read_mask, CheckpointResponseField::EVENTS_BCS)
118                .then(|| {
119                    value
120                        .events()
121                        .iter()
122                        .map(|event| event.event().map(Into::into).map_err(SdkFfiError::new))
123                        .collect::<Result<Vec<_>>>()
124                })
125                .transpose()?,
126        })
127    }
128}

130/// Define a handle object over a server-streaming checkpoint RPC.
131///
132/// The Rust API exposes these as a `Stream`, which has no uniffi equivalent,
133/// so the handle is pulled one item at a time with `next` and closed with
134/// `cancel`. The handle keeps the read mask the stream was opened with, since
135/// converting an item needs it.
136macro_rules! define_checkpoint_stream {
137    ($(#[$meta:meta])* $name:ident, $item:ty, $ffi_item:ty, $convert:expr) => {
138        $(#[$meta])*
139        ///
140        /// Call `next` in a loop to receive items; it returns `None` once the
141        /// stream is exhausted or `cancel` has been called.
142        #[derive(uniffi::Object)]
143        pub struct $name {
144            stream: StreamHandle<BoxStream<'static, GrpcResult<$item>>>,
145            read_mask: CheckpointResponseReadMask,
146        }
147
148        #[uniffi::export(async_runtime = "tokio")]
149        impl $name {
150            /// Get the next item from the stream.
151            ///
152            /// Returns `None` once the stream is exhausted or has been
153            /// canceled. Concurrent calls are serialized; there is no ordering
154            /// guarantee between them.
155            ///
156            /// An error from the connection or the server ends the stream:
157            /// there is no reconnect, so every later call returns `None`, and
158            /// a caller that wants to resume must open a new stream from the
159            /// last sequence number it received. An error converting a single
160            /// item, such as a BCS decode failure, only affects that item; the
161            /// next call continues with the following one.
162            pub async fn next(&self) -> Result<Option<$ffi_item>> {
163                self.stream
164                    .next()
165                    .await
166                    .transpose()?
167                    .map(|item| ($convert)(item, &self.read_mask))
168                    .transpose()
169            }
170
171            /// Cancel the stream, dropping the connection and unblocking a
172            /// pending `next`.
173            ///
174            /// Idempotent, and safe to call while `next` is pending — the
175            /// pending call drops the connection on its way out.
176            ///
177            /// Named `cancel` rather than `close` because a `close` method
178            /// collides with the disposal method uniffi generates for objects
179            /// in some languages.
180            pub fn cancel(&self) {
181                self.stream.cancel();
182            }
183
184            /// Whether the stream has been canceled.
185            pub fn is_canceled(&self) -> bool {
186                self.stream.is_canceled()
187            }
188        }
189
190        impl $name {
191            fn new(
192                stream: BoxStream<'static, GrpcResult<$item>>,
193                read_mask: CheckpointResponseReadMask,
194            ) -> Self {
195                Self {
196                    stream: StreamHandle::new(stream),
197                    read_mask,
198                }
199            }
200        }
201    };
202}

206    CheckpointStream,

212/// An item yielded by a filtered checkpoint stream: either a checkpoint with
213/// matching data, or a progress indicator emitted while the server scans
214/// checkpoints that have none.
215#[derive(uniffi::Enum)]
216pub enum CheckpointStreamItem {
217    /// A complete checkpoint with its transactions and events.
218    Checkpoint { checkpoint: CheckpointResponse },
219    /// A progress indicator sent during filtered scanning, carrying the
220    /// sequence number of the latest scanned checkpoint.
221    Progress { latest_scanned_sequence_number: u64 },
222}
223
224impl CheckpointStreamItem {
225    fn from_item(
226        value: iota_sdk::grpc_client::CheckpointStreamItem,
227        read_mask: &CheckpointResponseReadMask,
228    ) -> Result<Self> {
229        Ok(match value {
230            iota_sdk::grpc_client::CheckpointStreamItem::Checkpoint(checkpoint) => {
231                Self::Checkpoint {
232                    checkpoint: CheckpointResponse::from_response(&checkpoint, read_mask)?,
233                }
234            }
235            iota_sdk::grpc_client::CheckpointStreamItem::Progress {
236                latest_scanned_sequence_number,
237            } => Self::Progress {
238                latest_scanned_sequence_number,
239            },
240            _ => {
241                return Err(SdkFfiError::custom(
242                    "unsupported checkpoint stream item variant",
243                ));
244            }
245        })
246    }
247}

262    /// The optional `transactions_filter` and `events_filter` narrow the
263    /// transactions and events returned for the checkpoint.

389    /// Stream checkpoints across a range of sequence numbers, skipping

406    /// each checkpoint. If `None`, only the checkpoint summary is returned.

408        start_sequence_number = None,
409        end_sequence_number = None,

413        read_mask = None
