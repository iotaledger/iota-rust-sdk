33/// Response for a checkpoint query.
34///
35/// Which fields are populated depends on the read mask used for the query;
36/// the default read mask only includes the checkpoint summary, and every
37/// other field is `None`.
38///
39/// The `summary`, `signature`, `contents` and `events` fields are
40/// deserialized from BCS, so the read mask must include
41/// `CheckpointSummaryBcs`, `CheckpointSignature`, `CheckpointContentsBcs` and
42/// `EventsBcs` respectively for them to be populated; digest-only read masks
43/// populate only the digest fields.
44#[derive(uniffi::Record)]
45pub struct GrpcCheckpointResponse {
46    /// The checkpoint sequence number. Always available regardless of the
47    /// read mask.
48    pub sequence_number: u64,
49    /// The digest of the checkpoint summary.
50    pub summary_digest: Option<Arc<CheckpointDigest>>,
51    /// The checkpoint summary.
52    pub summary: Option<Arc<CheckpointSummary>>,
53    /// The aggregated validator signature of the checkpoint.
54    pub signature: Option<Arc<ValidatorAggregatedSignature>>,
55    /// The digest of the checkpoint contents.
56    pub contents_digest: Option<Arc<CheckpointContentsDigest>>,
57    /// The checkpoint contents.
58    pub contents: Option<Arc<CheckpointContents>>,
59    /// The transactions executed in the checkpoint. `None` unless the read
60    /// mask requests `Transactions` or one of its sub-fields (e.g.
61    /// `TransactionsEffectsBcs`); with a transactions filter that matches
62    /// nothing, an empty list.
63    pub transactions: Option<Vec<GrpcExecutedTransaction>>,
64    /// The events emitted in the checkpoint. `None` unless the read mask
65    /// requests `EventsBcs` (or `Events`); a checkpoint with no events, or an
66    /// events filter that matches nothing, yields an empty list.
67    pub events: Option<Vec<Event>>,
68}
69
70impl GrpcCheckpointResponse {
71    /// Convert a client response, using the read mask it was requested with
72    /// to tell fields that were not requested from fields that are empty.
73    fn from_response(
74        value: &iota_sdk::grpc_client::CheckpointResponse,
75        read_mask: &CheckpointResponseReadMask,
76    ) -> Result<Self> {
77        let summary = value.summary().ok();
78        let contents = value.contents().ok();
79        Ok(Self {
80            sequence_number: value.sequence_number(),
81            summary_digest: summary
82                .and_then(|summary| summary.digest.as_ref())
83                .map(iota_sdk::types::CheckpointDigest::try_from)
84                .transpose()?
85                .map(Into::into)
86                .map(Arc::new),
87            summary: summary
88                .filter(|summary| summary.bcs.is_some())
89                .map(|summary| summary.summary().map_err(SdkFfiError::new))
90                .transpose()?
91                .map(Into::into)
92                .map(Arc::new),
93            signature: value
94                .signature()
95                .ok()
96                .filter(|signature| signature.bcs.is_some())
97                .map(|signature| signature.signature().map_err(SdkFfiError::new))
98                .transpose()?
99                .map(Into::into)
100                .map(Arc::new),
101            contents_digest: contents
102                .and_then(|contents| contents.digest.as_ref())
103                .map(iota_sdk::types::CheckpointContentsDigest::try_from)
104                .transpose()?
105                .map(Into::into)
106                .map(Arc::new),
107            contents: contents
108                .filter(|contents| contents.bcs.is_some())
109                .map(|contents| contents.contents().map_err(SdkFfiError::new))
110                .transpose()?
111                .map(Into::into)
112                .map(Arc::new),
113            transactions: read_mask_requests(read_mask, SdkCheckpointResponseField::TRANSACTIONS)
114                .then(|| {
115                    value
116                        .executed_transactions()
117                        .iter()
118                        .map(TryInto::try_into)
119                        .collect::<Result<Vec<_>>>()
120                })
121                .transpose()?,
122            events: read_mask_requests(read_mask, SdkCheckpointResponseField::EVENTS_BCS)
123                .then(|| {
124                    value
125                        .events()
126                        .iter()
127                        .map(|event| event.event().map(Into::into).map_err(SdkFfiError::new))
128                        .collect::<Result<Vec<_>>>()
129                })
130                .transpose()?,
131        })
132    }
133}

135/// Define a handle object over a server-streaming checkpoint RPC.
136///
137/// The Rust API exposes these as a `Stream`, which has no uniffi equivalent,
138/// so the handle is pulled one item at a time with `next` and closed with
139/// `cancel`. The handle keeps the read mask the stream was opened with, since
140/// converting an item needs it.
141macro_rules! define_checkpoint_stream {
142    ($(#[$meta:meta])* $name:ident, $item:ty, $ffi_item:ty, $convert:expr) => {
143        $(#[$meta])*
144        ///
145        /// Call `next` in a loop to receive items; it returns `None` once the
146        /// stream is exhausted or `cancel` has been called.
147        #[derive(uniffi::Object)]
148        pub struct $name {
149            stream: StreamHandle<BoxStream<'static, GrpcResult<$item>>>,
150            read_mask: CheckpointResponseReadMask,
151        }
152
153        #[uniffi::export(async_runtime = "tokio")]
154        impl $name {
155            /// Get the next item from the stream.
156            ///
157            /// Returns `None` once the stream is exhausted or has been
158            /// canceled. Concurrent calls are serialized; there is no ordering
159            /// guarantee between them.
160            ///
161            /// An error from the connection or the server ends the stream:
162            /// there is no reconnect, so every later call returns `None`, and
163            /// a caller that wants to resume must open a new stream from the
164            /// last sequence number it received. An error converting a single
165            /// item, such as a BCS decode failure, only affects that item; the
166            /// next call continues with the following one.
167            pub async fn next(&self) -> Result<Option<$ffi_item>> {
168                self.stream
169                    .next()
170                    .await
171                    .transpose()?
172                    .map(|item| ($convert)(item, &self.read_mask))
173                    .transpose()
174            }
175
176            /// Cancel the stream, dropping the connection and unblocking a
177            /// pending `next`.
178            ///
179            /// Idempotent, and safe to call while `next` is pending — the
180            /// pending call drops the connection on its way out.
181            ///
182            /// Named `cancel` rather than `close` because a `close` method
183            /// collides with the disposal method uniffi generates for objects
184            /// in some languages.
185            pub fn cancel(&self) {
186                self.stream.cancel();
187            }
188
189            /// Whether the stream has been canceled.
190            pub fn is_canceled(&self) -> bool {
191                self.stream.is_canceled()
192            }
193        }
194
195        impl $name {
196            fn new(
197                stream: BoxStream<'static, GrpcResult<$item>>,
198                read_mask: CheckpointResponseReadMask,
199            ) -> Self {
200                Self {
201                    stream: StreamHandle::new(stream),
202                    read_mask,
203                }
204            }
205        }
206    };
207}

211    GrpcCheckpointStream,

217/// An item yielded by a filtered checkpoint stream: either a checkpoint with
218/// matching data, or a progress indicator emitted while the server scans
219/// checkpoints that have none.
220#[derive(uniffi::Enum)]
221pub enum GrpcCheckpointStreamItem {
222    /// A complete checkpoint with its transactions and events.
223    Checkpoint { checkpoint: GrpcCheckpointResponse },
224    /// A progress indicator sent during filtered scanning, carrying the
225    /// sequence number of the latest scanned checkpoint.
226    Progress { latest_scanned_sequence_number: u64 },
227}
228
229impl GrpcCheckpointStreamItem {
230    fn from_item(
231        value: iota_sdk::grpc_client::CheckpointStreamItem,
232        read_mask: &CheckpointResponseReadMask,
233    ) -> Result<Self> {
234        Ok(match value {
235            iota_sdk::grpc_client::CheckpointStreamItem::Checkpoint(checkpoint) => {
236                Self::Checkpoint {
237                    checkpoint: GrpcCheckpointResponse::from_response(&checkpoint, read_mask)?,
238                }
239            }
240            iota_sdk::grpc_client::CheckpointStreamItem::Progress {
241                latest_scanned_sequence_number,
242            } => Self::Progress {
243                latest_scanned_sequence_number,
244            },
245            _ => {
246                return Err(SdkFfiError::custom(
247                    "unsupported checkpoint stream item variant",
248                ));
249            }
250        })
251    }
252}

264impl GrpcClient {
265    /// Get the latest checkpoint.
266    ///
267    /// The optional `transactions_filter` and `events_filter` narrow the
268    /// transactions and events returned for the checkpoint.
269    ///
270    /// The optional `read_mask` controls which fields the server returns.
271    /// If `None`, only the checkpoint summary is returned.
272    #[uniffi::method(default(transactions_filter = None, events_filter = None, read_mask = None))]
273    pub async fn checkpoint_latest(
274        &self,
275        transactions_filter: Option<Arc<GrpcTransactionFilter>>,
276        events_filter: Option<Arc<GrpcEventFilter>>,
277        read_mask: Option<Vec<GrpcCheckpointResponseField>>,
278    ) -> Result<GrpcCheckpointResponse> {
279        let read_mask = crate::grpc::api::read_mask::<CheckpointResponseReadMask, _>(read_mask);
280        let response = self
281            .client()
282            .checkpoint_latest()
283            .transactions_filter(transactions_filter.as_deref().map(Into::into))
284            .events_filter(events_filter.as_deref().map(Into::into))
285            .read_mask(read_mask.clone())
286            .await?
287            .into_inner();
288        GrpcCheckpointResponse::from_response(&response, &read_mask)
289    }

345    /// Stream checkpoints across a range of sequence numbers.

388    /// Stream checkpoints across a range of sequence numbers, skipping
389    /// checkpoints with no data matching the filters.
390    ///
391    /// Unlike [`GrpcClient::checkpoints_stream`], the filters decide which
392    /// checkpoints are returned at all; checkpoints without any matching
393    /// transactions or events are skipped entirely. At least one of
394    /// `transactions_filter` or `events_filter` must be set.
395    ///
396    /// While the server scans non-matching checkpoints, the stream yields
397    /// progress items indicating the current scan position (default every
398    /// 2000ms, configurable via `progress_interval_ms`, minimum 500ms).
399    ///
400    /// If `start_sequence_number` is `None`, the stream starts from the latest
401    /// checkpoint. If `end_sequence_number` is `None`, the stream continues
402    /// indefinitely.
403    ///
404    /// The optional `read_mask` controls which fields the server returns for
405    /// each checkpoint. If `None`, only the checkpoint summary is returned.
406    #[uniffi::method(default(
407        start_sequence_number = None,
408        end_sequence_number = None,
409        transactions_filter = None,
410        events_filter = None,
411        progress_interval_ms = None,
412        read_mask = None
413    ))]
414    pub async fn checkpoints_stream_filtered(
415        &self,
416        start_sequence_number: Option<u64>,
417        end_sequence_number: Option<u64>,
418        transactions_filter: Option<Arc<GrpcTransactionFilter>>,
419        events_filter: Option<Arc<GrpcEventFilter>>,
420        progress_interval_ms: Option<u32>,
421        read_mask: Option<Vec<GrpcCheckpointResponseField>>,
422    ) -> Result<GrpcFilteredCheckpointStream> {
423        let read_mask = crate::grpc::api::read_mask::<CheckpointResponseReadMask, _>(read_mask);
424        let stream = self
425            .client()
426            .checkpoints_stream_filtered()
427            .start_sequence_number(start_sequence_number)
428            .end_sequence_number(end_sequence_number)
429            .transactions_filter(transactions_filter.as_deref().map(Into::into))
430            .events_filter(events_filter.as_deref().map(Into::into))
431            .progress_interval_ms(progress_interval_ms)
432            .read_mask(read_mask.clone())
433            .await?
434            .into_inner();
435        Ok(GrpcFilteredCheckpointStream::new(stream, read_mask))
436    }
