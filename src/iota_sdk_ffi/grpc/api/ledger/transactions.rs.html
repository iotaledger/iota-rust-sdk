27/// A transaction that has been executed, along with its signatures, effects,
28/// events and objects.
29///
30/// The `transaction`, `effects`, `events`, and input/output object fields are
31/// deserialized from BCS, so the read mask must include the matching
32/// `GrpcTransactionField` (`TransactionBcs`, `EffectsBcs`, `EventsEventsBcs`,
33/// `InputObjectsBcs`, `OutputObjectsBcs`), or its `GrpcCheckpointResponseField`
34/// / `GrpcSimulateField` counterpart, for them to be populated; digest-only
35/// read masks populate only the digest fields.
36#[derive(Clone, uniffi::Record)]
37pub struct GrpcExecutedTransaction {
38    /// The digest of the transaction.
39    pub digest: Option<Arc<TransactionDigest>>,
40    /// The transaction itself.
41    pub transaction: Option<Arc<Transaction>>,
42    /// The user signatures that authorized the execution of the transaction.
43    pub signatures: Option<Vec<Arc<UserSignature>>>,
44    /// The digest of the transaction effects.
45    pub effects_digest: Option<Arc<TransactionEffectsDigest>>,
46    /// The effects of the transaction.
47    pub effects: Option<Arc<TransactionEffects>>,
48    /// The digest of the transaction events.
49    pub events_digest: Option<Arc<TransactionEventsDigest>>,
50    /// The events emitted by the transaction, if any.
51    pub events: Option<Arc<TransactionEvents>>,
52    /// The sequence number of the checkpoint that includes the transaction.
53    pub checkpoint: Option<u64>,
54    /// Unix timestamp in milliseconds of the checkpoint that includes the
55    /// transaction.
56    pub timestamp_ms: Option<u64>,
57    /// The input objects used by the transaction.
58    pub input_objects: Option<Vec<Arc<Object>>>,
59    /// The output objects produced by the transaction.
60    pub output_objects: Option<Vec<Arc<Object>>>,
61}
62
63crate::grpc_batch_results! {
64    /// The results of a batch of transactions, in request order. Each item is
65    /// either the transaction or the error the server reported for it.
66    GrpcExecutedTransactionResults(GrpcExecutedTransaction, proto::transaction::ExecutedTransaction)
67}

69impl TryFrom<&proto::transaction::ExecutedTransaction> for GrpcExecutedTransaction {
70    type Error = SdkFfiError;
71
72    fn try_from(value: &proto::transaction::ExecutedTransaction) -> Result<Self> {
73        Ok(Self {
74            digest: value
75                .transaction
76                .as_ref()
77                .and_then(|transaction| transaction.digest.as_ref())
78                .map(iota_sdk::types::TransactionDigest::try_from)
79                .transpose()?
80                .map(Into::into)
81                .map(Arc::new),
82            transaction: value
83                .transaction
84                .as_ref()
85                .filter(|transaction| transaction.bcs.is_some())
86                .map(|transaction| transaction.transaction().map_err(SdkFfiError::new))
87                .transpose()?
88                .map(Into::into)
89                .map(Arc::new),
90            signatures: value
91                .signatures
92                .as_ref()
93                .map(Vec::<iota_sdk::types::UserSignature>::try_from)
94                .transpose()?
95                .map(|signatures| {
96                    signatures
97                        .into_iter()
98                        .map(Into::into)
99                        .map(Arc::new)
100                        .collect()
101                }),
102            effects_digest: value
103                .effects
104                .as_ref()
105                .and_then(|effects| effects.digest.as_ref())
106                .map(iota_sdk::types::TransactionEffectsDigest::try_from)
107                .transpose()?
108                .map(Into::into)
109                .map(Arc::new),
110            effects: value
111                .effects
112                .as_ref()
113                .filter(|effects| effects.bcs.is_some())
114                .map(|effects| effects.effects().map_err(SdkFfiError::new))
115                .transpose()?
116                .map(Into::into)
117                .map(Arc::new),
118            events_digest: value
119                .events
120                .as_ref()
121                .and_then(|events| events.digest.as_ref())
122                .map(iota_sdk::types::TransactionEventsDigest::try_from)
123                .transpose()?
124                .map(Into::into)
125                .map(Arc::new),
126            events: value
127                .events
128                .as_ref()
129                .filter(|events| {
130                    events
131                        .events
132                        .as_ref()
133                        .is_some_and(|events| events.events.iter().all(|event| event.bcs.is_some()))
134                })
135                .map(|events| events.events().map_err(SdkFfiError::new))
136                .transpose()?
137                .map(Into::into)
138                .map(Arc::new),
139            checkpoint: value.checkpoint,
140            timestamp_ms: value.timestamp.map(proto_to_timestamp_ms).transpose()?,
141            input_objects: value
142                .input_objects
143                .as_ref()
144                .filter(|objects| objects.objects.iter().all(|object| object.bcs.is_some()))
145                .map(Vec::<iota_sdk::types::Object>::try_from)
146                .transpose()?
147                .map(|objects| objects.into_iter().map(Into::into).map(Arc::new).collect()),
148            output_objects: value
149                .output_objects
150                .as_ref()
151                .filter(|objects| objects.objects.iter().all(|object| object.bcs.is_some()))
152                .map(Vec::<iota_sdk::types::Object>::try_from)
153                .transpose()?
154                .map(|objects| objects.into_iter().map(Into::into).map(Arc::new).collect()),
155        })
156    }
157}

159#[uniffi::export(async_runtime = "tokio")]
160impl GrpcClient {
161    /// Get transactions by their digests.
162    ///
163    /// Results are returned in the same order as the input digests, one per
164    /// digest. A transaction the serving node cannot return — because it is
165    /// not found or has been pruned — fails only its own item, which reading
166    /// throws with the server's error message. A transaction the server returns
167    /// but that cannot be decoded fails the whole call.
168    ///
169    /// The optional `read_mask` controls which fields the server returns.
170    /// If `None`, the transaction, signatures, checkpoint, and timestamp are
171    /// returned.
172    #[uniffi::method(default(read_mask = None))]
173    pub async fn transactions(
174        &self,
175        digests: Vec<Arc<TransactionDigest>>,
176        read_mask: Option<Vec<GrpcTransactionField>>,
177    ) -> Result<GrpcExecutedTransactionResults> {
178        let digests = digests.iter().map(|digest| ***digest).collect::<Vec<_>>();
179        GrpcExecutedTransactionResults::new(
180            self.client()
181                .transactions(digests)
182                .read_mask(crate::grpc::api::read_mask::<TransactionReadMask, _>(
183                    read_mask,
184                ))
185                .await?
186                .into_inner(),
187        )
188    }
189
190    /// Wait for the indexing (on the node) or finalization of a transaction by
191    /// its digest, polling the node until then. Returns an error after 60s.
192    pub async fn wait_for_transaction(
193        &self,
194        digest: &TransactionDigest,
195        wait_for: WaitForTransaction,
196    ) -> Result<()> {
197        Ok(TransactionBuilderExecutionClient::wait_for_transaction(
198            &self.client(),
199            **digest,
200            wait_for.into(),
201        )
202        .await?)
203    }
204}

206#[cfg(test)]
207mod tests {
208    use iota_sdk::{
209        grpc_client::GrpcError,
210        grpc_types::v1 as proto,
211        types::{TransactionDigest, TransactionEffectsDigest, TransactionEventsDigest},
212    };
213
214    use super::{GrpcExecutedTransaction, GrpcExecutedTransactionResults};
215
216    #[test]
217    fn digest_only_mask_populates_the_typed_digests() {
218        let transaction_digest = TransactionDigest::from([1; 32]);
219        let effects_digest = TransactionEffectsDigest::from([2; 32]);
220        let events_digest = TransactionEventsDigest::from([3; 32]);
221
222        let mut transaction = proto::transaction::Transaction::default();
223        transaction.digest = Some(transaction_digest.into());
224        let mut effects = proto::transaction::TransactionEffects::default();
225        effects.digest = Some(effects_digest.into());
226        let mut events = proto::transaction::TransactionEvents::default();
227        events.digest = Some(events_digest.into());
228
229        let mut value = proto::transaction::ExecutedTransaction::default();
230        value.transaction = Some(transaction);
231        value.effects = Some(effects);
232        value.events = Some(events);
233
234        let converted = GrpcExecutedTransaction::try_from(&value).unwrap();
235
236        assert_eq!(converted.digest.unwrap().0, transaction_digest);
237        assert_eq!(converted.effects_digest.unwrap().0, effects_digest);
238        assert_eq!(converted.events_digest.unwrap().0, events_digest);
239        assert!(converted.transaction.is_none());
240        assert!(converted.effects.is_none());
241        assert!(converted.events.is_none());
242    }
243
244    #[test]
245    fn item_error_fails_only_its_own_item() {
246        let results = GrpcExecutedTransactionResults::new(vec![
247            Err(GrpcError::EmptyRequest),
248            Ok(proto::transaction::ExecutedTransaction::default()),
249        ])
250        .unwrap();
251
252        assert_eq!(results.len(), 2);
253        assert!(results.get(0).is_err());
254        assert!(results.get(1).is_ok());
255        assert!(results.get(2).is_err());
256    }
257
258    #[test]
259    fn next_moves_past_errors_and_stops_at_the_end() {
260        let results = GrpcExecutedTransactionResults::new(vec![
261            Err(GrpcError::EmptyRequest),
262            Ok(proto::transaction::ExecutedTransaction::default()),
263        ])
264        .unwrap();
265
266        assert!(results.has_next());
267        assert!(results.next().is_err());
268        assert!(results.has_next());
269        assert!(results.next().is_ok());
270        assert!(!results.has_next());
271        assert!(results.next().is_err());
272        assert!(!results.has_next());
273        assert!(results.get(1).is_ok());
274    }
275
276    #[test]
277    fn undecodable_item_is_an_error() {
278        let mut transaction = proto::transaction::Transaction::default();
279        transaction.bcs = Some(proto::bcs::BcsData::from(vec![0xff, 0xff]));
280
281        let mut value = proto::transaction::ExecutedTransaction::default();
282        value.transaction = Some(transaction);
283
284        assert!(GrpcExecutedTransactionResults::new(vec![Ok(value)]).is_err());
285    }
