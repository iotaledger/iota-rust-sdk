4//! Foreign-language equivalent of the Rust APIs that take a `reqwest::Client`.
5//!
6//! The Rust APIs let callers hand over a fully built `reqwest::Client`. uniffi
7//! has no way to carry one across the boundary, so the bindings describe what
8//! they want instead and the client is built here.
9
10use crate::error::{Result, SdkFfiError};
11
12/// Sent as the `User-Agent` unless the caller overrides it.
13const USER_AGENT: &str = concat!(env!("CARGO_PKG_NAME"), "/", env!("CARGO_PKG_VERSION"));
14
15/// How the SDK should build the HTTP client backing a connection.
16#[derive(Debug, Default, uniffi::Record)]
17pub struct HttpClientOptions {
18    /// Additional DER-encoded CA certificates to trust, on top of the platform
19    /// trust store and the bundled roots.
20    #[uniffi(default = [])]
21    pub extra_root_certificates: Vec<Vec<u8>>,
22    /// Ignore the platform trust store, trusting only the bundled roots and
23    /// `extra_root_certificates`.
24    #[uniffi(default = false)]
25    pub exclude_platform_roots: bool,
26    /// Total request timeout in milliseconds. `None` leaves it unbounded.
27    #[uniffi(default = None)]
28    pub timeout_ms: Option<u64>,
29    /// Replaces the `User-Agent` the bindings would otherwise send.
30    #[uniffi(default = None)]
31    pub user_agent: Option<String>,
32}
33
34impl HttpClientOptions {
35    /// Build the described client.
36    pub(crate) fn build(&self) -> Result<reqwest::Client> {
37        let mut builder = base_builder();
38
39        if let Some(user_agent) = &self.user_agent {
40            builder = builder.user_agent(user_agent.clone());
41        }
42
43        self.apply_transport(builder)?
44            .build()
45            .map_err(SdkFfiError::new)
46    }
47
48    #[cfg(not(target_arch = "wasm32"))]
49    fn apply_transport(
50        &self,
51        mut builder: reqwest::ClientBuilder,
52    ) -> Result<reqwest::ClientBuilder> {
53        if let Some(timeout_ms) = self.timeout_ms {
54            builder = builder.timeout(std::time::Duration::from_millis(timeout_ms));
55        }
56
57        let mut roots = webpki_root_certs::TLS_SERVER_ROOT_CERTS
58            .iter()
59            .filter_map(|der| reqwest::Certificate::from_der(der).ok())
60            .collect::<Vec<_>>();
61        for der in &self.extra_root_certificates {
62            roots.push(reqwest::Certificate::from_der(der).map_err(SdkFfiError::new)?);
63        }
64
65        // Merging keeps the platform store and adds these as a floor. reqwest
66        // only supports that where `rustls-platform-verifier` accepts extra
67        // roots; elsewhere — Android in particular — the roots have to stand
68        // alone.
69        #[cfg(any(all(unix, not(target_os = "android")), target_os = "windows"))]
70        let merge_supported = true;
71        #[cfg(not(any(all(unix, not(target_os = "android")), target_os = "windows")))]
72        let merge_supported = false;
73
74        Ok(if self.exclude_platform_roots || !merge_supported {
75            builder.tls_certs_only(roots)
76        } else {
77            builder.tls_certs_merge(roots)
78        })
79    }
80
81    /// On wasm32 the browser owns both certificate verification and request
82    /// deadlines, and reqwest's wasm builder exposes neither. Asking for them
83    /// is refused rather than silently ignored, so that a caller never believes
84    /// it has pinned a root or bounded a request when it has not.
85    #[cfg(target_arch = "wasm32")]
86    fn apply_transport(&self, builder: reqwest::ClientBuilder) -> Result<reqwest::ClientBuilder> {
87        if self.exclude_platform_roots || !self.extra_root_certificates.is_empty() {
88            return Err(SdkFfiError::custom(
89                "custom root certificates are not supported on wasm32: \
90                 the browser controls certificate verification",
91            ));
92        }
93        if self.timeout_ms.is_some() {
94            return Err(SdkFfiError::custom(
95                "request timeouts are not supported on wasm32: \
96                 the browser controls request deadlines",
97            ));
98        }
99        Ok(builder)
100    }
101}
102
103/// `reqwest` is built with `rustls-no-provider` across this workspace, so a
104/// provider has to be installed before any client is built. The first caller
105/// wins, so an application that has already chosen one keeps it.
106#[cfg(not(target_arch = "wasm32"))]
107fn base_builder() -> reqwest::ClientBuilder {
108    let _ = rustls::crypto::ring::default_provider().install_default();
109    reqwest::Client::builder().user_agent(USER_AGENT)
110}
111
112/// On wasm32 the browser owns certificate verification, so there is no provider
113/// and no trust anchors to configure.
114#[cfg(target_arch = "wasm32")]
115fn base_builder() -> reqwest::ClientBuilder {
116    reqwest::Client::builder().user_agent(USER_AGENT)
117}
