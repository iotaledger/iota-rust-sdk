4use std::sync::Arc;
5
6use iota_sdk::crypto::Verifier;
7
8use crate::{
9    error::Result,
10    types::{address::Address, crypto::passkey::PasskeyAuthenticator},
11};
12
13#[derive(Debug, uniffi::Object)]
14#[uniffi::export(Debug)]
15pub struct PasskeyVerifier(pub iota_sdk::crypto::passkey::PasskeyVerifier);
16
17#[uniffi::export]
18impl PasskeyVerifier {
19    #[uniffi::constructor]
20    pub fn new() -> Self {
21        Self(iota_sdk::crypto::passkey::PasskeyVerifier::new())
22    }
23
24    /// Only accept signatures from the passkey whose public key derives to
25    /// `address`.
26    pub fn with_address(&self, address: &Address) -> Arc<Self> {
27        Arc::new(Self(self.0.clone().with_address(address.0)))
28    }
29
30    pub fn verify(&self, message: &[u8], authenticator: &PasskeyAuthenticator) -> Result<()> {
31        Ok(self.0.verify(message, &authenticator.0)?)
32    }
33}
