4/// The number of words in a BIP-39 mnemonic.
5#[derive(uniffi::Enum)]
6pub enum MnemonicLength {
7    Words12 = 12,
8    Words24 = 24,
9}
10
11impl From<MnemonicLength> for iota_sdk::crypto::mnemonic::MnemonicLength {
12    fn from(value: MnemonicLength) -> Self {
13        match value {
14            MnemonicLength::Words12 => Self::Words12,
15            MnemonicLength::Words24 => Self::Words24,
16        }
17    }
18}
19
20/// Generate a new BIP-39 mnemonic in English.
21/// Supported word counts are 12 and 24 (default).
22#[uniffi::export]
23pub fn generate_mnemonic(word_count: Option<MnemonicLength>) -> String {
24    iota_sdk::crypto::mnemonic::generate_mnemonic(word_count.map(Into::into))
25}
