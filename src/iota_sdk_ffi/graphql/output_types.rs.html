11/// A transaction argument used in programmable transactions.
12#[derive(uniffi::Enum)]
13pub enum GraphQLTransactionArgument {
14    /// Reference to the gas coin.
15    GasCoin,
16    /// An input to the programmable transaction block.
17    Input {
18        /// Index of the programmable transaction block input (0-indexed).
19        index: u32,
20    },
21    /// The result of another transaction command.
22    Result {
23        /// The index of the previous command (0-indexed) that returned this result.
24        cmd: u32,
25        /// If the previous command returns multiple values, this is the index of the individual
26        /// result among the multiple results from that command (also 0-indexed).
27        index: Option<u32>,
28    },
29}
30
31impl From<iota_sdk::graphql_client::TransactionArgument> for GraphQLTransactionArgument {
32    fn from(value: iota_sdk::graphql_client::TransactionArgument) -> Self {
33        match value {
34            iota_sdk::graphql_client::TransactionArgument::GasCoin => {
35                GraphQLTransactionArgument::GasCoin
36            }
37            iota_sdk::graphql_client::TransactionArgument::Input { index } => {
38                GraphQLTransactionArgument::Input { index }
39            }
40            iota_sdk::graphql_client::TransactionArgument::Result { cmd, index } => {
41                GraphQLTransactionArgument::Result { cmd, index }
42            }
43            _ => unimplemented!(
44                "a new TransactionArgument enum variant was added and needs to be handled"
45            ),
46        }
47    }
48}
49
50impl From<GraphQLTransactionArgument> for iota_sdk::graphql_client::TransactionArgument {
51    fn from(value: GraphQLTransactionArgument) -> Self {
52        match value {
53            GraphQLTransactionArgument::GasCoin => {
54                iota_sdk::graphql_client::TransactionArgument::GasCoin
55            }
56            GraphQLTransactionArgument::Input { index } => {
57                iota_sdk::graphql_client::TransactionArgument::Input { index }
58            }
59            GraphQLTransactionArgument::Result { cmd, index } => {
60                iota_sdk::graphql_client::TransactionArgument::Result { cmd, index }
61            }
62        }
63    }
64}

66/// A return value from a command in the dry run.
67#[derive(uniffi::Record)]
68pub struct GraphQLDryRunReturn {
69    /// The Move type of the return value.
70    pub type_tag: Arc<TypeTag>,
71    /// The BCS representation of the return value.
72    pub bcs: Vec<u8>,
73}
74
75impl From<iota_sdk::graphql_client::DryRunReturn> for GraphQLDryRunReturn {
76    fn from(value: iota_sdk::graphql_client::DryRunReturn) -> Self {
77        GraphQLDryRunReturn {
78            type_tag: Arc::new(value.type_tag.into()),
79            bcs: value.bcs,
80        }
81    }
82}

84/// A mutation to an argument that was mutably borrowed by a command.
85#[derive(uniffi::Record)]
86pub struct GraphQLDryRunMutation {
87    /// The transaction argument that was mutated.
88    pub input: GraphQLTransactionArgument,
89    /// The Move type of the mutated value.
90    pub type_tag: Arc<TypeTag>,
91    /// The BCS representation of the mutated value.
92    pub bcs: Vec<u8>,
93}
94
95impl From<iota_sdk::graphql_client::DryRunMutation> for GraphQLDryRunMutation {
96    fn from(value: iota_sdk::graphql_client::DryRunMutation) -> Self {
97        GraphQLDryRunMutation {
98            input: value.input.into(),
99            type_tag: Arc::new(value.type_tag.into()),
100            bcs: value.bcs,
101        }
102    }
103}

105/// Effects of a single command in the dry run, including mutated references and return values.
106#[derive(uniffi::Record)]
107pub struct GraphQLDryRunEffect {
108    /// Changes made to arguments that were mutably borrowed by this command.
109    pub mutated_references: Vec<GraphQLDryRunMutation>,
110    /// Return results of this command.
111    pub return_values: Vec<GraphQLDryRunReturn>,
112}
113
114impl From<iota_sdk::graphql_client::DryRunEffect> for GraphQLDryRunEffect {
115    fn from(value: iota_sdk::graphql_client::DryRunEffect) -> Self {
116        GraphQLDryRunEffect {
117            mutated_references: value
118                .mutated_references
119                .into_iter()
120                .map(Into::into)
121                .collect(),
122            return_values: value.return_values.into_iter().map(Into::into).collect(),
123        }
124    }
125}

127/// The result of a simulation (dry run), which includes the effects of the transaction, any errors
128/// that may have occurred, and intermediate results for each command.
129#[derive(uniffi::Record)]
130pub struct GraphQLDryRunResult {
131    /// The error that occurred during dry run execution, if any.
132    pub error: Option<String>,
133    /// The intermediate results for each command of the dry run execution, including contents of
134    /// mutated references and return values.
135    pub results: Vec<GraphQLDryRunEffect>,
136    /// The transaction that was dry run, without signatures.
137    pub transaction: Option<Arc<Transaction>>,
138    /// The effects of the transaction execution.
139    pub effects: Option<Arc<TransactionEffects>>,
140    /// The gas price to use. This is the reference gas price, or a higher price if an input object
141    /// is congested.
142    pub suggested_gas_price: Option<u64>,
143}
144
145impl From<iota_sdk::graphql_client::DryRunResult> for GraphQLDryRunResult {
146    fn from(value: iota_sdk::graphql_client::DryRunResult) -> Self {
147        GraphQLDryRunResult {
148            error: value.error,
149            results: value.results.into_iter().map(Into::into).collect(),
150            transaction: value.transaction.map(Into::into).map(Arc::new),
151            effects: value.effects.map(Into::into).map(Arc::new),
152            suggested_gas_price: value.suggested_gas_price,
153        }
154    }
