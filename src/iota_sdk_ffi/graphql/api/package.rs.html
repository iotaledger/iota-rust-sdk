24#[cfg_attr(not(target_arch = "wasm32"), uniffi::export(async_runtime = "tokio"))]

26impl GraphQLClient {
27    /// The package corresponding to the given address (at the optionally given version). When no
28    /// version is given, the package is loaded directly from the address given. Otherwise, the
29    /// address is translated before loading to point to the package whose original ID matches the
30    /// package at address, but whose version is version. For non-system packages, this might result
31    /// in a different address than address because different versions of a package, introduced by
32    /// upgrades, exist at distinct addresses.
33    ///
34    /// Note that this interpretation of version is different from a historical object read (the
35    /// interpretation of version for the object query).
36    #[uniffi::method(default(version = None))]
37    pub async fn package(
38        &self,
39        address: &Address,
40        version: Option<Arc<Version>>,
41    ) -> Result<Option<Arc<MovePackage>>> {
42        Ok(self
43            .client()
44            .package(**address)
45            .set_if_some(version.map(|v| **v), GetPackageQuery::version)
46            .await?
47            .map(Into::into)
48            .map(Arc::new))
49    }
50
51    /// Fetch all versions of package at address (packages that share this package's original ID),
52    /// optionally bounding the versions exclusively from below with afterVersion, or from above
53    /// with beforeVersion.
54    #[uniffi::method(default(pagination_filter = None, after_version = None, before_version = None))]
55    pub async fn package_versions(
56        &self,
57        address: &Address,
58        after_version: Option<Arc<Version>>,
59        before_version: Option<Arc<Version>>,
60        pagination_filter: Option<GraphQLPaginationFilter>,
61    ) -> Result<GraphQLMovePackagePage> {
62        Ok(self
63            .client()
64            .package_versions(**address)
65            .pagination(pagination_filter.map(Into::into).unwrap_or_default())
66            .set_if_some(
67                after_version.map(|v| **v),
68                ListPackageVersionsQuery::after_version,
69            )
70            .set_if_some(
71                before_version.map(|v| **v),
72                ListPackageVersionsQuery::before_version,
73            )
74            .await?
75            .map(Into::into)
76            .into())
77    }
78
79    /// Fetch the latest version of the package at address.
80    /// This corresponds to the package with the highest version that shares its original ID with
81    /// the package at address.
82    pub async fn package_latest(&self, address: &Address) -> Result<Option<Arc<MovePackage>>> {
83        Ok(self
84            .client()
85            .package_latest(**address)
86            .await?
87            .map(Into::into)
88            .map(Arc::new))
89    }
90
91    /// The Move packages that exist in the network, optionally filtered to be strictly before
92    /// beforeCheckpoint and/or strictly after afterCheckpoint.
93    ///
94    /// This query returns all versions of a given user package that appear between the specified
95    /// checkpoints, but only records the latest versions of system packages.
96    #[uniffi::method(default(pagination_filter = None, after_checkpoint = None, before_checkpoint = None))]
97    pub async fn packages(
98        &self,
99        after_checkpoint: Option<u64>,
100        before_checkpoint: Option<u64>,
101        pagination_filter: Option<GraphQLPaginationFilter>,
102    ) -> Result<GraphQLMovePackagePage> {
103        Ok(self
104            .client()
105            .packages()
106            .pagination(pagination_filter.map(Into::into).unwrap_or_default())
107            .set_if_some(after_checkpoint, ListPackagesQuery::after_checkpoint)
108            .set_if_some(before_checkpoint, ListPackagesQuery::before_checkpoint)
109            .await?
110            .map(Into::into)
111            .into())
112    }
113
114    /// Return the normalized Move function data for the provided package, module, and function.
115    #[uniffi::method(default(version = None))]
116    pub async fn normalized_move_function(
117        &self,
118        package: &Address,
119        module: &str,
120        function: &str,
121        version: Option<Arc<Version>>,
122    ) -> Result<Option<Arc<GraphQLMoveFunction>>> {
123        Ok(self
124            .client()
125            .normalized_move_function(**package, module, function)
126            .set_if_some(
127                version.map(|v| **v),
128                GetNormalizedMoveFunctionQuery::version,
129            )
130            .await?
131            .map(Into::into)
132            .map(Arc::new))
133    }
134
135    /// Return the normalized Move module data for the provided module.
136    // TODO: do we want to self paginate everything and return all the data, or keep pagination
137    // options?
138    #[expect(clippy::too_many_arguments)]
139    #[uniffi::method(default(
140        version = None,
141        pagination_filter_enums = None,
142        pagination_filter_friends = None,
143        pagination_filter_functions = None,
144        pagination_filter_structs = None,
145    ))]
146    pub async fn normalized_move_module(
147        &self,
148        package: &Address,
149        module: &str,
150        version: Option<Arc<Version>>,
151        pagination_filter_enums: Option<GraphQLPaginationFilter>,
152        pagination_filter_friends: Option<GraphQLPaginationFilter>,
153        pagination_filter_functions: Option<GraphQLPaginationFilter>,
154        pagination_filter_structs: Option<GraphQLPaginationFilter>,
155    ) -> Result<Option<GraphQLMoveModule>> {
156        Ok(self
157            .client()
158            .normalized_move_module(**package, module)
159            .set_if_some(version.map(|v| **v), GetNormalizedMoveModuleQuery::version)
160            .enums_pagination(pagination_filter_enums.map(Into::into).unwrap_or_default())
161            .friends_pagination(
162                pagination_filter_friends
163                    .map(Into::into)
164                    .unwrap_or_default(),
165            )
166            .functions_pagination(
167                pagination_filter_functions
168                    .map(Into::into)
169                    .unwrap_or_default(),
170            )
171            .structs_pagination(
172                pagination_filter_structs
173                    .map(Into::into)
174                    .unwrap_or_default(),
175            )
176            .await?
177            .map(Into::into))
178    }
