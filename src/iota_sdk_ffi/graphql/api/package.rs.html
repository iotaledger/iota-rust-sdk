20impl GraphQLClient {
21    /// The package corresponding to the given address (at the optionally given
22    /// version). When no version is given, the package is loaded directly
23    /// from the address given. Otherwise, the address is translated before
24    /// loading to point to the package whose original ID matches
25    /// the package at address, but whose version is version. For non-system
26    /// packages, this might result in a different address than address
27    /// because different versions of a package, introduced by upgrades,
28    /// exist at distinct addresses.
29    ///
30    /// Note that this interpretation of version is different from a historical
31    /// object read (the interpretation of version for the object query).
32    #[uniffi::method(default(version = None))]
33    pub async fn package(
34        &self,
35        address: &Address,
36        version: Option<Arc<Version>>,
37    ) -> Result<Option<Arc<MovePackage>>> {
38        Ok(self
39            .0
40            .read()
41            .await
42            .package(**address, version.map(|v| **v))
43            .await?
44            .map(Into::into)
45            .map(Arc::new))
46    }
47
48    /// Fetch all versions of package at address (packages that share this
49    /// package's original ID), optionally bounding the versions exclusively
50    /// from below with afterVersion, or from above with beforeVersion.
51    #[uniffi::method(default(pagination_filter = None, after_version = None, before_version = None))]
52    pub async fn package_versions(
53        &self,
54        address: &Address,
55        after_version: Option<Arc<Version>>,
56        before_version: Option<Arc<Version>>,
57        pagination_filter: Option<PaginationFilter>,
58    ) -> Result<MovePackagePage> {
59        Ok(self
60            .0
61            .read()
62            .await
63            .package_versions(
64                **address,
65                pagination_filter.map(Into::into).unwrap_or_default(),
66                after_version.map(|v| **v),
67                before_version.map(|v| **v),
68            )
69            .await?
70            .map(Into::into)
71            .into())
72    }
73
74    /// Fetch the latest version of the package at address.
75    /// This corresponds to the package with the highest version that shares its
76    /// original ID with the package at address.
77    pub async fn package_latest(&self, address: &Address) -> Result<Option<Arc<MovePackage>>> {
78        Ok(self
79            .0
80            .read()
81            .await
82            .package_latest(**address)
83            .await?
84            .map(Into::into)
85            .map(Arc::new))
86    }
87
88    /// The Move packages that exist in the network, optionally filtered to be
89    /// strictly before beforeCheckpoint and/or strictly after
90    /// afterCheckpoint.
91    ///
92    /// This query returns all versions of a given user package that appear
93    /// between the specified checkpoints, but only records the latest
94    /// versions of system packages.
95    #[uniffi::method(default(pagination_filter = None, after_checkpoint = None, before_checkpoint = None))]
96    pub async fn packages(
97        &self,
98        after_checkpoint: Option<u64>,
99        before_checkpoint: Option<u64>,
100        pagination_filter: Option<PaginationFilter>,
101    ) -> Result<MovePackagePage> {
102        Ok(self
103            .0
104            .read()
105            .await
106            .packages(
107                pagination_filter.map(Into::into).unwrap_or_default(),
108                after_checkpoint,
109                before_checkpoint,
110            )
111            .await?
112            .map(Into::into)
113            .into())
114    }

158            .0

165                pagination_filter_enums.map(Into::into).unwrap_or_default(),

172                pagination_filter_structs

176            .await?
177            .map(Into::into))
178    }
