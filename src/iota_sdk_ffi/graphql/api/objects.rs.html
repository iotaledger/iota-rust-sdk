4//! Objects API implementation.
5
6use std::sync::Arc;
7
8use crate::{
9    error::Result,
10    graphql::{
11        client::GraphQLClient,
12        pagination::ObjectPage,
13        query_types::{ObjectFilter, PaginationFilter},
14    },
15    types::{
16        object::{Object, ObjectId},
17        version::Version,
18    },
19};
20
21#[cfg_attr(not(target_arch = "wasm32"), uniffi::export(async_runtime = "tokio"))]
22#[cfg_attr(target_arch = "wasm32", uniffi::export)]
23impl GraphQLClient {
24    /// Return an object based on the provided `Address`.
25    ///
26    /// If the object does not exist (e.g., due to pruning), this will return
27    /// `Ok(None)`. Similarly, if this is not an object but an address, it
28    /// will return `Ok(None)`.
29    #[uniffi::method(default(version = None))]
30    pub async fn object(
31        &self,
32        object_id: &ObjectId,
33        version: Option<Arc<Version>>,
34    ) -> Result<Option<Arc<Object>>> {
35        Ok(self
36            .0
37            .read()
38            .await
39            .object(**object_id, version.map(|v| **v))
40            .await?
41            .map(Into::into)
42            .map(Arc::new))
43    }
44
45    /// Return a page of objects based on the provided parameters.
46    ///
47    /// Use this function together with the `ObjectFilter::owner` to get the
48    /// objects owned by an address.
49    #[uniffi::method(default(pagination_filter = None, filter = None))]
50    pub async fn objects(
51        &self,
52        filter: Option<ObjectFilter>,
53        pagination_filter: Option<PaginationFilter>,
54    ) -> Result<ObjectPage> {
55        Ok(self
56            .0
57            .read()
58            .await
59            .objects(
60                filter.map(Into::into),
61                pagination_filter.map(Into::into).unwrap_or_default(),
62            )
63            .await?
64            .map(Into::into)
65            .into())
66    }
67
68    /// Return the object's bcs content `Vec<u8>` based on the provided
69    /// `Address`.
70    pub async fn object_bcs(&self, object_id: &ObjectId) -> Result<Option<Vec<u8>>> {
71        Ok(self.0.read().await.object_bcs(**object_id).await?)
72    }
73
74    /// Return the BCS of an object that is a Move object.
75    ///
76    /// If the object does not exist (e.g., due to pruning), this will return
77    /// `Ok(None)`. Similarly, if this is not an object but an address, it
78    /// will return `Ok(None)`.
79    #[uniffi::method(default(version = None))]
80    pub async fn move_object_contents_bcs(
81        &self,
82        object_id: &ObjectId,
83        version: Option<Arc<Version>>,
84    ) -> Result<Option<Vec<u8>>> {
85        Ok(self
86            .0
87            .read()
88            .await
89            .move_object_contents_bcs(**object_id, version.map(|v| **v))
90            .await?)
91    }
92
93    /// Return the contents' JSON of an object that is a Move object.
94    ///
95    /// If the object does not exist (e.g., due to pruning), this will return
96    /// `Ok(None)`. Similarly, if this is not an object but an address, it
97    /// will return `Ok(None)`.
98    #[uniffi::method(default(version = None))]
99    pub async fn move_object_contents(
100        &self,
101        object_id: &ObjectId,
102        version: Option<Arc<Version>>,
103    ) -> Result<Option<serde_json::Value>> {
104        Ok(self
105            .0
106            .read()
107            .await
108            .move_object_contents(**object_id, version.map(|v| **v))
109            .await?)
110    }
111}
