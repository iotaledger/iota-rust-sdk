18impl GraphQLClient {
19    /// Execute a Move View Function with raw JSON arguments.
20    ///
21    /// This is an alternative to [`GraphQLClient::move_view_call`] that accepts
22    /// raw JSON values instead of typed arguments.
23    ///
24    /// A View Function is a function in a Move module with a return type that
25    /// does not alter the state of the ledger. When using this interface,
26    /// no transactions are submitted to the network for inclusion into the
27    /// ledger.
28    ///
29    /// # Arguments
30    /// * `package` - The package the Move function is defined in
31    /// * `module` - The module the Move function is defined in
32    /// * `function` - The name of the Move function
33    /// * `type_arguments` - The type arguments of the Move function
34    /// * `arguments` - The arguments to be passed into the Move function, in
35    ///   JSON format
36    ///
37    /// # Returns
38    /// A `GraphQLMoveViewResult` containing either execution results (return
39    /// values) or an error.
40    #[uniffi::method(default(type_arguments = None, arguments = None))]
41    pub async fn move_view_call_json(
42        &self,
43        package: &ObjectId,
44        module: String,
45        function: String,
46        type_arguments: Option<Vec<String>>,
47        arguments: Option<Vec<serde_json::Value>>,
48    ) -> Result<GraphQLMoveViewResult> {
49        Ok(self
50            .client()
51            .move_view_call_json(**package, module, function)
52            .set_if_some(type_arguments, MoveViewCallJsonQuery::type_arguments)
53            .set_if_some(arguments, MoveViewCallJsonQuery::arguments)
54            .await?
55            .into())
56    }
57
58    /// Execute a Move View Function.
59    ///
60    /// A View Function is a function in a Move module with a return type that
61    /// does not alter the state of the ledger. When using this interface,
62    /// no transactions are submitted to the network for inclusion into the
63    /// ledger.
64    ///
65    /// This method allows calling nearly any Move function with a return type
66    /// and any arguments. The function's result values are provided and
67    /// decoded using the appropriate Move type, then formatted in JSON.
68    ///
69    /// The use of this interface does not require signature checks (even for
70    /// functions that take Owned Objects as input) or gas coins, as it does
71    /// not alter ledger state. Spam attacks are dealt with at the RPC level
72    /// rather than execution level.
73    ///
74    /// # Arguments
75    /// * `package` - The package the Move function is defined in
76    /// * `module` - The module the Move function is defined in
77    /// * `function` - The name of the Move function
78    /// * `type_arguments` - The type arguments of the Move function
79    /// * `arguments` - The typed arguments to be passed into the Move function
80    ///
81    /// # Returns
82    /// A `GraphQLMoveViewResult` containing either execution results (return
83    /// values) or an error.
84    #[uniffi::method(default(type_arguments = None, arguments = None))]
85    pub async fn move_view_call(
86        &self,
87        package: &ObjectId,
88        module: String,
89        function: String,
90        type_arguments: Option<Vec<Arc<TypeTag>>>,
91        arguments: Option<Vec<Arc<MoveViewArg>>>,
92    ) -> Result<GraphQLMoveViewResult> {
93        let arguments = arguments.map(|args| args.iter().map(|arg| arg.to_json()).collect());
94        let type_arguments =
95            type_arguments.map(|tags| tags.iter().map(|t| t.to_string()).collect());
96
97        self.move_view_call_json(package, module, function, type_arguments, arguments)
98            .await
99    }
100}
