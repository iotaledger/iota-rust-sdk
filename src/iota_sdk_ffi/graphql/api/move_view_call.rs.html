13#[cfg_attr(not(target_arch = "wasm32"), uniffi::export(async_runtime = "tokio"))]
14#[cfg_attr(target_arch = "wasm32", uniffi::export)]
15impl GraphQLClient {
16    /// Execute a Move View Function with raw JSON arguments.
17    ///
18    /// This is an alternative to [`GraphQLClient::move_view_call`] that accepts
19    /// raw JSON values instead of typed arguments.
20    ///
21    /// A View Function is a function in a Move module with a return type that
22    /// does not alter the state of the ledger. When using this interface,
23    /// no transactions are submitted to the network for inclusion into the
24    /// ledger.
25    ///
26    /// # Arguments
27    /// * `function_name` - The Move function fully qualified name as
28    ///   `<package_id>::<module_name>::<function_name>`, e.g.,
29    ///   `0x533074f8e22e8ce1330d7e9d67c18966abb5a3d58dc2e2deea50e50bea4e87f4::shop::total_revenue`
30    /// * `type_arguments` - The type arguments of the Move function
31    /// * `arguments` - The arguments to be passed into the Move function, in
32    ///   JSON format
33    ///
34    /// # Returns
35    /// A `MoveViewResult` containing either execution results (return values)
36    /// or an error.
37    #[uniffi::method(default(type_arguments = None, arguments = None))]
38    pub async fn move_view_call_json(
39        &self,
40        function_name: String,
41        type_arguments: Option<Vec<String>>,
42        arguments: Option<Vec<serde_json::Value>>,
43    ) -> Result<MoveViewResult> {
44        Ok(self
45            .0
46            .read()
47            .await
48            .move_view_call_json(function_name, type_arguments, arguments)
49            .await?
50            .into())
51    }
52
53    /// Execute a Move View Function.
54    ///
55    /// A View Function is a function in a Move module with a return type that
56    /// does not alter the state of the ledger. When using this interface,
57    /// no transactions are submitted to the network for inclusion into the
58    /// ledger.
59    ///
60    /// This method allows calling nearly any Move function with a return type
61    /// and any arguments. The function's result values are provided and
62    /// decoded using the appropriate Move type, then formatted in JSON.
63    ///
64    /// The use of this interface does not require signature checks (even for
65    /// functions that take Owned Objects as input) or gas coins, as it does
66    /// not alter ledger state. Spam attacks are dealt with at the RPC level
67    /// rather than execution level.
68    ///
69    /// # Arguments
70    /// * `function_name` - The Move function fully qualified name as
71    ///   `<package_id>::<module_name>::<function_name>`, e.g.,
72    ///   `0x533074f8e22e8ce1330d7e9d67c18966abb5a3d58dc2e2deea50e50bea4e87f4::shop::total_revenue`
73    /// * `type_arguments` - The type arguments of the Move function
74    /// * `arguments` - The typed arguments to be passed into the Move function
75    ///
76    /// # Returns
77    /// A `MoveViewResult` containing either execution results (return values)
78    /// or an error.
79    #[uniffi::method(default(type_arguments = None, arguments = None))]
80    pub async fn move_view_call(
81        &self,
82        function_name: String,
83        type_arguments: Option<Vec<Arc<TypeTag>>>,
84        arguments: Option<Vec<Arc<MoveViewArg>>>,
85    ) -> Result<MoveViewResult> {
86        let arguments = arguments.map(|args| args.iter().map(|arg| arg.to_json()).collect());
87        let type_arguments =
88            type_arguments.map(|tags| tags.iter().map(|t| t.to_string()).collect());
89
90        self.move_view_call_json(function_name, type_arguments, arguments)
91            .await
92    }
93}
