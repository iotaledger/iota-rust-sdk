4//! IOTA Names API implementation.
5
6use std::sync::Arc;
7
8use crate::{
9    error::Result,
10    graphql::{
11        client::GraphQLClient, pagination::NameRegistrationPage, query_types::PaginationFilter,
12    },
13    types::{
14        address::Address,
15        iota_names::{Name, NameFormat},
16    },
17};
18
19#[cfg_attr(not(target_arch = "wasm32"), uniffi::export(async_runtime = "tokio"))]
20#[cfg_attr(target_arch = "wasm32", uniffi::export)]
21impl GraphQLClient {
22    /// Return the resolved address for the given name.
23    pub async fn iota_names_lookup(&self, name: &str) -> Result<Option<Arc<Address>>> {
24        Ok(self
25            .0
26            .read()
27            .await
28            .iota_names_lookup(name)
29            .await?
30            .map(Into::into)
31            .map(Arc::new))
32    }
33
34    /// Find all registration NFTs for the given address.
35    pub async fn iota_names_registrations(
36        &self,
37        address: &Address,
38        pagination_filter: PaginationFilter,
39    ) -> Result<NameRegistrationPage> {
40        Ok(self
41            .0
42            .read()
43            .await
44            .iota_names_registrations(**address, pagination_filter.into())
45            .await?
46            .map(Into::into)
47            .into())
48    }
49
50    /// Get the default name pointing to this address, if one exists.
51    pub async fn iota_names_default_name(
52        &self,
53        address: &Address,
54        format: Option<NameFormat>,
55    ) -> Result<Option<Arc<Name>>> {
56        Ok(self
57            .0
58            .read()
59            .await
60            .iota_names_default_name(**address, format.map(Into::into))
61            .await?
62            .map(Into::into)
63            .map(Arc::new))
64    }
65}
