4//! Network API implementation.
5
6use crate::{
7    error::Result,
8    graphql::{
9        client::GraphQLClient,
10        pagination::ValidatorPage,
11        query_types::{PaginationFilter, ProtocolConfigs},
12    },
13};
14
15#[cfg_attr(not(target_arch = "wasm32"), uniffi::export(async_runtime = "tokio"))]
16#[cfg_attr(target_arch = "wasm32", uniffi::export)]
17impl GraphQLClient {
18    /// Get the chain identifier.
19    pub async fn chain_id(&self) -> Result<String> {
20        Ok(self.0.read().await.chain_id().await?)
21    }
22
23    /// Get the reference gas price for the provided epoch or the last known one
24    /// if no epoch is provided.
25    ///
26    /// This will return `Ok(None)` if the epoch requested is not available in
27    /// the GraphQL service (e.g., due to pruning).
28    #[uniffi::method(default(epoch = None))]
29    pub async fn reference_gas_price(&self, epoch: Option<u64>) -> Result<Option<u64>> {
30        Ok(self.0.read().await.reference_gas_price(epoch).await?)
31    }
32
33    /// Get the protocol configuration.
34    #[uniffi::method(default(version = None))]
35    pub async fn protocol_config(&self, version: Option<u64>) -> Result<ProtocolConfigs> {
36        Ok(self.0.read().await.protocol_config(version).await?.into())
37    }
38
39    /// Get the list of active validators for the provided epoch, including
40    /// related metadata. If no epoch is provided, it will return the active
41    /// validators for the current epoch.
42    #[uniffi::method(default(epoch = None, pagination_filter = None))]
43    pub async fn active_validators(
44        &self,
45        epoch: Option<u64>,
46        pagination_filter: Option<PaginationFilter>,
47    ) -> Result<ValidatorPage> {
48        Ok(self
49            .0
50            .read()
51            .await
52            .active_validators(epoch, pagination_filter.map(Into::into).unwrap_or_default())
53            .await?
54            .map(Into::into)
55            .into())
56    }
57}
