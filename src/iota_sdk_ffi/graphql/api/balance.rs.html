4//! Balance API implementation.
5
6use crate::{error::Result, graphql::client::GraphQLClient, types::address::Address};
7
8#[cfg_attr(not(target_arch = "wasm32"), uniffi::export(async_runtime = "tokio"))]
9#[cfg_attr(target_arch = "wasm32", uniffi::export)]
10impl GraphQLClient {
11    /// Get the balance of all the coins owned by address for the provided coin
12    /// type. Coin type will default to `0x2::coin::Coin<0x2::iota::IOTA>`
13    /// if not provided.
14    #[uniffi::method(default(coin_type = None))]
15    pub async fn balance(
16        &self,
17        address: &Address,
18        coin_type: Option<String>,
19    ) -> Result<Option<u64>> {
20        Ok(self.0.read().await.balance(**address, coin_type).await?)
21    }
22}
