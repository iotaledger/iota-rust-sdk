19/// A stream that yields items from a paginated query with support for bidirectional pagination.
20pub struct PageStream<T, F, Fut> {
21    query_fn: F,
22    direction: Direction,
23    limit: Option<i32>,
24    start_cursor: Option<String>,
25    /// The cursor the page in flight or being yielded was requested with.
26    requested_cursor: Option<String>,
27    current_page: Option<(PageInfo, std::vec::IntoIter<T>)>,
28    current_future: Option<Pin<Box<Fut>>>,
29    finished: bool,
30}
31
32impl<T, F, Fut> PageStream<T, F, Fut> {
33    pub fn new(query_fn: F, pagination: PaginationFilter) -> Self {
34        Self {
35            query_fn,
36            direction: pagination.direction,
37            limit: pagination.limit,
38            start_cursor: pagination.cursor,
39            requested_cursor: None,
40            current_page: None,
41            current_future: None,
42            finished: false,
43        }
44    }
45}
46
47impl<T, F, Fut> Stream for PageStream<T, F, Fut>
48where
49    T: Clone + Unpin,
50    F: Fn(PaginationFilter) -> Fut,
51    F: Unpin,
52    Fut: Future<Output = Result<Page<T>, error::GraphQLError>>,
53{
54    type Item = Result<T, error::GraphQLError>;
55
56    fn poll_next(mut self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Option<Self::Item>> {
57        if self.finished {
58            return Poll::Ready(None);
59        }
60
61        loop {
62            let direction = self.direction.clone();
63            // If we have a current page, return the next item
64            if let Some((page_info, iter)) = &mut self.current_page {
65                if let Some(item) = iter.next() {
66                    return Poll::Ready(Some(Ok(item)));
67                }
68
69                let should_continue = match direction {
70                    Direction::Forward => page_info.has_next_page,
71                    Direction::Backward => page_info.has_previous_page,
72                };
73                if !should_continue {
74                    self.finished = true;
75                    return Poll::Ready(None);
76                }
77            }
78
79            // Get cursor from current page
80            let current_cursor = self
81                .current_page
82                .as_ref()
83                .and_then(|(page_info, _iter)| match self.direction {
84                    Direction::Forward => page_info
85                        .has_next_page
86                        .then(|| page_info.end_cursor.clone()),
87                    Direction::Backward => page_info
88                        .has_previous_page
89                        .then(|| page_info.start_cursor.clone()),
90                })
91                .flatten();
92
93            // If there's no future yet, create one
94            if self.current_future.is_none() {
95                let current_cursor = if self.current_page.is_none() {
96                    self.start_cursor.take()
97                } else {
98                    current_cursor
99                };
100                self.requested_cursor = current_cursor.clone();
101                let filter = PaginationFilter {
102                    direction: self.direction.clone(),
103                    cursor: current_cursor,
104                    limit: self.limit,
105                };
106                let future = (self.query_fn)(filter);
107                self.current_future = Some(Box::pin(future));
108            }
109
110            // Poll the future
111            match self.current_future.as_mut().unwrap().as_mut().poll(cx) {
112                Poll::Ready(Ok(page)) => {
113                    self.current_future = None;
114
115                    let (page_info, data) = page.into_parts();
116                    let (has_more, cursor) = match self.direction {
117                        Direction::Forward => {
118                            (page_info.has_next_page, page_info.end_cursor.as_ref())
119                        }
120                        Direction::Backward => {
121                            (page_info.has_previous_page, page_info.start_cursor.as_ref())
122                        }
123                    };
124                    let reason = match (has_more, cursor) {
125                        (true, None) => Some("the server announced more pages but gave no cursor"),
126                        (true, Some(cursor)) if self.requested_cursor.as_ref() == Some(cursor) => {
127                            Some("the server repeated the cursor the page was requested with")
128                        }
129                        _ => None,
130                    };
131                    if let Some(reason) = reason {
132                        self.finished = true;
133                        return Poll::Ready(Some(Err(error::GraphQLError::PageNotAdvanced(
134                            reason,
135                        ))));
136                    }
137                    // For backward pagination, we need to reverse the items
138                    let iter = match self.direction {
139                        Direction::Forward => data.into_iter(),
140                        Direction::Backward => {
141                            let mut vec = data;
142                            vec.reverse();
143                            vec.into_iter()
144                        }
145                    };
146                    self.current_page = Some((page_info, iter));
147                }
148                Poll::Ready(Err(e)) => {
149                    self.finished = true;
150                    self.current_future = None;
151                    return Poll::Ready(Some(Err(e)));
152                }
153                Poll::Pending => return Poll::Pending,
154            }
155        }
156    }
157}

159/// Creates a new `PageStream` for a paginated query.
160pub fn stream_paginated_query<T, F, Fut>(
161    query_fn: F,
162    pagination: PaginationFilter,
163) -> PageStream<T, F, Fut>
164where
165    F: Fn(PaginationFilter) -> Fut,
166    Fut: Future<Output = Result<Page<T>, error::GraphQLError>>,
167{
168    PageStream::new(query_fn, pagination)
169}

184    macro_rules! first_request_of {
185        ($operation:expr, $query:expr) => {
186            first_request_of!($operation, $query, backward_page())
187        };
188        ($operation:expr, $query:expr, $page:expr) => {
189            sent_variables($operation, |client| async move {
190                let _ = $query(client).pagination($page).stream().next().await;
191            })
192            .await
193        };
194    }
195
196    fn page(items: Vec<i32>, has_more: bool, cursor: Option<&str>) -> Page<i32> {
197        Page::new(
198            PageInfo {
199                has_previous_page: has_more,
200                has_next_page: has_more,
201                start_cursor: cursor.map(|c| format!("{c}-backward")),
202                end_cursor: cursor.map(|c| format!("{c}-forward")),
203            },
204            items,
205        )
206    }
207
208    /// Stream `pages` in order and return the items and the requests made.
209    async fn run(
210        pagination: PaginationFilter,
211        pages: Vec<GraphQLResult<Page<i32>>>,
212    ) -> (Vec<GraphQLResult<i32>>, Vec<PaginationFilter>) {
213        let pages = std::sync::Mutex::new(pages.into_iter());
214        let requests = std::sync::Mutex::new(Vec::new());
215        let items = stream_paginated_query(
216            |filter: PaginationFilter| {
217                requests.lock().unwrap().push(filter);
218                let page = pages.lock().unwrap().next().expect("no page left");
219                async move { page }
220            },
221            pagination,
222        )
223        .collect()
224        .await;
225        (items, requests.into_inner().unwrap())
226    }
227
228    fn values(items: Vec<GraphQLResult<i32>>) -> Vec<i32> {
229        items.into_iter().map(Result::unwrap).collect()
230    }

232    #[tokio::test]
233    async fn forward_pages_continue_from_the_end_cursor_with_the_same_limit() {
234        let pagination = PaginationFilter {
235            direction: Direction::Forward,
236            cursor: Some("start".to_owned()),
237            limit: Some(3),
238        };
239        let (items, requests) = run(
240            pagination,
241            vec![
242                Ok(page(vec![1, 2], true, Some("second"))),
243                Ok(page(vec![3], false, Some("third"))),
244            ],
245        )
246        .await;
247
248        assert_eq!(values(items), [1, 2, 3]);
249        assert_eq!(requests.len(), 2);
250        assert_eq!(requests[0].cursor.as_deref(), Some("start"));
251        assert_eq!(requests[1].cursor.as_deref(), Some("second-forward"));
252        assert!(requests.iter().all(|r| r.limit == Some(3)));
253        assert!(
254            requests
255                .iter()
256                .all(|r| matches!(r.direction, Direction::Forward))
257        );
258    }
