162#[derive(Clone, cynic::QueryFragment, Debug, Default)]
163#[cynic(schema = "rpc", graphql_type = "PageInfo")]
164/// Information about pagination in a connection.
165pub struct PageInfo {
166    /// When paginating backwards, are there more items?
167    pub has_previous_page: bool,
168    /// Are there more items when paginating forwards?
169    pub has_next_page: bool,
170    /// When paginating backwards, the cursor to continue.
171    pub start_cursor: Option<String>,
172    /// When paginating forwards, the cursor to continue.
173    pub end_cursor: Option<String>,
174}

176impl TryFrom<BigInt> for u64 {
177    type Error = error::Error;
178
179    fn try_from(value: BigInt) -> Result<Self, Self::Error> {
180        Ok(value.0.parse::<u64>()?)
181    }
