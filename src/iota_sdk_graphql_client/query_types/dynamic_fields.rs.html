110impl DynamicFieldValue {
111    /// Returns the JSON representation of the field value, if available.
112    pub fn field_value_json(&self) -> Option<JsonValue> {
113        match self {
114            DynamicFieldValue::MoveObject(mo) => {
115                mo.contents.as_ref().and_then(|mv| mv.json.clone())
116            }
117            DynamicFieldValue::MoveValue(mv) => mv.json.clone(),
118            _ => None,
119        }
120    }
121
122    /// Return the typename and bcs of this dynamic field value.
123    pub fn type_bcs(&self) -> Option<crate::DynamicFieldValue> {
124        match self {
125            DynamicFieldValue::MoveObject(mo) => {
126                mo.contents.as_ref().map(|o| crate::DynamicFieldValue {
127                    type_tag: TypeTag::from_str(&o.move_type.repr.clone())
128                        .expect("Invalid TypeTag"),
129                    bcs: base64ct::Base64::decode_vec(&o.bcs.0).expect("Invalid Base64"),
130                })
131            }
132            DynamicFieldValue::MoveValue(mv) => Some(crate::DynamicFieldValue {
133                type_tag: TypeTag::from_str(&mv.move_type.repr.clone()).expect("Invalid TypeTag"),
134                bcs: base64ct::Base64::decode_vec(&mv.bcs.0).expect("Invalid Base64"),
135            }),
136            _ => None,
137        }
138    }
139}
140
141impl DynamicField {
142    /// Returns the JSON representation of the field value, if available.
143    pub fn field_value_json(&self) -> Option<JsonValue> {
144        self.value.as_ref().and_then(|v| v.field_value_json())
145    }
146}

148impl TryFrom<DynamicField> for DynamicFieldOutput {
149    type Error = error::Error;
150
151    fn try_from(val: DynamicField) -> Result<Self, Self::Error> {
152        let typetag = TypeTag::from_str(
153            val.name
154                .as_ref()
155                .expect("There should be a name in this dynamic field")
156                .move_type
157                .repr
158                .as_str(),
159        )?;
160        Ok(DynamicFieldOutput {
161            name: crate::DynamicFieldName {
162                type_tag: typetag,
163                bcs: base64ct::Base64::decode_vec(val.name.as_ref().unwrap().bcs.0.as_ref())
164                    .unwrap(),
165                json: val.name.as_ref().unwrap().json.clone(),
166            },
167            value_as_json: val.field_value_json(),
168            value: val.value.and_then(|x| x.type_bcs()),
169        })
170    }
171}
