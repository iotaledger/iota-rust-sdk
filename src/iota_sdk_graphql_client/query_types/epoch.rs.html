51#[derive(Clone, cynic::QueryFragment, Debug)]
52#[cynic(schema = "rpc", graphql_type = "Epoch")]
53pub struct Epoch {
54    /// The epoch's id as a sequence number that starts at 0 and is incremented
55    /// by one at every epoch change.
56    pub epoch_id: u64,
57    /// The storage fees paid for transactions executed during the epoch.
58    pub fund_inflow: Option<BigInt>,
59    /// The storage fee rebates paid to users who deleted the data associated
60    /// with past transactions.
61    pub fund_outflow: Option<BigInt>,
62    /// The storage fund available in this epoch.
63    /// This fund is used to redistribute storage fees from past transactions
64    /// to future validators.
65    pub fund_size: Option<BigInt>,
66    /// A commitment by the committee at the end of epoch on the contents of the
67    /// live object set at that time. This can be used to verify state
68    /// snapshots.
69    pub live_object_set_digest: Option<String>,
70    /// The difference between the fund inflow and outflow, representing
71    /// the net amount of storage fees accumulated in this epoch.
72    pub net_inflow: Option<BigInt>,
73    /// The epoch's corresponding protocol configuration, including the feature
74    /// flags and the configuration options.
75    pub protocol_configs: Option<ProtocolConfigs>,
76    /// The minimum gas price that a quorum of validators are guaranteed to sign
77    /// a transaction for.
78    pub reference_gas_price: Option<BigInt>,
79    /// The epoch's starting timestamp.
80    pub start_timestamp: DateTime,
81    /// The epoch's ending timestamp. Note that this is available only on epochs
82    /// that have ended.
83    pub end_timestamp: Option<DateTime>,
84    /// The value of the `version` field of `0x5`, the
85    /// `0x3::iota::IotaSystemState` object.  This version changes whenever
86    /// the fields contained in the system state object (held in a dynamic
87    /// field attached to `0x5`) change.
88    pub system_state_version: Option<u64>,
89    /// The total number of checkpoints in this epoch.
90    pub total_checkpoints: Option<u64>,
91    /// The total amount of gas fees (in IOTA) that were paid in this epoch.
92    pub total_gas_fees: Option<BigInt>,
93    /// The total IOTA rewarded as stake.
94    pub total_stake_rewards: Option<BigInt>,
95    /// The total number of transaction in this epoch.
96    pub total_transactions: Option<u64>,
97    /// Validator related properties. For active validators, see
98    /// `active_validators` API.
99    pub validator_set: Option<ValidatorSet>,
100}

102#[derive(Clone, cynic::QueryFragment, Debug)]
103#[cynic(schema = "rpc", graphql_type = "ValidatorSet")]
104pub struct ValidatorSet {
105    /// Object ID of the `Table` storing the inactive staking pools.
106    pub inactive_pools_id: Option<ObjectId>,
107    /// Size of the inactive pools `Table`.
108    pub inactive_pools_size: Option<i32>,
109    /// Object ID of the wrapped object `TableVec` storing the pending active
110    /// validators.
111    pub pending_active_validators_id: Option<ObjectId>,
112    /// Size of the pending active validators table.
113    pub pending_active_validators_size: Option<i32>,
114    /// Validators that are pending removal from the active validator set,
115    /// expressed as indices in to `activeValidators`.
116    pub pending_removals: Option<Vec<i32>>,
117    /// Object ID of the `Table` storing the mapping from staking pool ids to
118    /// the addresses of the corresponding validators. This is needed
119    /// because a validator's address can potentially change but the object
120    /// ID of its pool will not.
121    pub staking_pool_mappings_id: Option<ObjectId>,
122    /// Size of the stake pool mappings `Table`.
123    pub staking_pool_mappings_size: Option<i32>,
124    /// Total amount of stake for all active validators at the beginning of the
125    /// epoch.
126    pub total_stake: Option<BigInt>,
127    /// Size of the validator candidates `Table`.
128    pub validator_candidates_size: Option<i32>,
129    /// Object ID of the `Table` storing the validator candidates.
130    pub validator_candidates_id: Option<ObjectId>,
131}
