31/// The coin metadata associated with the given coin type.
32#[derive(Clone, cynic::QueryFragment, Debug)]
33#[cynic(schema = "rpc", graphql_type = "CoinMetadata")]
34pub struct CoinMetadata {
35    /// The CoinMetadata object ID.
36    pub address: ObjectId,
37    /// The number of decimal places used to represent the token.
38    pub decimals: Option<i32>,
39    /// Optional description of the token, provided by the creator of the token.
40    pub description: Option<String>,
41    /// Icon URL of the coin.
42    pub icon_url: Option<String>,
43    /// Full, official name of the token.
44    pub name: Option<String>,
45    /// The token's identifying abbreviation.
46    pub symbol: Option<String>,
47    /// The overall quantity of tokens that will be issued.
48    pub supply: Option<BigInt>,
49    /// Version of the token.
50    pub version: u64,
51}
