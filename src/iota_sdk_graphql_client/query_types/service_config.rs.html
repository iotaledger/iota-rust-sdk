21// Information about the configuration of the GraphQL service.
22#[derive(Clone, cynic::QueryFragment, Debug)]
23#[cynic(schema = "rpc", graphql_type = "ServiceConfig")]
24pub struct ServiceConfig {
25    /// Default number of elements allowed on a single page of a connection.
26    pub default_page_size: i32,
27    /// List of all features that are enabled on this RPC service.
28    pub enabled_features: Vec<Feature>,
29    // TODO This field is retrieved as a string, instead of i32
30    /// Maximum estimated cost of a database query used to serve a GraphQL
31    /// request.  This is measured in the same units that the database uses
32    /// in EXPLAIN queries.
33    // pub max_db_query_cost: i32,
34    /// Maximum nesting allowed in struct fields when calculating the layout of
35    /// a single Move Type.
36    pub max_move_value_depth: i32,
37    /// The maximum number of output nodes in a GraphQL response.
38    /// Non-connection nodes have a count of 1, while connection nodes are
39    /// counted as the specified 'first' or 'last' number of items, or the
40    /// default_page_size as set by the server if those arguments are not
41    /// set. Counts accumulate multiplicatively down the query tree. For
42    /// example, if a query starts with a connection of first: 10 and has a
43    /// field to a connection with last: 20, the count at the second level
44    /// would be 200 nodes. This is then summed to the count of 10 nodes
45    /// at the first level, for a total of 210 nodes.
46    pub max_output_nodes: i32,
47    /// Maximum number of elements allowed on a single page of a connection.
48    pub max_page_size: i32,
49    /// The maximum depth a GraphQL query can be to be accepted by this service.
50    pub max_query_depth: i32,
51    /// The maximum number of nodes (field names) the service will accept in a
52    /// single query.
53    pub max_query_nodes: i32,
54    /// Maximum length of a query payload string.
55    pub max_query_payload_size: i32,
56    /// Maximum nesting allowed in type arguments in Move Types resolved by this
57    /// service.
58    pub max_type_argument_depth: i32,
59    /// Maximum number of type arguments passed into a generic instantiation of
60    /// a Move Type resolved by this service.
61    pub max_type_argument_width: i32,
62    /// Maximum number of structs that need to be processed when calculating the
63    /// layout of a single Move Type.
64    pub max_type_nodes: i32,
65    /// Maximum time in milliseconds spent waiting for a response from fullnode
66    /// after issuing a transaction to execute. Note that the transaction
67    /// may still succeed even in the case of a timeout. Transactions are
68    /// idempotent, so a transaction that times out should be resubmitted
69    /// until the network returns a definite response (success or failure, not
70    /// timeout).
71    pub mutation_timeout_ms: i32,
72    /// Maximum time in milliseconds that will be spent to serve one query
73    /// request.
74    pub request_timeout_ms: i32,
75}
76
77impl ServiceConfig {
78    /// Whether `feature` is among those this RPC service has enabled.
79    pub fn supports_feature(&self, feature: Feature) -> bool {
80        self.enabled_features.contains(&feature)
81    }
82}
