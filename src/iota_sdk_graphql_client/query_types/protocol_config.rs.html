35/// Information about the configuration of the protocol.
36/// Constants that control how the chain operates.
37/// These can only change during protocol upgrades which happen on epoch
38/// boundaries.
39#[derive(Clone, cynic::QueryFragment, Debug)]
40#[cynic(schema = "rpc", graphql_type = "ProtocolConfigs")]
41pub struct ProtocolConfigs {
42    /// The protocol is not required to change on every epoch boundary, so the
43    /// protocol version tracks which change to the protocol these configs
44    /// are from.
45    pub protocol_version: u64,
46    /// List all available feature flags and their values. Feature flags are a
47    /// form of boolean configuration that are usually used to gate features
48    /// while they are in development. Once a flag has been enabled, it is
49    /// rare for it to be disabled.
50    pub feature_flags: Vec<ProtocolConfigFeatureFlag>,
51    /// List all available configurations and their values. These configurations
52    /// can take any value (but they will all be represented in string
53    /// form), and do not include feature flags.
54    pub configs: Vec<ProtocolConfigAttr>,
55}
56
57/// Feature flags are a form of boolean configuration that are usually used to
58/// gate features while they are in development. Once a lag has been enabled, it
59/// is rare for it to be disabled.
60#[derive(Clone, cynic::QueryFragment, Debug)]
61#[cynic(schema = "rpc", graphql_type = "ProtocolConfigFeatureFlag")]
62pub struct ProtocolConfigFeatureFlag {
63    pub key: String,
64    pub value: bool,
65}
66
67/// A key-value protocol configuration attribute.
68#[derive(Clone, cynic::QueryFragment, Debug)]
69#[cynic(schema = "rpc", graphql_type = "ProtocolConfigAttr")]
70pub struct ProtocolConfigAttr {
71    pub key: String,
72    pub value: Option<String>,
73}
