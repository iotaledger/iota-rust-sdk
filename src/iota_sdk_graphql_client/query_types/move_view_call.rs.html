6/// GraphQL query for executing a Move View Function.
7///
8/// A View Function is a function in a Move module with a return type that does
9/// not alter the state of the ledger. When using the Move View Function
10/// interface, no transactions are submitted to the network for inclusion into
11/// the ledger.
12///
13/// Move View Functions are callable via this RPC method that supports type
14/// parameters and function arguments. The use of this interface does not
15/// require signature checks or gas coins, as it does not alter ledger state.
16/// Spam attacks are dealt with at the RPC level rather than execution level.
17///
18/// Returned results are resolved (Move types deserialized) and formatted in
19/// JSON.
20#[derive(cynic::QueryFragment, Debug)]
21#[cynic(schema = "rpc", graphql_type = "Query", variables = "MoveViewCallArgs")]
22pub struct MoveViewCallQuery {
23    #[arguments(functionName: $function_name, typeArgs: $type_arguments, arguments: $arguments)]
24    pub move_view_call: MoveViewResult,
25}
26
27/// The result of executing a Move View Function.
28///
29/// Execution errors are captured in the `error` field, in which case the
30/// `results` field will be `None`. On success, the `results` field will contain
31/// the return values of the Move view function, and the `error` field will be
32/// `None`.
33#[derive(cynic::QueryFragment, Debug)]
34#[cynic(schema = "rpc", graphql_type = "MoveViewResult")]
35pub struct MoveViewResult {
36    /// Execution error from executing the Move view function.
37    pub error: Option<String>,
38    /// The return values of the Move view function, resolved and formatted as
39    /// JSON.
40    pub results: Option<Vec<JsonValue>>,
41}
42
43/// Arguments for the Move View Call GraphQL query.
44///
45/// The function name should be fully qualified as
46/// `<package_id>::<module_name>::<function_name>`,
47/// e.g., `0x533074f8e22e8ce1330d7e9d67c18966abb5a3d58dc2e2deea50e50bea4e87f4::shop::total_revenue`.
48#[derive(cynic::QueryVariables, Debug)]
49pub struct MoveViewCallArgs {
50    /// The Move function fully qualified name as
51    /// `<package_id>::<module_name>::<function_name>`.
52    pub function_name: String,
53    /// The type arguments of the Move function.
54    pub type_arguments: Option<Vec<String>>,
55    /// The arguments to be passed into the Move function, in JSON format.
56    pub arguments: Option<Vec<JsonValue>>,
57}
