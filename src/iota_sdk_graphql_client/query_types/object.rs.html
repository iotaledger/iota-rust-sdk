55#[derive(Clone, cynic::InputObject, Debug, Default)]
56#[cynic(schema = "rpc", graphql_type = "ObjectFilter")]
57#[non_exhaustive]
58pub struct ObjectFilter {
59    #[cynic(rename = "type")]
60    pub type_tag: Option<String>,
61    pub owner: Option<Address>,
62    pub object_ids: Option<Vec<ObjectId>>,
63}
64
65impl ObjectFilter {
66    /// Filter by package, module, or fully qualified type, e.g. `"0x02"`,
67    /// `"0x02::coin"`, or `"0x02::coin::Coin"`.
68    pub fn with_type(mut self, type_tag: impl Into<Option<String>>) -> Self {
69        self.type_tag = type_tag.into();
70        self
71    }
72
73    /// Filter by the address owning the object.
74    pub fn with_owner(mut self, owner: impl Into<Option<Address>>) -> Self {
75        self.owner = owner.into();
76        self
77    }
78
79    /// Filter by object ids.
80    pub fn with_object_ids(mut self, object_ids: impl Into<Option<Vec<ObjectId>>>) -> Self {
81        self.object_ids = object_ids.into();
82        self
83    }
84}
