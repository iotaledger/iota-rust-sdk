72mod base64 {
73    use base64ct::Encoding;
74
75    use crate::error::{GraphQLError, GraphQLResult};

77    /// Decodes a base64 string from a response into bytes.
78    pub(crate) fn decode(input: &str) -> GraphQLResult<Vec<u8>> {
79        base64ct::Base64::decode_vec(input).map_err(|e| GraphQLError::Parse(e.into()))
80    }
