20impl Client {
21    /// Return a stream of objects based on the (optional) object filter.
22    pub fn objects_stream(
23        &self,
24        filter: impl Into<Option<ObjectFilter>>,
25        streaming_direction: Direction,
26    ) -> impl Stream<Item = Result<Object>> + '_ {
27        let filter = filter.into();
28        stream_paginated_query(
29            move |pag_filter| self.objects(filter.clone(), pag_filter),
30            streaming_direction,
31        )
32    }

34    /// Return an object based on the provided [`Address`](iota_types::Address).
35    ///
36    /// If the object does not exist (e.g., due to pruning), this will return
37    /// `Ok(None)`. Similarly, if this is not an object but an address, it
38    /// will return `Ok(None)`.
39    pub async fn object(
40        &self,
41        object_id: ObjectId,
42        version: impl Into<Option<Version>>,
43    ) -> Result<Option<Object>> {
44        let operation = ObjectQuery::build(ObjectQueryArgs {
45            object_id,
46            version: version.into().map(|v| v.as_u64()),
47        });
48
49        let response = self.run_query(&operation).await?;
50
51        let obj = response.object;
52        let bcs = obj
53            .and_then(|o| o.bcs)
54            .map(|bcs| base64ct::Base64::decode_vec(bcs.0.as_str()))
55            .transpose()?;
56
57        let object = bcs
58            .map(|b| bcs::from_bytes::<iota_types::Object>(&b))
59            .transpose()?;
60
61        Ok(object)
62    }
63
64    /// Return a page of objects based on the provided parameters.
65    ///
66    /// Use this function together with
67    /// [`ObjectFilter::with_owner`] to get the objects owned by an address.
68    ///
69    /// # Example
70    ///
71    /// ```rust,ignore
72    /// let filter = ObjectFilter::default().with_owner(Address::from_str("test").unwrap());
73    ///
74    /// let owned_objects = client.objects(filter, PaginationFilter::default()).await;
75    /// ```
76    pub async fn objects(
77        &self,
78        filter: impl Into<Option<ObjectFilter>>,
79        pagination_filter: PaginationFilter,
80    ) -> Result<Page<Object>> {
81        let pagination = self.pagination_filter(pagination_filter).await;
82        let operation = ObjectsQuery::build(ObjectsQueryArgs {
83            after: pagination.after,
84            before: pagination.before,
85            filter: filter.into(),
86            first: pagination.first,
87            last: pagination.last,
88        });
89
90        let response = self.run_query(&operation).await?;
91
92        let oc = response.objects;
93        let page_info = oc.page_info;
94        let bcs = oc
95            .nodes
96            .iter()
97            .map(|o| &o.bcs)
98            .filter_map(|b64| {
99                b64.as_ref()
100                    .map(|b| base64ct::Base64::decode_vec(b.0.as_str()))
101            })
102            .collect::<Result<Vec<_>, base64ct::Error>>()?;
103        let objects = bcs
104            .iter()
105            .map(|b| bcs::from_bytes::<iota_types::Object>(b))
106            .collect::<Result<Vec<_>, bcs::Error>>()?;
107
108        Ok(Page::new(page_info, objects))
109    }

111    /// Return the object's bcs content [`Vec<u8>`] based on the provided
112    /// [`Address`](iota_types::Address).
113    pub async fn object_bcs(&self, object_id: ObjectId) -> Result<Option<Vec<u8>>> {
114        let operation = ObjectQuery::build(ObjectQueryArgs {
115            object_id,
116            version: None,
117        });
118
119        let response = self.run_query(&operation).await.unwrap();
120
121        Ok(response
122            .object
123            .and_then(|o| {
124                o.bcs
125                    .map(|bcs| base64ct::Base64::decode_vec(bcs.0.as_str()))
126            })
127            .transpose()?)
128    }
129
130    /// Return the contents JSON of an object that is a Move object.
131    ///
132    /// If the object does not exist (e.g., due to pruning), this will return
133    /// `Ok(None)`. Similarly, if this is not an object but an address, it
134    /// will return `Ok(None)`.
135    pub async fn move_object_contents(
136        &self,
137        object_id: ObjectId,
138        version: impl Into<Option<Version>>,
139    ) -> Result<Option<serde_json::Value>> {
140        let operation = ObjectQuery::build(ObjectQueryArgs {
141            object_id,
142            version: version.into().map(|v| v.as_u64()),
143        });
144
145        let response = self.run_query(&operation).await?;
146
147        Ok(response
148            .object
149            .and_then(|o| o.as_move_object)
150            .and_then(|o| o.contents)
151            .and_then(|mv| mv.json))
152    }
153
154    /// Return the BCS of an object that is a Move object.
155    ///
156    /// If the object does not exist (e.g., due to pruning), this will return
157    /// `Ok(None)`. Similarly, if this is not an object but an address, it
158    /// will return `Ok(None)`.
159    pub async fn move_object_contents_bcs(
160        &self,
161        object_id: ObjectId,
162        version: impl Into<Option<Version>>,
163    ) -> Result<Option<Vec<u8>>> {
164        let operation = ObjectQuery::build(ObjectQueryArgs {
165            object_id,
166            version: version.into().map(|v| v.as_u64()),
167        });
168
169        let response = self.run_query(&operation).await?;
170
171        Ok(response
172            .object
173            .and_then(|o| o.as_move_object)
174            .and_then(|o| o.contents)
175            .map(|bcs| base64ct::Base64::decode_vec(bcs.bcs.0.as_str()))
176            .transpose()?)
177    }
