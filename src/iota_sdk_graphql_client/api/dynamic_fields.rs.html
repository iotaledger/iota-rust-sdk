23impl Client {
24    /// Get a stream of dynamic fields for the provided address. Note that this
25    /// will also fetch dynamic fields on wrapped objects.
26    pub fn dynamic_fields_stream(
27        &self,
28        address: Address,
29        streaming_direction: Direction,
30    ) -> impl Stream<Item = Result<DynamicFieldOutput>> + '_ {
31        stream_paginated_query(
32            move |filter| self.dynamic_fields(address, filter),
33            streaming_direction,
34        )
35    }

37    /// Access a dynamic field on an object using its name. Names are arbitrary
38    /// Move values whose type have copy, drop, and store, and are specified
39    /// using their type, and their BCS contents, Base64 encoded.
40    ///
41    /// The `name` argument can be either a [`BcsName`](crate::BcsName) for
42    /// passing raw bcs bytes or a type that implements Serialize.
43    ///
44    /// This returns [`DynamicFieldOutput`] which contains the name, the value
45    /// as json, and object.
46    ///
47    /// # Example
48    /// ```rust,ignore
49    /// 
50    /// let client = iota_graphql_client::Client::new_testnet();
51    /// let address = ObjectId::system().into();
52    /// let df = client.dynamic_field_with_name(address, "u64", 2u64).await.unwrap();
53    ///
54    /// # alternatively, pass in the bcs bytes
55    /// let bcs = base64ct::Base64::decode_vec("AgAAAAAAAAA=").unwrap();
56    /// let df = client.dynamic_field(address, "u64", BcsName(bcs)).await.unwrap();
57    /// ```
58    pub async fn dynamic_field(
59        &self,
60        address: Address,
61        type_tag: TypeTag,
62        name: impl Into<NameValue>,
63    ) -> Result<Option<DynamicFieldOutput>> {
64        let bcs = name.into().0;
65        let operation = DynamicFieldQuery::build(DynamicFieldArgs {
66            address,
67            name: crate::query_types::DynamicFieldName {
68                type_tag: type_tag.to_string(),
69                bcs: crate::query_types::Base64(base64ct::Base64::encode_string(&bcs)),
70            },
71        });
72
73        let response = self.run_query(&operation).await?;
74
75        let result = response
76            .owner
77            .and_then(|o| o.dynamic_field)
78            .map(|df| df.try_into())
79            .transpose()?;
80
81        Ok(result)
82    }
83
84    /// Access a dynamic object field on an object using its name. Names are
85    /// arbitrary Move values whose type have copy, drop, and store, and are
86    /// specified using their type, and their BCS contents, Base64 encoded.
87    ///
88    /// The `name` argument can be either a [`BcsName`](crate::BcsName) for
89    /// passing raw bcs bytes or a type that implements Serialize.
90    ///
91    /// This returns [`DynamicFieldOutput`] which contains the name, the value
92    /// as json, and object.
93    pub async fn dynamic_object_field(
94        &self,
95        address: Address,
96        type_tag: TypeTag,
97        name: impl Into<NameValue>,
98    ) -> Result<Option<DynamicFieldOutput>> {
99        let bcs = name.into().0;
100        let operation = DynamicObjectFieldQuery::build(DynamicFieldArgs {
101            address,
102            name: crate::query_types::DynamicFieldName {
103                type_tag: type_tag.to_string(),
104                bcs: crate::query_types::Base64(base64ct::Base64::encode_string(&bcs)),
105            },
106        });
107
108        let response = self.run_query(&operation).await?;
109
110        let result: Option<DynamicFieldOutput> = response
111            .owner
112            .and_then(|o| o.dynamic_object_field)
113            .map(|df| df.try_into())
114            .transpose()?;
115        Ok(result)
116    }

118    /// Get a page of dynamic fields for the provided address. Note that this
119    /// will also fetch dynamic fields on wrapped objects.
120    ///
121    /// This returns [`Page`] of [`DynamicFieldOutput`]s.
122    pub async fn dynamic_fields(
123        &self,
124        address: Address,
125        pagination_filter: PaginationFilter,
126    ) -> Result<Page<DynamicFieldOutput>> {
127        let pagination = self.pagination_filter(pagination_filter).await;
128        let operation = DynamicFieldsOwnerQuery::build(DynamicFieldConnectionArgs {
129            address,
130            after: pagination.after.as_deref(),
131            before: pagination.before.as_deref(),
132            first: pagination.first,
133            last: pagination.last,
134        });
135        let response = self.run_query(&operation).await?;
136
137        let DynamicFieldsOwnerQuery { owner: Some(dfs) } = response else {
138            return Ok(Page::new_empty());
139        };
140
141        Ok(Page::new(
142            dfs.dynamic_fields.page_info,
143            dfs.dynamic_fields
144                .nodes
145                .into_iter()
146                .map(TryInto::try_into)
147                .collect::<Result<Vec<_>>>()?,
148        ))
149    }
150}
