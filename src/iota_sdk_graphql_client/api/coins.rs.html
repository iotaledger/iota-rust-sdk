19impl Client {
20    /// Get the list of coins for the specified address as a stream.
21    ///
22    /// If `coin_type` is not provided, all coins will be returned. For IOTA
23    /// coins, pass in the coin type: `0x2::iota::IOTA`.
24    pub fn coins_stream(
25        &self,
26        address: Address,
27        coin_type: impl Into<Option<StructTag>>,
28        streaming_direction: Direction,
29    ) -> impl Stream<Item = Result<Coin>> + '_ {
30        let coin_type = coin_type.into();
31        stream_paginated_query(
32            move |filter| self.coins(address, coin_type.clone(), filter),
33            streaming_direction,
34        )
35    }

43        stream_paginated_query(

49    /// Get the list of coins for the specified address.
50    ///
51    /// If `coin_type` is not provided, all coins will be returned. For IOTA
52    /// coins, pass in the coin type: `0x2::iota::IOTA`.
53    pub async fn coins(
54        &self,
55        owner: Address,
56        coin_type: impl Into<Option<StructTag>>,
57        pagination_filter: PaginationFilter,
58    ) -> Result<Page<Coin>> {
59        let filter = ObjectFilter {
60            type_tag: Some(
61                coin_type
62                    .into()
63                    .map(StructTag::new_coin)
64                    .unwrap_or_else(|| {
65                        StructTag::new(
66                            Address::FRAMEWORK,
67                            Identifier::from_static("coin"),
68                            Identifier::from_static("Coin"),
69                            Default::default(),
70                        )
71                    })
72                    .to_string(),
73            ),
74            owner: Some(owner),
75            object_ids: None,
76        };
77        let response = self.objects(filter, pagination_filter).await?;
78
79        Ok(Page::new(
80            response.page_info,
81            response
82                .data
83                .iter()
84                .flat_map(Coin::try_from_object)
85                .collect::<Vec<_>>(),
86        ))
87    }
88
89    /// Get the list of gas coins for the specified address.
90    pub async fn gas_coins(
91        &self,
92        owner: Address,
93        pagination_filter: PaginationFilter,
94    ) -> Result<Page<Coin>> {
95        self.coins(owner, StructTag::new_gas(), pagination_filter)
96            .await
97    }
98
99    /// Get the coin metadata for the coin type.
100    pub async fn coin_metadata(&self, coin_type: &str) -> Result<Option<CoinMetadata>> {
101        let operation = CoinMetadataQuery::build(CoinMetadataArgs { coin_type });
102        let response = self.run_query(&operation).await?;
103
104        Ok(response.coin_metadata)
105    }
106
107    /// Get total supply for the coin type.
108    pub async fn total_supply(&self, coin_type: &str) -> Result<Option<u64>> {
109        let coin_metadata = self.coin_metadata(coin_type).await?;
110
111        coin_metadata
112            .and_then(|c| c.supply)
113            .map(|c| c.try_into())
114            .transpose()
115    }
116}

146    #[tokio::test]
147    async fn test_coins_stream() {
148        let client = test_client();
149        let faucet = match client.rpc_server().as_str() {
150            LOCAL_HOST => FaucetClient::new_localnet(),
151            // The testnet and devnet faucets are web-only and expose no
152            // programmatic gas endpoint, so this test can only fund an
153            // address on localnet.
154            _ => return,
155        };
156        let key = Ed25519PublicKey::random_with(rand::thread_rng());
157        let address = key.derive_address();
158        faucet
159            .request_and_wait_for_finalized(address, &client)
160            .await
161            .unwrap();
162
163        const MAX_RETRIES: u32 = 10;
164        const RETRY_DELAY: time::Duration = time::Duration::from_secs(1);
165
166        // Retry until the expected number of coins is streamed. Indexer lag can
167        // return a successful but incomplete page, so retry on a low count as
168        // well as on a stream error, re-counting from scratch each attempt.
169        let mut num_coins = 0;
170        for attempt in 0..MAX_RETRIES {
171            num_coins = 0;
172            let mut stream = client.coins_stream(address, None, Direction::default());
173            let mut errored = false;
174            while let Some(result) = stream.next().await {
175                match result {
176                    Ok(_) => num_coins += 1,
177                    Err(_) => {
178                        errored = true;
179                        break;
180                    }
181                }
182            }
183
184            if !errored && num_coins >= NUM_COINS_FROM_FAUCET {
185                break;
186            }
187            if attempt < MAX_RETRIES - 1 {
188                time::sleep(RETRY_DELAY).await;
189            }
190        }
191
192        assert!(
193            num_coins >= NUM_COINS_FROM_FAUCET,
194            "expected at least {NUM_COINS_FROM_FAUCET} coins for {address}, got {num_coins}"
195        );
196    }
