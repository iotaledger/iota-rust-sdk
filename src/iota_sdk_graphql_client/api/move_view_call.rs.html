15define_query! {
16    /// Query for [`GraphQLClient::move_view_call_json`]. Await it to send the request.
17    pub struct MoveViewCallJsonQuery {
18        client: GraphQLClient,
19        function_name: String,
20        type_arguments: Option<Vec<String>>,
21        arguments: Option<Vec<serde_json::Value>>,
22    }
23    output: GraphQLResult<MoveViewResult>;
24}
25
26impl MoveViewCallJsonQuery {
27    /// Set the type arguments of the Move function.
28    pub fn type_arguments(mut self, type_arguments: Vec<String>) -> Self {
29        self.type_arguments = Some(type_arguments);
30        self
31    }
32
33    /// Set the arguments passed into the Move function, in JSON format.
34    pub fn arguments(mut self, arguments: Vec<serde_json::Value>) -> Self {
35        self.arguments = Some(arguments);
36        self
37    }
38
39    async fn send(self) -> GraphQLResult<MoveViewResult> {
40        let operation = MoveViewCallQueryFragment::build(MoveViewCallArgs {
41            function_name: self.function_name,
42            type_arguments: self.type_arguments,
43            arguments: self.arguments,
44        });
45        let response = self.client.run_query(&operation).await?;
46
47        Ok(response.move_view_call)
48    }
49}

51define_query! {
52    /// Query for [`GraphQLClient::move_view_call`]. Await it to send the request.
53    pub struct MoveViewCallQuery {
54        client: GraphQLClient,
55        function_name: String,
56        type_arguments: Option<Vec<TypeTag>>,
57        arguments: Option<Vec<serde_json::Value>>,
58    }
59    output: GraphQLResult<MoveViewResult>;
60}
61
62impl MoveViewCallQuery {
63    /// Set the type arguments of the Move function.
64    pub fn type_arguments(mut self, type_arguments: Vec<TypeTag>) -> Self {
65        self.type_arguments = Some(type_arguments);
66        self
67    }
68
69    /// Set the type arguments of the Move function from Rust types, e.g. `generics::<(u64,
70    /// String)>()`.
71    pub fn generics<G: MoveTypes>(mut self) -> Self {
72        self.type_arguments = Some(G::type_tags());
73        self
74    }
75
76    /// Set the typed arguments passed into the Move function, replacing the ones set so far. A
77    /// single argument is wrapped in a list or tuple.
78    pub fn arguments(mut self, arguments: impl MoveViewArgList) -> Self {
79        self.arguments = Some(arguments.to_json_vec());
80        self
81    }
82
83    /// Append a single typed argument passed into the Move function.
84    ///
85    /// A collection is appended as one argument, i.e. a Move vector; use
86    /// [`arguments`](Self::arguments) to pass a collection as the whole argument list.
87    pub fn argument(mut self, argument: impl MoveViewArg) -> Self {
88        self.arguments
89            .get_or_insert_default()
90            .push(argument.to_json());
91        self
92    }
93
94    async fn send(self) -> GraphQLResult<MoveViewResult> {
95        let type_arguments = self
96            .type_arguments
97            .map(|tags| tags.into_iter().map(|t| t.to_string()).collect());
98        MoveViewCallJsonQuery {
99            client: self.client,
100            function_name: self.function_name,
101            type_arguments,
102            arguments: self.arguments,
103        }
104        .await
105    }
106}

108/// The fully qualified name of a Move function, as `<package_id>::<module_name>::<function_name>`.
109fn function_name(package: ObjectId, module: &str, function: &str) -> String {
110    format!("{package}::{module}::{function}")
111}

113impl GraphQLClient {
114    /// Execute a Move View Function with raw JSON arguments.
115    ///
116    /// This is an alternative to [`GraphQLClient::move_view_call`] that accepts raw JSON values
117    /// instead of typed arguments.
118    ///
119    /// A View Function is a function in a Move module with a return type that does not alter the
120    /// state of the ledger. When using this interface, no transactions are submitted to the network
121    /// for inclusion into the ledger.
122    ///
123    /// The function is `function` in the module `module` of the package `package`. Set its type
124    /// arguments with [`type_arguments`](MoveViewCallJsonQuery::type_arguments) and its JSON
125    /// arguments with [`arguments`](MoveViewCallJsonQuery::arguments).
126    ///
127    /// Resolves to a `MoveViewResult` containing either execution results (return values) or an
128    /// error.
129    pub fn move_view_call_json(
130        &self,
131        package: impl Into<ObjectId>,
132        module: impl Into<String>,
133        function: impl Into<String>,
134    ) -> MoveViewCallJsonQuery {
135        MoveViewCallJsonQuery {
136            client: self.clone(),
137            function_name: function_name(package.into(), &module.into(), &function.into()),
138            type_arguments: None,
139            arguments: None,
140        }
141    }
142
143    /// Execute a Move View Function.
144    ///
145    /// A View Function is a function in a Move module with a return type that does not alter the
146    /// state of the ledger. When using this interface, no transactions are submitted to the network
147    /// for inclusion into the ledger.
148    ///
149    /// This method allows calling nearly any Move function with a return type and any arguments.
150    /// The function's result values are provided and decoded using the appropriate Move type, then
151    /// formatted in JSON.
152    ///
153    /// The use of this interface does not require signature checks (even for functions that take
154    /// Owned Objects as input) or gas coins, as it does not alter ledger state. Spam attacks are
155    /// dealt with at the RPC level rather than execution level.
156    ///
157    /// The function is `function` in the module `module` of the package `package`. Set its typed
158    /// arguments with [`arguments`](MoveViewCallQuery::arguments) and its type arguments with
159    /// [`type_arguments`](MoveViewCallQuery::type_arguments).
160    ///
161    /// # Example
162    /// ```rust,ignore
163    /// // The `view_demo` package published on testnet, and the shared
164    /// // `view_demo::shop::Shop` created when it was published.
165    /// let package = ObjectId::from_str(
166    ///     "0x533074f8e22e8ce1330d7e9d67c18966abb5a3d58dc2e2deea50e50bea4e87f4",
167    /// )?;
168    /// let shop = ObjectId::from_str(
169    ///     "0x9d5ce0da7531d56ffecced5efb7e19ccad0e191071041267cc8134a3e5a6cd20",
170    /// )?;
171    ///
172    /// // Single argument: wrap in a list or tuple
173    /// let result = client
174    ///     .move_view_call(package, "shop", "total_revenue")
175    ///     .arguments((shop,))
176    ///     .await?;
177    ///
178    /// // Or append it
179    /// let result = client
180    ///     .move_view_call(package, "shop", "total_revenue")
181    ///     .argument(shop)
182    ///     .await?;
183    /// ```
184    ///
185    /// Resolves to a `MoveViewResult` containing either execution results (return values) or an
186    /// error.
187    pub fn move_view_call(
188        &self,
189        package: impl Into<ObjectId>,
190        module: impl Into<String>,
191        function: impl Into<String>,
192    ) -> MoveViewCallQuery {
193        MoveViewCallQuery {
194            client: self.clone(),
195            function_name: function_name(package.into(), &module.into(), &function.into()),
196            type_arguments: None,
197            arguments: None,
198        }
199    }
200}

202/// A trait which defines a single argument for a Move View Function call.
203#[diagnostic::on_unimplemented(message = "Provided value is not a valid Move view argument.")]
204pub trait MoveViewArg {
205    /// Convert this argument to a JSON value for the GraphQL API.
206    fn to_json(self) -> serde_json::Value;
207}
208
209// Macro for types that convert to JSON Number
210macro_rules! impl_move_view_arg_number {
211    ($($ty:ty),* $(,)?) => {
212        $(
213            impl MoveViewArg for $ty {
214                fn to_json(self) -> serde_json::Value {
215                    serde_json::Value::Number(self.into())
216                }
217            }
218
219            impl MoveViewArg for &$ty {
220                fn to_json(self) -> serde_json::Value {
221                    (*self).to_json()
222                }
223            }
224        )*
225    };
226}
227
228// Macro for types that convert to JSON String via to_string()
229macro_rules! impl_move_view_arg_string {
230    ($($ty:ty),* $(,)?) => {
231        $(
232            impl MoveViewArg for $ty {
233                fn to_json(self) -> serde_json::Value {
234                    serde_json::Value::String(self.to_string())
235                }
236            }
237
238            impl MoveViewArg for &$ty {
239                fn to_json(self) -> serde_json::Value {
240                    (*self).to_json()
241                }
242            }
243        )*
244    };
245}

351/// A trait which defines a list of arguments for a Move View Function call.
352#[diagnostic::on_unimplemented(
353    message = "Provided value is not a valid list of Move view arguments.",
354    note = "Expected a tuple, vector, array, or slice of types that implement `MoveViewArg`."
355)]
356pub trait MoveViewArgList: sealed::Sealed {
357    /// Convert the arguments to a vector of JSON values.
358    fn to_json_vec(self) -> Vec<serde_json::Value>;
359}

361// Single element tuple implementation
362impl<T: MoveViewArg> MoveViewArgList for (T,) {
363    fn to_json_vec(self) -> Vec<serde_json::Value> {
364        vec![self.0.to_json()]
365    }
366}

369    fn to_json_vec(self) -> Vec<serde_json::Value> {
370        self.into_iter().map(|v| v.to_json()).collect()
371    }
372}

374impl<const N: usize, T: MoveViewArg> MoveViewArgList for [T; N] {
375    fn to_json_vec(self) -> Vec<serde_json::Value> {
376        self.into_iter().map(|v| v.to_json()).collect()
377    }
378}

380impl<T> MoveViewArgList for &[T]
381where
382    for<'a> &'a T: MoveViewArg,
383{
384    fn to_json_vec(self) -> Vec<serde_json::Value> {
385        self.iter().map(|v| v.to_json()).collect()
386    }
387}
388
389// Tuple implementations using a macro
390macro_rules! impl_move_view_args_tuple {
391    ($(($n:tt, $T:ident)),*) => {
392        impl<$($T),+> sealed::Sealed for ($($T),+) {}
393
394        impl<$($T),+> MoveViewArgList for ($($T),+)
395        where $($T: MoveViewArg),+
396        {
397            fn to_json_vec(self) -> Vec<serde_json::Value> {
398                vec![
399                    $(
400                        self.$n.to_json()
401                    ),+
402                ]
403            }
404        }
405    };
406}

421    use iota_types::{Address, ObjectId, TypeTag};

423    use crate::test_utils::sent_variables;

425    #[tokio::test]
426    async fn move_view_calls_send_the_function_type_arguments_and_arguments() {
427        let vars = sent_variables("MoveViewCallQueryFragment", |client| async move {
428            let _ = client
429                .move_view_call(Address::FRAMEWORK, "coin", "value")
430                .arguments((21u64,))
431                .type_arguments(vec![TypeTag::U64])
432                .await;
433        })
434        .await;
435        assert_eq!(
436            vars["functionName"],
437            format!("{}::coin::value", ObjectId::from(Address::FRAMEWORK))
438        );
439        assert_eq!(vars["typeArguments"], serde_json::json!(["u64"]));
440        assert_eq!(vars["arguments"], serde_json::json!(["21"]));
441
442        let vars = sent_variables("MoveViewCallQueryFragment", |client| async move {
443            let _ = client
444                .move_view_call_json(Address::FRAMEWORK, "coin", "value")
445                .type_arguments(vec!["u64".to_owned()])
446                .arguments(vec![serde_json::json!("21")])
447                .await;
448        })
449        .await;
450        assert_eq!(
451            vars["functionName"],
452            format!("{}::coin::value", ObjectId::from(Address::FRAMEWORK))
453        );
454        assert_eq!(vars["typeArguments"], serde_json::json!(["u64"]));
455        assert_eq!(vars["arguments"], serde_json::json!(["21"]));
456
457        let vars = sent_variables("MoveViewCallQueryFragment", |client| async move {
458            let _ = client
459                .move_view_call(Address::FRAMEWORK, "coin", "value")
460                .await;
461        })
462        .await;
463        assert!(vars["typeArguments"].is_null());
464        assert!(vars["arguments"].is_null());
465
466        let vars = sent_variables("MoveViewCallQueryFragment", |client| async move {
467            let _ = client
468                .move_view_call_json(Address::FRAMEWORK, "coin", "value")
469                .await;
470        })
471        .await;
472        assert!(vars["typeArguments"].is_null());
473        assert!(vars["arguments"].is_null());
474    }

476    #[tokio::test]
477    async fn move_view_calls_append_arguments_and_take_type_arguments_from_generics() {
478        let vars = sent_variables("MoveViewCallQueryFragment", |client| async move {
479            let _ = client
480                .move_view_call(Address::FRAMEWORK, "coin", "value")
481                .arguments((1u8, "a"))
482                .argument(u64::MAX)
483                .argument(vec![2u8, 3])
484                .generics::<(u64, String)>()
485                .await;
486        })
487        .await;
488        assert_eq!(
489            vars["arguments"],
490            serde_json::json!([1, "a", u64::MAX.to_string(), [2, 3]])
491        );
492        assert_eq!(
493            vars["typeArguments"],
494            serde_json::json!(["u64", "0x1::string::String"])
495        );
496
497        let vars = sent_variables("MoveViewCallQueryFragment", |client| async move {
498            let _ = client
499                .move_view_call(Address::FRAMEWORK, "coin", "value")
500                .argument(21u64)
501                .await;
502        })
503        .await;
504        assert_eq!(vars["arguments"], serde_json::json!(["21"]));
505    }
