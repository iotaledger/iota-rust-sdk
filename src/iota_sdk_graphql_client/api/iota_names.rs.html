5//! IOTA Names API implementation.
6
7use std::str::FromStr;
8
9use cynic::QueryBuilder;
10use iota_types::{
11    Address,
12    iota_names::{NameFormat, NameRegistration, name::Name},
13};
14
15use crate::{
16    Client,
17    error::{Error, Kind, Result},
18    pagination::{Page, PaginationFilter},
19    query_types::{
20        IotaNamesAddressDefaultNameQuery, IotaNamesAddressRegistrationsQuery,
21        IotaNamesDefaultNameArgs, IotaNamesDefaultNameQuery, IotaNamesRegistrationsArgs,
22        IotaNamesRegistrationsQuery, ResolveIotaNamesAddressArgs, ResolveIotaNamesAddressQuery,
23    },
24};
25
26impl Client {
27    /// Return the resolved address for the given name.
28    pub async fn iota_names_lookup(&self, name: &str) -> Result<Option<Address>> {
29        let operation = ResolveIotaNamesAddressQuery::build(ResolveIotaNamesAddressArgs {
30            name: name.to_owned(),
31        });
32        let response = self.run_query(&operation).await?;
33
34        let ResolveIotaNamesAddressQuery {
35            resolve_iota_names_address: Some(address),
36        } = response
37        else {
38            return Ok(None);
39        };
40
41        Ok(Some(address.address))
42    }
43
44    /// Find all registration NFTs for the given address.
45    pub async fn iota_names_registrations(
46        &self,
47        address: Address,
48        pagination_filter: PaginationFilter,
49    ) -> Result<Page<NameRegistration>> {
50        let pagination = self.pagination_filter(pagination_filter).await;
51        let operation = IotaNamesAddressRegistrationsQuery::build(IotaNamesRegistrationsArgs {
52            address,
53            after: pagination.after,
54            before: pagination.before,
55            first: pagination.first,
56            last: pagination.last,
57        });
58        let response = self.run_query(&operation).await?;
59
60        let IotaNamesAddressRegistrationsQuery {
61            address:
62                Some(IotaNamesRegistrationsQuery {
63                    iota_names_registrations,
64                }),
65        } = response
66        else {
67            return Ok(Page::new_empty());
68        };
69
70        Ok(Page::new(
71            iota_names_registrations.page_info,
72            iota_names_registrations
73                .nodes
74                .into_iter()
75                .map(TryInto::try_into)
76                .collect::<Result<Vec<_>>>()?,
77        ))
78    }
79
80    /// Get the default name pointing to this address, if one exists.
81    pub async fn iota_names_default_name(
82        &self,
83        address: Address,
84        format: impl Into<Option<NameFormat>>,
85    ) -> Result<Option<Name>> {
86        let operation = IotaNamesAddressDefaultNameQuery::build(IotaNamesDefaultNameArgs {
87            address,
88            format: format.into().map(Into::into),
89        });
90        let response = self.run_query(&operation).await?;
91
92        let IotaNamesAddressDefaultNameQuery {
93            address:
94                Some(IotaNamesDefaultNameQuery {
95                    iota_names_default_name: Some(name),
96                }),
97        } = response
98        else {
99            return Ok(None);
100        };
101
102        Ok(Some(Name::from_str(&name).map_err(|_| {
103            Error::from_error(Kind::Parse, format!("invalid name: {name}"))
104        })?))
105    }
106}
