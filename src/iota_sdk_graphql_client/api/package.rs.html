26define_query! {
27    /// Query for [`GraphQLClient::package_versions`]. Await it to send the request.
28    #[derive(Clone)]
29    pub struct ListPackageVersionsQuery {
30        client: GraphQLClient,
31        address: Address,
32        pagination: PaginationFilter,
33        after_version: Option<Version>,
34        before_version: Option<Version>,
35    }
36    output: GraphQLResult<Page<MovePackage>>;
37}
38
39impl ListPackageVersionsQuery {
40    /// Set the page to fetch.
41    pub fn pagination(mut self, pagination: PaginationFilter) -> Self {
42        self.pagination = pagination;
43        self
44    }
45
46    /// Stream every item, page by page, starting at the pagination's cursor and in its direction,
47    /// with its limit as the page size.
48    pub fn stream(self) -> impl Stream<Item = GraphQLResult<MovePackage>> + Unpin {
49        let pagination = self.pagination.clone();
50        stream_paginated_query(move |page| self.clone().pagination(page).send(), pagination)
51    }
52
53    /// Only return versions after this one.
54    pub fn after_version(mut self, after_version: Version) -> Self {
55        self.after_version = Some(after_version);
56        self
57    }
58
59    /// Only return versions before this one.
60    pub fn before_version(mut self, before_version: Version) -> Self {
61        self.before_version = Some(before_version);
62        self
63    }
64
65    fn operation<'a>(
66        address: Address,
67        after_version: Option<Version>,
68        before_version: Option<Version>,
69        pagination: &'a PaginationFilterResponse,
70    ) -> cynic::Operation<PackageVersionsQueryFragment, PackageVersionsArgs<'a>> {
71        PackageVersionsQueryFragment::build(PackageVersionsArgs {
72            address,
73            after: pagination.after.as_deref(),
74            before: pagination.before.as_deref(),
75            first: pagination.first,
76            last: pagination.last,
77            filter: Some(MovePackageVersionFilter {
78                after_version: after_version.map(|v| v.as_u64()),
79                before_version: before_version.map(|v| v.as_u64()),
80            }),
81        })
82    }
83
84    async fn send(self) -> GraphQLResult<Page<MovePackage>> {
85        let Self {
86            client,
87            pagination,
88            address,
89            after_version,
90            before_version,
91        } = self;
92        let pagination = client.pagination_filter(pagination).await;
93        let response = client
94            .run_query(&Self::operation(
95                address,
96                after_version,
97                before_version,
98                &pagination,
99            ))
100            .await?;
101
102        let pc = response.package_versions;
103        let page_info = pc.page_info;
104        let bcs = pc
105            .nodes
106            .iter()
107            .map(|p| {
108                let b64 = p
109                    .bcs
110                    .as_ref()
111                    .ok_or(GraphQLError::EmptyResponseField("package bcs"))?;
112                crate::base64::decode(b64.0.as_str())
113            })
114            .collect::<crate::error::GraphQLResult<Vec<_>>>()?;
115        let packages = bcs
116            .iter()
117            .map(|b| package_from_bcs(b))
118            .collect::<GraphQLResult<Vec<_>>>()?;
119
120        Ok(Page::new(page_info, packages))
121    }
122}

218    /// Query for [`GraphQLClient::normalized_move_module`]. Await it to send the request.
219    pub struct GetNormalizedMoveModuleQuery {
220        client: GraphQLClient,
221        package: Address,
222        module: String,
223        version: Option<Version>,
224        enums_pagination: PaginationFilter,
225        friends_pagination: PaginationFilter,
226        functions_pagination: PaginationFilter,
227        structs_pagination: PaginationFilter,
228    }
229    output: GraphQLResult<Option<MoveModule>>;
230}

232struct ModulePagination {
233    enums: PaginationFilterResponse,
234    friends: PaginationFilterResponse,
235    functions: PaginationFilterResponse,
236    structs: PaginationFilterResponse,
237}
238
239impl GetNormalizedMoveModuleQuery {
240    /// Set the package version.
241    pub fn version(mut self, version: Version) -> Self {
242        self.version = Some(version);
243        self
244    }
245
246    /// Set the page of the module's enums to fetch.
247    pub fn enums_pagination(mut self, pagination: PaginationFilter) -> Self {
248        self.enums_pagination = pagination;
249        self
250    }
251
252    /// Set the page of the module's friends to fetch.
253    pub fn friends_pagination(mut self, pagination: PaginationFilter) -> Self {
254        self.friends_pagination = pagination;
255        self
256    }
257
258    /// Set the page of the module's functions to fetch.
259    pub fn functions_pagination(mut self, pagination: PaginationFilter) -> Self {
260        self.functions_pagination = pagination;
261        self
262    }
263
264    /// Set the page of the module's structs to fetch.
265    pub fn structs_pagination(mut self, pagination: PaginationFilter) -> Self {
266        self.structs_pagination = pagination;
267        self
268    }
269
270    fn operation<'a>(
271        package: Address,
272        module: &'a str,
273        version: Option<Version>,
274        pagination: &'a ModulePagination,
275    ) -> cynic::Operation<NormalizedMoveModuleQueryFragment, NormalizedMoveModuleQueryArgs<'a>>
276    {
277        let ModulePagination {
278            enums,
279            friends,
280            functions,
281            structs,
282        } = pagination;
283        NormalizedMoveModuleQueryFragment::build(NormalizedMoveModuleQueryArgs {
284            package,
285            module,
286            version: version.map(|v| v.as_u64()),
287            after_enums: enums.after.as_deref(),
288            after_functions: functions.after.as_deref(),
289            after_structs: structs.after.as_deref(),
290            after_friends: friends.after.as_deref(),
291            before_enums: enums.before.as_deref(),
292            before_functions: functions.before.as_deref(),
293            before_structs: structs.before.as_deref(),
294            before_friends: friends.before.as_deref(),
295            first_enums: enums.first,
296            first_functions: functions.first,
297            first_structs: structs.first,
298            first_friends: friends.first,
299            last_enums: enums.last,
300            last_functions: functions.last,
301            last_structs: structs.last,
302            last_friends: friends.last,
303        })
304    }
305
306    async fn send(self) -> GraphQLResult<Option<MoveModule>> {
307        let Self {
308            client,
309            package,
310            module,
311            version,
312            enums_pagination,
313            friends_pagination,
314            functions_pagination,
315            structs_pagination,
316        } = self;
317        let pagination = ModulePagination {
318            enums: client.pagination_filter(enums_pagination).await,
319            friends: client.pagination_filter(friends_pagination).await,
320            functions: client.pagination_filter(functions_pagination).await,
321            structs: client.pagination_filter(structs_pagination).await,
322        };
323        let response = client
324            .run_query(&Self::operation(package, &module, version, &pagination))
325            .await?;
326
327        Ok(response.package.and_then(|p| p.module).map(Into::into))
328    }

332    /// Query for [`GraphQLClient::package`]. Await it to send the request.
333    pub struct GetPackageQuery {
334        client: GraphQLClient,
335        address: Address,
336        version: Option<Version>,
337    }
338    output: GraphQLResult<Option<MovePackage>>;
339}
340
341impl GetPackageQuery {
342    /// Set the package version. Without it, the package is loaded from the given address.
343    pub fn version(mut self, version: Version) -> Self {
344        self.version = Some(version);
345        self
346    }
347
348    async fn send(self) -> GraphQLResult<Option<MovePackage>> {
349        let operation = PackageQueryFragment::build(PackageArgs {
350            address: self.address,
351            version: self.version.map(|v| v.as_u64()),
352        });
353
354        let response = self.client.run_query(&operation).await?;
355
356        response
357            .package
358            .map(|x| x.bcs.ok_or(GraphQLError::EmptyResponseField("package bcs")))
359            .transpose()?
360            .map(|bcs| crate::base64::decode(bcs.0.as_str()))
361            .transpose()?
362            .map(|bcs| package_from_bcs(&bcs))
363            .transpose()
364    }
365}

433impl GraphQLClient {
434    /// The package corresponding to the given address (at the optionally given version). When no
435    /// version is given, the package is loaded directly from the address given. Otherwise, the
436    /// address is translated before loading to point to the package whose original ID matches the
437    /// package at address, but whose version is version. For non-system packages, this might result
438    /// in a different address than address because different versions of a package, introduced by
439    /// upgrades, exist at distinct addresses.
440    ///
441    /// Note that this interpretation of version is different from a historical object read (the
442    /// interpretation of version for the object query).
443    pub fn package(&self, address: Address) -> GetPackageQuery {
444        GetPackageQuery {
445            client: self.clone(),
446            address,
447            version: None,
448        }
449    }
450
451    /// Fetch all versions of package at address (packages that share this package's original ID),
452    /// optionally bounding the versions exclusively with
453    /// [`ListPackageVersionsQuery::after_version`] and
454    /// [`ListPackageVersionsQuery::before_version`].
455    pub fn package_versions(&self, address: Address) -> ListPackageVersionsQuery {
456        ListPackageVersionsQuery {
457            client: self.clone(),
458            address,
459            pagination: PaginationFilter::default(),
460            after_version: None,
461            before_version: None,
462        }
463    }
464
465    /// Fetch the latest version of the package at address.
466    /// This corresponds to the package with the highest version that shares its original ID with
467    /// the package at address.
468    pub fn package_latest(&self, address: Address) -> GetPackageLatestQuery {
469        GetPackageLatestQuery {
470            client: self.clone(),
471            address,
472        }
473    }
474
475    /// The Move packages that exist in the network, optionally bounded exclusively with
476    /// [`ListPackagesQuery::after_checkpoint`] and [`ListPackagesQuery::before_checkpoint`].
477    ///
478    /// This query returns all versions of a given user package that appear between the specified
479    /// checkpoints, but only records the latest versions of system packages.
480    pub fn packages(&self) -> ListPackagesQuery {
481        ListPackagesQuery {
482            client: self.clone(),
483            pagination: PaginationFilter::default(),
484            after_checkpoint: None,
485            before_checkpoint: None,
486        }
487    }
488
489    /// Return the normalized Move function data for the provided package, module, and function.
490    pub fn normalized_move_function(
491        &self,
492        package: Address,
493        module: impl Into<String>,
494        function: impl Into<String>,
495    ) -> GetNormalizedMoveFunctionQuery {
496        GetNormalizedMoveFunctionQuery {
497            client: self.clone(),
498            package,
499            module: module.into(),
500            function: function.into(),
501            version: None,
502        }
503    }
504
505    /// Return the normalized Move module data for the provided module.
506    // TODO: do we want to self paginate everything and return all the data, or keep pagination
507    // options?
508    pub fn normalized_move_module(
509        &self,
510        package: Address,

512    ) -> GetNormalizedMoveModuleQuery {
513        GetNormalizedMoveModuleQuery {
514            client: self.clone(),

520            functions_pagination: PaginationFilter::default(),
521            structs_pagination: PaginationFilter::default(),
522        }
523    }

526/// Decode an object's BCS and return it as a package.
527fn package_from_bcs(bcs: &[u8]) -> GraphQLResult<MovePackage> {
528    bcs::from_bytes::<Object>(bcs)
529        .map_err(iota_types::BcsError::new)?
530        .data
531        .into_opt_package()
532        .ok_or_else(|| GraphQLError::Deserialization("object is not a package".into()))
533}

723    #[tokio::test]
724    async fn normalized_move_module_sends_each_pagination_to_its_field() {
725        let page = |direction, cursor: &str, limit| PaginationFilter {
726            direction,
727            cursor: Some(cursor.to_owned()),
728            limit: Some(limit),
729        };
730        let vars = sent_variables("NormalizedMoveModuleQueryFragment", |client| async move {
731            let _ = client
732                .normalized_move_module(Address::FRAMEWORK, "coin")
733                .version(Version::from_u64(4))
734                .enums_pagination(page(Direction::Forward, "enums", 1))
735                .friends_pagination(page(Direction::Backward, "friends", 2))
736                .functions_pagination(page(Direction::Forward, "functions", 3))
737                .structs_pagination(page(Direction::Backward, "structs", 5))
738                .await;
739        })
740        .await;
741        assert_eq!(vars["package"], Address::FRAMEWORK.to_string());
742        assert_eq!(vars["module"], "coin");
743        assert_eq!(vars["version"], 4);
744        for (member, direction, limit) in [
745            ("Enums", Direction::Forward, 1),
746            ("Friends", Direction::Backward, 2),
747            ("Functions", Direction::Forward, 3),
748            ("Structs", Direction::Backward, 5),
749        ] {
750            let cursor = member.to_lowercase();
751            let (used, unused) = match direction {
752                Direction::Forward => (("after", "first"), ("before", "last")),
753                Direction::Backward => (("before", "last"), ("after", "first")),
754            };
755            assert_eq!(vars[format!("{}{member}", used.0)], cursor);
756            assert_eq!(vars[format!("{}{member}", used.1)], limit);
757            assert!(vars[format!("{}{member}", unused.0)].is_null());
758            assert!(vars[format!("{}{member}", unused.1)].is_null());
759        }
760    }
