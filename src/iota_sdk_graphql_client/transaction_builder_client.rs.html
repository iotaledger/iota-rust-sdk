23impl Client {
24    /// Create a new [`TransactionBuilder`] with the given sender address.
25    pub fn transaction_builder(&self, sender: Address) -> TransactionBuilder<&Self> {
26        TransactionBuilder::new(sender).with_client(self)
27    }

34impl TransactionBuilderLedgerClient for Client {
35    async fn object(
36        &self,
37        object_id: ObjectId,
38        version: impl Into<Option<Version>>,
39    ) -> Result<Option<Object>, Self::Error> {
40        self.object(object_id, version).await
41    }
42
43    async fn objects(
44        &self,
45        struct_tag: Option<StructTag>,
46        owner: Address,
47        cursor: Option<Vec<u8>>,
48        limit: Option<usize>,
49    ) -> Result<ObjectsPage, Self::Error> {
50        // GraphQL cursors are base64 ASCII, so round-tripping through
51        // Vec<u8> is lossless. Caller-supplied cursors must come from a
52        // prior call to this method; anything else is rejected here
53        // rather than panicked on.
54        let cursor = cursor
55            .map(String::from_utf8)
56            .transpose()
57            .map_err(|e| crate::error::Error::from_error(crate::error::Kind::Parse, e))?;
58        let page = self
59            .objects(
60                ObjectFilter {
61                    type_tag: struct_tag.map(|tag| tag.to_string()),
62                    owner: Some(owner),
63                    object_ids: None,
64                },
65                PaginationFilter {
66                    direction: Direction::Forward,
67                    cursor,
68                    limit: limit.map(|v| v as _),
69                },
70            )
71            .await?;
72        let (page_info, data) = page.into_parts();
73        let next_cursor = page_info
74            .has_next_page
75            .then_some(page_info.end_cursor)
76            .flatten()
77            .map(String::into_bytes);
78        Ok(ObjectsPage { data, next_cursor })
79    }
80
81    async fn protocol_config(&self) -> Result<ProtocolConfig, Self::Error> {
82        let cfg = crate::Client::protocol_config(self, None).await?;
83        let attributes = cfg
84            .configs
85            .into_iter()
86            .filter_map(|attr| attr.value.map(|v| (attr.key, v)))
87            .collect();
88        Ok(ProtocolConfig { attributes })
89    }
90
91    async fn reference_gas_price(
92        &self,
93        epoch: impl Into<Option<u64>>,
94    ) -> Result<Option<u64>, Self::Error> {
95        self.reference_gas_price(epoch).await
96    }
97}

99impl TransactionBuilderSimulationClient for Client {
100    type DryRunResult = DryRunResult;
101
102    async fn estimate_transaction_budget(
103        &self,
104        transaction: &Transaction,
105    ) -> Result<Option<u64>, Self::Error> {
106        let res = self.dry_run_transaction(transaction, true).await?;
107        Ok(res.effects.map(|effects| match effects {
108            TransactionEffects::V1(v1) => v1.gas_cost_summary.gas_used(),
109            _ => unimplemented!(
110                "a new TransactionEffects enum variant was added and needs to be handled"
111            ),
112        }))
113    }
114
115    async fn dry_run_transaction(
116        &self,
117        transaction: &Transaction,
118        skip_checks: bool,
119    ) -> Result<Self::DryRunResult, Self::Error> {
120        (*self).dry_run_transaction(transaction, skip_checks).await
121    }
122}

124impl TransactionBuilderExecutionClient for Client {
125    async fn execute_transaction(
126        &self,
127        signatures: &[UserSignature],
128        transaction: &Transaction,
129        wait_for: impl Into<Option<WaitForTransaction>>,
130    ) -> Result<TransactionEffects, Self::Error> {
131        self.execute_transaction(signatures, transaction, wait_for)
132            .await
133    }
134
135    async fn wait_for_transaction(
136        &self,
137        digest: TransactionDigest,
138        wait_for: WaitForTransaction,
139    ) -> Result<(), Self::Error> {
140        self.wait_for_transaction(digest, wait_for, None).await
141    }
142
143    async fn transaction_effects(
144        &self,
145        digest: TransactionDigest,
146    ) -> Result<Option<TransactionEffects>, Self::Error> {
147        self.transaction_effects(digest).await
148    }
