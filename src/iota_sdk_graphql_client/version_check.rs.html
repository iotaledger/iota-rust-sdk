4//! Rejects responses from an indexer that requires a newer SDK than this one.
5//!
6//! The indexer's GraphQL service stamps every response with the lowest `iota-sdk-graphql-client`
7//! version able to decode it, in the [`X_IOTA_SDK_GRAPHQL_CLIENT_VERSION_MIN`] header. The check
8//! runs on the response headers, before any body is decoded, so an outdated client fails with a
9//! clear error instead of tripping over a value it does not know.
10
11use reqwest::header::HeaderMap;
12use semver::Version;
13
14use crate::error::{GraphQLError, GraphQLResult};
15
16/// Version of this crate, which the indexer's minimum SDK version is compared against.
17pub const CRATE_VERSION: &str = env!("CARGO_PKG_VERSION");
18
19/// Lowest `iota-sdk-graphql-client` version able to decode the server's responses, as a semantic
20/// version. Clients below it fail their requests instead of decoding values they do not know.
21pub const X_IOTA_SDK_GRAPHQL_CLIENT_VERSION_MIN: &str = "x-iota-sdk-graphql-client-version-min";
22
23/// Fails with [`GraphQLError::OutdatedClient`] when the response headers name a minimum SDK version
24/// newer than this crate.
25///
26/// A response without the header, or with one that is not a semantic version, passes.
27pub(crate) fn check_client_version(response_headers: &HeaderMap) -> GraphQLResult<()> {
28    // `CARGO_PKG_VERSION` comes from this crate's manifest, which cargo only accepts with a valid
29    // semantic version.
30    let current = Version::parse(CRATE_VERSION).expect("CARGO_PKG_VERSION is a valid version");
31    check_against(response_headers, &current)
32}
33
34fn check_against(response_headers: &HeaderMap, current: &Version) -> GraphQLResult<()> {
35    let Some(minimum) = response_headers
36        .get(X_IOTA_SDK_GRAPHQL_CLIENT_VERSION_MIN)
37        .and_then(|value| value.to_str().ok())
38        .and_then(|value| Version::parse(value).ok())
39    else {
40        return Ok(());
41    };
42
43    if current.cmp_precedence(&minimum).is_ge() {
44        return Ok(());
45    }
46
47    Err(GraphQLError::OutdatedClient {
48        minimum: minimum.to_string(),
49        current: current.to_string(),
50    })
51}

53#[cfg(test)]
54mod tests {
55    use reqwest::header::HeaderValue;
56
57    use super::*;
58
59    fn headers_with_minimum(minimum: &str) -> HeaderMap {
60        let mut headers = HeaderMap::new();
61        headers.insert(
62            X_IOTA_SDK_GRAPHQL_CLIENT_VERSION_MIN,
63            HeaderValue::from_str(minimum).unwrap(),
64        );
65        headers
66    }
67
68    fn check(headers: &HeaderMap, current: &str) -> GraphQLResult<()> {
69        check_against(headers, &Version::parse(current).unwrap())
70    }
71
72    #[test]
73    fn passes_when_client_matches_minimum() {
74        check(&headers_with_minimum("1.2.3"), "1.2.3").unwrap();
75    }
76
77    #[test]
78    fn passes_when_client_is_newer() {
79        check(&headers_with_minimum("1.2.3"), "1.3.0").unwrap();
80        check(&headers_with_minimum("1.0.0-beta.1"), "1.0.0").unwrap();
81    }
82
83    #[test]
84    fn fails_when_client_is_older() {
85        let error = check(&headers_with_minimum("1.3.0"), "1.2.9").unwrap_err();
86        let GraphQLError::OutdatedClient { minimum, current } = &error else {
87            panic!("expected GraphQLError::OutdatedClient, got {error:?}");
88        };
89        assert_eq!(minimum, "1.3.0");
90        assert_eq!(current, "1.2.9");
91        assert_eq!(
92            error.to_string(),
93            "the indexer requires a newer IOTA SDK, upgrade it (needs iota-sdk-graphql-client >= 1.3.0, this SDK uses 1.2.9)"
94        );
95    }
96
97    #[test]
98    fn fails_when_client_is_older_pre_release() {
99        let error = check(&headers_with_minimum("1.0.0-beta.2"), "1.0.0-beta.1").unwrap_err();
100        assert!(
101            matches!(error, GraphQLError::OutdatedClient { .. }),
102            "{error:?}"
103        );
104    }
105
106    #[test]
107    fn ignores_build_metadata() {
108        check(&headers_with_minimum("1.2.3+build"), "1.2.3").unwrap();
109        let error = check(&headers_with_minimum("1.2.4+build"), "1.2.3").unwrap_err();
110        assert!(
111            matches!(error, GraphQLError::OutdatedClient { .. }),
112            "{error:?}"
113        );
114    }
115
116    #[test]
117    fn passes_without_header() {
118        check(&HeaderMap::new(), "1.2.3").unwrap();
119    }
120
121    #[test]
122    fn passes_with_unparsable_header() {
123        check(&headers_with_minimum("latest"), "1.2.3").unwrap();
124    }
125
126    #[test]
127    fn crate_version_is_valid_semver() {
128        Version::parse(CRATE_VERSION).unwrap();
129    }
130}
