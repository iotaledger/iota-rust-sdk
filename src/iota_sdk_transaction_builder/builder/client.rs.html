11/// Determines what to wait for after executing a transaction.
12#[derive(Default)]
13#[non_exhaustive]
14pub enum WaitForTransaction {
15    /// Indicates that the transaction effects will be usable in subsequent
16    /// transactions (you can reference objects created by this transaction),
17    /// and that the transaction itself is indexed on the fullnode, so queries
18    /// served by the fullnode, such as gRPC, will find it.
19    ///
20    /// **Warning:** This does not guarantee the transaction is indexed on the
21    /// indexer, so queries served by an indexer, such as GraphQL, may not
22    /// find it yet. Use [`WaitForTransaction::Finalized`] with those clients.
23    IndexedOnNode,
24    /// Indicates that the transaction has been included in a checkpoint, and
25    /// all queries may include it.
26    #[default]
27    Finalized,
28}

30/// One page of objects plus an optional cursor for the next page. See
31/// [`TransactionBuilderLedgerClient::objects`].
32#[derive(Clone, Debug)]
33pub struct ObjectsPage {
34    /// The objects in this page.
35    pub data: Vec<Object>,
36    /// Opaque continuation cursor for fetching the next page; `None` when no
37    /// further pages exist. Pass it back as the `cursor` argument to
38    /// [`TransactionBuilderLedgerClient::objects`] to advance.
39    pub next_cursor: Option<Vec<u8>>,
40}

42/// Transport-neutral view of the chain's protocol configuration: a flat
43/// map of attribute name to value, parsed by callers as needed.
44#[derive(Clone, Debug, Default)]
45pub struct ProtocolConfig {
46    pub(crate) attributes: BTreeMap<String, String>,
47}
48
49impl ProtocolConfig {
50    /// Builds a config from attributes keyed by their canonical protocol name
51    /// (e.g. `"max_gas_payment_objects"`).
52    pub fn new(attributes: BTreeMap<String, String>) -> Self {
53        Self { attributes }
54    }
55
56    /// All available configuration attributes, keyed by their canonical
57    /// protocol name (e.g. `"max_gas_payment_objects"`).
58    pub fn attributes(&self) -> &BTreeMap<String, String> {
59        &self.attributes
60    }
61
62    /// Looks up one attribute by its canonical protocol name.
63    pub fn attribute(&self, name: &str) -> Option<&str> {
64        self.attributes.get(name).map(String::as_str)
65    }
66}

68/// Base trait shared by the transaction builder client traits, carrying the
69/// client's error type.
70pub trait TransactionBuilderClientBase {
71    /// The error type for this client.
72    type Error: 'static + std::error::Error + Send + Sync;
73}

75/// Read-only access to ledger state: everything the Transaction Builder needs
76/// to resolve and build a transaction
77/// ([`finish_with_budget`](crate::TransactionBuilder::finish_with_budget)).
78pub trait TransactionBuilderLedgerClient: TransactionBuilderClientBase {
79    /// Fetch an object
80    fn object(
81        &self,
82        object_id: ObjectId,
83        version: impl Into<Option<Version>>,
84    ) -> impl std::future::Future<Output = Result<Option<Object>, Self::Error>>;
85
86    /// Fetch several objects at once, returning them in the order they were
87    /// requested with `None` in place of any object that does not exist.
88    ///
89    /// The default impl calls [`object`](Self::object) once per entry, costing
90    /// one round trip each. Clients whose transport can fetch a batch (such as
91    /// gRPC's `GetObjects`) should override it.
92    fn objects_by_id(
93        &self,
94        object_ids: &[(ObjectId, Option<Version>)],
95    ) -> impl std::future::Future<Output = Result<Vec<Option<Object>>, Self::Error>> {
96        async move {
97            let mut objects = Vec::with_capacity(object_ids.len());
98            for (object_id, version) in object_ids {
99                objects.push(self.object(*object_id, *version).await?);
100            }
101            Ok(objects)
102        }
103    }
104
105    /// Fetch one page of objects matching the filter, returning the page
106    /// contents and a continuation cursor (when more pages exist).
107    ///
108    /// The cursor is opaque to callers — both GraphQL (base64-encoded
109    /// JSON/BCS) and gRPC (`prost::bytes::Bytes` page token) formats fit
110    /// into `Option<Vec<u8>>`. Pass `None` to start from the beginning;
111    /// pass the cursor returned by a previous call to advance.
112    fn objects(
113        &self,
114        struct_tag: Option<StructTag>,
115        owner: Address,
116        cursor: Option<Vec<u8>>,
117        limit: Option<usize>,
118    ) -> impl std::future::Future<Output = Result<ObjectsPage, Self::Error>>;
119
120    /// Fetch the chain's protocol configuration.
121    fn protocol_config(
122        &self,
123    ) -> impl std::future::Future<Output = Result<ProtocolConfig, Self::Error>>;
124
125    /// Get the reference gas price
126    fn reference_gas_price(
127        &self,
128        epoch: impl Into<Option<u64>>,
129    ) -> impl std::future::Future<Output = Result<Option<u64>, Self::Error>>;
130}

132/// Transaction simulation: dry runs and the gas budget estimation built on
133/// them ([`finish`](crate::TransactionBuilder::finish),
134/// [`dry_run`](crate::TransactionBuilder::dry_run)).
135pub trait TransactionBuilderSimulationClient: TransactionBuilderClientBase {
136    /// The result of a dry run.
137    type DryRunResult;
138
139    /// Estimate the gas budget needed for a transaction, typically by
140    /// simulating it and reading the gas cost from the result. `Ok(None)`
141    /// means no estimate is available;
142    /// [`finish`](crate::TransactionBuilder::finish) then fails unless a
143    /// budget was set explicitly.
144    fn estimate_transaction_budget(
145        &self,
146        transaction: &Transaction,
147    ) -> impl std::future::Future<Output = Result<Option<u64>, Self::Error>>;
148
149    /// Dry run a transaction
150    fn dry_run_transaction(
151        &self,
152        transaction: &Transaction,
153        skip_checks: bool,
154    ) -> impl std::future::Future<Output = Result<Self::DryRunResult, Self::Error>>;
155}

157/// Transaction execution: submitting a transaction and tracking its result
158/// ([`execute`](crate::TransactionBuilder::execute)).
159pub trait TransactionBuilderExecutionClient: TransactionBuilderClientBase {
160    /// Execute a transaction
161    ///
162    /// If the wait fails, the error is returned although the transaction was
163    /// executed, and its effects are lost. To keep them, pass `None` and call
164    /// [`wait_for_transaction`](Self::wait_for_transaction) afterwards.
165    fn execute_transaction(
166        &self,
167        signatures: &[UserSignature],
168        transaction: &Transaction,
169        wait_for: impl Into<Option<WaitForTransaction>>,
170    ) -> impl std::future::Future<Output = Result<TransactionEffects, Self::Error>>;
171
172    /// Wait for the indexing or finalization of a transaction by its digest.
173    fn wait_for_transaction(
174        &self,
175        digest: TransactionDigest,
176        wait_for: WaitForTransaction,
177    ) -> impl std::future::Future<Output = Result<(), Self::Error>>;
178
179    /// Fetch the effects of an executed transaction
180    fn transaction_effects(
181        &self,
182        digest: TransactionDigest,
183    ) -> impl std::future::Future<Output = Result<Option<TransactionEffects>, Self::Error>>;
184}

186/// A full transaction builder client: ledger reads, simulation, and execution.
187///
188/// This is a blanket alias — do not implement it directly. Implement
189/// [`TransactionBuilderLedgerClient`], [`TransactionBuilderSimulationClient`],
190/// and [`TransactionBuilderExecutionClient`] instead, and this trait is
191/// implemented automatically.
192pub trait TransactionBuilderClient:
193    TransactionBuilderLedgerClient
194    + TransactionBuilderSimulationClient
195    + TransactionBuilderExecutionClient
196{
197}

405    /// Balance, in NANOS, of every fabricated coin. Large enough to cover any
406    /// gas budget the builder might estimate in a doc test or example.
407    const FABRICATED_COIN_BALANCE: u64 = 1_000_000_000_000;
408
409    /// Build a fabricated gas coin (`0x2::coin::Coin<0x2::iota::IOTA>`) with
410    /// the given id, owner and balance.
411    ///
412    /// The contents are the BCS layout the coin resolution code expects: the
413    /// 32-byte object id followed by the little-endian `u64` balance.
414    fn fabricated_coin(object_id: ObjectId, owner: Owner, balance: u64) -> Object {
415        let mut contents = Vec::with_capacity(ObjectId::LENGTH + std::mem::size_of::<u64>());
416        contents.extend_from_slice(object_id.as_ref());
417        contents.extend_from_slice(&balance.to_le_bytes());
418        let move_struct = MoveStruct::new(
419            StructTag::new_gas_coin().into(),
420            Version::from_u64(1),
421            contents,
422        )
423        .expect("contents always contain a full object id");
424        Object::new(
425            ObjectData::Struct(move_struct),
426            owner,
427            TransactionDigest::ZERO,
428            0,
429        )
430    }

432    /// A test client that implements the transaction builder client traits by
433    /// fabricating objects on demand.
434    ///
435    /// It is useful for building transactions in tests, examples, and doc tests
436    /// where a live network connection is not available. Object lookups resolve
437    /// to a synthesized gas coin owned by an address (shared system objects
438    /// such as the system state object resolve as shared), and gas
439    /// selection always finds a single funded coin. This is enough to drive
440    /// [`finish`](crate::TransactionBuilder::finish) to completion, but the
441    /// resulting transaction references made-up objects and cannot be executed
442    /// — [`execute_transaction`](TransactionBuilderExecutionClient::execute_transaction) returns
443    /// an error.
444    #[derive(Clone, Copy, Debug, Default)]
445    pub struct TestClient;
446
447    /// Error type for [`TestClient`].
448    #[derive(Clone, Debug, thiserror::Error)]
449    #[error("TestClientError: {0}")]
450    pub struct TestClientError(pub String);
451
452    impl TransactionBuilderClientBase for TestClient {
453        type Error = TestClientError;
454    }

456    impl TransactionBuilderLedgerClient for TestClient {
457        async fn object(
458            &self,
459            object_id: ObjectId,
460            _version: impl Into<Option<Version>>,
461        ) -> Result<Option<Object>, Self::Error> {
462            // System objects (e.g. the system state object used by staking) are
463            // shared; everything else resolves as an address-owned coin.
464            let owner = if object_id == ObjectId::SYSTEM_STATE || object_id == ObjectId::CLOCK {
465                Owner::Shared(Version::from_u64(1))
466            } else {
467                Owner::Address(Address::ZERO)
468            };
469            Ok(Some(fabricated_coin(
470                object_id,
471                owner,
472                FABRICATED_COIN_BALANCE,
473            )))
474        }
475
476        async fn objects(
477            &self,
478            _struct_tag: Option<StructTag>,
479            owner: Address,
480            _cursor: Option<Vec<u8>>,
481            _limit: Option<usize>,
482        ) -> Result<ObjectsPage, Self::Error> {
483            // A single funded gas coin owned by the requested owner is enough for
484            // the builder's automatic gas selection. Its id is a fixed sentinel
485            // that won't collide with the object ids used in examples.
486            let gas_coin_id = ObjectId::from_bytes([0xee; ObjectId::LENGTH])
487                .expect("32 bytes is a valid object id");
488            let owner = Owner::Address(owner);
489            Ok(ObjectsPage {
490                data: vec![fabricated_coin(gas_coin_id, owner, FABRICATED_COIN_BALANCE)],
491                next_cursor: None,
492            })
493        }
494
495        async fn reference_gas_price(
496            &self,
497            _epoch: impl Into<Option<u64>>,
498        ) -> Result<Option<u64>, Self::Error> {
499            Ok(Some(1000))
500        }
501
502        async fn protocol_config(&self) -> Result<super::ProtocolConfig, Self::Error> {
503            let mut config = super::ProtocolConfig::default();
504            config
505                .attributes
506                .insert(BASE_TX_COST_FIXED_KEY.to_owned(), "1000".to_owned());
507            config
508                .attributes
509                .insert(MAX_GAS_PAYMENT_OBJECTS_KEY.to_owned(), "256".to_owned());
510            Ok(config)
511        }
512    }

561    /// A [`TestClient`] that records how the builder asked for objects, and can
562    /// report chosen ids as missing.
563    #[derive(Clone, Default)]
564    pub struct RecordingClient {
565        /// The ids of each `objects_by_id` call, in call order.
566        pub batches: std::sync::Arc<std::sync::Mutex<Vec<Vec<ObjectId>>>>,
567        /// The ids of each single-object `object` call, in call order.
568        pub singles: std::sync::Arc<std::sync::Mutex<Vec<ObjectId>>>,
569        /// Ids to report as missing instead of fabricating an object.
570        pub missing: Vec<ObjectId>,
571    }

573    impl RecordingClient {
574        /// Returns the ids of each `objects_by_id` call, in call order.
575        pub fn batches(&self) -> Vec<Vec<ObjectId>> {
576            self.batches.lock().unwrap().clone()
577        }
578
579        /// Returns the ids of each single-object `object` call, in call order.
580        pub fn singles(&self) -> Vec<ObjectId> {
581            self.singles.lock().unwrap().clone()
582        }
583    }

590        async fn object(
591            &self,
592            object_id: ObjectId,
593            version: impl Into<Option<Version>>,
594        ) -> Result<Option<Object>, Self::Error> {
595            self.singles.lock().unwrap().push(object_id);
596            if self.missing.contains(&object_id) {
597                return Ok(None);
598            }
599            crate::TestClient.object(object_id, version).await

602        async fn objects_by_id(

604            object_ids: &[(ObjectId, Option<Version>)],
605        ) -> Result<Vec<Option<Object>>, Self::Error> {
606            self.batches
607                .lock()
608                .unwrap()
609                .push(object_ids.iter().map(|(id, _)| *id).collect());
610            let mut objects = Vec::with_capacity(object_ids.len());
611            for (object_id, _) in object_ids {
612                objects.push(if self.missing.contains(object_id) {
613                    None
614                } else {
615                    crate::TestClient.object(*object_id, None).await?
616                });
617            }
618            Ok(objects)
619        }
