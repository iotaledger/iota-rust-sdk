21#[derive(Debug, thiserror::Error)]
22#[non_exhaustive]
23enum VersionParsingErrorKind {
24    #[error("failed to parse {} version into a number", idx_to_segment_name(*segment_idx))]
25    InvalidVersionSegment {
26        segment_idx: usize,
27        source: std::num::ParseIntError,
28    },
29    #[error(
30        "invalid amount of version segments. A valid SemVer has exactly three: \"<major>.<minor>.<patch>\""
31    )]
32    InvalidNumberOfSegments,
33    #[error("an empty string cannot be a valid SemVer")]
34    Empty,
35}

37/// Parsing a [GasStationVersion] out of a string failed.
38#[derive(Debug, thiserror::Error)]
39#[error("failed to parse a valid SemVer out of `{input}`")]
40pub struct VersionParsingError {
41    /// The input string.
42    input: String,
43    #[source]
44    kind: VersionParsingErrorKind,
45}

47fn idx_to_segment_name(idx: usize) -> &'static str {
48    assert!(idx < 3);
49
50    ["major", "minor", "patch"][idx]
51}

53/// Errors returned by the [`GasStation`] sponsor.
54#[derive(Debug, thiserror::Error)]
55#[non_exhaustive]
56pub enum GasStationError {
57    /// A request path could not be joined onto the configured base URL.
58    #[error(transparent)]
59    InvalidUrl(<Url as FromStr>::Err),
60    /// A request to the gas station could not be completed.
61    #[error("request to gas station `{gas_station_url}` failed: {source}")]
62    Request {
63        /// The underlying transport error.
64        source: reqwest::Error,
65        /// The URL that was requested.
66        gas_station_url: Url,
67    },
68    /// The gas station answered with something unusable.
69    #[
70        error("invalid gas station response from {gas_station_url}{}",
71        .message.as_deref().map(|msg| format!(": {msg}")).unwrap_or_default())
72    ]
73    Response {
74        /// The error the gas station reported, when it reported one.
75        message: Option<String>,
76        /// The URL that was requested.
77        gas_station_url: Url,
78    },
79    /// The gas station is too old for this client.
80    #[error(
81        "invalid gas-station version: got version `{version}`, but at least version `{min_required_version}` is required"
82    )]
83    UnsupportedVersion {
84        /// The minimum IOTA gas-station version needed for this operation.
85        min_required_version: GasStationVersion,
86        /// The actual IOTA gas-station's version.
87        version: GasStationVersion,
88    },
89    /// The version the gas station reported could not be parsed.
90    #[error(transparent)]
91    VersionParsing(VersionParsingError),
92    /// The transaction could not be serialized for the gas station.
93    #[error("BCS serialization error: {0}")]
94    Bcs(bcs::Error),
95}

97/// The IOTA gas station, sponsoring transactions over its HTTP API.
98///
99/// A station is configured once and reused for any number of transactions;
100/// build one with [`GasStation::builder`].
101///
102/// # Example
103///
104/// ```no_run
105/// # use std::time::Duration;
106/// use iota_sdk_transaction_builder::GasStation;
107/// use reqwest::header::{AUTHORIZATION, HeaderValue};
108///
109/// # fn main() -> eyre::Result<()> {
110/// let station = GasStation::builder("http://0.0.0.0:9527".parse()?)
111///     .header(AUTHORIZATION, HeaderValue::from_static("Bearer token"))
112///     .reservation_duration(Duration::from_secs(60))
113///     .build();
114/// # Ok(())
115/// # }
116/// ```
117#[derive(Clone, Debug)]
118pub struct GasStation {
119    url: Url,
120    http_client: reqwest::Client,
121    headers: HeaderMap<HeaderValue>,
122    reservation_duration: Duration,
123    /// The version check result, kept so the station is probed once rather
124    /// than once per reservation. Racing callers may both probe, which is
125    /// harmless.
126    checked_version: OnceLock<GasStationVersion>,
127}
128
129impl GasStation {
130    /// Start configuring a gas station reachable at `url`.
131    pub fn builder(url: Url) -> GasStationBuilder {
132        GasStationBuilder {
133            url,
134            http_client: None,
135            headers: HeaderMap::default(),
136            reservation_duration: DEFAULT_RESERVATION_DURATION,
137        }
138    }
139
140    /// A gas station at `url` with no headers and a default HTTP client.
141    pub fn new(url: Url) -> Self {
142        Self::builder(url).build()
143    }
144
145    /// The URL this station is reached at.
146    pub fn url(&self) -> &Url {
147        &self.url
148    }
149
150    /// The HTTP client this station sends its requests with.
151    pub fn http_client(&self) -> &reqwest::Client {
152        &self.http_client
153    }
154}
155
156/// Duration of the gas allocation when the caller sets none.
157const DEFAULT_RESERVATION_DURATION: Duration = Duration::from_secs(60);

159/// Configures a [`GasStation`].
160#[derive(Clone, Debug)]
161pub struct GasStationBuilder {
162    url: Url,
163    http_client: Option<reqwest::Client>,
164    headers: HeaderMap<HeaderValue>,
165    reservation_duration: Duration,
166}
167
168impl GasStationBuilder {
169    /// Send requests with this HTTP client instead of a default one.
170    ///
171    /// Set this to control timeouts, proxies, TLS roots or connection pooling;
172    /// without it the station builds a [`reqwest::Client`] with reqwest's own
173    /// defaults.
174    pub fn http_client(mut self, client: reqwest::Client) -> Self {
175        self.http_client = Some(client);
176        self
177    }
178
179    /// Add a header sent with every request to the gas station.
180    pub fn header(mut self, name: HeaderName, value: HeaderValue) -> Self {
181        self.headers.append(name, value);
182        self
183    }
184
185    /// Add headers sent with every request to the gas station.
186    pub fn headers(mut self, headers: impl IntoIterator<Item = (HeaderName, HeaderValue)>) -> Self {
187        for (name, value) in headers {
188            self.headers.append(name, value);
189        }
190        self
191    }
192
193    /// How long the station should hold the gas it reserves. Defaults to 60
194    /// seconds.
195    pub fn reservation_duration(mut self, duration: Duration) -> Self {
196        self.reservation_duration = duration;
197        self
198    }
199
200    /// Build the gas station.
201    pub fn build(self) -> GasStation {
202        let Self {
203            url,
204            http_client,
205            mut headers,
206            reservation_duration,
207        } = self;
208        headers
209            .entry(reqwest::header::CONTENT_TYPE)
210            .or_insert_with(|| HeaderValue::from_static("application/json"));
211
212        GasStation {
213            url,
214            http_client: http_client.unwrap_or_default(),
215            headers,
216            reservation_duration,
217            checked_version: OnceLock::new(),
218        }
219    }
220}

222/// The version of an IOTA gas station.
223#[derive(Clone, Debug, Eq, Hash, PartialEq)]
224pub struct GasStationVersion {
225    version_core: [u8; 3],
226    // Suffix without leading '-'.
227    suffix: Option<String>,
228}
229
230impl GasStationVersion {
231    const MIN: Self = Self::new(0, 3, 0);
232
233    const fn new(major: u8, minor: u8, patch: u8) -> Self {
234        Self {
235            version_core: [major, minor, patch],
236            suffix: None,
237        }
238    }
239}
240
241impl PartialOrd for GasStationVersion {
242    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
243        Some(self.cmp(other))
244    }
245}
246
247impl Ord for GasStationVersion {
248    fn cmp(&self, other: &Self) -> std::cmp::Ordering {
249        self.version_core.cmp(&other.version_core)
250    }
251}
252
253impl std::fmt::Display for GasStationVersion {
254    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
255        let v = self.version_core;
256        write!(f, "{}.{}.{}", v[0], v[1], v[2])?;
257        if let Some(suffix) = self.suffix.as_deref() {
258            write!(f, "-{suffix}")?;
259        }
260
261        Ok(())
262    }
263}
264
265impl FromStr for GasStationVersion {
266    type Err = VersionParsingError;
267
268    // Disable this lint as looping over a range allows for checking that we have at
269    // least 3 segments.
270    #[allow(clippy::needless_range_loop)]
271    fn from_str(s: &str) -> Result<Self, Self::Err> {
272        if s.is_empty() {
273            return Err(VersionParsingError {
274                input: String::default(),
275                kind: VersionParsingErrorKind::Empty,
276            });
277        }
278
279        let (version_core_str, maybe_suffix) = if let Some((version, suffix)) = s.split_once('-') {
280            (version, Some(suffix))
281        } else {
282            (s, None)
283        };
284
285        let mut segments = version_core_str.split('.');
286        let mut version_core = [0; 3];
287        for i in 0..3 {
288            let segment = segments.next().ok_or_else(|| VersionParsingError {
289                input: s.to_owned(),
290                kind: VersionParsingErrorKind::InvalidNumberOfSegments,
291            })?;
292            let parsed_segment = segment.parse().map_err(|parse_int_e| VersionParsingError {
293                input: s.to_owned(),
294                kind: VersionParsingErrorKind::InvalidVersionSegment {
295                    segment_idx: i,
296                    source: parse_int_e,
297                },
298            })?;
299            version_core[i] = parsed_segment;
300        }
301        // Check if there would be more segments than 3.
302        if segments.next().is_some() {
303            return Err(VersionParsingError {
304                input: s.to_owned(),
305                kind: VersionParsingErrorKind::InvalidNumberOfSegments,
306            });
307        }
308
309        Ok(Self {
310            version_core,
311            suffix: maybe_suffix.map(String::from),
312        })
313    }
314}

335#[derive(Debug, Deserialize)]
336#[serde(rename_all = "camelCase")]
337struct GasObjectRef {
338    /// The object id of this object.
339    pub object_id: ObjectId,
340    /// The version of this object.
341    #[serde(deserialize_with = "deserialize_readable_u64")]
342    pub version: u64,
343    /// The digest of this object.
344    pub digest: ObjectDigest,
345}
346
347fn deserialize_readable_u64<'de, D: serde::Deserializer<'de>>(
348    deserializer: D,
349) -> Result<u64, D::Error> {
350    #[derive(Deserialize)]
351    #[serde(untagged)]
352    enum NumOrString {
353        Num(i64),
354        String(String),
355    }
356
357    match NumOrString::deserialize(deserializer)? {
358        NumOrString::Num(num) => num
359            .try_into()
360            .map_err(|e: std::num::TryFromIntError| serde::de::Error::custom(e.to_string())),
361        NumOrString::String(s) => s
362            .parse()
363            .map_err(|e: std::num::ParseIntError| serde::de::Error::custom(e.to_string())),
364    }
365}

381impl GasStation {
382    fn endpoint(&self, kind: GasStationRequestKind) -> Result<Url, GasStationError> {
383        self.url
384            .join(kind.as_path())
385            .map_err(GasStationError::InvalidUrl)
386    }
387
388    async fn gas_station_version(&self) -> Result<GasStationVersion, GasStationError> {
389        let url = self.endpoint(GasStationRequestKind::Version)?;
390        let response = self
391            .http_client
392            .request(reqwest::Method::GET, url.clone())
393            .headers(self.headers.clone())
394            .send()
395            .await
396            .map_err(|e| GasStationError::Request {
397                source: e,
398                gas_station_url: url.clone(),
399            })?
400            .error_for_status()
401            .map_err(|e| GasStationError::Request {
402                source: e,
403                gas_station_url: url.clone(),
404            })?;
405
406        // A string in the format <PKG VERSION>-<GIT REVISION>.
407        let mut version_info = String::from_utf8(
408            response
409                .bytes()
410                .await
411                .map_err(|_| GasStationError::Response {
412                    message: None,
413                    gas_station_url: url.clone(),
414                })?
415                .to_vec(),
416        )
417        .map_err(|_| GasStationError::Response {
418            message: None,
419            gas_station_url: url.clone(),
420        })?;
421
422        // We only care about the version.
423        // Using `rfind` instead of `find` because the pkg's version might have a suffix
424        // like "-alpha".
425        let separator_idx = version_info
426            .rfind('-')
427            .ok_or_else(|| GasStationError::Response {
428                message: None,
429                gas_station_url: url,
430            })?;
431        version_info.truncate(separator_idx);
432
433        let version = version_info
434            .parse()
435            .map_err(GasStationError::VersionParsing)?;
436
437        Ok(version)
438    }
439
440    /// Check that the station is new enough, probing it only the first time.
441    async fn check_version(&self) -> Result<(), GasStationError> {
442        if self.checked_version.get().is_some() {
443            return Ok(());
444        }
445
446        let version = self.gas_station_version().await?;
447        if version < GasStationVersion::MIN {
448            return Err(GasStationError::UnsupportedVersion {
449                min_required_version: GasStationVersion::MIN,
450                version,
451            });
452        }
453        let _ = self.checked_version.set(version);
454
455        Ok(())
456    }
457
458    async fn reserve(&self, gas_budget: u64) -> Result<GasReservation, GasStationError> {
459        self.check_version().await?;
460
461        let url = self.endpoint(GasStationRequestKind::ReserveGas)?;
462        let response = self
463            .http_client
464            .request(reqwest::Method::POST, url.clone())
465            .json(&ReserveGasRequest {
466                gas_budget,
467                reserve_duration_secs: self.reservation_duration.as_secs(),
468            })
469            .headers(self.headers.clone())
470            .send()
471            .await
472            .map_err(|e| GasStationError::Request {
473                source: e,
474                gas_station_url: url.clone(),
475            })?
476            .error_for_status()
477            .map_err(|e| GasStationError::Request {
478                source: e,
479                gas_station_url: url.clone(),
480            })?;
481
482        let res: ReserveGasResponse =
483            response
484                .json()
485                .await
486                .map_err(|e| GasStationError::Request {
487                    source: e,
488                    gas_station_url: url.clone(),
489                })?;
490
491        res.result.ok_or(GasStationError::Response {
492            message: res.error,
493            gas_station_url: url,
494        })
495    }
496
497    /// Execute a transaction against a reservation and return the gas
498    /// station's own JSON effects.
499    ///
500    /// The station reports effects in a JSON-RPC shape that cannot be turned
501    /// back into [`TransactionEffects`](iota_types::TransactionEffects); use
502    /// this when that JSON is what you want, and
503    /// [`execute_with_gas_sponsor`](crate::TransactionBuilder::execute_with_gas_sponsor)
504    /// when you want typed effects.
505    pub async fn execute_reserved_json(
506        &self,
507        reservation_id: u64,
508        transaction: &Transaction,
509        signature: &UserSignature,
510    ) -> Result<serde_json::Value, GasStationError> {
511        let url = self.endpoint(GasStationRequestKind::ExecuteTx)?;
512
513        let tx_bytes = base64ct::Base64::encode_string(
514            &bcs::to_bytes(transaction).map_err(GasStationError::Bcs)?,
515        );
516
517        let response = self
518            .http_client
519            .request(reqwest::Method::POST, url.clone())
520            .headers(self.headers.clone())
521            .json(&ExecuteTxRequest {
522                reservation_id,
523                tx_bytes,
524                user_sig: signature.to_base64(),
525                request_type: "waitForLocalExecution".to_owned(),
526            })
527            .send()
528            .await
529            .map_err(|e| GasStationError::Request {
530                source: e,
531                gas_station_url: url.clone(),
532            })?
533            .error_for_status()
534            .map_err(|e| GasStationError::Request {
535                source: e,
536                gas_station_url: url.clone(),
537            })?;
538
539        let res: ExecuteTxResponse =
540            response
541                .json()
542                .await
543                .map_err(|e| GasStationError::Request {
544                    source: e,
545                    gas_station_url: url.clone(),
546                })?;
547
548        res.effects.ok_or(GasStationError::Response {
549            message: res.error,
550            gas_station_url: url,
551        })
552    }
553}

555impl GasSponsor for GasStation {
556    type Error = GasStationError;
557    type Reservation = u64;
558
559    async fn reserve_gas(
560        &self,
561        transaction: &Transaction,
562    ) -> Result<(Self::Reservation, SponsoredGas), Self::Error> {
563        let Transaction::V1(v1) = transaction else {
564            unimplemented!("a new Transaction enum variant was added and needs to be handled")
565        };
566        let GasReservation {
567            sponsor_address,
568            reservation_id,
569            gas_coins,
570        } = self.reserve(v1.gas_payment.budget).await?;
571
572        let gas = SponsoredGas {
573            owner: sponsor_address,
574            objects: gas_coins
575                .into_iter()
576                .map(|obj_ref| ObjectReference {
577                    object_id: obj_ref.object_id,
578                    version: Version::from_u64(obj_ref.version),
579                    digest: obj_ref.digest,
580                })
581                .collect(),
582        };
583
584        Ok((reservation_id, gas))
585    }
586
587    async fn execute_reserved(
588        &self,
589        reservation: Self::Reservation,
590        transaction: &Transaction,
591        signature: &UserSignature,
592    ) -> Result<TransactionDigest, Self::Error> {
593        let effects = self
594            .execute_reserved_json(reservation, transaction, signature)
595            .await?;
596
597        effects
598            .get("transactionDigest")
599            .ok_or_else(|| GasStationError::Response {
600                message: Some("Missing transaction digest".to_owned()),
601                gas_station_url: self.url.clone(),
602            })
603            .and_then(|v| {
604                TransactionDigest::deserialize(v).map_err(|e| GasStationError::Response {
605                    message: Some(e.to_string()),
606                    gas_station_url: self.url.clone(),
607                })
608            })
609    }
610}
