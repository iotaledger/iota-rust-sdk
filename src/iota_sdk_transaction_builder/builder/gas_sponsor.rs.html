4//! Defines the [`GasSponsor`] trait, which allows users to implement any
5//! service that pays the gas for a transaction it did not send.
6
7use std::future::Future;
8
9use iota_types::{Address, ObjectReference, Transaction, TransactionDigest, UserSignature};
10
11/// The gas payment a sponsor committed to for one transaction.
12#[derive(Clone, Debug)]
13pub struct SponsoredGas {
14    /// The address paying for the transaction.
15    pub owner: Address,
16    /// The gas coins the sponsor put up.
17    pub objects: Vec<ObjectReference>,
18}
19
20/// A service that pays the gas for someone else's transaction.
21///
22/// A sponsor hands out gas against a budget and later executes the transaction
23/// that spends it, so it both funds and submits. Implement this to sponsor
24/// through a service the SDK does not ship; for the IOTA gas station, use
25/// [`GasStation`](crate::GasStation).
26///
27/// Pass an implementation to
28/// [`execute_with_gas_sponsor`](crate::TransactionBuilder::execute_with_gas_sponsor).
29pub trait GasSponsor {
30    /// The error that can occur while sponsoring.
31    type Error: 'static + std::error::Error + Send + Sync;
32
33    /// Whatever the sponsor needs to recognize a reservation it handed out.
34    /// The builder passes it back untouched and never interprets it.
35    type Reservation;
36
37    /// Reserve gas covering the transaction's gas needs.
38    fn reserve_gas(
39        &self,
40        transaction: &Transaction,
41    ) -> impl Future<Output = Result<(Self::Reservation, SponsoredGas), Self::Error>>;
42
43    /// Execute a transaction against gas reserved by
44    /// [`reserve_gas`](Self::reserve_gas).
45    ///
46    /// The transaction pays with the [`SponsoredGas`] that reservation
47    /// returned, and `signature` is the sender's; the sponsor supplies its own.
48    fn execute_reserved(
49        &self,
50        reservation: Self::Reservation,
51        transaction: &Transaction,
52        signature: &UserSignature,
53    ) -> impl Future<Output = Result<TransactionDigest, Self::Error>>;
54}
