4//! Defines the [`TransactionSigner`] trait, which allows users to implement any type which can sign
5//! a transaction asynchronously.
6
7use std::future::Future;
8
9use iota_crypto::{
10    IotaSigner, SignatureError, ed25519::Ed25519PrivateKey, secp256k1::Secp256k1PrivateKey,
11    secp256r1::Secp256r1PrivateKey, simple::SimpleKeypair,
12};
13use iota_types::{MoveAuthenticator, Object, Transaction, UserSignature};
14
15/// Defines a type which can sign a transaction asynchronously.
16///
17/// This trait can be implemented downstream to enable signing when using the
18/// [`TransactionBuilder`](crate::TransactionBuilder)
19/// [`execute`](crate::TransactionBuilder::execute) function.
20pub trait TransactionSigner {
21    /// The error that can occur during signing.
22    type Error: 'static + std::error::Error + Send + Sync;
23
24    /// Sign a transaction and return a [`UserSignature`].
25    fn sign(
26        &self,
27        transaction: &Transaction,
28    ) -> impl Future<Output = Result<UserSignature, Self::Error>>;
29
30    /// Whether the signer uses the objects the transaction reads, such as a hardware wallet showing
31    /// what the transaction does. When `true`, the builder gathers them for
32    /// [`sign_with_objects`](Self::sign_with_objects).
33    fn uses_objects(&self) -> bool {
34        false
35    }
36
37    /// Sign a transaction given the objects it reads: its gas coins and immutable, owned or
38    /// receiving inputs at the versions it references, and its shared inputs as the builder fetched
39    /// them, which may change before the transaction executes. `objects` is empty when the signer
40    /// does not use them or the builder has no client to fetch them. The default ignores them and
41    /// calls [`sign`](Self::sign).
42    fn sign_with_objects(
43        &self,
44        transaction: &Transaction,
45        _objects: &[Object],
46    ) -> impl Future<Output = Result<UserSignature, Self::Error>> {
47        self.sign(transaction)
48    }
49}
50
51macro_rules! impl_basic_signer {
52    ($($signer:ident),*) => {
53        $(
54        impl TransactionSigner for $signer {
55            type Error = SignatureError;
56
57            async fn sign(&self, transaction: &Transaction) -> Result<UserSignature, Self::Error> {
58                self.sign_transaction(transaction)
59            }
60        }
61        )*
62    };
63}
64
65impl_basic_signer!(
66    Ed25519PrivateKey,
67    Secp256k1PrivateKey,
68    Secp256r1PrivateKey,
69    SimpleKeypair
70);
71
72impl TransactionSigner for MoveAuthenticator {
73    type Error = SignatureError;
74
75    async fn sign(&self, _transaction: &Transaction) -> Result<UserSignature, Self::Error> {
76        Ok(UserSignature::MoveAuthenticator(self.clone()))
77    }
78}

87    /// Clear-signs with the objects the builder gathers, so the device shows what the transaction
88    /// does. Without them, or with an IOTA app too old to clear-sign, it blind-signs with a
89    /// warning, which the app only allows when blind signing is enabled in its settings.
90    ///
91    /// With the Speculos transport, signing blocks the thread polling it until the emulator
92    /// answers.
93    impl TransactionSigner for LedgerSigner {
94        type Error = LedgerError;
95
96        async fn sign(&self, transaction: &Transaction) -> Result<UserSignature, Self::Error> {
97            self.sign_transaction(transaction).await
98        }
99
100        fn uses_objects(&self) -> bool {
101            true
102        }
103
104        async fn sign_with_objects(
105            &self,
106            transaction: &Transaction,
107            objects: &[Object],
108        ) -> Result<UserSignature, Self::Error> {
109            if objects.is_empty() {
110                tracing::warn!("the transaction's objects are unavailable, blind signing instead");
111            }
112            self.sign_transaction_with_objects(transaction, objects)
113                .await
114        }
115    }

134        /// Answers each command with its final result at once and records how many parameters each
135        /// signing request carries: three when the input objects are sent, two otherwise.
136        struct FakeDevice {
137            sign_parameters: Arc<Mutex<Vec<usize>>>,
138        }
139
140        impl Transport for FakeDevice {
141            fn exchange<'a>(&'a self, command: &'a ApduCommand) -> TransportFuture<'a, ApduAnswer> {
142                Box::pin(async move {
143                    let public_key = Ed25519PublicKey::new([7; 32]);
144                    let result = match command.ins() {
145                        0x00 => vec![1, 0, 1],
146                        0x02 => [
147                            &[32][..],
148                            public_key.bytes(),
149                            &[32],
150                            public_key.derive_address().as_bytes(),
151                        ]
152                        .concat(),
153                        0x03 => {
154                            // The start request is a tag byte and one hash per parameter
155                            let parameters = (command.data().len() - 1) / 32;
156                            self.sign_parameters.lock().unwrap().push(parameters);
157                            vec![9; 64]
158                        }
159                        ins => panic!("unexpected instruction {ins:#04x}"),
160                    };
161                    ApduAnswer::from_bytes([&[1][..], &result, &[0x90, 0x00]].concat())
162                })
163            }
164        }
165
166        async fn sign_parameters(
167            sign: impl AsyncFnOnce(LedgerSigner, Transaction, Object),
168        ) -> Vec<usize> {
169            let sign_parameters = Arc::new(Mutex::new(Vec::new()));
170            let signer = Ledger::new(FakeDevice {
171                sign_parameters: sign_parameters.clone(),
172            })
173            .into_signer("m/44'/4218'/0'/0'/0'".parse().unwrap())
174            .await
175            .unwrap();
176            let transaction = bcs::from_bytes(&hex_decode(TRANSACTION)).unwrap();
177            let gas_coin = bcs::from_bytes(&hex_decode(GAS_COIN)).unwrap();
178            sign(signer, transaction, gas_coin).await;
179            sign_parameters.lock().unwrap().clone()
180        }
181
182        fn hex_decode(hex: &str) -> Vec<u8> {
183            (0..hex.len())
184                .step_by(2)
185                .map(|i| u8::from_str_radix(&hex[i..i + 2], 16).unwrap())
186                .collect()
187        }

248    /// Records the objects it is given and returns an arbitrary signature.
249    #[derive(Default)]
250    struct RecordingSigner {
251        objects: Mutex<Option<Vec<Object>>>,
252    }
253
254    impl TransactionSigner for RecordingSigner {
255        type Error = SignatureError;
256
257        async fn sign(&self, _: &Transaction) -> Result<UserSignature, Self::Error> {
258            unreachable!("the builder signs through `sign_with_objects`")
259        }
260
261        fn uses_objects(&self) -> bool {
262            true
263        }
264
265        async fn sign_with_objects(
266            &self,
267            _: &Transaction,
268            objects: &[Object],
269        ) -> Result<UserSignature, Self::Error> {
270            *self.objects.lock().unwrap() = Some(objects.to_vec());
271            Err(SignatureError::from_source("recorded"))
272        }
273    }

275    #[tokio::test]
276    async fn execute_passes_the_objects_the_transaction_reads() {
277        let coin = ObjectId::from_bytes([1; ObjectId::LENGTH]).unwrap();
278        let mut builder = TransactionBuilder::new(Address::ZERO).with_client(TestClient);
279        builder.transfer_objects(Address::new([2; Address::LENGTH]), [coin]);
280
281        let signer = RecordingSigner::default();
282        builder.execute(&signer, None).await.unwrap_err();
283
284        // the selected gas coin and the transferred coin
285        let objects = signer.objects.lock().unwrap().take().unwrap();
286        let mut ids = objects.iter().map(Object::id).collect::<Vec<_>>();
287        ids.sort();
288        let gas_coin = ObjectId::from_bytes([0xee; ObjectId::LENGTH]).unwrap();
289        assert_eq!(ids, [coin, gas_coin]);
290    }
