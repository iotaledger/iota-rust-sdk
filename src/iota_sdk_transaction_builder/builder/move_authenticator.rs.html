4//! Defines the [`MoveAuthenticator`] builder that represents a call to an
5//! authenticator function in move which can authorize a transaction as part of
6//! Account Abstraction.
7
8use iota_types::{
9    Input, MoveAuthenticator, MoveAuthenticatorV1, ObjectId, ObjectReference, Owner,
10    SharedObjectReference, TypeTag,
11};
12
13use crate::{
14    PTBArgumentList, TransactionBuilderLedgerClient, error::Error, types::MoveTypes,
15    unresolved::InputKind,
16};
17
18/// A function call to authorize a transaction via move.
19#[derive(Clone, Debug)]
20pub struct MoveAuthenticatorBuilder {
21    /// Input objects or primitive values
22    call_args: Vec<InputKind>,
23    /// Type arguments for the Move authenticate function
24    type_args: Vec<TypeTag>,
25    /// The account ID, which is the sender of the transaction.
26    account_id: ObjectId,
27}
28
29impl MoveAuthenticatorBuilder {
30    /// Create a new Move Authenticator builder from an account ID, which is the
31    /// sender of a transaction that this will be used to authenticate.
32    pub fn new(account_id: ObjectId) -> Self {
33        Self {
34            call_args: Default::default(),
35            type_args: Default::default(),
36            account_id,
37        }
38    }
39
40    /// Set the move authenticator call inputs.
41    pub fn call_args<I: PTBArgumentList>(mut self, call_args: I) -> Self {
42        self.call_args = call_args.inputs();
43        self
44    }
45
46    /// Set the move authenticator call type parameters.
47    pub fn generics<G: MoveTypes>(mut self) -> Self {
48        self.type_args = G::type_tags();
49        self
50    }
51
52    /// Set the move authenticator call type parameters manually.
53    pub fn type_args(mut self, tags: impl IntoIterator<Item = TypeTag>) -> Self {
54        self.type_args = tags.into_iter().collect();
55        self
56    }
57
58    /// Resolve this move authenticator builder into a [`MoveAuthenticator`]
59    /// which can be used to execute the given transaction.
60    pub async fn finish(
61        self,
62        client: impl TransactionBuilderLedgerClient,
63    ) -> Result<MoveAuthenticator, Error> {
64        let account = client
65            .object(self.account_id, None)
66            .await
67            .map_err(Error::client)?
68            .ok_or_else(|| Error::Input(format!("missing account {}", self.account_id)))?;
69
70        let mut call_args = Vec::new();
71        for input in self.call_args {
72            call_args.push(match input {
73                InputKind::ImmutableOrOwned(object_id)
74                | InputKind::Input(Input::ImmutableOrOwned(ObjectReference {
75                    object_id, ..
76                })) => {
77                    let obj = client
78                        .object(object_id, None)
79                        .await
80                        .map_err(Error::client)?
81                        .ok_or_else(|| {
82                            Error::InvalidMoveAuthArg(format!("missing object {object_id}"))
83                        })?;
84                    if !obj.owner().is_immutable() {
85                        return Err(Error::InvalidMoveAuthArg(
86                            "call arguments must not be owned".to_owned(),
87                        ));
88                    }
89                    Input::ImmutableOrOwned(obj.object_ref())
90                }
91                InputKind::Shared { object_id, mutable }
92                | InputKind::Input(Input::Shared(SharedObjectReference {
93                    object_id,
94                    mutable,
95                    ..
96                })) => {
97                    let obj = client
98                        .object(object_id, None)
99                        .await
100                        .map_err(Error::client)?
101                        .ok_or_else(|| {
102                            Error::InvalidMoveAuthArg(format!("missing object {object_id}"))
103                        })?;
104
105                    let Owner::Shared(version) = obj.owner() else {
106                        return Err(Error::InvalidMoveAuthArg(format!(
107                            "object {object_id} was passed as shared, but is not"
108                        )));
109                    };
110
111                    Input::Shared(SharedObjectReference {
112                        object_id,
113                        initial_shared_version: *version,
114                        mutable,
115                    })
116                }
117                InputKind::Receiving(_) | InputKind::Input(Input::Receiving(_)) => {
118                    return Err(Error::InvalidMoveAuthArg(
119                        "call arguments must not be receiving".to_owned(),
120                    ));
121                }
122                InputKind::Input(input) => input.clone(),
123            })
124        }
125        Ok(match account.owner() {
126            Owner::Immutable => {
127                MoveAuthenticator::V1(MoveAuthenticatorV1::new_with_immutable_account_object(
128                    call_args,
129                    self.type_args,
130                    account.object_ref(),
131                ))
132            }
133            Owner::Shared(version) => {
134                MoveAuthenticator::V1(MoveAuthenticatorV1::new_with_shared_account_object(
135                    call_args,
136                    self.type_args,
137                    SharedObjectReference {
138                        object_id: account.id(),
139                        initial_shared_version: *version,
140                        mutable: false,
141                    },
142                ))
143            }
144            _ => {
145                return Err(Error::InvalidMoveAuthAccount(
146                    "account must be immutable or shared".to_owned(),
147                ));
148            }
149        })
150    }
151}
