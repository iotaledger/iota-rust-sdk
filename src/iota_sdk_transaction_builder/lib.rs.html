4//! # IOTA Transaction Builder
5//!
6//! This crate contains the [TransactionBuilder], which allows for simple
7//! construction of Programmable Transactions which can be executed on the IOTA
8//! network.
9//!
10//! The builder is designed to allow for a lot of flexibility while also
11//! reducing the necessary boilerplate code. It uses a type-state pattern to
12//! ensure the proper flow through the various functions. It is chainable via
13//! mutable references.
14//!
15//! ## Online vs. Offline Builder
16//!
17//! The Transaction Builder can be used with or without a client implementing
18//! [TransactionBuilderLedgerClient]. When one is provided via the
19//! [with_client](TransactionBuilder::with_client) method, the resulting builder
20//! will use it to find and validate provided IDs. A ledger-only client can
21//! build transactions with an explicit gas budget via
22//! [finish_with_budget](TransactionBuilder::finish_with_budget). Clients that
23//! also implement [TransactionBuilderSimulationClient] enable
24//! [dry_run](TransactionBuilder::dry_run) and
25//! [finish](TransactionBuilder::finish) with automatic gas budget estimation;
26//! clients that additionally implement [TransactionBuilderExecutionClient]
27//! enable [execute](TransactionBuilder::execute).
28//!
29//! ### Example with Client Resolution
30//!
31//! ```
32//! # use std::str::FromStr;
33//! # use iota_sdk_transaction_builder::TestClient;
34//! use iota_sdk_transaction_builder::TransactionBuilder;
35//! use iota_types::{Address, ObjectId, Transaction};
36//!
37//! # #[tokio::main(flavor = "current_thread")]
38//! # async fn main() -> eyre::Result<()> {
39//!
40//! let sender =
41//!     Address::from_str("0xda1820edf693ee32b5729907b9b2ec8e64980ee8c008c17e89cfb4e5ecd72151")?;
42//! let to_address =
43//!     Address::from_str("0x0000a4984bd495d4346fa208ddff4f5d5e5ad48c21dec631ddebc99809f16900")?;
44//!
45//! # let client = TestClient;
46//! let mut builder = TransactionBuilder::new(sender).with_client(client);
47//!
48//! let coin =
49//!     ObjectId::from_str("0xe0e45ecb12ddca5f0d5192d2ee9e7f711959aa98614f9905e1e25c612ffd99a2")?;
50//!
51//! builder.send_coins([coin], to_address, 50000000000u64);
52//!
53//! let txn: Transaction = builder.finish().await?;
54//! # Ok(())
55//! # }
56//! ```
57//!
58//! ### Example without Client Resolution
59//!
60//! ```
61//! # use std::str::FromStr;
62//! use iota_sdk_transaction_builder::TransactionBuilder;
63//! use iota_types::{Address, ObjectDigest, ObjectId, ObjectReference, Transaction, Version};
64//!
65//! let sender =
66//!     Address::from_str("0xda1820edf693ee32b5729907b9b2ec8e64980ee8c008c17e89cfb4e5ecd72151")?;
67//! let to_address =
68//!     Address::from_str("0x0000a4984bd495d4346fa208ddff4f5d5e5ad48c21dec631ddebc99809f16900")?;
69//!
70//! let mut builder = TransactionBuilder::new(sender);
71//!
72//! let coin = ObjectReference {
73//!     object_id: ObjectId::from_str(
74//!         "0xe0e45ecb12ddca5f0d5192d2ee9e7f711959aa98614f9905e1e25c612ffd99a2",
75//!     )?,
76//!     digest: ObjectDigest::from_str("hSAGU3ZwDwxptd17ZK1QPDdJLhvPMfpSxe1p892GFVn")?,
77//!     version: Version::from_u64(545110774),
78//! };
79//! let gas_coin = ObjectReference {
80//!     object_id: ObjectId::from_str(
81//!         "0x65beb18e282d1f33a39bffa84ff92ec4d2fec0350ba6f7e5a568afff72d651db",
82//!     )?,
83//!     digest: ObjectDigest::from_str("8ahH5RXFnK1jttQEWTypYX7MRzLuQDEXk7fhMHCyZekX")?,
84//!     version: Version::from_u64(473053810),
85//! };
86//!
87//! builder
88//!     .send_coins([coin], to_address, 50000000000u64)
89//!     .gas([gas_coin])
90//!     .gas_budget(1000000000)
91//!     .gas_price(100);
92//!
93//! let txn: Transaction = builder.finish()?;
94//! # Result::<_, eyre::Error>::Ok(())
95//! ```
96//!
97//! NOTE: It is possible to provide an [ObjectId](iota_types::ObjectId) to an
98//! offline client builder, but this will cause the builder to fail when calling
99//! `finish`.
100//!
101//! ## Methods
102//!
103//! There are three kinds of methods available:
104//!
105//! ### Commands
106//!
107//! Each command method adds one or more commands to the final transaction. Some
108//! commands have optional follow-up methods. All command results can be
109//! assigned a name via [assign](TransactionBuilder::assign). Assigning a name
110//! to a command allows them to be used later in the transaction via the
111//! [assigned] method.
112//!
113//! - [move_call](TransactionBuilder::move_call): Call a move function.
114//!     - `arguments`: Add arguments to the move call.
115//!     - `generics`: Add generic types to the move call using types that
116//!       implement [MoveType](types::MoveType).
117//!     - `type_tags`: Add generic types directly using the
118//!       [TypeTag](iota_types::TypeTag).
119//! - [send_iota](TransactionBuilder::send_iota): Send IOTA coins to a recipient
120//!   address.
121//! - [send_coins](TransactionBuilder::send_coins): Send coins of any type to a
122//!   recipient address.
123//! - [pay](TransactionBuilder::pay): Send coins of any type to several
124//!   recipients, each paired with the amount to send.
125//! - [pay_iota](TransactionBuilder::pay_iota): Send IOTA coins from the gas
126//!   coin to several recipients.
127//! - [merge_coins](TransactionBuilder::merge_coins): Merge a list of coins into
128//!   a single primary coin.
129//! - [split_coins](TransactionBuilder::split_coins): Split a coin into coins of
130//!   various amounts.
131//! - [transfer_objects](TransactionBuilder::transfer_objects): Send objects to
132//!   a recipient address.
133//! - [publish_package](TransactionBuilder::publish_package): Publish a move
134//!   package.
135//!     - `package_id`: Name the package ID returned by the publish call.
136//! - [upgrade](TransactionBuilder::upgrade): Upgrade a move package.
137//! - [make_move_vec](TransactionBuilder::make_move_vec): Create a move
138//!   `vector`.
139//!
140//! ### Metadata
141//!
142//! These methods set various metadata which may be needed for the execution.
143//!
144//! - [gas](TransactionBuilder::gas): Add gas coins to pay for the execution.
145//! - [gas_refs](TransactionBuilder::gas_refs): Add gas coins that the caller
146//!   has already resolved to references.
147//! - [gas_budget](TransactionBuilder::gas_budget): Set the maximum gas budget
148//!   to spend.
149//! - [gas_price](TransactionBuilder::gas_price): Set the gas price.
150//! - [sponsor](TransactionBuilder::sponsor): Set the gas sponsor address.
151//! - [gas_station_sponsor](TransactionBuilder::gas_station_sponsor): Set the
152//!   gas station URL. See [Gas Station](crate#gas-station) for more info.
153//! - [expiration](TransactionBuilder::expiration): Set the transaction
154//!   expiration epoch.
155//!
156//! ### Other
157//!
158//! Many other methods exist, either to get data or allow for development on top
159//! of the builder. Typically, these methods should not be needed, but they are
160//! made available for special circumstances.
161//!
162//! - [apply_argument](TransactionBuilder::apply_argument)
163//! - [apply_arguments](TransactionBuilder::apply_arguments)
164//! - [input](TransactionBuilder::input)
165//! - [pure_bytes](TransactionBuilder::pure_bytes)
166//! - [pure](TransactionBuilder::pure)
167//! - [command](TransactionBuilder::command)
168//! - [assigned_command](TransactionBuilder::assigned_command)
169//!
170//! ## Finalization and Execution
171//!
172//! There are several ways to finish the builder. First, the
173//! [finish](TransactionBuilder::finish) method can be used to return the
174//! resulting [Transaction](iota_types::Transaction), which can be manually
175//! serialized, executed, etc. On a client without simulation support, use
176//! [finish_with_budget](TransactionBuilder::finish_with_budget) instead and
177//! provide the gas budget explicitly.
178//!
179//! Additionally, when a client is provided, the builder can directly
180//! [dry_run](TransactionBuilder::dry_run) or
181//! [execute](TransactionBuilder::execute) the transaction.
182//!
183//! When the gas payment is decided elsewhere,
184//! [finish_kind](TransactionBuilder::finish_kind) returns just the
185//! [TransactionKind](iota_types::TransactionKind): the inputs are resolved with
186//! the client, but no gas coins are selected, no budget is estimated and no gas
187//! price is fetched.
188//!
189//! When the transaction is resolved, the builder will try to ensure a valid
190//! state by de-duplicating and converting appropriate inputs into references to
191//! the gas coin. This means that the same input can be passed multiple times
192//! and the final transaction will only contain one instance. However, in some
193//! cases an invalid state can still be reached. For instance, if a coin is used
194//! both for gas and as part of a group of coins, i.e. when transferring
195//! objects, the transaction can not possibly be valid.
196//!
197//! ### Defaults
198//!
199//! When a client is provided, the builder can set some values by default. The
200//! following are the default behaviors for each metadata value.
201//!
202//! - Gas: One page of coins owned by the sender.
203//! - Gas Budget: A dry run will be used to estimate.
204//! - Gas Price: The current reference gas price.
205//!
206//! ## Gas Station
207//!
208//! The Transaction Builder supports executing via a
209//! [Gas Station](https://github.com/iotaledger/gas-station). To do so, the URL
210//! must be provided via
211//! [gas_station_sponsor](TransactionBuilder::gas_station_sponsor). Additional
212//! configuration can then be provided via
213//! [gas_reservation_duration](TransactionBuilder::gas_reservation_duration) and
214//! [add_gas_station_header](TransactionBuilder::add_gas_station_header).
215//!
216//! By default the request will contain the header `Content-Type:
217//! application/json`
218//!
219//! When this data has been set, calling [execute](TransactionBuilder::execute)
220//! will request gas from and send the resulting transaction to this endpoint
221//! instead of using the client.
222//!
223//! ## Traits and Helpers
224//!
225//! This crate provides several traits which enable the functionality of the
226//! builder. Often, when providing arguments, functions will accept either a
227//! single [PTBArgument] or a [PTBArgumentList].
228//!
229//! [PTBArgument] is implemented for any type implementing
230//! [MoveArg](types::MoveArg) as well as:
231//! - [unresolved::Argument]: Arguments returned by various builder functions.
232//!   Distinct from [iota_types::Argument], which cannot be used.
233//! - [Input](iota_types::Input): A resolved input.
234//! - [ObjectId](iota_types::ObjectId): An object's ID. Can only be used when a
235//!   client is provided. This will be assumed immutable or owned.
236//! - [ObjectReference](iota_types::ObjectReference): An object's reference.
237//!   This will be assumed immutable or owned.
238//! - [Assigned](builder::ptb_arguments::Assigned): A reference to the result of
239//!   a previous assigned command, set with
240//!   [assign](TransactionBuilder::assign).
241//! - [Shared]: Allows specifying shared immutable move objects.
242//! - [SharedMut]: Allows specifying shared mutable move objects.
243//! - [Receiving]: Allows specifying receiving move objects.
244//!
245//! [PTBArgumentList] is implemented for collection types, and represents a set
246//! of arguments. For move calls, this enables tuples of rust values to
247//! represent the parameters defined in the smart contract. For calls like
248//! [merge_coins](TransactionBuilder::merge_coins), this can represent a list of
249//! coins.
250//!
251//! [MoveArg](types::MoveArg) represents types that can be serialized and
252//! provided to the transaction as pure bytes.
253//!
254//! [MoveType](types::MoveType) defines the type tag for a rust type, so that it
255//! can be used for generic arguments.
256//!
257//! ### Example
258//!
259//! The following function is defined in move in `vec_map`:
260//!
261//! ```ignore
262//! public fun from_keys_values<K: copy, V>(mut keys: vector<K>, mut values: vector<V>): VecMap<K, V>
263//! ```
264//!
265//! ```ignore
266//! builder
267//!     .move_call(Address::TWO, "vec_map", "from_keys_values")
268//!     .generics::<(Address, u64)>()
269//!     .arguments(([address1, address2], [10000000u64, 20000000u64]));
270//! ```
271//!
272//! ### Custom Type
273//!
274//! In order to use a custom type, implement [MoveType](types::MoveType) and
275//! [MoveArg](types::MoveArg).
276//!
277//! ```
278//! # use std::str::FromStr;
279//! # use iota_sdk_transaction_builder::types::{MoveArg, MoveType, PureBytes};
280//! # use iota_types::TypeTag;
281//! #[derive(serde::Serialize)]
282//! struct MyStruct {
283//!     val1: String,
284//!     val2: u64,
285//! }
286//!
287//! impl MoveType for MyStruct {
288//!     fn type_tag() -> TypeTag {
289//!         TypeTag::from_str("0x0::my_module::MyStruct").unwrap()
290//!     }
291//! }
292//!
293//! impl MoveArg for MyStruct {
294//!     fn pure_bytes(self) -> PureBytes {
295//!         PureBytes(bcs::to_bytes(&self).unwrap())
296//!     }
297//! }
298//! ```

541        /// Serves gas coins in three single-coin pages and counts the
542        /// `objects` calls, to pin how far automatic gas selection paginates.
543        struct PagingClient {
544            pages_served: Arc<AtomicUsize>,
545        }
546
547        impl TransactionBuilderClientBase for PagingClient {
548            type Error = TestClientError;
549        }

551        impl TransactionBuilderLedgerClient for PagingClient {
552            async fn object(
553                &self,
554                object_id: ObjectId,
555                version: impl Into<Option<Version>>,
556            ) -> Result<Option<Object>, Self::Error> {
557                TestClient.object(object_id, version).await
558            }
559
560            async fn objects(
561                &self,
562                _struct_tag: Option<StructTag>,
563                _owner: iota_types::Address,
564                cursor: Option<Vec<u8>>,
565                _limit: Option<usize>,
566            ) -> Result<ObjectsPage, Self::Error> {
567                self.pages_served.fetch_add(1, Ordering::SeqCst);
568                let page = cursor.map_or(0, |c| c[0]);
569                let coin_id = ObjectId::from_bytes([0xa0 + page; ObjectId::LENGTH]).unwrap();
570                let coin = TestClient.object(coin_id, None).await?.unwrap();
571                Ok(ObjectsPage {
572                    data: vec![coin],
573                    next_cursor: (page < 2).then(|| vec![page + 1]),
574                })
575            }
576
577            async fn reference_gas_price(
578                &self,
579                epoch: impl Into<Option<u64>>,
580            ) -> Result<Option<u64>, Self::Error> {
581                TestClient.reference_gas_price(epoch).await
582            }
583        }

614        async fn an_explicit_budget_stops_gas_selection_at_the_covering_page() {
615            // `finish_with_budget` sets the budget before resolution, so gas
616            // selection stops as soon as the first page covers it.
617            let pages_served = Arc::new(AtomicUsize::new(0));
618            let builder = super::builder_with(PagingClient {
619                pages_served: pages_served.clone(),

686    #[test]
687    fn test_transaction_to_builder_roundtrip() {
688        let sender: Address = "0xc574ea804d9c1a27c886312e96c0e2c9cfd71923ebaeb3000d04b5e65fca2793"
689            .parse()
690            .unwrap();
691        let sponsor: Address = "0x0000a4984bd495d4346fa208ddff4f5d5e5ad48c21dec631ddebc99809f16900"
692            .parse()
693            .unwrap();
694        let recipient = Address::random_with(rand::thread_rng());
695        let coin = ObjectReference::new(
696            "0x19406ea4d9609cd9422b85e6bf2486908f790b778c757aff805241f3f609f9b4"
697                .parse()
698                .unwrap(),
699            Version::from_u64(2),
700            "7opR9rFUYivSTqoJHvFb9p6p54THyHTatMG6id4JKZR9"
701                .parse()
702                .unwrap(),
703        );
704        let gas_coin = ObjectReference::new(
705            "0xd8792bce2743e002673752902c0e7348dfffd78638cb5367b0b85857bceb9821"
706                .parse()
707                .unwrap(),
708            Version::from_u64(2),
709            "2ZigdvsZn5BMeszscPQZq9z8ebnS2FpmAuRbAi9ednCk"
710                .parse()
711                .unwrap(),
712        );
713
714        // Build a transaction with multiple commands and a sponsor to exercise
715        // inputs, gas, and the various builder fields.
716        let mut tx = TransactionBuilder::new(sender);
717        tx.transfer_objects(recipient, vec![coin]);
718        tx.split_coins(crate::unresolved::Argument::Gas, [42u64]);
719        tx.gas([gas_coin]);
720        tx.gas_price(1000);
721        tx.gas_budget(5_000_000);
722        tx.sponsor(sponsor);
723        tx.expiration(123);
724
725        let original = tx.finish().unwrap();
726
727        let rebuilt: TransactionBuilder = TransactionBuilder::try_from(original.clone()).unwrap();
728        let roundtrip = rebuilt.finish().unwrap();
729
730        assert_eq!(original, roundtrip);
731    }
