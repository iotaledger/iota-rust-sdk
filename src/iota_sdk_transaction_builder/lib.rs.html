4//! # IOTA Transaction Builder
5//!
6//! This crate contains the [TransactionBuilder], which allows for simple
7//! construction of Programmable Transactions which can be executed on the IOTA
8//! network.
9//!
10//! The builder is designed to allow for a lot of flexibility while also
11//! reducing the necessary boilerplate code. It uses a type-state pattern to
12//! ensure the proper flow through the various functions. It is chainable via
13//! mutable references.
14//!
15//! ## Online vs. Offline Builder
16//!
17//! The Transaction Builder can be used with or without a client implementing
18//! [TransactionBuilderLedgerClient]. When one is provided via the
19//! [with_client](TransactionBuilder::with_client) method, the resulting builder
20//! will use it to find and validate provided IDs. A ledger-only client can
21//! build transactions with an explicit gas budget via
22//! [finish_with_budget](TransactionBuilder::finish_with_budget). Clients that
23//! also implement [TransactionBuilderSimulationClient] enable
24//! [dry_run](TransactionBuilder::dry_run) and
25//! [finish](TransactionBuilder::finish) with automatic gas budget estimation;
26//! clients that additionally implement [TransactionBuilderExecutionClient]
27//! enable [execute](TransactionBuilder::execute).
28//!
29//! ### Example with Client Resolution
30//!
31//! ```
32//! # use std::str::FromStr;
33//! # use iota_sdk_transaction_builder::TestClient;
34//! use iota_sdk_transaction_builder::TransactionBuilder;
35//! use iota_types::{Address, ObjectId, Transaction};
36//!
37//! # #[tokio::main(flavor = "current_thread")]
38//! # async fn main() -> eyre::Result<()> {
39//!
40//! let sender =
41//!     Address::from_str("0xda1820edf693ee32b5729907b9b2ec8e64980ee8c008c17e89cfb4e5ecd72151")?;
42//! let to_address =
43//!     Address::from_str("0x0000a4984bd495d4346fa208ddff4f5d5e5ad48c21dec631ddebc99809f16900")?;
44//!
45//! # let client = TestClient;
46//! let mut builder = TransactionBuilder::new(sender).with_client(client);
47//!
48//! let coin =
49//!     ObjectId::from_str("0xe0e45ecb12ddca5f0d5192d2ee9e7f711959aa98614f9905e1e25c612ffd99a2")?;
50//!
51//! builder.send_coins([coin], to_address, 50000000000u64);
52//!
53//! let txn: Transaction = builder.finish().await?;
54//! # Ok(())
55//! # }
56//! ```
57//!
58//! ### Example without Client Resolution
59//!
60//! ```
61//! # use std::str::FromStr;
62//! use iota_sdk_transaction_builder::TransactionBuilder;
63//! use iota_types::{Address, ObjectDigest, ObjectId, ObjectReference, Transaction, Version};
64//!
65//! let sender =
66//!     Address::from_str("0xda1820edf693ee32b5729907b9b2ec8e64980ee8c008c17e89cfb4e5ecd72151")?;
67//! let to_address =
68//!     Address::from_str("0x0000a4984bd495d4346fa208ddff4f5d5e5ad48c21dec631ddebc99809f16900")?;
69//!
70//! let mut builder = TransactionBuilder::new(sender);
71//!
72//! let coin = ObjectReference {
73//!     object_id: ObjectId::from_str(
74//!         "0xe0e45ecb12ddca5f0d5192d2ee9e7f711959aa98614f9905e1e25c612ffd99a2",
75//!     )?,
76//!     digest: ObjectDigest::from_str("hSAGU3ZwDwxptd17ZK1QPDdJLhvPMfpSxe1p892GFVn")?,
77//!     version: Version::from_u64(545110774),
78//! };
79//! let gas_coin = ObjectReference {
80//!     object_id: ObjectId::from_str(
81//!         "0x65beb18e282d1f33a39bffa84ff92ec4d2fec0350ba6f7e5a568afff72d651db",
82//!     )?,
83//!     digest: ObjectDigest::from_str("8ahH5RXFnK1jttQEWTypYX7MRzLuQDEXk7fhMHCyZekX")?,
84//!     version: Version::from_u64(473053810),
85//! };
86//!
87//! builder
88//!     .send_coins([coin], to_address, 50000000000u64)
89//!     .gas([gas_coin])
90//!     .gas_budget(1000000000)
91//!     .gas_price(100);
92//!
93//! let txn: Transaction = builder.finish()?;
94//! # Result::<_, eyre::Error>::Ok(())
95//! ```
96//!
97//! NOTE: It is possible to provide an [ObjectId](iota_types::ObjectId) to an
98//! offline client builder, but this will cause the builder to fail when calling
99//! `finish`.
100//!
101//! ## Methods
102//!
103//! There are three kinds of methods available:
104//!
105//! ### Commands
106//!
107//! Each command method adds one or more commands to the final transaction. Some
108//! commands have optional follow-up methods. All command results can be
109//! assigned a name via [assign](TransactionBuilder::assign). Assigning a name
110//! to a command allows them to be used later in the transaction via the
111//! [assigned] method.
112//!
113//! - [move_call](TransactionBuilder::move_call): Call a move function.
114//!     - `arguments`: Add arguments to the move call.
115//!     - `generics`: Add generic types to the move call using types that
116//!       implement [MoveType](types::MoveType).
117//!     - `type_tags`: Add generic types directly using the
118//!       [TypeTag](iota_types::TypeTag).
119//! - [send_iota](TransactionBuilder::send_iota): Send IOTA coins to a recipient
120//!   address.
121//! - [send_coins](TransactionBuilder::send_coins): Send coins of any type to a
122//!   recipient address.
123//! - [pay](TransactionBuilder::pay): Send coins of any type to several
124//!   recipients, each paired with the amount to send.
125//! - [pay_iota](TransactionBuilder::pay_iota): Send IOTA coins from the gas
126//!   coin to several recipients.
127//! - [merge_coins](TransactionBuilder::merge_coins): Merge a list of coins into
128//!   a single primary coin.
129//! - [split_coins](TransactionBuilder::split_coins): Split a coin into coins of
130//!   various amounts.
131//! - [transfer_objects](TransactionBuilder::transfer_objects): Send objects to
132//!   a recipient address.
133//! - [publish_package](TransactionBuilder::publish_package): Publish a move
134//!   package.
135//!     - `package_id`: Name the package ID returned by the publish call.
136//! - [upgrade](TransactionBuilder::upgrade): Upgrade a move package.
137//! - [make_move_vec](TransactionBuilder::make_move_vec): Create a move
138//!   `vector`.
139//!
140//! ### Metadata
141//!
142//! These methods set various metadata which may be needed for the execution.
143//!
144//! - [gas](TransactionBuilder::gas): Add gas coins to pay for the execution.
145//! - [gas_refs](TransactionBuilder::gas_refs): Add gas coins that the caller
146//!   has already resolved to references.
147//! - [gas_budget](TransactionBuilder::gas_budget): Set the maximum gas budget
148//!   to spend.
149//! - [gas_price](TransactionBuilder::gas_price): Set the gas price.
150//! - [sponsor](TransactionBuilder::sponsor): Set the gas sponsor address.
151//! - [expiration](TransactionBuilder::expiration): Set the transaction
152//!   expiration epoch.
153//!
154//! ### Other
155//!
156//! Many other methods exist, either to get data or allow for development on top
157//! of the builder. Typically, these methods should not be needed, but they are
158//! made available for special circumstances.
159//!
160//! - [apply_argument](TransactionBuilder::apply_argument)
161//! - [apply_arguments](TransactionBuilder::apply_arguments)
162//! - [input](TransactionBuilder::input)
163//! - [pure_bytes](TransactionBuilder::pure_bytes)
164//! - [pure](TransactionBuilder::pure)
165//! - [command](TransactionBuilder::command)
166//! - [assigned_command](TransactionBuilder::assigned_command)
167//!
168//! ## Finalization and Execution
169//!
170//! There are several ways to finish the builder. First, the
171//! [finish](TransactionBuilder::finish) method can be used to return the
172//! resulting [Transaction](iota_types::Transaction), which can be manually
173//! serialized, executed, etc. On a client without simulation support, use
174//! [finish_with_budget](TransactionBuilder::finish_with_budget) instead and
175//! provide the gas budget explicitly.
176//!
177//! Additionally, when a client is provided, the builder can directly
178//! [dry_run](TransactionBuilder::dry_run) or
179//! [execute](TransactionBuilder::execute) the transaction.
180//!
181//! When the gas payment is decided elsewhere,
182//! [finish_kind](TransactionBuilder::finish_kind) returns just the
183//! [TransactionKind](iota_types::TransactionKind): the inputs are resolved with
184//! the client, but no gas coins are selected, no budget is estimated and no gas
185//! price is fetched.
186//!
187//! When the transaction is resolved, the builder will try to ensure a valid
188//! state by de-duplicating and converting appropriate inputs into references to
189//! the gas coin. This means that the same input can be passed multiple times
190//! and the final transaction will only contain one instance. However, in some
191//! cases an invalid state can still be reached. For instance, if a coin is used
192//! both for gas and as part of a group of coins, i.e. when transferring
193//! objects, the transaction can not possibly be valid.
194//!
195//! ### Defaults
196//!
197//! When a client is provided, the builder can set some values by default. The
198//! following are the default behaviors for each metadata value.
199//!
200//! - Gas: One page of coins owned by the sender.
201//! - Gas Budget: A dry run will be used to estimate.
202//! - Gas Price: The current reference gas price.
203//!
204//! ## Gas Sponsorship
205//!
206//! A transaction's gas can be paid by someone other than its sender, in two
207//! ways depending on who holds the sponsor's key.
208//!
209//! When you hold it, set the sponsor's address with
210//! [sponsor](TransactionBuilder::sponsor) — the gas coins are drawn from it —
211//! and call
212//! [execute_with_sponsor_signer](TransactionBuilder::execute_with_sponsor_signer),
213//! which signs as both parties and submits through the client.
214//!
215//! When a service holds it, pass a [GasSponsor] to
216//! [execute_with_gas_sponsor](TransactionBuilder::execute_with_gas_sponsor). It
217//! supplies the whole gas payment and submits the transaction itself, so the
218//! sender's own coins are never looked up; setting gas coins or a
219//! [sponsor](TransactionBuilder::sponsor) address on the same builder is
220//! rejected.
221//!
222//! [GasStation] implements [GasSponsor] for the
223//! [IOTA gas station](https://github.com/iotaledger/gas-station) and is enabled
224//! by the `gas-station` feature. A station is configured once — with its URL
225//! and, typically, an authorization header — and reused for any number of
226//! transactions:
227//!
228//! ```no_run
229//! # use iota_sdk_transaction_builder::GasStation;
230//! use iota_sdk_transaction_builder::{HeaderValue, header::AUTHORIZATION};
231//!
232//! # fn main() -> eyre::Result<()> {
233//! let station = GasStation::builder("http://0.0.0.0:9527".parse()?)
234//!     .header(AUTHORIZATION, HeaderValue::from_static("Bearer token"))
235//!     .build();
236//! # Ok(())
237//! # }
238//! ```
239//!
240//! Pass [http_client](GasStationBuilder::http_client) to control timeouts,
241//! proxies or TLS roots; otherwise reqwest's defaults are used. Requests carry
242//! `Content-Type: application/json` unless a header overrides it.
243//!
244//! Implement [GasSponsor] yourself to sponsor through a service this crate does
245//! not ship.
246//!
247//! ## Traits and Helpers
248//!
249//! This crate provides several traits which enable the functionality of the
250//! builder. Often, when providing arguments, functions will accept either a
251//! single [PTBArgument] or a [PTBArgumentList].
252//!
253//! [PTBArgument] is implemented for any type implementing
254//! [MoveArg](types::MoveArg) as well as:
255//! - [unresolved::Argument]: Arguments returned by various builder functions.
256//!   Distinct from [iota_types::Argument], which cannot be used.
257//! - [Input](iota_types::Input): A resolved input.
258//! - [ObjectId](iota_types::ObjectId): An object's ID. Can only be used when a
259//!   client is provided. This will be assumed immutable or owned.
260//! - [ObjectReference](iota_types::ObjectReference): An object's reference.
261//!   This will be assumed immutable or owned.
262//! - [Assigned]: A reference to the result of a previous assigned command, set
263//!   with [assign](TransactionBuilder::assign).
264//! - [Shared]: Allows specifying shared immutable move objects.
265//! - [SharedMut]: Allows specifying shared mutable move objects.
266//! - [Receiving]: Allows specifying receiving move objects.
267//!
268//! [PTBArgumentList] is implemented for collection types, and represents a set
269//! of arguments. For move calls, this enables tuples of rust values to
270//! represent the parameters defined in the smart contract. For calls like
271//! [merge_coins](TransactionBuilder::merge_coins), this can represent a list of
272//! coins.
273//!
274//! [MoveArg](types::MoveArg) represents types that can be serialized and
275//! provided to the transaction as pure bytes.
276//!
277//! [MoveType](types::MoveType) defines the type tag for a rust type, so that it
278//! can be used for generic arguments.
279//!
280//! ### Example
281//!
282//! The following function is defined in move in `vec_map`:
283//!
284//! ```ignore
285//! public fun from_keys_values<K: copy, V>(mut keys: vector<K>, mut values: vector<V>): VecMap<K, V>
286//! ```
287//!
288//! ```ignore
289//! builder
290//!     .move_call(Address::TWO, "vec_map", "from_keys_values")
291//!     .generics::<(Address, u64)>()
292//!     .arguments(([address1, address2], [10000000u64, 20000000u64]));
293//! ```
294//!
295//! ### Custom Type
296//!
297//! In order to use a custom type, implement [MoveType](types::MoveType) and
298//! [MoveArg](types::MoveArg).
299//!
300//! ```
301//! # use std::str::FromStr;
302//! # use iota_sdk_transaction_builder::types::{MoveArg, MoveType, PureBytes};
303//! # use iota_types::TypeTag;
304//! #[derive(serde::Serialize)]
305//! struct MyStruct {
306//!     val1: String,
307//!     val2: u64,
308//! }
309//!
310//! impl MoveType for MyStruct {
311//!     fn type_tag() -> TypeTag {
312//!         TypeTag::from_str("0x0::my_module::MyStruct").unwrap()
313//!     }
314//! }
315//!
316//! impl MoveArg for MyStruct {
317//!     fn pure_bytes(self) -> PureBytes {
318//!         PureBytes(bcs::to_bytes(&self).unwrap())
319//!     }
320//! }
321//! ```

577    mod gas_pagination {
578        use std::sync::{
579            Arc,
580            atomic::{AtomicUsize, Ordering},
581        };
582
583        use iota_types::{Object, ObjectId, StructTag, Transaction, Version};
584
585        use crate::{
586            ObjectsPage, ProtocolConfig, TestClient, TestClientError, TransactionBuilderClientBase,
587            TransactionBuilderLedgerClient, TransactionBuilderSimulationClient,
588        };
589
590        /// Serves gas coins in three single-coin pages and counts the
591        /// `objects` calls, to pin how far automatic gas selection paginates.
592        struct PagingClient {
593            pages_served: Arc<AtomicUsize>,
594        }
595
596        impl TransactionBuilderClientBase for PagingClient {
597            type Error = TestClientError;
598        }
599
600        impl TransactionBuilderLedgerClient for PagingClient {
601            async fn object(
602                &self,
603                object_id: ObjectId,
604                version: impl Into<Option<Version>>,
605            ) -> Result<Option<Object>, Self::Error> {
606                TestClient.object(object_id, version).await
607            }
608
609            async fn objects(
610                &self,
611                _struct_tag: Option<StructTag>,
612                _owner: iota_types::Address,
613                cursor: Option<Vec<u8>>,
614                _limit: Option<usize>,
615            ) -> Result<ObjectsPage, Self::Error> {
616                self.pages_served.fetch_add(1, Ordering::SeqCst);
617                let page = cursor.map_or(0, |c| c[0]);
