10/// An identifier indicating the unresolved index of an input.
11pub type InputId = usize;
12
13#[derive(Clone, Debug, Eq, PartialEq)]
14pub struct Input {
15    pub kind: InputKind,
16    pub is_gas: bool,
17}
18
19impl Input {
20    pub fn object_id(&self) -> Option<&ObjectId> {
21        match &self.kind {
22            InputKind::ImmutableOrOwned(object_id)
23            | InputKind::Shared { object_id, .. }
24            | InputKind::Receiving(object_id) => Some(object_id),
25            InputKind::Input(input) => match input {
26                iota_types::Input::Pure(..) => None,
27                iota_types::Input::ImmutableOrOwned(ObjectReference { object_id, .. })
28                | iota_types::Input::Shared(SharedObjectReference { object_id, .. })
29                | iota_types::Input::Receiving(ObjectReference { object_id, .. }) => {
30                    Some(object_id)
31                }
32                _ => unimplemented!("a new enum variant was added and needs to be handled"),
33            },
34        }
35    }
36}
37
38#[derive(Clone, Debug, Eq, PartialEq)]
39#[non_exhaustive]
40pub enum InputKind {
41    ImmutableOrOwned(ObjectId),
42    Shared { object_id: ObjectId, mutable: bool },
43    Receiving(ObjectId),
44    Input(iota_types::Input),
45}
46
47impl InputKind {
48    pub fn object_id(&self) -> Option<ObjectId> {
49        if let Self::ImmutableOrOwned(object_id)
50        | Self::Receiving(object_id)
51        | Self::Shared { object_id, .. }
52        | Self::Input(
53            iota_types::Input::ImmutableOrOwned(ObjectReference { object_id, .. })
54            | iota_types::Input::Receiving(ObjectReference { object_id, .. })
55            | iota_types::Input::Shared(SharedObjectReference { object_id, .. }),
56        ) = self
57        {
58            Some(*object_id)
59        } else {
60            None
61        }
62    }
63}

65#[derive(Clone, Debug, derive_more::From)]
66#[non_exhaustive]
67pub enum Command {
68    MoveCall(MoveCall),
69    TransferObjects(TransferObjects),
70    SplitCoins(SplitCoins),
71    MergeCoins(MergeCoins),
72    Publish(Publish),
73    MakeMoveVector(MakeMoveVector),
74    Upgrade(Upgrade),
75}
76
77impl Command {
78    pub fn resolve(self, input_map: &HashMap<InputId, u16>) -> iota_types::Command {
79        match self {
80            Command::MoveCall(move_call) => {
81                iota_types::Command::MoveCall(move_call.resolve(input_map))
82            }
83            Command::TransferObjects(transfer_objects) => {
84                iota_types::Command::TransferObjects(transfer_objects.resolve(input_map))
85            }
86            Command::SplitCoins(split_coins) => {
87                iota_types::Command::SplitCoins(split_coins.resolve(input_map))
88            }
89            Command::MergeCoins(merge_coins) => {
90                iota_types::Command::MergeCoins(merge_coins.resolve(input_map))
91            }
92            Command::Publish(publish) => iota_types::Command::Publish(publish.resolve()),
93            Command::MakeMoveVector(make_move_vector) => {
94                iota_types::Command::MakeMoveVector(make_move_vector.resolve(input_map))
95            }
96            Command::Upgrade(upgrade) => iota_types::Command::Upgrade(upgrade.resolve(input_map)),
97        }
98    }
99
100    /// The arguments of this command that cannot be replaced by
101    /// [`Argument::Gas`].
102    ///
103    /// [`Argument::Gas`] stands for the whole gas payment — every gas coin
104    /// smashed into one — so substituting it for an argument that names a
105    /// single coin changes what the command operates on. The positions left
106    /// out take a coin without consuming it: a coin split from or merged into
107    /// is still there afterwards.
108    pub(crate) fn arguments_that_cannot_be_gas(&self) -> &[Argument] {
109        match self {
110            Command::MoveCall(MoveCall { arguments, .. }) => arguments,
111            Command::TransferObjects(TransferObjects { objects, .. }) => objects,
112            Command::MergeCoins(MergeCoins { coins_to_merge, .. }) => coins_to_merge,
113            _ => &[],
114        }
115    }
116}
117
118impl From<iota_types::Command> for Command {
119    fn from(cmd: iota_types::Command) -> Self {
120        match cmd {
121            iota_types::Command::MoveCall(c) => Self::MoveCall(MoveCall {
122                package: c.package,
123                module: c.module,
124                function: c.function,
125                type_arguments: c.type_arguments,
126                arguments: c.arguments.into_iter().map(Into::into).collect(),
127            }),
128            iota_types::Command::TransferObjects(c) => Self::TransferObjects(TransferObjects {
129                objects: c.objects.into_iter().map(Into::into).collect(),
130                address: c.address.into(),
131            }),
132            iota_types::Command::SplitCoins(c) => Self::SplitCoins(SplitCoins {
133                coin: c.coin.into(),
134                amounts: c.amounts.into_iter().map(Into::into).collect(),
135            }),
136            iota_types::Command::MergeCoins(c) => Self::MergeCoins(MergeCoins {
137                coin: c.coin.into(),
138                coins_to_merge: c.coins_to_merge.into_iter().map(Into::into).collect(),
139            }),
140            iota_types::Command::Publish(c) => Self::Publish(Publish {
141                modules: c.modules,
142                dependencies: c.dependencies,
143            }),
144            iota_types::Command::MakeMoveVector(c) => Self::MakeMoveVector(MakeMoveVector {
145                type_tag: c.type_tag,
146                elements: c.elements.into_iter().map(Into::into).collect(),
147            }),
148            iota_types::Command::Upgrade(c) => Self::Upgrade(Upgrade {
149                modules: c.modules,
150                dependencies: c.dependencies,
151                package: c.package,
152                ticket: c.ticket.into(),
153            }),
154            _ => unimplemented!("a new Command enum variant was added and needs to be handled"),
155        }
156    }
157}

168impl MoveCall {

170        iota_types::MoveCall {
171            package: self.package,
172            module: self.module,
173            function: self.function,
174            type_arguments: self.type_arguments,

294#[derive(Clone, Copy, Debug)]
295#[non_exhaustive]
296pub enum Argument {
297    Gas,
298    Input(InputId),
299    Result(u16),
300    NestedResult(u16, u16),
301}
302
303impl Argument {
304    fn resolve(self, input_map: &HashMap<InputId, u16>) -> iota_types::Argument {
305        match self {
306            Argument::Gas => iota_types::Argument::Gas,
307            Argument::Input(i) => input_map
308                .get(&i)
309                .map(|i| iota_types::Argument::Input(*i))
310                .unwrap_or(iota_types::Argument::Gas),
311            Argument::Result(i) => iota_types::Argument::Result(i),
312            Argument::NestedResult(i1, i2) => iota_types::Argument::NestedResult(i1, i2),
313        }
314    }
315}
316
317impl From<iota_types::Argument> for Argument {
318    fn from(arg: iota_types::Argument) -> Self {
319        match arg {
320            iota_types::Argument::Gas => Self::Gas,
321            iota_types::Argument::Input(i) => Self::Input(i as InputId),
322            iota_types::Argument::Result(i) => Self::Result(i),
323            iota_types::Argument::NestedResult(i1, i2) => Self::NestedResult(i1, i2),
324            _ => unimplemented!("a new Argument enum variant was added and needs to be handled"),
325        }
326    }
