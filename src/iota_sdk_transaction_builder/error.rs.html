5//! Transaction Builder errors.
6
7use base64ct::Error as Base64Error;
8use iota_types::{ObjectId, TransactionDigest};
9
10use crate::builder::gas_station::{GasStationVersion, VersionParsingError};
11
12#[derive(Debug, thiserror::Error)]
13#[non_exhaustive]
14#[allow(missing_docs)]
15pub enum Error {
16    #[error("Conversion error due to input issue: {0}")]
17    Input(String),
18    #[error("Gas object should be an immutable or owned object")]
19    WrongGasObject,
20    #[error("gas coin {object_id} cannot also be passed to a command")]
21    GasCoinAsArgument { object_id: ObjectId },
22    #[error(
23        "transferring gas coin {transferred} would also transfer gas coin {missing} due to gas smashing; if this is intentional, add coin {missing} to the transferred objects"
24    )]
25    IncompleteGasTransfer {
26        transferred: ObjectId,
27        missing: ObjectId,
28    },
29    #[error("only one command can transfer the gas coin")]
30    GasCoinTransferredMoreThanOnce,
31    #[error("BCS serialization error: {0}")]
32    Bcs(bcs::Error),
33    #[error("Decoding error: {0}")]
34    Decoding(#[from] Base64Error),
35    #[error("Missing object id")]
36    MissingObjectId,
37    #[error("Missing version for object {0}")]
38    MissingVersion(ObjectId),
39    #[error("Missing digest for object {0}")]
40    MissingDigest(ObjectId),
41    #[error("Missing transaction for digest {0}")]
42    MissingTransaction(TransactionDigest),
43    #[error("Missing gas objects")]
44    MissingGasObjects,
45    #[error("Missing gas budget")]
46    MissingGasBudget,
47    #[error("Missing gas price")]
48    MissingGasPrice,
49    #[error("Missing object kind for object {0}")]
50    MissingObjectKind(ObjectId),
51    #[error("Missing initial shared version for object {0}")]
52    MissingInitialSharedVersion(ObjectId),
53    #[error("Missing pure value")]
54    MissingPureValue,
55    #[error("Missing gas station data")]
56    MissingGasStationData,
57    #[error("Unknown shared object mutability for object {0}")]
58    SharedObjectMutability(ObjectId),
59    #[error("Unsupported literal")]
60    UnsupportedLiteral,
61    #[error("only programmable transactions can be converted into a TransactionBuilder")]
62    UnsupportedTransactionKind,
63    #[error("Invalid account for move authenticator: {0}")]
64    InvalidMoveAuthAccount(String),
65    #[error("Invalid argument for move authenticator: {0}")]
66    InvalidMoveAuthArg(String),
67    #[error(transparent)]
68    InvalidUrl(<reqwest::Url as std::str::FromStr>::Err),
69    #[error("Request to gas station `{gas_station_url}` failed: {source}")]
70    GasStationRequest {
71        source: reqwest::Error,
72        gas_station_url: reqwest::Url,
73    },
74    #[
75        error("Invalid gas station response from {gas_station_url}{}", 
76        .message.as_deref().map(|msg| format!(": {msg}")).unwrap_or_default())
77    ]
78    GasStationResponse {
79        message: Option<String>,
80        gas_station_url: reqwest::Url,
81    },
82    #[error(
83        "invalid gas-station version: got version `{version}`, but at least version `{min_required_version}` is required"
84    )]
85    InvalidGasStationVersion {
86        /// The minimum IOTA gas-station version needed for this operation.
87        min_required_version: GasStationVersion,
88        /// The actual IOTA gas-station's version.
89        version: GasStationVersion,
90    },
91    #[error(transparent)]
92    VersionParsing(VersionParsingError),
93    #[error(transparent)]
94    Signature(Box<dyn std::error::Error + Send + Sync>),
95    #[error(transparent)]
96    Client(Box<dyn std::error::Error + Send + Sync>),
97    #[error("Failed to dry run transaction: {0}")]
98    DryRun(String),
99}
100
101impl Error {
102    /// Create a client error
103    pub fn client<E: 'static + std::error::Error + Send + Sync>(e: E) -> Self {
104        Self::Client(Box::new(e))
105    }
106
107    /// Create a signature error
108    pub fn signature<E: 'static + std::error::Error + Send + Sync>(e: E) -> Self {
109        Self::Signature(Box::new(e))
110    }
111}
