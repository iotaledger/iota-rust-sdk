5//! Transaction Builder errors.
6
7use base64ct::Error as Base64Error;
8use iota_types::{Address, ObjectId, TransactionDigest};
9
10#[derive(Debug, thiserror::Error)]
11#[non_exhaustive]
12#[allow(missing_docs)]
13pub enum TransactionBuilderError {
14    #[error("Conversion error due to input issue: {0}")]
15    Input(String),
16    #[error("Gas object should be an immutable or owned object")]
17    WrongGasObject,
18    #[error("gas coin {object_id} cannot also be passed to a command")]
19    GasCoinAsArgument { object_id: ObjectId },
20    #[error(
21        "transferring gas coin {transferred} would also transfer gas coin {missing} due to gas smashing; if this is intentional, add coin {missing} to the transferred objects"
22    )]
23    IncompleteGasTransfer {
24        transferred: ObjectId,
25        missing: ObjectId,
26    },
27    #[error("only one command can transfer the gas coin")]
28    GasCoinTransferredMoreThanOnce,
29    #[error("BCS serialization error: {0}")]
30    Bcs(bcs::Error),
31    #[error("Decoding error: {0}")]
32    Decoding(#[from] Base64Error),
33    #[error("Missing object id")]
34    MissingObjectId,
35    #[error("Missing version for object {0}")]
36    MissingVersion(ObjectId),
37    #[error("Missing digest for object {0}")]
38    MissingDigest(ObjectId),
39    #[error("Missing transaction for digest {0}")]
40    MissingTransaction(TransactionDigest),
41    #[error("Missing gas objects")]
42    MissingGasObjects,
43    #[error("Missing gas budget")]
44    MissingGasBudget,
45    #[error("Missing gas price")]
46    MissingGasPrice,
47    #[error("Missing object kind for object {0}")]
48    MissingObjectKind(ObjectId),
49    #[error("Missing initial shared version for object {0}")]
50    MissingInitialSharedVersion(ObjectId),
51    #[error("Missing pure value")]
52    MissingPureValue,
53    #[error("Missing protocol value: {name}")]
54    MissingProtocolValue { name: String },
55    #[error("Invalid protocol value: {name} = {value}")]
56    InvalidProtocolValue { name: String, value: String },
57    #[error("Unknown shared object mutability for object {0}")]
58    SharedObjectMutability(ObjectId),
59    #[error("Unsupported literal")]
60    UnsupportedLiteral,
61    #[error("only programmable transactions can be converted into a TransactionBuilder")]
62    UnsupportedTransactionKind,
63    #[error("Invalid account for move authenticator: {0}")]
64    InvalidMoveAuthAccount(String),
65    #[error("Invalid argument for move authenticator: {0}")]
66    InvalidMoveAuthArg(String),
67    #[error("gas coins were set on the transaction, but the gas sponsor supplies the gas payment")]
68    SponsorGasConflict,
69    #[error(
70        "sponsor {sponsor} was set on the transaction, but the gas sponsor supplies the gas payment"
71    )]
72    SponsorAddressConflict { sponsor: Address },
73    #[error(transparent)]
74    GasSponsor(Box<dyn std::error::Error + Send + Sync>),
75    #[error(transparent)]
76    Signature(Box<dyn std::error::Error + Send + Sync>),
77    #[error(transparent)]
78    Client(Box<dyn std::error::Error + Send + Sync>),
79    #[error("Failed to dry run transaction: {0}")]
80    DryRun(String),
81}
82
83impl TransactionBuilderError {
84    /// Create a client error
85    pub fn client<E: 'static + std::error::Error + Send + Sync>(e: E) -> Self {
86        Self::Client(Box::new(e))
87    }
88
89    /// Create a signature error
90    pub fn signature<E: 'static + std::error::Error + Send + Sync>(e: E) -> Self {
91        Self::Signature(Box::new(e))
92    }
93
94    /// Create a gas sponsor error
95    pub fn gas_sponsor<E: 'static + std::error::Error + Send + Sync>(e: E) -> Self {
96        Self::GasSponsor(Box::new(e))
97    }
98}
