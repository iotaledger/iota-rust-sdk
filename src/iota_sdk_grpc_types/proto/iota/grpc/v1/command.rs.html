14impl TryFrom<iota_types::transaction::Argument> for Argument {
15    type Error = GrpcConversionError;
16
17    fn try_from(arg: iota_types::transaction::Argument) -> Result<Self, Self::Error> {
18        let kind = match arg {
19            iota_types::transaction::Argument::Gas => argument::Kind::GasCoin(argument::GasCoin {}),
20            iota_types::transaction::Argument::Input(idx) => {
21                argument::Kind::Input(argument::Input {
22                    index: Some(idx as u32),
23                })
24            }
25            iota_types::transaction::Argument::Result(idx) => {
26                argument::Kind::Result(argument::Result {
27                    index: Some(idx as u32),
28                    nested_result_index: None,
29                })
30            }
31            iota_types::transaction::Argument::NestedResult(idx, nested_idx) => {
32                argument::Kind::Result(argument::Result {
33                    index: Some(idx as u32),
34                    nested_result_index: Some(nested_idx as u32),
35                })
36            }
37            _ => {
38                return Err(GrpcConversionError::UnsupportedArgumentType {
39                    arg_type: format!("{arg:?}"),
40                });
41            }
42        };
43
44        Ok(Self { kind: Some(kind) })
45    }
46}

48impl TryFrom<&Argument> for iota_types::transaction::Argument {
49    type Error = TryFromProtoError;
50
51    fn try_from(value: &Argument) -> Result<Self, Self::Error> {
52        match &value.kind {
53            Some(argument::Kind::GasCoin(_)) => Ok(iota_types::transaction::Argument::Gas),
54            Some(argument::Kind::Input(input)) => {
55                let index = input
56                    .index
57                    .ok_or_else(|| TryFromProtoError::missing("argument.input.index"))?;
58                Ok(iota_types::transaction::Argument::Input(index as u16))
59            }
60            Some(argument::Kind::Result(result)) => {
61                let index = result
62                    .index
63                    .ok_or_else(|| TryFromProtoError::missing("argument.result.index"))?;
64                match result.nested_result_index {
65                    Some(nested_idx) => Ok(iota_types::transaction::Argument::NestedResult(
66                        index as u16,
67                        nested_idx as u16,
68                    )),
69                    None => Ok(iota_types::transaction::Argument::Result(index as u16)),
70                }
71            }
72            Some(argument::Kind::Unknown(_)) => Err(TryFromProtoError::invalid(
73                "argument.kind",
74                "unknown argument type",
75            )),
76            None => Err(TryFromProtoError::missing("argument.kind")),
77        }
78    }
79}
