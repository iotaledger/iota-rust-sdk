11impl DynamicField {
12    /// Get the JSON rendering of the dynamic field's "name".
13    ///
14    /// **Read mask:** `"json_name"` (see
15    /// [`DYNAMIC_FIELD_JSON_NAME`](crate::read_masks::DYNAMIC_FIELD_JSON_NAME)).
16    pub fn json_name(&self) -> Result<serde_json::Value, TryFromProtoError> {
17        self.json_name
18            .as_ref()
19            .map(crate::proto::prost_to_json)
20            .ok_or_else(|| TryFromProtoError::missing(DynamicField::JSON_NAME_FIELD.name))
21    }
22}
