19impl TryFrom<&Digest> for iota_types::Digest {
20    type Error = TryFromProtoError;
21
22    fn try_from(value: &Digest) -> Result<Self, Self::Error> {
23        iota_types::Digest::from_bytes(&value.digest)
24            .map_err(|e| TryFromProtoError::invalid("digest", e))
25    }
26}

28/// The domain-specific digest newtypes are transparent wrappers around
29/// [`iota_types::Digest`], so their proto conversions just delegate to it.
30macro_rules! impl_digest_wrapper_conversion {
31    ($wrapper:ty) => {
32        impl From<$wrapper> for Digest {
33            fn from(value: $wrapper) -> Self {
34                iota_types::Digest::from(value).into()
35            }
36        }
37
38        impl TryFrom<&Digest> for $wrapper {
39            type Error = TryFromProtoError;
40
41            fn try_from(value: &Digest) -> Result<Self, Self::Error> {
42                iota_types::Digest::try_from(value).map(Into::into)
43            }
44        }
45    };
46}

108impl TryFrom<&ObjectReference> for iota_types::ObjectReference {
109    type Error = TryFromProtoError;
110
111    fn try_from(value: &ObjectReference) -> Result<Self, Self::Error> {
112        let object_id = value
113            .object_id
114            .as_ref()
115            .ok_or_else(|| TryFromProtoError::missing(ObjectReference::OBJECT_ID_FIELD.name))?
116            .object_id()
117            .map_err(|e| e.nested(ObjectReference::OBJECT_ID_FIELD.name))?;
118
119        let version = value
120            .version
121            .ok_or_else(|| TryFromProtoError::missing(ObjectReference::VERSION_FIELD.name))?;
122
123        let digest = value
124            .digest
125            .as_ref()
126            .ok_or_else(|| TryFromProtoError::missing(ObjectReference::DIGEST_FIELD.name))?;
127
128        let digest = digest
129            .try_into()
130            .map_err(|e| TryFromProtoError::invalid(ObjectReference::DIGEST_FIELD.name, e))?;
131
132        Ok(iota_types::ObjectReference {
133            object_id,
134            version: version.into(),
135            digest,
136        })
137    }
138}

189// Owner conversions
190impl From<iota_types::Owner> for Owner {
191    fn from(value: iota_types::Owner) -> Self {
192        let kind = match value {
193            iota_types::Owner::Address(address) => Some(owner::Kind::AddressOwner(address.into())),
194            iota_types::Owner::Object(object_id) => {
195                Some(owner::Kind::ObjectOwner(object_id.into()))
196            }
197            iota_types::Owner::Shared(version) => Some(owner::Kind::Shared(version.as_u64())),
198            iota_types::Owner::Immutable => Some(owner::Kind::Immutable(true)),
199            // `iota_types::Owner` is `#[non_exhaustive]`; an unknown future
200            // variant maps to an unset oneof ("unspecified ownership kind").
201            _ => None,
202        };
203
204        Self { kind }
205    }
206}

208impl TryFrom<&Owner> for iota_types::Owner {
209    type Error = TryFromProtoError;
210
211    fn try_from(value: &Owner) -> Result<Self, Self::Error> {
212        match &value.kind {
213            Some(owner::Kind::AddressOwner(address)) => Ok(iota_types::Owner::Address(
214                address
215                    .address()
216                    .map_err(|e| e.nested(Owner::ADDRESS_OWNER_FIELD.name))?,
217            )),
218            Some(owner::Kind::ObjectOwner(object_id)) => Ok(iota_types::Owner::Object(
219                object_id
220                    .object_id()
221                    .map_err(|e| e.nested(Owner::OBJECT_OWNER_FIELD.name))?,
222            )),
223            Some(owner::Kind::Shared(version)) => Ok(iota_types::Owner::Shared((*version).into())),
224            Some(owner::Kind::Immutable(_)) => Ok(iota_types::Owner::Immutable),
225            None => Err(TryFromProtoError::missing("kind")),
226        }
227    }
228}

237impl From<&iota_types::TypeTag> for TypeTag {
238    fn from(ty: &iota_types::TypeTag) -> Self {
239        let type_tag = match ty {
240            iota_types::TypeTag::Bool => type_tag::TypeTag::BoolTag(true),
241            iota_types::TypeTag::U8 => type_tag::TypeTag::U8Tag(true),
242            iota_types::TypeTag::U16 => type_tag::TypeTag::U16Tag(true),
243            iota_types::TypeTag::U32 => type_tag::TypeTag::U32Tag(true),
244            iota_types::TypeTag::U64 => type_tag::TypeTag::U64Tag(true),
245            iota_types::TypeTag::U128 => type_tag::TypeTag::U128Tag(true),
246            iota_types::TypeTag::U256 => type_tag::TypeTag::U256Tag(true),
247            iota_types::TypeTag::Address => type_tag::TypeTag::AddressTag(true),
248            iota_types::TypeTag::Signer => type_tag::TypeTag::SignerTag(true),
249            iota_types::TypeTag::Vector(inner) => {
250                type_tag::TypeTag::VectorTag(Box::new(TypeTagVector {
251                    inner_type: Some(Box::new(inner.as_ref().into())),
252                }))
253            }
254            iota_types::TypeTag::Struct(struct_tag) => {
255                type_tag::TypeTag::StructTag(TypeTagStruct {
256                    struct_tag: struct_tag.to_canonical_string(true),
257                })
258            }
259        };
260
261        Self {
262            type_tag: Some(type_tag),
263        }
264    }
265}

267impl TryFrom<&TypeTag> for iota_types::TypeTag {
268    type Error = TryFromProtoError;
269
270    fn try_from(value: &TypeTag) -> Result<Self, Self::Error> {
271        match &value.type_tag {
272            Some(type_tag::TypeTag::BoolTag(_)) => Ok(iota_types::TypeTag::Bool),
273            Some(type_tag::TypeTag::U8Tag(_)) => Ok(iota_types::TypeTag::U8),
274            Some(type_tag::TypeTag::U16Tag(_)) => Ok(iota_types::TypeTag::U16),
275            Some(type_tag::TypeTag::U32Tag(_)) => Ok(iota_types::TypeTag::U32),
276            Some(type_tag::TypeTag::U64Tag(_)) => Ok(iota_types::TypeTag::U64),
277            Some(type_tag::TypeTag::U128Tag(_)) => Ok(iota_types::TypeTag::U128),
278            Some(type_tag::TypeTag::U256Tag(_)) => Ok(iota_types::TypeTag::U256),
279            Some(type_tag::TypeTag::AddressTag(_)) => Ok(iota_types::TypeTag::Address),
280            Some(type_tag::TypeTag::SignerTag(_)) => Ok(iota_types::TypeTag::Signer),
281            Some(type_tag::TypeTag::VectorTag(inner)) => {
282                let inner_type = inner
283                    .inner_type
284                    .as_ref()
285                    .ok_or_else(|| TryFromProtoError::missing("type_tag.vector.inner_type"))?;
286                let inner_sdk: iota_types::TypeTag = inner_type.as_ref().try_into()?;
287                Ok(iota_types::TypeTag::Vector(Box::new(inner_sdk)))
288            }
289            Some(type_tag::TypeTag::StructTag(struct_tag)) => {
290                let parsed: iota_types::StructTag = struct_tag
291                    .struct_tag
292                    .parse()
293                    .map_err(|e| TryFromProtoError::invalid("type_tag.struct_tag", e))?;
294                Ok(iota_types::TypeTag::Struct(Box::new(parsed)))
295            }
296            None => Err(TryFromProtoError::missing("type_tag")),
297        }
298    }
299}
