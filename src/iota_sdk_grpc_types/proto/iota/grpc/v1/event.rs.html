14// TryFrom implementations for Event
15impl TryFrom<&Event> for iota_types::Event {
16    type Error = TryFromProtoError;
17
18    fn try_from(value: &Event) -> Result<Self, Self::Error> {
19        let bcs = value
20            .bcs
21            .as_ref()
22            .ok_or_else(|| TryFromProtoError::missing(Event::BCS_FIELD.name))?;
23
24        bcs.deserialize::<VersionedEvent>()
25            .map_err(|e| TryFromProtoError::invalid(Event::BCS_FIELD.name, e))?
26            .try_into_v1()
27            .map_err(|_| {
28                TryFromProtoError::invalid(Event::BCS_FIELD.name, "unsupported Event version")
29            })
30    }
31}
32
33// Convenience methods for Event (delegate to TryFrom)
34impl Event {
35    /// Deserialize the full event from BCS.
36    ///
37    /// **Read mask:** `bcs` relative to this event (see [`EVENT_BCS`]).
38    /// Full path depends on context — e.g. `"events.bcs"` for checkpoint
39    /// top-level events, `"events.events.bcs"` for transaction events.
40    ///
41    /// [`EVENT_BCS`]: crate::read_masks::EVENT_BCS
42    pub fn event(&self) -> Result<iota_types::Event, TryFromProtoError> {
43        self.try_into()
44    }
45
46    /// Get the package ID of the Move module that emitted this event.
47    ///
48    /// **Read mask:** `package_id` relative to this event (see
49    /// [`EVENT_PACKAGE_ID`]).
50    ///
51    /// [`EVENT_PACKAGE_ID`]: crate::read_masks::EVENT_PACKAGE_ID
52    pub fn package_id(&self) -> Result<iota_types::ObjectId, TryFromProtoError> {
53        get_inner_field!(self.package_id, Self::PACKAGE_ID_FIELD, object_id)
54    }
55
56    /// Get the module name of the Move module that emitted this event.
57    ///
58    /// **Read mask:** `module` relative to this event (see [`EVENT_MODULE`]).
59    ///
60    /// [`EVENT_MODULE`]: crate::read_masks::EVENT_MODULE
61    pub fn module_name(&self) -> Result<iota_types::Identifier, TryFromProtoError> {
62        self.module
63            .as_deref()
64            .ok_or_else(|| TryFromProtoError::missing(Self::MODULE_FIELD.name))?
65            .parse()
66            .map_err(|e: iota_types::TypeParseError| {
67                TryFromProtoError::invalid(
68                    Self::MODULE_FIELD.name,
69                    format!("invalid identifier format {e}"),
70                )
71            })
72    }
73
74    /// Get the sender address of the transaction that emitted this event.
75    ///
76    /// **Read mask:** `sender` relative to this event (see [`EVENT_SENDER`]).
77    ///
78    /// [`EVENT_SENDER`]: crate::read_masks::EVENT_SENDER
79    pub fn sender(&self) -> Result<iota_types::Address, TryFromProtoError> {
80        get_inner_field!(self.sender, Self::SENDER_FIELD, try_into)
81    }
82
83    /// Get the type of the event emitted.
84    ///
85    /// **Read mask:** `event_type` relative to this event (see
86    /// [`EVENT_TYPE`]).
87    ///
88    /// [`EVENT_TYPE`]: crate::read_masks::EVENT_TYPE
89    pub fn type_name(&self) -> Result<iota_types::StructTag, TryFromProtoError> {
90        self.event_type
91            .as_deref()
92            .ok_or_else(|| TryFromProtoError::missing(Self::EVENT_TYPE_FIELD.name))?
93            .parse()
94            .map_err(|e: iota_types::TypeParseError| {
95                TryFromProtoError::invalid(
96                    Self::EVENT_TYPE_FIELD.name,
97                    format!("invalid struct tag format {e}"),
98                )
99            })
100    }
101
102    /// Get the raw BCS bytes of the event contents/data only.
103    ///
104    /// This is the serialized event data without the metadata (package, module,
105    /// sender, type).
106    ///
107    /// **Read mask:** `bcs_contents` relative to this event (see
108    /// [`EVENT_BCS_CONTENTS`]).
109    ///
110    /// [`EVENT_BCS_CONTENTS`]: crate::read_masks::EVENT_BCS_CONTENTS
111    pub fn bcs_contents(&self) -> Result<&[u8], TryFromProtoError> {
112        self.bcs_contents
113            .as_ref()
114            .map(BcsData::as_bytes)
115            .ok_or_else(|| TryFromProtoError::missing(Self::BCS_CONTENTS_FIELD.name))
116    }
117
118    /// Get the JSON contents of the event.
119    ///
120    /// **Read mask:** `json_contents` relative to this event (see
121    /// [`EVENT_JSON_CONTENTS`]).
122    ///
123    /// [`EVENT_JSON_CONTENTS`]: crate::read_masks::EVENT_JSON_CONTENTS
124    pub fn json_contents(&self) -> Result<serde_json::Value, TryFromProtoError> {
125        self.json_contents
126            .as_ref()
127            .map(crate::proto::prost_to_json)
128            .ok_or_else(|| TryFromProtoError::missing(Self::JSON_CONTENTS_FIELD.name))
129    }
130}
