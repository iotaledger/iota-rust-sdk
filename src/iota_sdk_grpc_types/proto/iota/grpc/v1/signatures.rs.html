24impl TryFrom<&ValidatorAggregatedSignature> for iota_types::ValidatorAggregatedSignature {
25    type Error = TryFromProtoError;
26
27    fn try_from(value: &ValidatorAggregatedSignature) -> Result<Self, Self::Error> {
28        let bcs = value.bcs.as_ref().ok_or_else(|| {
29            TryFromProtoError::missing(ValidatorAggregatedSignature::BCS_FIELD.name)
30        })?;
31        bcs.deserialize::<VersionedValidatorAggregatedSignature>()
32            .map_err(|e| TryFromProtoError::invalid(ValidatorAggregatedSignature::BCS_FIELD, e))?
33            .try_into_v1()
34            .map_err(|_| {
35                TryFromProtoError::invalid(
36                    ValidatorAggregatedSignature::BCS_FIELD,
37                    "unsupported ValidatorAggregatedSignature version",
38                )
39            })
40    }
41}

54impl TryFrom<iota_types::UserSignature> for UserSignature {
55    type Error = GrpcConversionError;
56
57    fn try_from(value: iota_types::UserSignature) -> Result<Self, Self::Error> {
58        Ok(Self {
59            bcs: Some(BcsData::serialize(&value).map_err(|e| {
60                GrpcConversionError::BcsSerializationFailed {
61                    message: e.to_string(),
62                }
63            })?),
64        })
65    }
66}

68impl TryFrom<&UserSignature> for iota_types::UserSignature {
69    type Error = TryFromProtoError;
70
71    fn try_from(value: &UserSignature) -> Result<Self, Self::Error> {
72        let bcs = value
73            .bcs
74            .as_ref()
75            .ok_or_else(|| TryFromProtoError::missing(UserSignature::BCS_FIELD.name))?;
76        BcsData::deserialize(bcs)
77            .map_err(|e| TryFromProtoError::invalid(UserSignature::BCS_FIELD, e))
78    }
79}
