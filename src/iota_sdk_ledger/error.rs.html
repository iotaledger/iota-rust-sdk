10/// Error returned when communicating with a Ledger device fails.
11#[derive(Debug, thiserror::Error)]
12#[non_exhaustive]
13pub enum LedgerError {
14    /// The device answered a command with an error.
15    #[error(transparent)]
16    Apdu(#[from] ApduError),
17    /// The IOTA app is not installed on the device.
18    #[error("IOTA app not found; install the IOTA app on the Ledger device")]
19    AppNotFound,
20    /// The IOTA app cannot clear-sign the transaction, and blind signing is disabled in its
21    /// settings.
22    #[error(
23        "transaction cannot be clear-signed and blind signing is disabled in the IOTA app's settings"
24    )]
25    BlindSigningDisabled,
26    /// The IOTA app is older than 1.0, which introduced clear signing.
27    #[error("IOTA app {0} does not support clear signing; update the app")]
28    ClearSigningUnsupported(AppVersion),
29    /// The IOTA app only allows paths under `m/44'/4218'` and `m/44'/1'`.
30    #[error("the IOTA app does not allow this derivation path")]
31    DerivationPathNotAllowed,
32    /// No Ledger device is connected.
33    #[error("Ledger device not found; connect the device")]
34    DeviceNotFound,
35    /// The device's response could not be decoded.
36    #[error("invalid response from the device: {0}")]
37    InvalidResponse(String),
38    /// The device returns a different key for the signer's derivation path than when the signer was
39    /// created, because another device is connected or the device was unlocked with a different
40    /// seed or passphrase.
41    #[error(
42        "the device returned a different key for the signer's derivation path; use the same device, seed and passphrase as when the signer was created"
43    )]
44    KeyMismatch,
45    /// The objects passed for clear signing lack an owned or immutable input or a gas coin of the
46    /// transaction at the version it references.
47    #[error(
48        "object {0} is missing from the objects passed for clear signing, or not at the version the transaction references"
49    )]
50    MissingObject(ObjectId),
51    /// A command payload exceeds what the protocol can encode.
52    #[error("payload is too long for the device")]
53    PayloadTooLong,
54    /// A transaction, message or object could not be serialized.
55    #[error(transparent)]
56    Serialization(BcsError),
57    /// The transport failed to exchange a command with the device.
58    #[error(transparent)]
59    Transport(Box<dyn std::error::Error + Send + Sync>),
60}
61
62impl LedgerError {
63    /// Create a transport error from a message or from an underlying error.
64    pub fn transport(source: impl Into<Box<dyn std::error::Error + Send + Sync>>) -> Self {
65        Self::Transport(source.into())
66    }
67
68    pub(crate) fn invalid_response(message: impl Into<String>) -> Self {
69        Self::InvalidResponse(message.into())
70    }
71}

73/// Error the device answered a command with, as reported by the APDU status word.
74#[derive(Clone, Copy, Debug, Eq, PartialEq, thiserror::Error)]
75#[non_exhaustive]
76pub enum ApduError {
77    /// The device is locked.
78    #[error("device locked; unlock the Ledger device")]
79    DeviceLocked,
80    /// The device runs another app than the IOTA app.
81    #[error("device not ready; open the IOTA app on the Ledger device")]
82    DeviceNotReady,
83    /// The app on the device panicked.
84    #[error("device panic")]
85    DevicePanic,
86    /// A status word without a dedicated variant.
87    #[error("device returned status word {0:#06x}")]
88    Other(u16),
89    /// The user rejected the request on the device.
90    #[error("the user refused the operation on the device")]
91    UserRefused,
92}
93
94impl ApduError {
95    /// Map an APDU status word to an error, or `None` for success.
96    pub(crate) fn from_status_word(sw: u16) -> Option<Self> {
97        Some(match sw {
98            0x9000 => return None,
99            0x5515 => Self::DeviceLocked,
100            0xe000 => Self::DevicePanic,
101            // User cancelled, user denied
102            0x6985 | 0x5501 => Self::UserRefused,
103            // Bad class, instruction or parameters: another app is running
104            0x6e00..=0x6e02 => Self::DeviceNotReady,
105            sw => Self::Other(sw),
106        })
107    }
108}
