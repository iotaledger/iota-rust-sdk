4//! Commands handled by the device's operating system (BOLOS) rather than by the IOTA app.
5
6use super::{Reader, command, exec};
7use crate::{ApduError, LedgerError, Transport};
8
9/// Instruction class of the commands the Ledger SDK answers inside every app and the dashboard:
10/// reading the running app's name and exiting it.
11const CLA_APP: u8 = 0xb0;
12/// Instruction class of the dashboard's commands, such as opening an app.
13const CLA_DASHBOARD: u8 = 0xe0;
14
15const INS_GET_APP_NAME: u8 = 0x01;
16const INS_EXIT_APP: u8 = 0xa7;
17const INS_OPEN_APP: u8 = 0xd8;
18
19/// Status word for an invalid counter syscall error, returned when the app to open is not
20/// installed.
21const SW_INVALID_COUNTER: u16 = 0x6807;
22
23/// Read the name of the app running on the device; the dashboard is named `BOLOS`.
24pub(crate) async fn app_name(transport: &dyn Transport) -> Result<String, LedgerError> {
25    // Dashboard:
26    // HID => b001000000
27    // HID <= 0105|424f4c4f53|05|322e302e30|9000
28    // B O L O S      2 . 0 . 0
29    //
30    // "IOTA"
31    // HID => b001000000
32    // HID <= 0104|494f5441|05|302e372e30|0102|9000
33    // I O T A      0 . 7 . 0
34    let response = exec(transport, command(CLA_APP, INS_GET_APP_NAME, Vec::new())).await?;
35
36    let mut reader = Reader::new(&response);
37    // The format byte is always 0x01; the app version and flags follow the name
38    reader.u8()?;
39    String::from_utf8(reader.length_prefixed()?.to_vec())
40        .map_err(|_| LedgerError::invalid_response("app name is not UTF-8"))
41}
42
43/// Open `app` from the dashboard.
44pub(crate) async fn open_app(transport: &dyn Transport, app: &str) -> Result<(), LedgerError> {
45    // The APDU length byte carries the name's length, so the data holds the bare name
46    match exec(
47        transport,
48        command(CLA_DASHBOARD, INS_OPEN_APP, app.as_bytes().to_vec()),
49    )
50    .await
51    {
52        Ok(_) => Ok(()),
53        Err(LedgerError::Apdu(ApduError::Other(SW_INVALID_COUNTER))) => {
54            Err(LedgerError::AppNotFound)
55        }
56        Err(e) => Err(e),
57    }
58}
59
60/// Close the running app and return to the dashboard.
61pub(crate) async fn exit_app(transport: &dyn Transport) -> Result<(), LedgerError> {
62    exec(transport, command(CLA_APP, INS_EXIT_APP, Vec::new()))
63        .await
64        .map(drop)
65}
