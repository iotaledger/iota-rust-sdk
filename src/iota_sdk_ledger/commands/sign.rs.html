22/// Sign a transaction, passing its input objects for clear signing when there are any.
23pub(crate) async fn transaction(
24    transport: &dyn Transport,
25    path: &DerivationPath,
26    transaction: &Transaction,
27    objects: &[Object],
28) -> Result<Ed25519Signature, LedgerError> {
29    let objects = if objects.is_empty() {
30        None
31    } else {
32        let objects = referenced_objects(transaction, objects)?;
33        let count = u32::try_from(objects.len()).map_err(|_| LedgerError::PayloadTooLong)?;
34        let mut packed = count.to_le_bytes().to_vec();
35        for object in objects {
36            write_u32_prefixed(&mut packed, &bcs_to_bytes(object)?)?;
37        }
38        Some(packed)
39    };
40    intent_message(
41        transport,
42        path,
43        &bcs_to_bytes(&transaction.intent_message())?,
44        objects.as_deref(),
45    )
46    .await
47}
48
49/// The objects among `objects` that `transaction` references as an owned or immutable input or a
50/// gas coin, in the order it references them, as the device looks them up by digest. Fails if any
51/// of them is missing or at another version.
52fn referenced_objects<'a>(
53    transaction: &Transaction,
54    objects: &'a [Object],
55) -> Result<Vec<&'a Object>, LedgerError> {
56    let transaction = match transaction {
57        Transaction::V1(transaction) => transaction,
58        _ => unimplemented!("a new Transaction enum variant was added and needs to be handled"),
59    };
60    let inputs = match &transaction.kind {
61        TransactionKind::Programmable(programmable) => programmable.inputs.as_slice(),
62        _ => &[],
63    };
64    let by_digest = objects
65        .iter()
66        .map(|object| (object.digest(), object))
67        .collect::<HashMap<_, _>>();
68    inputs
69        .iter()
70        .filter_map(|input| match input {
71            Input::ImmutableOrOwned(reference) => Some(reference),
72            _ => None,
73        })
74        .chain(&transaction.gas_payment.objects)
75        .map(|reference| {
76            by_digest
77                .get(reference.digest())
78                .copied()
79                .ok_or(LedgerError::MissingObject(*reference.object_id()))
80        })
81        .collect()
82}

84/// Sign a personal message.
85pub(crate) async fn personal_message(
86    transport: &dyn Transport,
87    path: &DerivationPath,
88    message: &PersonalMessage<'_>,
89) -> Result<Ed25519Signature, LedgerError> {
90    intent(transport, path, Intent::personal_message(), &message.0).await
91}

93/// Sign `message` under `intent`.
94pub(crate) async fn intent<T: Serialize + ?Sized>(
95    transport: &dyn Transport,
96    path: &DerivationPath,
97    intent: Intent,
98    message: &T,
99) -> Result<Ed25519Signature, LedgerError> {
100    let message = IntentMessage::new(intent, message);
101    intent_message(transport, path, &bcs_to_bytes(&message)?, None).await
102}
103
104/// Sign a BCS-serialized intent message; the device hashes it itself.
105async fn intent_message(
106    transport: &dyn Transport,
107    path: &DerivationPath,
108    intent_message: &[u8],
109    packed_objects: Option<&[u8]>,
110) -> Result<Ed25519Signature, LedgerError> {
111    let mut message = Vec::with_capacity(4 + intent_message.len());
112    write_u32_prefixed(&mut message, intent_message)?;
113    let path = encode_path(path)?;
114
115    let mut parameters: Vec<&[u8]> = vec![&message, &path];
116    parameters.extend(packed_objects);
117    let response = block_protocol::send(transport, Instruction::Sign, &parameters)
118        .await
119        .map_err(app::map_app_error)?;
120
121    Ok(Ed25519Signature::new(Reader::new(&response).array()?))
122}
123
124fn bcs_to_bytes<T: Serialize + ?Sized>(value: &T) -> Result<Vec<u8>, LedgerError> {
125    bcs::to_bytes(value).map_err(|e| LedgerError::Serialization(BcsError::new(e)))
126}
127
128/// Append `bytes` prefixed with their length as a little-endian `u32`.
129fn write_u32_prefixed(buf: &mut Vec<u8>, bytes: &[u8]) -> Result<(), LedgerError> {
130    let len = u32::try_from(bytes.len()).map_err(|_| LedgerError::PayloadTooLong)?;
131    buf.extend_from_slice(&len.to_le_bytes());
132    buf.extend_from_slice(bytes);
133    Ok(())
134}

136#[cfg(test)]
137mod tests {
138    use iota_types::{
139        Address, GasPayment, MoveStruct, ObjectData, ObjectId, Owner, ProgrammableTransaction,
140        StructTag, TransactionDigest, TransactionExpiration, TransactionV1, Version,
141    };
142
143    use super::*;
144
145    /// An IOTA coin with every byte of its id set to `id`.
146    fn coin(id: u8, version: u64) -> Object {
147        let mut contents = [id; ObjectId::LENGTH].to_vec();
148        contents.extend_from_slice(&1_000u64.to_le_bytes());
149        let coin = MoveStruct::new(
150            StructTag::new_gas_coin().into(),
151            Version::from_u64(version),
152            contents,
153        )
154        .unwrap();
155        Object::new(
156            ObjectData::Struct(coin),
157            Owner::Address(Address::ZERO),
158            TransactionDigest::default(),
159            0,
160        )
161    }
162
163    /// A transaction that takes `input` and pays for gas with `gas`.
164    fn transaction(input: &Object, gas: &Object) -> Transaction {
165        Transaction::V1(TransactionV1 {
166            kind: TransactionKind::Programmable(ProgrammableTransaction {
167                inputs: vec![
168                    Input::ImmutableOrOwned(input.object_ref()),
169                    Input::Pure(vec![]),
170                ],
171                commands: vec![],
172            }),
173            sender: Address::ZERO,
174            gas_payment: GasPayment {
175                objects: vec![gas.object_ref()],
176                owner: Address::ZERO,
177                price: 1_000,
178                budget: 1_000_000,
179            },
180            expiration: TransactionExpiration::None,
181        })
182    }
183
184    #[test]
185    fn keeps_only_the_referenced_objects() {
186        let (input, gas) = (coin(1, 1), coin(2, 1));
187        let objects = [coin(3, 1), gas.clone(), input.clone()];
188        assert_eq!(
189            referenced_objects(&transaction(&input, &gas), &objects).unwrap(),
190            [&input, &gas]
191        );
192    }
193
194    #[test]
195    fn rejects_a_missing_gas_coin() {
196        let (input, gas) = (coin(1, 1), coin(2, 1));
197        assert!(matches!(
198            referenced_objects(&transaction(&input, &gas), &[input]),
199            Err(LedgerError::MissingObject(id)) if id == gas.id()
200        ));
201    }
202
203    #[test]
204    fn rejects_an_input_at_another_version() {
205        let (input, gas) = (coin(1, 1), coin(2, 1));
206        assert!(matches!(
207            referenced_objects(&transaction(&input, &gas), &[coin(1, 2), gas]),
208            Err(LedgerError::MissingObject(id)) if id == input.id()
209        ));
210    }
211}
