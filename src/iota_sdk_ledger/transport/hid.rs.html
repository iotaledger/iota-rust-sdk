15impl Ledger {
16    /// List the Ledger devices attached over USB.
17    pub fn hid_devices() -> Result<Vec<HidDeviceInfo>, LedgerError> {
18        let api = HidApi::new().map_err(LedgerError::transport)?;
19        Ok(TransportNativeHID::list_ledgers(&api)
20            .map(HidDeviceInfo::from)
21            .collect())
22    }
23
24    /// Connect to the first Ledger device attached over USB.
25    pub fn new_native_hid() -> Result<Self, LedgerError> {
26        HidConnection::open_first()
27            .and_then(HidTransport::spawn)
28            .map(Self::new)
29    }
30
31    /// Connect to a Ledger device from [`Ledger::hid_devices`].
32    ///
33    /// The device's path usually changes when it switches apps, so with several Ledger devices
34    /// attached, [`Ledger::open_app`] and [`Ledger::exit_app`] cannot find it again and fail with
35    /// [`LedgerError::DeviceNotFound`].
36    pub fn new_native_hid_device(device: &HidDeviceInfo) -> Result<Self, LedgerError> {
37        HidConnection::open(device)
38            .and_then(HidTransport::spawn)
39            .map(Self::new)
40    }
41}

43/// A Ledger device connected over USB.
44#[derive(Clone, Debug, Eq, PartialEq)]
45pub struct HidDeviceInfo {
46    path: CString,
47    product: Option<String>,
48}

50impl HidDeviceInfo {
51    /// The operating system's path to the device. It usually changes when the device switches apps.
52    pub fn path(&self) -> String {
53        self.path.to_string_lossy().into_owned()
54    }
55
56    /// The product name the device reports, such as `Nano S Plus`.
57    pub fn product(&self) -> Option<&str> {
58        self.product.as_deref()
59    }
60}

62impl From<&DeviceInfo> for HidDeviceInfo {

64        Self {
65            path: info.path().to_owned(),
66            product: info.product_string().map(str::to_owned),
67        }
68    }

71/// Talks to a Ledger device connected over USB.
72///
73/// The connection lives on a thread of its own that handles requests in the order they are sent, so
74/// waiting for the device does not block the thread polling the call.
75struct HidTransport {
76    requests: mpsc::Sender<Request>,
77}
78
79/// A request to the thread holding the connection, answered through the enclosed sender.
80enum Request {
81    Exchange(
82        ApduCommand,
83        oneshot::Sender<Result<ApduAnswer, LedgerError>>,
84    ),
85    Reconnect(oneshot::Sender<Result<(), LedgerError>>),
86}
87
88impl HidTransport {
89    /// Move `connection` to a new thread, which ends once the transport is dropped and the requests
90    /// already sent are answered.
91    fn spawn(mut connection: HidConnection) -> Result<Self, LedgerError> {
92        let (requests, receiver) = mpsc::channel();
93        thread::Builder::new()
94            .name("ledger-hid".to_owned())
95            .spawn(move || {
96                for request in receiver {
97                    // A caller that dropped its call no longer waits for the answer.
98                    match request {
99                        Request::Exchange(command, answer) => {
100                            let _ = answer.send(connection.exchange(&command));
101                        }
102                        Request::Reconnect(answer) => {
103                            let _ = answer.send(connection.reconnect());
104                        }
105                    }
106                }
107            })
108            .map_err(LedgerError::transport)?;
109        Ok(Self { requests })
110    }
111
112    /// Send a request to the connection's thread and wait for its answer.
113    async fn request<T>(
114        &self,
115        request: impl FnOnce(oneshot::Sender<Result<T, LedgerError>>) -> Request,
116    ) -> Result<T, LedgerError> {
117        let (sender, receiver) = oneshot::channel();
118        // If the thread has stopped, the request is dropped with its sender and the receiver
119        // reports that it was canceled.
120        let _ = self.requests.send(request(sender));
121        receiver.await.map_err(LedgerError::transport)?
122    }
123}
124
125impl Transport for HidTransport {
126    fn exchange<'a>(&'a self, command: &'a ApduCommand) -> TransportFuture<'a, ApduAnswer> {
127        Box::pin(self.request(|answer| Request::Exchange(command.clone(), answer)))
128    }
129
130    fn reconnect(&mut self) -> TransportFuture<'_, ()> {
131        Box::pin(self.request(Request::Reconnect))
132    }
133}

135/// A blocking connection to a Ledger device connected over USB.
136struct HidConnection {
137    /// `None` while reconnecting: the previous handle is closed before a new one is opened, since
138    /// two open handles to the same device can lock it.
139    device: Option<TransportNativeHID>,
140    path: CString,
141}
142
143impl HidConnection {
144    /// Open the first Ledger device found.
145    fn open_first() -> Result<Self, LedgerError> {
146        let api = HidApi::new().map_err(LedgerError::transport)?;
147        let info = TransportNativeHID::list_ledgers(&api)
148            .next()
149            .ok_or(LedgerError::DeviceNotFound)?;
150        Self::open_info(&api, info)
151    }
152
153    /// Open `device`, which must still be connected at the same path.
154    fn open(device: &HidDeviceInfo) -> Result<Self, LedgerError> {
155        let api = HidApi::new().map_err(LedgerError::transport)?;
156        let info = TransportNativeHID::list_ledgers(&api)
157            .find(|info| info.path() == device.path.as_c_str())
158            .ok_or(LedgerError::DeviceNotFound)?;
159        Self::open_info(&api, info)
160    }
161
162    fn open_info(api: &HidApi, info: &DeviceInfo) -> Result<Self, LedgerError> {
163        let device = TransportNativeHID::open_device(api, info).map_err(|e| match e {
164            LedgerHIDError::DeviceNotFound => LedgerError::DeviceNotFound,
165            e => LedgerError::transport(e),
166        })?;
167        Ok(Self {
168            device: Some(device),
169            path: info.path().to_owned(),
170        })
171    }
172
173    /// Exchange the command, blocking until the device answers.
174    fn exchange(&self, command: &ApduCommand) -> Result<ApduAnswer, LedgerError> {
175        let device = self.device.as_ref().ok_or(LedgerError::DeviceNotFound)?;
176        let answer = device
177            .exchange(&ledger_transport::APDUCommand {
178                cla: command.cla,
179                ins: command.ins,
180                p1: command.p1,
181                p2: command.p2,
182                data: command.data.as_slice(),
183            })
184            .map_err(LedgerError::transport)?;
185        Ok(ApduAnswer {
186            data: answer.data().to_vec(),
187            status_word: answer.retcode(),
188        })
189    }
190
191    /// Reopen the device at its previous path, or, as the path usually changes when the device
192    /// switches apps, the only Ledger device connected. Ledger devices share a serial number, so
193    /// with several connected the device cannot be told apart and this fails.
194    fn reconnect(&mut self) -> Result<(), LedgerError> {
195        self.device = None;
196        let api = HidApi::new().map_err(LedgerError::transport)?;
197        let ledgers = TransportNativeHID::list_ledgers(&api).collect::<Vec<_>>();
198        let info = match ledgers
199            .iter()
200            .find(|info| info.path() == self.path.as_c_str())
201        {
202            Some(info) => info,
203            None => match ledgers.as_slice() {
204                [only] => only,
205                _ => return Err(LedgerError::DeviceNotFound),
206            },
207        };
208        *self = Self::open_info(&api, info)?;
209        Ok(())
210    }
211}
