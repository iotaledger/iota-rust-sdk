11/// An APDU command sent to the device.
12#[derive(Clone, Debug, Eq, PartialEq)]
13pub struct ApduCommand {
14    pub(crate) cla: u8,
15    pub(crate) ins: u8,
16    pub(crate) p1: u8,
17    pub(crate) p2: u8,
18    pub(crate) data: Vec<u8>,
19}
20
21impl ApduCommand {
22    /// The instruction class.
23    pub fn cla(&self) -> u8 {
24        self.cla
25    }
26
27    /// The instruction code.
28    pub fn ins(&self) -> u8 {
29        self.ins
30    }
31
32    /// The first instruction parameter.
33    pub fn p1(&self) -> u8 {
34        self.p1
35    }
36
37    /// The second instruction parameter.
38    pub fn p2(&self) -> u8 {
39        self.p2
40    }
41
42    /// The command payload.
43    pub fn data(&self) -> &[u8] {
44        &self.data
45    }
46
47    /// The command as sent over the wire: `cla ins p1 p2 len data`.
48    pub fn to_bytes(&self) -> Vec<u8> {
49        let mut bytes = Vec::with_capacity(5 + self.data.len());
50        bytes.extend([self.cla, self.ins, self.p1, self.p2, self.data.len() as u8]);
51        bytes.extend_from_slice(&self.data);
52        bytes
53    }
54}

56/// An APDU response received from the device.
57#[derive(Clone, Debug, Eq, PartialEq)]
58pub struct ApduAnswer {
59    pub(crate) data: Vec<u8>,
60    pub(crate) status_word: u16,
61}
62
63impl ApduAnswer {
64    /// Parse a response as received over the wire: the payload followed by a big-endian two-byte
65    /// status word.
66    pub fn from_bytes(mut bytes: Vec<u8>) -> Result<Self, LedgerError> {
67        let Some(split) = bytes.len().checked_sub(2) else {
68            return Err(LedgerError::transport("APDU response is too short"));
69        };
70        let status_word = u16::from_be_bytes([bytes[split], bytes[split + 1]]);
71        bytes.truncate(split);
72        Ok(Self {
73            data: bytes,
74            status_word,
75        })
76    }
77
78    /// The response payload.
79    pub fn data(&self) -> &[u8] {
80        &self.data
81    }
82
83    /// The status word; `0x9000` means success.
84    pub fn status_word(&self) -> u16 {
85        self.status_word
86    }
87}

109/// A link to a Ledger device that exchanges APDUs.
110///
111/// The crate provides transports for USB HID (behind the `hid` feature) and the Speculos emulator
112/// (behind the `speculos` feature); implement this trait to talk to the device over another link.
113pub trait Transport: MaybeSendSync {
114    /// Send a command and wait for the device's answer.
115    fn exchange<'a>(&'a self, command: &'a ApduCommand) -> TransportFuture<'a, ApduAnswer>;
116
117    /// Re-establish the link after the device switched apps, which closes the previous connection
118    /// on some transports.
119    ///
120    /// While waiting for an app to start, [`Ledger`](crate::Ledger) calls this at intervals until
121    /// the app answers.
122    fn reconnect(&mut self) -> TransportFuture<'_, ()> {
123        Box::pin(async { Ok(()) })
124    }
125
126    /// Whether the device behind this transport can open and close apps.
127    ///
128    /// Return `false` for emulators such as Speculos, which run a single app and report a
129    /// placeholder app name unless configured otherwise.
130    fn can_switch_apps(&self) -> bool {
131        true
132    }
133}
