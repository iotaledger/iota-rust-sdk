4use std::time::{SystemTime, UNIX_EPOCH};
5
6use serde::{Deserialize, Serialize};
7use sqlx::{PgPool, Row};
8
9#[derive(Clone, Debug, Deserialize, Serialize)]
10pub struct ProgressState {
11    pub next_checkpoint: u64,
12    pub updated_at_ms: u128,
13}
14
15impl Default for ProgressState {
16    fn default() -> Self {
17        Self {
18            next_checkpoint: 0,
19            updated_at_ms: now_ms(),
20        }
21    }
22}
23
24pub async fn load(pool: &PgPool, progress_key: &str) -> anyhow::Result<ProgressState> {
25    let row = sqlx::query(
26        r#"
27        SELECT next_checkpoint, updated_at_ms
28        FROM indexer_progress
29        WHERE progress_key = $1
30        "#,
31    )
32    .bind(progress_key)
33    .fetch_optional(pool)
34    .await?;
35
36    let Some(row) = row else {
37        return Ok(ProgressState::default());
38    };
39
40    let next_checkpoint_i64: i64 = row.try_get("next_checkpoint")?;
41    let updated_at_ms_i64: i64 = row.try_get("updated_at_ms")?;
42    Ok(ProgressState {
43        next_checkpoint: u64::try_from(next_checkpoint_i64)?,
44        updated_at_ms: u128::try_from(updated_at_ms_i64)?,
45    })
46}
47
48pub async fn store(pool: &PgPool, progress_key: &str, next_checkpoint: u64) -> anyhow::Result<()> {
49    sqlx::query(
50        r#"
51        INSERT INTO indexer_progress (progress_key, next_checkpoint, updated_at_ms)
52        VALUES ($1, $2, $3)
53        ON CONFLICT(progress_key)
54        DO UPDATE SET
55            next_checkpoint = excluded.next_checkpoint,
56            updated_at_ms = excluded.updated_at_ms
57        "#,
58    )
59    .bind(progress_key)
60    .bind(i64::try_from(next_checkpoint)?)
61    .bind(i64::try_from(now_ms())?)
62    .execute(pool)
63    .await?;
64
65    Ok(())
66}
67
68fn now_ms() -> u128 {
69    SystemTime::now()
70        .duration_since(UNIX_EPOCH)
71        .unwrap_or_default()
72        .as_millis()
73}
