41    pub fn verify_member_signature(
42        &self,
43        message: &[u8],
44        member_public_key: &PublicKey,
45        signature: &MultisigMemberSignature,
46    ) -> Result<(), SignatureError> {
47        match (member_public_key, signature) {
48            #[cfg(not(feature = "ed25519"))]
49            (PublicKey::Ed25519(_), MultisigMemberSignature::Ed25519(_)) => Err(
50                SignatureError::from_source("support for ed25519 is not enabled"),
51            ),
52            #[cfg(feature = "ed25519")]
53            (
54                PublicKey::Ed25519(ed25519_public_key),
55                MultisigMemberSignature::Ed25519(ed25519_signature),
56            ) => crate::ed25519::Ed25519VerifyingKey::new(ed25519_public_key)?
57                .verify(message, ed25519_signature),
58            #[cfg(not(feature = "secp256k1"))]
59            (PublicKey::Secp256k1(_), MultisigMemberSignature::Secp256k1(_)) => Err(
60                SignatureError::from_source("support for secp256k1 is not enabled"),
61            ),
62            #[cfg(feature = "secp256k1")]
63            (
64                PublicKey::Secp256k1(k1_public_key),
65                MultisigMemberSignature::Secp256k1(k1_signature),
66            ) => crate::secp256k1::Secp256k1VerifyingKey::new(k1_public_key)?
67                .verify(message, k1_signature),
68            #[cfg(not(feature = "secp256r1"))]
69            (PublicKey::Secp256r1(_), MultisigMemberSignature::Secp256r1(_)) => Err(
70                SignatureError::from_source("support for secp256r1 is not enabled"),
71            ),
72            #[cfg(feature = "secp256r1")]
73            (
74                PublicKey::Secp256r1(r1_public_key),
75                MultisigMemberSignature::Secp256r1(r1_signature),
76            ) => crate::secp256r1::Secp256r1VerifyingKey::new(r1_public_key)?
77                .verify(message, r1_signature),
78            #[cfg(not(feature = "passkey"))]
79            (PublicKey::Passkey(_), MultisigMemberSignature::Passkey(_)) => Err(
80                SignatureError::from_source("support for passkey is not enabled"),
81            ),
82            #[cfg(feature = "passkey")]
83            (
84                PublicKey::Passkey(passkey_public_key),
85                MultisigMemberSignature::Passkey(passkey_authenticator),
86            ) => {
87                let passkey_verifier = self.passkey_verifier().cloned().unwrap_or_default();
88
89                // verify that the member public key and the authenticator's key match
90                if passkey_public_key != &passkey_authenticator.public_key() {
91                    return Err(SignatureError::from_source(
92                        "member passkey public key does not match signature",
93                    ));
94                }
95
96                passkey_verifier.verify(message, passkey_authenticator)
97            }
98            _ => Err(SignatureError::from_source(
99                "member and signature scheme do not match",
100            )),
101        }
102    }
103}

121impl Verifier<MultisigAggregatedSignature> for MultisigVerifier {
122    fn verify(
123        &self,
124        message: &[u8],
125        signature: &MultisigAggregatedSignature,
126    ) -> Result<(), SignatureError> {
127        signature
128            .validate()
129            .map_err(|e| SignatureError::from_source(format!("invalid multisig: {e}")))?;
130
131        if let Some(address) = &self.address
132            && signature.committee().derive_address() != *address
133        {
134            return Err(SignatureError::from_source(
135                "Invalid address derived from pks",
136            ));
137        }
138
139        if !self.accept_passkey_in_multisig
140            && signature.contains_signature_scheme(SignatureScheme::PasskeyAuthenticator)
141        {
142            return Err(SignatureError::from_source(
143                "Passkey sig not supported inside multisig",
144            ));
145        }
146
147        let indices = signature
148            .indices()
149            .map_err(|e| SignatureError::from_source(format!("invalid multisig: {e}")))?;
150
151        let mut weight: ThresholdUnit = 0;
152        for (member_idx, member_signature) in indices.into_iter().zip(signature.signatures()) {
153            let member = signature
154                .committee()
155                .members()
156                .get(member_idx as usize)
157                .ok_or_else(|| SignatureError::from_source("Invalid public keys index"))?;
158
159            if self.additional_multisig_checks
160                && member.public_key().scheme() != member_signature.scheme()
161            {
162                return Err(SignatureError::from_source(format!(
163                    "Invalid sig for pk={} address={:?} error=signature/pubkey type mismatch",
164                    member.public_key().to_base64(),
165                    member.public_key().derive_address(),
166                )));
167            }
168
169            self.verify_member_signature(message, member.public_key(), member_signature)
170                .map_err(|e| {
171                    SignatureError::from_source(format!(
172                        "Invalid sig for pk={} address={:?} error={e}",
173                        member.public_key().to_base64(),
174                        member.public_key().derive_address(),
175                    ))
176                })?;
177
178            weight = weight
179                .checked_add(member.weight() as ThresholdUnit)
180                .ok_or(SignatureError::from_source("Weight overflow"))?;
181        }
182
183        if weight >= signature.committee().threshold() {
184            Ok(())
185        } else {
186            Err(SignatureError::from_source(format!(
187                "Insufficient weight={weight:?} threshold={:?}",
188                signature.committee().threshold()
189            )))
190        }
191    }
192}

194impl Verifier<UserSignature> for MultisigVerifier {
195    fn verify(&self, message: &[u8], signature: &UserSignature) -> Result<(), SignatureError> {
196        let UserSignature::Multisig(signature) = signature else {
197            return Err(SignatureError::from_source("not a multisig signature"));
198        };
199
200        self.verify(message, signature)
201    }

234impl Verifier<UserSignature> for UserSignatureVerifier {
235    fn verify(&self, message: &[u8], signature: &UserSignature) -> Result<(), SignatureError> {
236        match signature {
237            UserSignature::Simple(signature) => {
238                crate::simple::SimpleVerifier.verify(message, signature)
239            }
240            UserSignature::Multisig(signature) => self.inner.verify(message, signature),
241            #[cfg(not(feature = "passkey"))]
242            UserSignature::PasskeyAuthenticator(_) => Err(SignatureError::from_source(
243                "support for passkey is not enabled",
244            )),
245            #[cfg(feature = "passkey")]
246            UserSignature::PasskeyAuthenticator(authenticator) => {
247                crate::passkey::PasskeyVerifier::default().verify(message, authenticator)
248            }
249            UserSignature::MoveAuthenticator(_) => Err(SignatureError::from_source(
250                "move authenticators cannot be verified",
251            )),
252            _ => Err(SignatureError::from_source("unknown signature scheme")),
253        }
254    }
255}

259#[derive(Clone, Debug, PartialEq)]
260pub struct MultisigAggregator {
261    committee: MultisigCommittee,
262    signatures: std::collections::BTreeMap<usize, MultisigMemberSignature>,
263    signed_weight: u16,
264    message: Vec<u8>,
265    verifier: MultisigVerifier,
266}
267
268impl MultisigAggregator {
269    pub fn new_with_transaction(
270        committee: MultisigCommittee,
271        transaction: &iota_types::Transaction,
272    ) -> Self {
273        Self {
274            committee,
275            signatures: Default::default(),
276            signed_weight: 0,
277            message: transaction.signing_digest().to_vec(),
278            verifier: Default::default(),
279        }
280    }
281
282    pub fn new_with_message(
283        committee: MultisigCommittee,
284        message: &iota_types::PersonalMessage<'_>,
285    ) -> Self {
286        Self {
287            committee,
288            signatures: Default::default(),
289            signed_weight: 0,
290            message: message.signing_digest().to_vec(),
291            verifier: Default::default(),
292        }
293    }
294
295    pub fn verifier(&self) -> &MultisigVerifier {
296        &self.verifier
297    }
298
299    pub fn verifier_mut(&mut self) -> &mut MultisigVerifier {
300        &mut self.verifier
301    }
302
303    pub fn add_signature(&mut self, signature: UserSignature) -> Result<(), SignatureError> {
304        use std::collections::btree_map::Entry;
305
306        let (public_key, signature) = multisig_pubkey_and_signature_from_user_signature(signature)?;
307        let member_idx = self
308            .committee
309            .members()
310            .iter()
311            .position(|member| member.public_key() == &public_key)
312            .ok_or_else(|| {
313                SignatureError::from_source(
314                    "provided signature does not belong to committee member",
315                )
316            })?;
317
318        self.verifier()
319            .verify_member_signature(&self.message, &public_key, &signature)?;
320
321        match self.signatures.entry(member_idx) {
322            Entry::Vacant(v) => {
323                v.insert(signature);
324            }
325            Entry::Occupied(_) => {
326                return Err(SignatureError::from_source(
327                    "duplicate signature from same committee member",
328                ));
329            }
330        }
331
332        self.signed_weight = self
333            .signed_weight
334            .checked_add(self.committee.members()[member_idx].weight() as ThresholdUnit)
335            .ok_or_else(|| SignatureError::from_source("signed weight overflow"))?;
336
337        Ok(())
338    }
339
340    pub fn finish(&self) -> Result<MultisigAggregatedSignature, SignatureError> {
341        if self.signed_weight < self.committee.threshold() {
342            return Err(SignatureError::from_source(
343                "insufficient signature weight to reach threshold",
344            ));
345        }
346
347        let (signatures, bitmap) = self.signatures.clone().into_iter().fold(
348            (Vec::new(), 0),
349            |(mut signatures, mut bitmap), (member_idx, signature)| {
350                bitmap |= 1 << member_idx;
351                signatures.push(signature);
352                (signatures, bitmap)
353            },
354        );
355
356        Ok(MultisigAggregatedSignature::new_unchecked(
357            signatures,
358            bitmap,
359            self.committee.clone(),
360        ))
361    }
362}

364fn multisig_pubkey_and_signature_from_user_signature(
365    signature: UserSignature,
366) -> Result<(PublicKey, MultisigMemberSignature), SignatureError> {
367    use iota_types::SimpleSignature;
368    match signature {
369        UserSignature::Simple(SimpleSignature::Ed25519 {
370            signature,
371            public_key,
372        }) => Ok((public_key.into(), signature.into())),
373        UserSignature::Simple(SimpleSignature::Secp256k1 {
374            signature,
375            public_key,
376        }) => Ok((public_key.into(), signature.into())),
377        UserSignature::Simple(SimpleSignature::Secp256r1 {
378            signature,
379            public_key,
380        }) => Ok((public_key.into(), signature.into())),
381        UserSignature::PasskeyAuthenticator(passkey_authenticator) => Ok((
382            passkey_authenticator.clone().into(),
383            passkey_authenticator.into(),
384        )),
385        UserSignature::Multisig(_) | UserSignature::MoveAuthenticator(_) => {
386            Err(SignatureError::from_source("invalid signature scheme"))
387        }
388        _ => Err(SignatureError::from_source("unknown signature scheme")),
389    }
390}
