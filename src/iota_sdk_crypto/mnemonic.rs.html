4use bip39::Mnemonic;
5
6#[derive(Clone, Copy, Debug)]
7#[non_exhaustive]
8pub enum MnemonicLength {
9    Words12 = 12,
10    Words24 = 24,
11}
12
13/// Generate a new BIP-39 mnemonic in English.
14/// Supported word counts are 12 and 24 (default).
15pub fn generate_mnemonic(word_count: impl Into<Option<MnemonicLength>>) -> String {
16    let count = word_count.into().unwrap_or(MnemonicLength::Words24) as usize;
17    Mnemonic::generate(count)
18        .expect("mnemonic generation failed") // Safe to unwrap since the word count is controlled
19        .to_string()
20}
