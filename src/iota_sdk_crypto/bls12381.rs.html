12const DST_G1: &[u8] = b"BLS_SIG_BLS12381G1_XMD:SHA-256_SSWU_RO_NUL_";

26#[derive(Clone, zeroize::Zeroize, zeroize::ZeroizeOnDrop)]
27pub struct Bls12381PrivateKey(SecretKey);

29impl PartialEq for Bls12381PrivateKey {
30    fn eq(&self, other: &Self) -> bool {
31        zeroize::Zeroizing::new(self.0.to_bytes()) == zeroize::Zeroizing::new(other.0.to_bytes())
32    }
33}

35impl Eq for Bls12381PrivateKey {}

61impl Bls12381PrivateKey {
62    /// The length of an bls12381 private key in bytes.
63    pub const LENGTH: usize = 32;
64
65    pub fn new(bytes: [u8; Self::LENGTH]) -> Result<Self, SignatureError> {
66        SecretKey::from_bytes(&bytes)
67            .map_err(BlstError)
68            .map_err(SignatureError::from_source)
69            .map(Self)
70    }
71
72    pub fn scheme(&self) -> SignatureScheme {
73        SignatureScheme::Bls12381
74    }
75
76    pub fn verifying_key(&self) -> Bls12381VerifyingKey {
77        let verifying_key = self.0.sk_to_pk();
78        Bls12381VerifyingKey(verifying_key)
79    }
80
81    pub fn public_key(&self) -> Bls12381PublicKey {
82        self.verifying_key().public_key()
83    }
84
85    pub fn random_with<R>(mut rng: R) -> Self
86    where
87        R: rand_core::RngCore + rand_core::CryptoRng,
88    {
89        let mut buf: [u8; Self::LENGTH] = [0; Self::LENGTH];
90        rng.fill_bytes(&mut buf);
91        let secret_key = SecretKey::key_gen(&buf, &[]).unwrap();
92        Self(secret_key)
93    }
94
95    /// Generate a new private key using the operating system's random number
96    /// generator.
97    #[cfg(feature = "rand")]
98    #[cfg_attr(doc_cfg, doc(cfg(feature = "rand")))]
99    pub fn random() -> Self {
100        Self::random_with(rand_core::OsRng)
101    }
102
103    pub fn sign_checkpoint_summary(&self, summary: &CheckpointSummary) -> ValidatorSignature {
104        let message = summary.signing_message();
105        let signature = self.sign(&message);
106        ValidatorSignature {
107            epoch: summary.epoch,
108            public_key: self.public_key(),
109            signature,
110        }
111    }
112}
113
114impl Signer<Bls12381Signature> for Bls12381PrivateKey {
115    fn try_sign(&self, msg: &[u8]) -> Result<Bls12381Signature, SignatureError> {
116        let signature = self.0.sign(msg, DST_G1, &[]);
117        Ok(Bls12381Signature::new(signature.to_bytes()))
118    }
119}

121#[derive(Debug)]
122pub struct Bls12381VerifyingKey(pub(crate) PublicKey);
123
124impl Bls12381VerifyingKey {
125    pub fn new(public_key: &Bls12381PublicKey) -> Result<Self, SignatureError> {
126        PublicKey::key_validate(public_key.bytes())
127            .map(Self)
128            .map_err(BlstError)
129            .map_err(SignatureError::from_source)
130    }
131
132    pub fn public_key(&self) -> Bls12381PublicKey {
133        Bls12381PublicKey::new(self.0.to_bytes())
134    }
135}
136
137impl Verifier<Bls12381Signature> for Bls12381VerifyingKey {
138    fn verify(&self, message: &[u8], signature: &Bls12381Signature) -> Result<(), SignatureError> {
139        let signature = Signature::sig_validate(signature.bytes(), true)
140            .map_err(BlstError)
141            .map_err(SignatureError::from_source)?;
142
143        let err = signature.verify(true, message, DST_G1, &[], &self.0, false);
144        if err == blst::BLST_ERROR::BLST_SUCCESS {
145            Ok(())
146        } else {
147            Err(SignatureError::from_source(BlstError(err)))
148        }
149    }
150}
