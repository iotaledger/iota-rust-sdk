62impl Bls12381PrivateKey {
63    /// The length of an bls12381 private key in bytes.
64    pub const LENGTH: usize = 32;
65
66    pub fn new(bytes: [u8; Self::LENGTH]) -> Result<Self, SignatureError> {
67        SecretKey::from_bytes(&bytes)
68            .map_err(BlstError)
69            .map_err(SignatureError::from_source)
70            .map(Self)
71    }
72
73    pub fn scheme(&self) -> SignatureScheme {
74        SignatureScheme::Bls12381
75    }
76
77    pub fn verifying_key(&self) -> Bls12381VerifyingKey {
78        let verifying_key = self.0.sk_to_pk();
79        Bls12381VerifyingKey(verifying_key)
80    }
81
82    pub fn public_key(&self) -> Bls12381PublicKey {
83        self.verifying_key().public_key()
84    }
85
86    pub fn random_with<R>(mut rng: R) -> Self
87    where
88        R: rand_core::RngCore + rand_core::CryptoRng,
89    {
90        let mut buf: [u8; Self::LENGTH] = [0; Self::LENGTH];
91        rng.fill_bytes(&mut buf);
92        let secret_key = SecretKey::key_gen(&buf, &[]).unwrap();
93        Self(secret_key)
94    }
95
96    /// Generate a new private key using the operating system's random number
97    /// generator.
98    #[cfg(feature = "rand")]
99    #[cfg_attr(doc_cfg, doc(cfg(feature = "rand")))]
100    pub fn random() -> Self {
101        Self::random_with(rand_core::OsRng)
102    }
103
104    /// Sign a proof that this key's holder also controls `address`, which a
105    /// validator submits alongside its public key so the network can check
106    /// that the key is not someone else's.
107    pub fn generate_proof_of_possession(&self, address: Address) -> Bls12381Signature {
108        self.sign(&self.public_key().proof_of_possession_message(address))
109    }
110
111    pub fn sign_checkpoint_summary(&self, summary: &CheckpointSummary) -> ValidatorSignature {
112        let message = summary.signing_message();
113        let signature = self.sign(&message);
114        ValidatorSignature {
115            epoch: summary.epoch,
116            public_key: self.public_key(),
117            signature,
118        }
119    }
120}

122impl crate::ToFromBytes for Bls12381PrivateKey {
123    type Error = crate::PrivateKeyError;
124    type ByteArray = [u8; Self::LENGTH];
125
126    /// Return the raw 32-byte private key
127    fn to_bytes(&self) -> Self::ByteArray {
128        self.0.to_bytes()
129    }
130
131    fn from_bytes(bytes: impl AsRef<[u8]>) -> Result<Self, Self::Error> {
132        let bytes = bytes.as_ref();
133        let bytes: [u8; Self::LENGTH] = bytes.try_into().map_err(|_| {
134            crate::PrivateKeyError::InvalidScheme("invalid bls12381 key length".to_string())
135        })?;
136
137        Self::new(bytes).map_err(|e| crate::PrivateKeyError::InvalidScheme(e.to_string()))
138    }
139}

141impl Signer<Bls12381Signature> for Bls12381PrivateKey {
142    fn try_sign(&self, msg: &[u8]) -> Result<Bls12381Signature, SignatureError> {
143        let signature = self.0.sign(msg, DST_G1, &[]);
144        Ok(Bls12381Signature::new(signature.to_bytes()))
145    }
146}

148#[derive(Debug)]
149pub struct Bls12381VerifyingKey(pub(crate) PublicKey);
150
151impl Bls12381VerifyingKey {
152    pub fn new(public_key: &Bls12381PublicKey) -> Result<Self, SignatureError> {
153        PublicKey::key_validate(public_key.bytes())
154            .map(Self)
155            .map_err(BlstError)
156            .map_err(SignatureError::from_source)
157    }
158
159    pub fn public_key(&self) -> Bls12381PublicKey {
160        Bls12381PublicKey::new(self.0.to_bytes())
161    }
162
163    /// Check a proof of possession produced by
164    /// [`Bls12381PrivateKey::generate_proof_of_possession`] for `address`.
165    pub fn verify_proof_of_possession(
166        &self,
167        address: Address,
168        proof: &Bls12381Signature,
169    ) -> Result<(), SignatureError> {
170        self.verify(
171            &self.public_key().proof_of_possession_message(address),
172            proof,
173        )
174    }
175}
176
177impl Verifier<Bls12381Signature> for Bls12381VerifyingKey {
178    fn verify(&self, message: &[u8], signature: &Bls12381Signature) -> Result<(), SignatureError> {
179        let signature = Signature::sig_validate(signature.bytes(), true)
180            .map_err(BlstError)
181            .map_err(SignatureError::from_source)?;
182
183        let err = signature.verify(true, message, DST_G1, &[], &self.0, false);
184        if err == blst::BLST_ERROR::BLST_SUCCESS {
185            Ok(())
186        } else {
187            Err(SignatureError::from_source(BlstError(err)))
188        }
189    }
190}
