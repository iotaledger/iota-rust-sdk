45impl Secp256k1PrivateKey {
46    /// The length of an secp256k1 private key in bytes.
47    pub const LENGTH: usize = 32;
48
49    pub fn new(bytes: [u8; Self::LENGTH]) -> Result<Self, SignatureError> {
50        // Validate that the bytes represent a well-formed secp256k1 private key.
51        Secp256k1KeyPair::from_bytes(&bytes).map_err(SignatureError::from_source)?;
52        Ok(Self(bytes))
53    }
54
55    pub fn scheme(&self) -> SignatureScheme {
56        SignatureScheme::Secp256k1
57    }
58
59    /// Reconstruct the fastcrypto keypair, which performs all signing.
60    fn keypair(&self) -> Secp256k1KeyPair {
61        Secp256k1KeyPair::from_bytes(&self.0).expect("validated on construction")
62    }
63
64    pub fn verifying_key(&self) -> Secp256k1VerifyingKey {
65        Secp256k1VerifyingKey(self.keypair().public().clone())
66    }
67
68    pub fn public_key(&self) -> Secp256k1PublicKey {
69        Secp256k1PublicKey::new(
70            self.keypair()
71                .public()
72                .as_ref()
73                .try_into()
74                .expect("secp256k1 public key must be 33 bytes"),
75        )
76    }
77
78    #[cfg(feature = "rand")]
79    #[cfg_attr(doc_cfg, doc(cfg(feature = "rand")))]
80    pub fn random_with<R>(mut rng: R) -> Self
81    where
82        R: rand_core::CryptoRng,
83    {
84        // Almost every 32-byte value is a valid secp256k1 private key, but a
85        // few (zero, or values at/above the curve order) are not. Draw fresh
86        // bytes until we get a valid key; in practice this never repeats.
87        loop {
88            let mut buf: [u8; Self::LENGTH] = [0; Self::LENGTH];
89            rng.fill_bytes(&mut buf);
90            if let Ok(key) = Self::new(buf) {
91                return key;
92            }
93        }
94    }
95
96    /// Generate a new private key using the operating system's random number
97    /// generator.
98    #[cfg(feature = "rand")]
99    pub fn random() -> Self {
100        Self::random_with(rand_core::UnwrapErr(getrandom_4::SysRng))
101    }

103    /// Deserialize PKCS#8 private key from ASN.1 DER-encoded data (binary
104    /// format).
105    #[cfg(feature = "pem")]
106    pub fn from_der(bytes: &[u8]) -> Result<Self, SignatureError> {
107        k256::pkcs8::DecodePrivateKey::from_pkcs8_der(bytes)
108            .map(Self::from_k256)
109            .map_err(SignatureError::from_source)
110    }
111
112    /// Serialize this private key as DER-encoded PKCS#8
113    #[cfg(feature = "pem")]
114    pub fn to_der(&self) -> Result<Vec<u8>, SignatureError> {
115        use k256::pkcs8::EncodePrivateKey;
116
117        self.to_k256()
118            .to_pkcs8_der()
119            .map_err(SignatureError::from_source)
120            .map(|der| der.as_bytes().to_owned())
121    }
122
123    /// Deserialize PKCS#8-encoded private key from PEM.
124    #[cfg(feature = "pem")]
125    pub fn from_pem(s: &str) -> Result<Self, SignatureError> {
126        k256::pkcs8::DecodePrivateKey::from_pkcs8_pem(s)
127            .map(Self::from_k256)
128            .map_err(SignatureError::from_source)
129    }
130
131    /// Serialize this private key as PEM-encoded PKCS#8
132    #[cfg(feature = "pem")]
133    pub fn to_pem(&self) -> Result<String, SignatureError> {
134        use pkcs8::EncodePrivateKey;
135
136        self.to_k256()
137            .to_pkcs8_pem(pkcs8::LineEnding::default())
138            .map_err(SignatureError::from_source)
139            .map(|pem| (*pem).to_owned())
140    }
141
142    #[cfg(feature = "pem")]
143    pub(crate) fn from_k256(private_key: k256::ecdsa::SigningKey) -> Self {
144        // A k256 SigningKey is by construction a valid secp256k1 scalar.
145        Self(private_key.to_bytes().into())
146    }
147
148    /// Re-expose the raw private key through k256, used only as a PKCS#8/PEM
149    /// codec. Don't sign with the returned key — use this type's `Signer` impl
150    /// (`try_sign`) instead, so signing goes through fastcrypto.
151    #[cfg(feature = "pem")]
152    fn to_k256(&self) -> k256::ecdsa::SigningKey {
153        k256::ecdsa::SigningKey::from_slice(&self.0).expect("validated on construction")
154    }

157impl crate::ToFromBytes for Secp256k1PrivateKey {
158    type Error = crate::PrivateKeyError;
159    type ByteArray = [u8; Self::LENGTH];
160
161    /// Return the raw 32-byte private key
162    fn to_bytes(&self) -> Self::ByteArray {
163        self.0
164    }
165
166    fn from_bytes(bytes: impl AsRef<[u8]>) -> Result<Self, Self::Error> {
167        let bytes = bytes.as_ref();
168        if bytes.len() != Self::LENGTH {
169            return Err(SignatureError::from_source("invalid secp256k1 key length").into());
170        }
171
172        let mut arr = [0u8; Self::LENGTH];
173        arr.copy_from_slice(bytes);
174        Ok(Self::new(arr)?)
175    }
176}

182#[cfg(feature = "mnemonic")]
183impl crate::FromMnemonic for Secp256k1PrivateKey {
184    type Error = crate::PrivateKeyError;
185
186    fn from_mnemonic(
187        phrase: &str,
188        account_index: impl Into<Option<u64>>,
189        password: impl Into<Option<String>>,
190    ) -> Result<Self, Self::Error>
191    where
192        Self: Sized,
193    {
194        let path = format!(
195            "m/{}'/{}'/0'/0/{}",
196            crate::DERIVATION_PATH_PURPOSE_SECP256K1,
197            crate::DERIVATION_PATH_COIN_TYPE,
198            account_index.into().unwrap_or_default()
199        );
200        Self::from_mnemonic_with_path(phrase, path, password)
201    }
202
203    fn from_mnemonic_with_path(
204        phrase: &str,
205        path: String,
206        password: impl Into<Option<String>>,
207    ) -> Result<Self, Self::Error>
208    where
209        Self: Sized,
210    {
211        use std::str::FromStr;
212
213        use crate::ToFromBytes;
214
215        let mnemonic = bip39::Mnemonic::parse_in_normalized(bip39::Language::English, phrase)
216            .map_err(|e| crate::PrivateKeyError::Bip39(e.to_string()))?;
217        let seed = mnemonic.to_seed(password.into().unwrap_or_default());
218        let child_xprv = bip32::DerivationPath::from_str(&path)
219            .and_then(|path| bip32::XPrv::derive_from_path(seed, &path))
220            .map_err(|e| crate::PrivateKeyError::Bip32(e.to_string()))?;
221        Self::from_bytes(child_xprv.private_key().to_bytes())
222    }
223}

339impl Verifier<SimpleSignature> for Secp256k1VerifyingKey {
340    fn verify(&self, message: &[u8], signature: &SimpleSignature) -> Result<(), SignatureError> {
341        let SimpleSignature::Secp256k1 {
342            signature,
343            public_key,
344        } = signature
345        else {
346            return Err(SignatureError::from_source("not a secp256k1 signature"));
347        };
348
349        if public_key.bytes() != self.public_key().bytes() {
350            return Err(SignatureError::from_source(
351                "public_key in signature does not match",
352            ));
353        }
354
355        <Self as Verifier<Secp256k1Signature>>::verify(self, message, signature)
356    }
357}

506    // The secp256k1 order n. A signature `(r, s)` has a malleable counterpart
507    // `(r, n - s)`; fastcrypto (libsecp256k1) only accepts the low-S form.
508    const SECP256K1_ORDER: [u8; 32] = [
509        0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF,
510        0xFE, 0xBA, 0xAE, 0xDC, 0xE6, 0xAF, 0x48, 0xA0, 0x3B, 0xBF, 0xD2, 0x5E, 0x8C, 0xD0, 0x36,
511        0x41, 0x41,
512    ];

514    // Big-endian 256-bit subtraction `a - b`.
515    fn sub_be(a: &[u8; 32], b: &[u8; 32]) -> [u8; 32] {
516        let mut result = [0u8; 32];
517        let mut borrow = 0i16;
518        for i in (0..32).rev() {
519            let mut diff = a[i] as i16 - b[i] as i16 - borrow;
520            if diff < 0 {
521                diff += 256;
522                borrow = 1;
523            } else {
524                borrow = 0;
525            }
526            result[i] = diff as u8;
527        }
528        result
529    }

531    #[test]
532    fn verify_rejects_high_s() {
533        let key = [
534            172, 12, 96, 180, 207, 143, 111, 151, 81, 57, 242, 89, 74, 5, 150, 51, 56, 111, 245,
535            150, 182, 30, 149, 178, 29, 255, 188, 27, 48, 241, 151, 193,
536        ];
537        let signer = Secp256k1PrivateKey::new(key).unwrap();
538        let verifying_key = signer.verifying_key();
539
540        let message = b"hello";
541        // fastcrypto produces a low-S signature, which verifies.
542        let sig: Secp256k1Signature = signer.try_sign(message.as_slice()).unwrap();
543        verifying_key.verify(message, &sig).unwrap();
544
545        // Malleate to the equivalent high-S signature `(r, n - s)`.
546        let bytes = sig.bytes();
547        let mut r = [0u8; 32];
548        let mut s = [0u8; 32];
549        r.copy_from_slice(&bytes[..32]);
550        s.copy_from_slice(&bytes[32..]);
551        let high_s = sub_be(&SECP256K1_ORDER, &s);
552
553        let mut malleated = [0u8; 64];
554        malleated[..32].copy_from_slice(&r);
555        malleated[32..].copy_from_slice(&high_s);
556        let malleated = Secp256k1Signature::new(malleated);
557
558        // The high-S variant must be rejected even though it is algebraically valid.
559        verifying_key.verify(message, &malleated).unwrap_err();
560    }
