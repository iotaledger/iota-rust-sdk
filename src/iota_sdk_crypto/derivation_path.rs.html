4use std::{fmt, str::FromStr};
5
6/// Bit set on hardened indices.
7const HARDENED: u32 = 1 << 31;
8
9/// A BIP-32 derivation path, parsed from a string like
10/// `m/44'/4218'/0'/0'/0'`; `'` or `h` marks a hardened index.
11#[derive(Clone, Eq, Hash, PartialEq)]
12pub struct DerivationPath(Vec<u32>);
13
14impl DerivationPath {
15    /// The path's indices, with bit 31 set on hardened ones.
16    pub fn indices(&self) -> &[u32] {
17        &self.0
18    }
19}
20
21impl FromStr for DerivationPath {
22    type Err = DerivationPathError;
23
24    fn from_str(path: &str) -> Result<Self, Self::Err> {
25        let mut components = path.split('/');
26        if components.next() != Some("m") {
27            return Err(DerivationPathError(path.to_owned()));
28        }
29        components
30            .map(|component| {
31                let (index, hardened) = match component.strip_suffix(['\'', 'h']) {
32                    Some(index) => (index, true),
33                    None => (component, false),
34                };
35                let index = index.parse::<u32>().ok().filter(|i| *i < HARDENED)?;
36                Some(if hardened { index | HARDENED } else { index })
37            })
38            .collect::<Option<Vec<_>>>()
39            .map(Self)
40            .ok_or_else(|| DerivationPathError(path.to_owned()))
41    }
42}
43
44impl fmt::Display for DerivationPath {
45    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
46        f.write_str("m")?;
47        for index in &self.0 {
48            if index & HARDENED == 0 {
49                write!(f, "/{index}")?;
50            } else {
51                write!(f, "/{}'", index & !HARDENED)?;
52            }
53        }
54        Ok(())
55    }
56}
57
58impl fmt::Debug for DerivationPath {
59    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
60        write!(f, "DerivationPath({self})")
61    }
62}
63
64/// Error returned when a string is not a valid derivation path.
65#[derive(Debug, thiserror::Error)]
66#[error("invalid derivation path: {0:?}")]
67pub struct DerivationPathError(String);

69#[cfg(test)]
70mod tests {
71    #[cfg(target_arch = "wasm32")]
72    use wasm_bindgen_test::wasm_bindgen_test as test;
73
74    use super::*;
75
76    #[test]
77    fn parse() {
78        let path: DerivationPath = "m/44'/4218'/0h/1/2".parse().unwrap();
79        assert_eq!(
80            path.indices(),
81            [44 | HARDENED, 4218 | HARDENED, HARDENED, 1, 2]
82        );
83        assert_eq!(path.to_string(), "m/44'/4218'/0'/1/2");
84    }
85
86    #[test]
87    fn reject_invalid() {
88        for path in [
89            "",
90            "44'/4218'",
91            "m/",
92            "m//0",
93            "m/x",
94            "m/0''",
95            "m/2147483648",
96            "M/0",
97        ] {
98            assert!(path.parse::<DerivationPath>().is_err(), "{path:?}");
99        }
100    }
