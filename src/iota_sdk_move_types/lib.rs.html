4//! Rust representations of Move types used by the IOTA blockchain.
5//!
6//! Each top-level module corresponds to a system package, identified by the
7//! address constants on [`iota_types::Address`]:
8//!
9//! - [`move_stdlib`]    — `0x1`, the Move standard library
10//! - [`iota_framework`] — `0x2`, the IOTA framework
11//! - [`iota_system`]    — `0x3`, the IOTA system package
12//! - [`stardust`]       — `0x107a`, the Stardust migration package
13//!
14//! Inside each package, every Move source module is mirrored 1:1 as a Rust
15//! `pub mod`. Generic Move types stay generic in Rust (with a
16//! `PhantomData<T>` placeholder for phantom parameters).

40/// A Rust type that knows its Move type tag.
41///
42/// The type argument `T` of a generic mirror such as `Coin<T>` must implement
43/// this trait, so that `Coin<IOTA>` can check that an object really is a
44/// `0x2::coin::Coin<0x2::iota::IOTA>`.
45///
46/// Markers like [`IOTA`](iota_framework::iota::IOTA) implement it by hand.
47/// Every `MoveObject` implements it automatically, so an object mirror can
48/// also be a type argument, as in `Display<Coin<IOTA>>`.
49///
50/// To use your own coin type, define an empty marker struct that derives
51/// `Deserialize` and implement this trait for it:
52///
53/// ```
54/// #[derive(serde::Deserialize)]
55/// struct FOO;
56///
57/// impl iota_sdk_move_types::MoveType for FOO {
58///     fn type_tag() -> iota_types::TypeTag {
59///         "0x123::foo::FOO".parse().unwrap()
60///     }
61/// }
62/// ```
63///
64/// For coin types only known at runtime, use the
65/// `try_from_object_with_type` constructors instead, which take the
66/// expected [`TypeTag`](iota_types::TypeTag) as a value.
67pub trait MoveType {
68    /// The Move type tag this type represents (e.g. `0x2::iota::IOTA`).
69    fn type_tag() -> iota_types::TypeTag;
70}
71
72/// An ordered list of [`MoveType`]s: `()`, a single type, or a tuple, such as
73/// the type arguments of a Move function call.
74pub trait MoveTypes {
75    /// Get the type tags.
76    fn type_tags() -> Vec<iota_types::TypeTag>;
77}

79/// A Rust mirror of a Move object that can be decoded from an
80/// [`Object`](iota_types::Object).
81///
82/// [`struct_tag`](Self::struct_tag) is the type of the objects it decodes,
83/// e.g. `0x2::coin::Coin<0x2::iota::IOTA>` for `Coin<IOTA>`. Every
84/// `MoveObject` is also a [`MoveType`] with that same tag.
85#[cfg(feature = "serde")]
86pub trait MoveObject:
87    Sized + for<'a> TryFrom<&'a iota_types::Object, Error = FromObjectError>
88{
89    /// The Move struct tag of the objects this type mirrors.
90    fn struct_tag() -> iota_types::StructTag;
91}
92
93#[cfg(feature = "serde")]
94impl<T: MoveObject> MoveType for T {
95    fn type_tag() -> iota_types::TypeTag {
96        iota_types::TypeTag::Struct(Box::new(T::struct_tag()))
97    }
98}

100/// Error returned when converting an `Object` into a typed mirror.
101///
102/// Every mirror's `TryFrom<&Object>` (and `try_from_object_with_type`)
103/// conversion returns this on failure: the object either isn't a Move
104/// struct, carries a type tag that doesn't match the expected type, or has
105/// BCS contents that fail to decode.
106#[cfg(feature = "serde")]
107#[derive(Debug, thiserror::Error)]
108#[non_exhaustive]
109pub enum FromObjectError {
110    /// The object is a package, not a Move struct.
111    #[error("object is not a Move struct")]
112    NotAMoveStruct,
113    /// The Move struct's type tag does not match the expected type.
114    #[error("object's type tag does not match expected type")]
115    WrongType,
116    /// BCS decoding of the struct contents failed.
117    #[error("bcs decoding failed: {0}")]
118    Bcs(#[from] iota_types::BcsError),
119}
120
121/// Decode the BCS contents of `object`, provided it is a Move struct tagged
122/// exactly `expected`.
123#[cfg(feature = "serde")]
124fn decode_move_struct<T: serde::de::DeserializeOwned>(
125    object: &iota_types::Object,
126    expected: &iota_types::StructTag,
127) -> Result<T, FromObjectError> {
128    let move_struct = object
129        .as_opt_struct()
130        .ok_or(FromObjectError::NotAMoveStruct)?;
131    if move_struct.struct_tag() != expected {
132        return Err(FromObjectError::WrongType);
133    }
134    Ok(bcs::from_bytes(move_struct.contents()).map_err(iota_types::BcsError::new)?)
135}

164            "0x2::coin::CoinMetadata<0x2::iota::IOTA>"

171            Coin::<IOTA>::type_tag(),

173        );
