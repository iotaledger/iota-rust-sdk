6/// Types from `0x107a::irc27`.
7pub mod irc27 {
8    use iota_types::Address;
9
10    use crate::{
11        iota_framework::{url::Url, vec_map::VecMap},
12        move_stdlib::{fixed_point32::FixedPoint32, string},
13    };
14
15    /// Rust version of the Move `stardust::irc27::Irc27Metadata` type.
16    ///
17    /// The IRC27 NFT metadata standard schema.
18    #[derive(Clone, Debug, Eq, PartialEq)]
19    #[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]
20    #[cfg_attr(
21        all(test, not(target_arch = "wasm32")),
22        derive(iota_bcs_schema::MoveShape)
23    )]
24    pub struct Irc27Metadata {
25        /// Version of the metadata standard.
26        pub version: string::String,
27        /// The media type (MIME) of the asset.
28        pub media_type: string::String,
29        /// URL pointing to the NFT file location.
30        pub uri: Url,
31        /// Alphanumeric text string defining the human identifiable name for
32        /// the NFT.
33        pub name: string::String,
34        /// The human-readable collection name of the NFT.
35        pub collection_name: Option<string::String>,
36        /// Royalty payment addresses mapped to the payout percentage.
37        pub royalties: VecMap<Address, FixedPoint32>,
38        /// The human-readable name of the NFT creator.
39        pub issuer_name: Option<string::String>,
40        /// The human-readable description of the NFT.
41        pub description: Option<string::String>,
42        /// Additional attributes following [OpenSea metadata standards].
43        ///
44        /// [OpenSea metadata standards]: https://docs.opensea.io/docs/metadata-standards
45        pub attributes: VecMap<string::String, string::String>,
46        /// Legacy non-standard metadata fields.
47        pub non_standard_fields: VecMap<string::String, string::String>,
48    }
49
50    #[cfg(feature = "serde")]
51    impl Irc27Metadata {
52        /// Decode an [`Irc27Metadata`] from BCS bytes without verifying any
53        /// on-chain type tag (the metadata is usually nested inside an
54        /// [`Nft`](super::nft::Nft), not stored as a top-level object).
55        pub fn from_bcs(bytes: &[u8]) -> Result<Self, bcs::Error> {
56            bcs::from_bytes(bytes)
57        }
58    }
59}

232        all(test, not(target_arch = "wasm32")),
233        derive(iota_bcs_schema::MoveShape)
234    )]
235    pub struct BasicOutput<T> {
236        /// Hash of the `outputId` that was migrated.
237        pub id: UID,
238        /// The amount of coins held by the output.
239        pub balance: Balance<T>,
240        /// A `Bag` of native tokens keyed by stringified asset type.
241        pub native_tokens: Bag,
242        /// The storage deposit return unlock condition.
243        pub storage_deposit_return_uc: Option<StorageDepositReturnUnlockCondition>,
244        /// The timelock unlock condition.
245        pub timelock_uc: Option<TimelockUnlockCondition>,
246        /// The expiration unlock condition.
247        pub expiration_uc: Option<ExpirationUnlockCondition>,
248        /// The metadata feature.
249        pub metadata: Option<Vec<u8>>,
250        /// The tag feature.
251        pub tag: Option<Vec<u8>>,
252        /// The sender feature.
253        pub sender: Option<Address>,
254    }
