4//! Internal macros generating the `Object` constructors that every `key`
5//! Move-object mirror shares.
6//!
7//! The mirrors differ only in their type and (for generic mirrors) their
8//! single type parameter, so the [`MoveObject`](crate::MoveObject) impl and
9//! the `TryFrom<&Object>` / `try_from_object_with_type` bodies are otherwise
10//! identical boilerplate. `from_bcs` and the field accessors stay hand-written
11//! per type.
12//!
13//! The struct tag comes from the [`StructTag`] constructor derived from the
14//! mirror's name as `new_<name:snake>` — the same `paste` snake-casing that
15//! generated that constructor in the first place, applied to the same
16//! identifier. `TryFrom<&Object>` accepts exactly the tag `struct_tag` returns.
17//!
18//! [`StructTag`]: iota_types::StructTag
19
20/// Generate the [`MoveObject`](crate::MoveObject) impl and the
21/// `TryFrom<&Object>` constructor for a non-generic mirror.
22macro_rules! impl_try_from_object {
23    ($ty:ident $(,)?) => {
24        #[cfg(feature = "serde")]
25        impl $crate::MoveObject for $ty {
26            fn struct_tag() -> ::iota_types::StructTag {
27                ::paste::paste! { ::iota_types::StructTag::[< new_ $ty:snake >]() }
28            }
29        }
30
31        #[cfg(feature = "serde")]
32        #[doc = concat!(
33            "Decode a [`",
34            stringify!($ty),
35            "`] from an on-chain object, validating that the object's Move type tag matches."
36        )]
37        impl TryFrom<&::iota_types::Object> for $ty {
38            type Error = $crate::FromObjectError;
39
40            fn try_from(object: &::iota_types::Object) -> Result<Self, Self::Error> {
41                $crate::decode_move_struct(object, &<Self as $crate::MoveObject>::struct_tag())
42            }
43        }
44    };
45}
46
47/// Generate the [`MoveObject`](crate::MoveObject) impl,
48/// `try_from_object_with_type` and the `TryFrom<&Object>` constructor for a
49/// mirror with a single type parameter.
50macro_rules! impl_try_from_object_generic {
51    ($ty:ident<$param:ident> $(,)?) => {
52        #[cfg(feature = "serde")]
53        impl<$param> $crate::MoveObject for $ty<$param>
54        where
55            $param: ::serde::de::DeserializeOwned + $crate::MoveType,
56        {
57            fn struct_tag() -> ::iota_types::StructTag {
58                ::paste::paste! {
59                    ::iota_types::StructTag::[< new_ $ty:snake >](
60                        <$param as $crate::MoveType>::type_tag(),
61                    )
62                }
63            }
64        }
65
66        #[cfg(feature = "serde")]
67        impl<$param> $ty<$param>
68        where
69            $param: ::serde::de::DeserializeOwned,
70        {
71            #[doc = concat!(
72                "Decode a [`",
73                stringify!($ty),
74                "`] from an on-chain object, validating its Move type tag and that its type parameter equals `type_param`.\n\nEscape hatch for type parameters only known at runtime; nothing ties `type_param` to `",
75                stringify!($param),
76                "`. Prefer the `TryFrom` impl when the type parameter is known at compile time."
77            )]
78            pub fn try_from_object_with_type(
79                object: &::iota_types::Object,
80                type_param: &::iota_types::TypeTag,
81            ) -> Result<Self, $crate::FromObjectError> {
82                ::paste::paste! {
83                    $crate::decode_move_struct(
84                        object,
85                        &::iota_types::StructTag::[< new_ $ty:snake >](type_param.clone()),
86                    )
87                }
88            }
89        }
90
91        #[cfg(feature = "serde")]
92        #[doc = concat!(
93            "Decode a [`",
94            stringify!($ty),
95            "`] from an on-chain object, validating its full Move type tag including the type parameter."
96        )]
97        impl<$param> TryFrom<&::iota_types::Object> for $ty<$param>
98        where
99            $param: ::serde::de::DeserializeOwned + $crate::MoveType,
100        {
101            type Error = $crate::FromObjectError;
102
103            fn try_from(object: &::iota_types::Object) -> Result<Self, Self::Error> {
104                $crate::decode_move_struct(object, &<Self as $crate::MoveObject>::struct_tag())
105            }
106        }
107    };
