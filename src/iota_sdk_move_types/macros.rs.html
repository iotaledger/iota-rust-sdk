4//! Internal macros generating the `Object` constructors that every `key`
5//! Move-object mirror shares.
6//!
7//! The mirrors differ only in their type and (for generic mirrors) their
8//! single type parameter, so the [`MoveObject`](crate::MoveObject) impl and
9//! the `TryFrom<&Object>` / `try_from_object_with_type` bodies are otherwise
10//! identical boilerplate. `from_bcs` and the field accessors stay hand-written
11//! per type.
12//!
13//! The struct tag comes from the [`StructTag`] constructor derived from the
14//! mirror's name as `new_<name:snake>` — the same `paste` snake-casing that
15//! generated that constructor in the first place, applied to the same
16//! identifier. `TryFrom<&Object>` accepts exactly the tag `struct_tag` returns.
17//!
18//! [`StructTag`]: iota_types::StructTag
19
20/// Generate the [`MoveObject`](crate::MoveObject) impl and the
21/// `TryFrom<&Object>` constructor for a non-generic mirror.
22macro_rules! impl_try_from_object {
23    ($ty:ident $(,)?) => {
24        #[cfg(feature = "serde")]
25        impl $crate::MoveObject for $ty {
26            fn struct_tag() -> ::iota_types::StructTag {
27                ::paste::paste! { ::iota_types::StructTag::[< new_ $ty:snake >]() }
28            }
29        }
30
31        #[cfg(feature = "serde")]
32        #[doc = concat!(
33            "Decode a [`",
34            stringify!($ty),
35            "`] from an on-chain object, validating that the object's Move type tag matches."
36        )]
37        impl TryFrom<&::iota_types::Object> for $ty {
38            type Error = $crate::FromObjectError;
39
40            fn try_from(object: &::iota_types::Object) -> Result<Self, Self::Error> {
41                $crate::decode_move_struct(object, &<Self as $crate::MoveObject>::struct_tag())
42            }
43        }
44    };
45}

47/// Generate the [`MoveObject`](crate::MoveObject) impl,
48/// `try_from_object_with_type` and the `TryFrom<&Object>` constructor for a
49/// mirror with a single type parameter.
50///
51/// Mark the parameter `phantom` when it is phantom in Move (it only shows up
52/// in `PhantomData` fields of the mirror). `try_from_object_with_type` is then
53/// only implemented on `$ty<()>`, so callers don't have to name a type
54/// parameter that the runtime type tag already decides.
55macro_rules! impl_try_from_object_generic {
56    ($ty:ident<phantom $param:ident> $(,)?) => {
57        impl_try_from_object_generic!(@move_object $ty<$param>);
58
59        #[cfg(feature = "serde")]
60        impl $ty<()> {
61            #[doc = concat!(
62                "Decode a [`",
63                stringify!($ty),
64                "`] from an on-chain object, validating its Move type tag and that its type parameter equals `type_param`.\n\nFor type parameters only known at runtime; the result is `",
65                stringify!($ty),
66                "<()>` because the type parameter is phantom. Prefer the `TryFrom` impl when the type parameter is known at compile time."
67            )]
68            pub fn try_from_object_with_type(
69                object: &::iota_types::Object,
70                type_param: &::iota_types::TypeTag,
71            ) -> Result<Self, $crate::FromObjectError> {
72                impl_try_from_object_generic!(@decode_with_type $ty, object, type_param)
73            }
74        }
75    };
76    ($ty:ident<$param:ident> $(,)?) => {
77        impl_try_from_object_generic!(@move_object $ty<$param>);
78
79        #[cfg(feature = "serde")]
80        impl<$param> $ty<$param>
81        where
82            $param: ::serde::de::DeserializeOwned,
83        {
84            #[doc = concat!(
85                "Decode a [`",
86                stringify!($ty),
87                "`] from an on-chain object, validating its Move type tag and that its type parameter equals `type_param`.\n\nEscape hatch for type parameters only known at runtime; nothing ties `type_param` to `",
88                stringify!($param),
89                "`. Prefer the `TryFrom` impl when the type parameter is known at compile time."
90            )]
91            pub fn try_from_object_with_type(
92                object: &::iota_types::Object,
93                type_param: &::iota_types::TypeTag,
94            ) -> Result<Self, $crate::FromObjectError> {
95                impl_try_from_object_generic!(@decode_with_type $ty, object, type_param)
96            }
97        }
98    };
99    (@decode_with_type $ty:ident, $object:ident, $type_param:ident) => {
100        ::paste::paste! {
101            $crate::decode_move_struct(
102                $object,
103                &::iota_types::StructTag::[< new_ $ty:snake >]($type_param.clone()),
104            )
105        }
106    };
107    (@move_object $ty:ident<$param:ident>) => {
108        #[cfg(feature = "serde")]
109        impl<$param> $crate::MoveObject for $ty<$param>
110        where
111            $param: ::serde::de::DeserializeOwned + $crate::MoveType,
112        {
113            fn struct_tag() -> ::iota_types::StructTag {
114                ::paste::paste! {
115                    ::iota_types::StructTag::[< new_ $ty:snake >](
116                        <$param as $crate::MoveType>::type_tag(),
117                    )
118                }
119            }
120        }
121
122        #[cfg(feature = "serde")]
123        #[doc = concat!(
124            "Decode a [`",
125            stringify!($ty),
126            "`] from an on-chain object, validating its full Move type tag including the type parameter."
127        )]
128        impl<$param> TryFrom<&::iota_types::Object> for $ty<$param>
129        where
130            $param: ::serde::de::DeserializeOwned + $crate::MoveType,
131        {
132            type Error = $crate::FromObjectError;
133
134            fn try_from(object: &::iota_types::Object) -> Result<Self, Self::Error> {
135                $crate::decode_move_struct(object, &<Self as $crate::MoveObject>::struct_tag())
136            }
137        }
138    };
