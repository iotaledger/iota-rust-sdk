4//! The single context over the compiled protobuf descriptors.
5//!
6//! `protox` produces a [`DescriptorPool`] that already resolves names, nesting, map entries and
7//! options across files, so this type adds only what the pool cannot know: where a message lands in
8//! the generated Rust module tree, which version module a package belongs to, and which messages
9//! reference each other in a cycle.
10//!
11//! Protobuf names come in two shapes: the pool uses bare names (`iota.grpc.v1.types.Digest`), while
12//! `FieldDescriptorProto::type_name` and the boxing config carry a leading dot. Everything here
13//! accepts either and normalizes on the way in.
14

31pub(crate) struct Context {
32    pool: DescriptorPool,
33    /// Types resolved to a path outside the generated code, such as the `google.protobuf`
34    /// well-known types.
35    extern_paths: ExternPaths,
36    /// Accessor annotations, keyed as [`parse_proto_accessors_from_pool`] produces them.
37    accessors: AccessorMap,
38    /// Messages marked `field_mask_transparent`, keyed by fully qualified name with a leading dot.
39    transparent: HashMap<String, TransparentInfo>,
40}
41
42impl Context {
43    pub(crate) fn build(pool: DescriptorPool) -> Self {
44        let accessors = parse_proto_accessors_from_pool(&pool);
45        let transparent = parse_transparent_messages_from_pool(&pool);
46
47        Self {
48            pool,
49            extern_paths: ExternPaths::new(&[], true).expect("built-in extern paths are valid"),
50            accessors,
51            transparent,
52        }
53    }
54
55    /// True when the type resolves outside the generated code and so has nothing generated for it.
56    pub(crate) fn is_extern(&self, full_type_name: &str) -> bool {
57        self.extern_paths
58            .resolve_ident(&format!(".{}", full_type_name.trim_start_matches('.')))
59            .is_some()
60    }
61
62    /// Resolves a protobuf type to a Rust path relative to `package`, the form accessor generation
63    /// emits (`super::super::types::Digest`).
64    ///
65    /// Kept separate from [`Self::message_path`], which produces absolute `crate::` paths:
66    /// accessors are written into a `_accessor_impls` module and address their types relatively
67    /// from there.
68    pub(crate) fn accessor_type_path(&self, package: &str, pb_ident: &str) -> String {
69        // protoc should always give fully qualified identifiers.
70        assert_eq!(".", &pb_ident[..1]);
71
72        if let Some(proto_ident) = self.extern_paths.resolve_ident(pb_ident) {
73            return proto_ident;
74        }
75
76        let mut local_path = package.split('.').peekable();
77
78        // If no package is specified the start of the package name will be '.' and split will
79        // return an empty string ("") which breaks resolution
80        // The fix to this is to ignore the first item if it is empty.
81        if local_path.peek().is_some_and(|s| s.is_empty()) {
82            local_path.next();
83        }
84
85        let mut ident_path = pb_ident[1..].split('.');
86        let ident_type = ident_path.next_back().unwrap();
87        let mut ident_path = ident_path.peekable();
88
89        // Skip path elements in common.
90        while local_path.peek().is_some() && local_path.peek() == ident_path.peek() {
91            local_path.next();
92            ident_path.next();
93        }
94
95        local_path
96            .map(|_| "super".to_string())
97            .chain(ident_path.map(to_snake))
98            .chain(std::iter::once(to_upper_camel(ident_type)))
99            .join("::")
100    }

102    /// Accessor annotations of every message, from the `iota.grpc` accessor options.
103    pub(crate) fn accessor_map(&self) -> &AccessorMap {
104        &self.accessors
105    }
106
107    /// Information about a `field_mask_transparent` wrapper, if the message is one. Takes a fully
108    /// qualified name with or without the leading dot.
109    pub(crate) fn transparent(&self, full_type_name: &str) -> Option<&TransparentInfo> {
110        self.transparent
111            .get(&format!(".{}", full_type_name.trim_start_matches('.')))
112    }

114    /// Every package of every compiled file, sorted, without a leading dot.
115    pub(crate) fn packages(&self) -> Vec<String> {
116        let mut packages: Vec<String> = self
117            .pool
118            .files()
119            .map(|file| file.package_name().to_owned())
120            .collect();
121        packages.sort();
122        packages.dedup();
123        packages
124    }
125
126    /// Root messages of a package in the order codegen emits them: files sorted by name, messages
127    /// in declaration order. Nested messages are reached through
128    /// [`MessageDescriptor::child_messages`].
129    pub(crate) fn messages_in_declaration_order(&self, package: &str) -> Vec<MessageDescriptor> {
130        let mut files: Vec<_> = self
131            .pool
132            .files()
133            .filter(|file| file.package_name() == package)
134            .collect();
135        files.sort_by(|a, b| a.name().cmp(b.name()));
136
137        files.iter().flat_map(|file| file.messages()).collect()
138    }
139
140    /// Every message of a package, nested ones included, in the order the proto files declare them:
141    /// files by name, then each message followed by its own nested messages. Map entry messages are
142    /// synthetic and have no generated type, so they are skipped.
143    pub(crate) fn all_messages_in_declaration_order(
144        &self,
145        package: &str,
146    ) -> Vec<MessageDescriptor> {
147        fn push_with_children(message: MessageDescriptor, out: &mut Vec<MessageDescriptor>) {
148            if message.is_map_entry() {
149                return;
150            }
151            let children: Vec<_> = message.child_messages().collect();
152            out.push(message);
153            for child in children {
154                push_with_children(child, out);
155            }
156        }
157
158        let mut messages = Vec::new();
159        for root in self.messages_in_declaration_order(package) {
160            push_with_children(root, &mut messages);
161        }
162        messages
163    }

165    /// Resolves a message by fully qualified name, with or without the leading dot. Panics when
166    /// unknown: every message a field can reference is compiled together with the field's own file.
167    pub(crate) fn resolve(&self, full_type_name: &str) -> MessageDescriptor {
168        let name = full_type_name.trim_start_matches('.');
169        self.pool
170            .get_message_by_name(name)
171            .unwrap_or_else(|| panic!("unknown protobuf message type {full_type_name}"))
172    }
173
174    /// Path to the message type as prost generates it. Messages of `from_package` that are not
175    /// nested are left unqualified — the generated file is included into that very module.
176    pub(crate) fn message_path(
177        &self,
178        message: &MessageDescriptor,
179        from_package: &str,
180    ) -> TokenStream {
181        let name = format_ident!("{}", message.name());
182        let outer = outer_modules(message);
183
184        if outer.is_empty() && message.package_name() == from_package {
185            return quote! { #name };
186        }
187
188        let root = package_path(message.package_name());
189        quote! { #root #(::#outer)* ::#name }
190    }
191
192    /// Path to the field path builder generated for the message.
193    pub(crate) fn builder_path(
194        &self,
195        message: &MessageDescriptor,
196        from_package: &str,
197    ) -> TokenStream {
198        let builder = format_ident!("{}FieldPathBuilder", message.name());
199        let outer = outer_modules(message);
200
201        if message.package_name() == from_package {
202            // Builders are generated into `_field_impls` of the file being written, mirroring the
203            // message nesting.
204            return quote! { #(#outer::)* #builder };
205        }
206
207        let root = package_path(message.package_name());
208        if !outer.is_empty() {
209            // A nested message's builder has no path from outside its own package: `_field_impls`
210            // is private, and at the package root the module holding the builder is shadowed by the
211            // prost module of the same name, so the glob re-export never surfaces it.
212            //
213            // No proto references such a builder across packages today. Making `_field_impls` `pub`
214            // is the one-line fix if one ever does — hence failing loudly here rather than emitting
215            // a path that does not resolve.
216            panic!(
217                "cannot reference the field path builder of nested message {} from package {from_package}: \
218                 nested builders are only addressable inside their own package",
219                message.full_name()
220            );
221        }
222
223        quote! { #root::#builder }
224    }

226    /// True when `to` reaches back to `from` through message-typed fields, i.e. the two sit on a
227    /// reference cycle. Used to stop `MessageFields::FIELDS` from recursing forever.
228    ///
229    /// The walk is scoped to the package `from` belongs to: cycle breaking has always been decided
230    /// per package, and a path leaving the package cannot close a cycle the generated code would
231    /// recurse through.
232    pub(crate) fn has_reference_cycle(&self, from: &str, to: &str) -> bool {
233        let from = from.trim_start_matches('.');
234        let package = self.resolve(from).package_name().to_owned();
235        self.can_reach(
236            to.trim_start_matches('.'),
237            from,
238            &package,
239            &mut HashSet::new(),
240        )
241    }
242
243    fn can_reach(
244        &self,
245        source: &str,
246        target: &str,
247        package: &str,
248        visited: &mut HashSet<String>,
249    ) -> bool {
250        if source == target {
251            return true;
252        }
253        if !visited.insert(source.to_owned()) {
254            return false;
255        }
256        let Some(message) = self.pool.get_message_by_name(source) else {
257            return false;
258        };
259        if message.package_name() != package {
260            return false;
261        }
262
263        message.fields().any(|field| match field.kind() {
264            Kind::Message(inner) => self.can_reach(inner.full_name(), target, package, visited),
265            _ => false,
266        })
267    }
268}

270/// Convert proto package to rust module path.
271/// Generated rust modules live in `crate::proto`.
272fn package_path(package: &str) -> TokenStream {
273    let path = package.split('.').collect::<Vec<_>>().join("::");
274    let module = if let Some(path) = path.strip_prefix("iota::grpc::v") {
275        // `crate::proto::iota::grpc::v1` is re-exported as `crate::v1`
276        format!("crate::v{path}")
277    } else if let Some(path) = path.strip_prefix("google") {
278        // `crate::proto::google` is re-exported as `crate::google`
279        format!("crate::google{path}")
280    } else {
281        // generic case
282        format!("crate::proto::{path}")
283    };
284    let path: syn::Path = syn::parse_str(module.as_str()).unwrap();
285    quote! { #path }
286}
287
288/// snake_case modules of the enclosing messages, outermost first.
289fn outer_modules(message: &MessageDescriptor) -> Vec<proc_macro2::Ident> {
290    let mut outer = Vec::new();
291    let mut parent = message.parent_message();
292    while let Some(message) = parent {
293        outer.push(format_ident!("{}", to_snake(message.name())));
294        parent = message.parent_message();
295    }
296    outer.reverse();
297    outer
298}

304    fn context() -> Context {
305        let fixtures =
306            std::path::PathBuf::from(std::env!("CARGO_MANIFEST_DIR")).join("tests/fixtures");
307        let mut compiler = protox::Compiler::new([&fixtures]).unwrap();
308        compiler
309            .open_files([
310                "iota/grpc/v1/a.proto",
311                "iota/grpc/v1/b.proto",
312                "iota/grpc/v2/a.proto",
313            ])
314            .unwrap();
315        Context::build(compiler.descriptor_pool())
316    }

318    fn message_path(full_type_name: &str, from_package: &str) -> String {
319        let context = context();
320        let message = context.resolve(full_type_name);
321        context
322            .message_path(&message, from_package)
323            .to_string()
324            .replace(' ', "")
325    }
