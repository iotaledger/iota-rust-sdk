5//! Proto build tool for generating gRPC types with field constants.
6//!
7//! Everything is written into `iota-sdk-grpc-types/src/proto/generated` and
8//! committed, so a run that changes the output fails the build. The pipeline
9//! is:
10//!
11//! 1. [`discover_proto_files`] — collect the `.proto` sources in a stable
12//!    order.
13//! 2. [`compile_proto_files`] — compile them with `protox` into a descriptor
14//!    pool (name resolution, options) and a file descriptor set.
15//! 3. [`generate_tonic`] — prost/tonic structs, clients and servers.
16//! 4. [`add_license_headers`] — license headers and `super::google` rewrites on
17//!    the files prost just wrote.
18//! 5. Accessor generation — `with_`/`set_`/getter methods driven by the
19//!    `iota.grpc.*_accessors` proto options.
20//! 6. Service method paths, then field constants and `MessageFields` impls,
21//!    which together back read masks and field path builders.
22//! 7. [`verify_generated_files_committed`] — fail if the output drifted from
23//!    what is committed.
24

36/// Fields that prost boxes in the generated structs, which accessor generation
37/// has to mirror in its signatures.
38///
39/// These are deliberately not passed to `Config::boxed`: prost boxes every
40/// non-repeated message field lying on a reference cycle by itself, before it
41/// consults its own config (`prost-build`, `Context::should_box_impl`), so
42/// configuring them changes nothing. The list exists only as input to our own
43/// accessor codegen.
44///
45/// Known gap, deliberately left alone:
46/// `.iota.grpc.v1.types.TypeTagVector.inner_type` is boxed by prost and missing
47/// here. No accessor is generated for it, so the omission is inert — adding it
48/// would change generated output.
49const PROST_BOXED_FIELDS: &[&str] = &[
50    ".iota.grpc.v1.filter.EventFilter.negation",
51    ".iota.grpc.v1.filter.TransactionFilter.negation",
52    ".iota.grpc.v1.filter.NotEventFilter.filter",
53    ".iota.grpc.v1.filter.NotTransactionFilter.filter",
54    ".iota.grpc.v1.types.TypeTag.vector_tag",
55];
56
57/// Fields whose `MessageFields::FIELDS` must stop recursing, because the field
58/// closes a reference cycle.
59const BOXED_TYPES_FIELD_INFO: &[&str] = &[
60    ".iota.grpc.v1.filter.AllEventFilter.filters",
61    ".iota.grpc.v1.filter.AnyEventFilter.filters",
62    ".iota.grpc.v1.filter.NotEventFilter.filter",
63    ".iota.grpc.v1.filter.AllTransactionFilter.filters",
64    ".iota.grpc.v1.filter.AnyTransactionFilter.filters",
65    ".iota.grpc.v1.filter.NotTransactionFilter.filter",
66    ".iota.grpc.v1.types.TypeTagVector.inner_type",
67];
68
69/// Fields whose accessors take and return boxed values: the ones prost boxes,
70/// plus those where a boxed signature is simply more ergonomic.
71const BOXED_TYPES_ACCESSOR: &[&str] = &[
72    ".iota.grpc.v1.filter.EventFilter.negation",
73    ".iota.grpc.v1.filter.TransactionFilter.negation",
74    ".iota.grpc.v1.filter.NotEventFilter.filter",
75    ".iota.grpc.v1.filter.NotTransactionFilter.filter",
76    ".iota.grpc.v1.types.TypeTag.vector_tag",
77];

79fn main() {
80    let root_dir = PathBuf::from(std::env!("CARGO_MANIFEST_DIR"));
81
82    let proto_dir = root_dir
83        .join("../iota-sdk-grpc-types/proto")
84        .canonicalize()
85        .unwrap();
86    let out_dir = root_dir
87        .join("../iota-sdk-grpc-types/src/proto")
88        .canonicalize()
89        .unwrap()
90        .join("generated");
91
92    // The codegen entry points take owned paths, so widen the lists once here.
93    let prost_boxed_fields = owned_paths(PROST_BOXED_FIELDS);
94    let boxed_types_field_info = owned_paths(BOXED_TYPES_FIELD_INFO);
95    let boxed_types_accessor = owned_paths(BOXED_TYPES_ACCESSOR);
96
97    reset_out_dir(&out_dir);
98
99    let proto_files = discover_proto_files(&proto_dir);
100    let descriptor_pool = compile_proto_files(&proto_dir, &proto_files);
101
102    generate_tonic(&proto_dir, &proto_files, &out_dir);
103    add_license_headers(&out_dir);
104
105    codegen::generate_service_methods::generate_service_method_paths(&descriptor_pool, &out_dir);
106
107    let context = Context::build(descriptor_pool);
108
109    codegen::accessors::generate_accessors(
110        &context,
111        &out_dir,
112        &prost_boxed_fields,
113        &boxed_types_accessor,
114    );
115
116    codegen::generate_fields::generate_field_info(&context, &out_dir, &boxed_types_field_info);
117
118    verify_generated_files_committed(&out_dir);
119}

121fn owned_paths(paths: &[&str]) -> Vec<String> {
122    paths.iter().map(|path| (*path).to_owned()).collect()
123}
124
125/// Clears the output directory so that removed protos cannot leave stale
126/// generated files behind.
127fn reset_out_dir(out_dir: &Path) {
128    if out_dir.exists() {
129        std::fs::remove_dir_all(out_dir).unwrap();
130    }
131    std::fs::create_dir_all(out_dir).unwrap();
132}
133
134fn discover_proto_files(proto_dir: &Path) -> Vec<PathBuf> {
135    let proto_ext = std::ffi::OsStr::new("proto");
136    let mut proto_files = walkdir::WalkDir::new(proto_dir)
137        .into_iter()
138        .filter_map(|entry| {
139            (|| {
140                let entry = entry?;
141                if entry.file_type().is_dir() {
142                    return Ok(None);
143                }
144
145                let path = entry.into_path();
146                if path.extension() != Some(proto_ext) {
147                    return Ok(None);
148                }
149
150                Ok(Some(path))
151            })()
152            .transpose()
153        })
154        .collect::<Result<Vec<_>, walkdir::Error>>()
155        .unwrap();
156    // `walkdir` traversal order depends on the OS (and is non-deterministic on
157    // Linux), so sort the paths to keep `compile_protos` output stable across
158    // runs — without this, messages from different .proto files in the same
159    // package can swap positions in the generated module.
160    proto_files.sort();
161
162    proto_files
163}

165fn compile_proto_files(proto_dir: &Path, proto_files: &[PathBuf]) -> DescriptorPool {
166    let mut compiler_init = protox::Compiler::new([proto_dir]).unwrap();
167    let compiler = compiler_init
168        .include_source_info(true)
169        .include_imports(true)
170        .open_files(proto_files)
171        .unwrap();
172
173    compiler.descriptor_pool()
174}
175
176fn generate_tonic(proto_dir: &Path, proto_files: &[PathBuf], out_dir: &Path) {
177    tonic_prost_build::configure()
178        .build_client(true)
179        .build_server(true)
180        .bytes(".")
181        .message_attribute(".iota.grpc", "#[non_exhaustive]")
182        .enum_attribute(".iota.grpc", "#[non_exhaustive]")
183        .btree_map(".")
184        .out_dir(out_dir)
185        .compile_protos(proto_files, &[proto_dir.to_path_buf()])
186        .unwrap();
187}

189/// Adds IOTA license headers to the tonic-generated files and rewrites the
190/// `super::google` paths prost emits, which do not resolve from the module the
191/// generated files are included into.
192fn add_license_headers(out_dir: &Path) {
193    let google_import_regex = regex::Regex::new(r"(?:super::)+google").unwrap();
194
195    for entry in std::fs::read_dir(out_dir).unwrap() {
196        let entry = entry.unwrap();
197        let path = entry.path();
198
199        // ignore google proto files
200        if path.to_str().unwrap().contains("google") {
201            continue;
202        }
203
204        if path.extension().and_then(|s| s.to_str()) == Some("rs")
205            && !path
206                .file_name()
207                .unwrap()
208                .to_str()
209                .unwrap()
210                .contains("field_info")
211        {
212            let mut content = std::fs::read_to_string(&path).unwrap();
213
214            // Add license header if missing
215            if !content.starts_with("// Copyright") {
216                content = format!(
217                    "// Copyright (c) Mysten Labs, Inc.\n// Modifications Copyright (c) 2026 IOTA Stiftung\n// SPDX-License-Identifier: Apache-2.0\n\n{content}"
218                );
219            }
220
221            // Replace all occurrences of super::google with crate::google (any number of
222            // super::)
223            content = google_import_regex
224                .replace_all(&content, "crate::google")
225                .to_string();
226            std::fs::write(&path, content).unwrap();
227        }
228    }
229}

231/// Fails the build when a run changed the generated files, which are committed.
232fn verify_generated_files_committed(out_dir: &Path) {
233    let status = std::process::Command::new("git")
234        .arg("diff")
235        .arg("--exit-code")
236        .arg("--")
237        .arg(out_dir)
238        .status();
239    match status {
240        Ok(status) if !status.success() => {
241            eprintln!("Generated protobuf files have uncommitted changes. Please commit them.");
242            std::process::exit(2); // Custom exit code for uncommitted changes
243        }
244        Err(error) => panic!("failed to run `git diff`: {error}"),
245        Ok(_) => {}
246    }
