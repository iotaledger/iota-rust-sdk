5//! Proto build tool for generating gRPC types with field constants.
6//!
7//! Everything is written into `iota-sdk-grpc-types/src/proto/generated` and
8//! committed, so a run that changes the output fails the build. The pipeline
9//! is:
10//!
11//! 1. [`discover_proto_files`] — collect the `.proto` sources in a stable
12//!    order.
13//! 2. [`compile_proto_files`] — compile them with `protox` into a descriptor
14//!    pool (name resolution, options) and a file descriptor set.
15//! 3. [`generate_tonic`] — prost/tonic structs, clients and servers.
16//! 4. [`add_license_headers`] — license headers and `super::google` rewrites on
17//!    the files prost just wrote.
18//! 5. [`remove_eq_hash_derives`] — drop prost's automatic `Eq`/`Hash` from the
19//!    messages listed in [`STRIP_EQ_HASH_MESSAGES`].
20//! 6. Accessor generation — `with_`/`set_`/getter methods driven by the
21//!    `iota.grpc.*_accessors` proto options.
22//! 7. Service method paths, then field constants and `MessageFields` impls,
23//!    which together back read masks and field path builders.
24//! 8. [`verify_generated_files_committed`] — fail if the output drifted from
25//!    what is committed.

41/// Fields that prost boxes in the generated structs, which accessor generation
42/// has to mirror in its signatures.
43///
44/// These are deliberately not passed to `Config::boxed`: prost boxes every
45/// non-repeated message field lying on a reference cycle by itself, before it
46/// consults its own config (`prost-build`, `Context::should_box_impl`), so
47/// configuring them changes nothing. The list exists only as input to our own
48/// accessor codegen.
49///
50/// Known gap, deliberately left alone:
51/// `.iota.grpc.v1.types.TypeTagVector.inner_type` is boxed by prost and missing
52/// here. No accessor is generated for it, so the omission is inert — adding it
53/// would change generated output.
54const PROST_BOXED_FIELDS: &[&str] = &[
55    ".iota.grpc.v1.filter.EventFilter.negation",
56    ".iota.grpc.v1.filter.TransactionFilter.negation",
57    ".iota.grpc.v1.filter.NotEventFilter.filter",
58    ".iota.grpc.v1.filter.NotTransactionFilter.filter",
59    ".iota.grpc.v1.types.TypeTag.vector_tag",
60];
61
62/// Fields whose `MessageFields::FIELDS` must stop recursing, because the field
63/// closes a reference cycle.
64const BOXED_TYPES_FIELD_INFO: &[&str] = &[
65    ".iota.grpc.v1.filter.AllEventFilter.filters",
66    ".iota.grpc.v1.filter.AnyEventFilter.filters",
67    ".iota.grpc.v1.filter.NotEventFilter.filter",
68    ".iota.grpc.v1.filter.AllTransactionFilter.filters",
69    ".iota.grpc.v1.filter.AnyTransactionFilter.filters",
70    ".iota.grpc.v1.filter.NotTransactionFilter.filter",
71    ".iota.grpc.v1.types.TypeTagVector.inner_type",
72];
73
74/// Fields whose accessors take and return boxed values: the ones prost boxes,
75/// plus those where a boxed signature is simply more ergonomic.
76const BOXED_TYPES_ACCESSOR: &[&str] = &[
77    ".iota.grpc.v1.filter.EventFilter.negation",
78    ".iota.grpc.v1.filter.TransactionFilter.negation",
79    ".iota.grpc.v1.filter.NotEventFilter.filter",
80    ".iota.grpc.v1.filter.NotTransactionFilter.filter",
81    ".iota.grpc.v1.types.TypeTag.vector_tag",
82];

84/// Messages whose automatic `Eq`/`Hash` derives are stripped after generation.
85///
86/// prost derives `Eq, Hash` for every message whose fields are all hashable.
87/// For these BCS carriers the digest is the identity, so the derives add
88/// nothing; dropping them now keeps a later decoded `json`
89/// (`google.protobuf.Value`, which holds an `f64` and is not hashable) field
90/// from silently removing the derives and breaking downstream `Eq`/`Hash` uses.
91///
92/// Only the intended roots are listed; [`eq_hash_strip_closure`] extends the
93/// set to every message that transitively contains one, since those would
94/// otherwise still derive `Eq`/`Hash` around a now-non-`Eq` field and fail to
95/// compile.
96///
97/// Entries are fully-qualified proto names so a message is addressed by its own
98/// package, never by a bare name that another package could share.
99const STRIP_EQ_HASH_MESSAGES: &[&str] = &[
100    "iota.grpc.v1.transaction.Transaction",
101    "iota.grpc.v1.transaction.TransactionEffects",
102];

104fn main() {
105    let root_dir = PathBuf::from(std::env!("CARGO_MANIFEST_DIR"));
106
107    let proto_dir = root_dir
108        .join("../iota-sdk-grpc-types/proto")
109        .canonicalize()
110        .unwrap();
111    let out_dir = root_dir
112        .join("../iota-sdk-grpc-types/src/proto")
113        .canonicalize()
114        .unwrap()
115        .join("generated");
116
117    // The codegen entry points take owned paths, so widen the lists once here.
118    let prost_boxed_fields = owned_paths(PROST_BOXED_FIELDS);
119    let boxed_types_field_info = owned_paths(BOXED_TYPES_FIELD_INFO);
120    let boxed_types_accessor = owned_paths(BOXED_TYPES_ACCESSOR);
121
122    reset_out_dir(&out_dir);
123
124    let proto_files = discover_proto_files(&proto_dir);
125    let descriptor_pool = compile_proto_files(&proto_dir, &proto_files);
126
127    generate_tonic(&proto_dir, &proto_files, &out_dir);
128    add_license_headers(&out_dir);
129    remove_eq_hash_derives(
130        &out_dir,
131        &descriptor_pool,
132        &eq_hash_strip_closure(&descriptor_pool, STRIP_EQ_HASH_MESSAGES),
133    );
134
135    codegen::generate_service_methods::generate_service_method_paths(&descriptor_pool, &out_dir);
136
137    let context = Context::build(descriptor_pool);
138
139    codegen::accessors::generate_accessors(
140        &context,
141        &out_dir,
142        &prost_boxed_fields,
143        &boxed_types_accessor,
144    );
145
146    codegen::generate_fields::generate_field_info(&context, &out_dir, &boxed_types_field_info);
147
148    verify_generated_files_committed(&out_dir);
149}

151fn owned_paths(paths: &[&str]) -> Vec<String> {
152    paths.iter().map(|path| (*path).to_owned()).collect()
153}
154
155/// Clears the output directory so that removed protos cannot leave stale
156/// generated files behind.
157fn reset_out_dir(out_dir: &Path) {
158    if out_dir.exists() {
159        std::fs::remove_dir_all(out_dir).unwrap();
160    }
161    std::fs::create_dir_all(out_dir).unwrap();
162}

164fn discover_proto_files(proto_dir: &Path) -> Vec<PathBuf> {
165    let proto_ext = std::ffi::OsStr::new("proto");
166    let mut proto_files = walkdir::WalkDir::new(proto_dir)
167        .into_iter()
168        .filter_map(|entry| {
169            (|| {
170                let entry = entry?;
171                if entry.file_type().is_dir() {
172                    return Ok(None);
173                }
174
175                let path = entry.into_path();
176                if path.extension() != Some(proto_ext) {
177                    return Ok(None);
178                }
179
180                Ok(Some(path))
181            })()
182            .transpose()
183        })
184        .collect::<Result<Vec<_>, walkdir::Error>>()
185        .unwrap();
186    // `walkdir` traversal order depends on the OS (and is non-deterministic on
187    // Linux), so sort the paths to keep `compile_protos` output stable across
188    // runs — without this, messages from different .proto files in the same
189    // package can swap positions in the generated module.
190    proto_files.sort();
191
192    proto_files
193}

195fn compile_proto_files(proto_dir: &Path, proto_files: &[PathBuf]) -> DescriptorPool {
196    let mut compiler_init = protox::Compiler::new([proto_dir]).unwrap();
197    let compiler = compiler_init
198        .include_source_info(true)
199        .include_imports(true)
200        .open_files(proto_files)
201        .unwrap();
202
203    compiler.descriptor_pool()
204}
205
206fn generate_tonic(proto_dir: &Path, proto_files: &[PathBuf], out_dir: &Path) {
207    tonic_prost_build::configure()
208        .build_client(true)
209        .build_server(true)
210        .bytes(".")
211        .message_attribute(".iota.grpc", "#[non_exhaustive]")
212        .enum_attribute(".iota.grpc", "#[non_exhaustive]")
213        .message_attribute(".google.rpc", "#[non_exhaustive]")
214        .enum_attribute(".google.rpc", "#[non_exhaustive]")
215        .btree_map(".")
216        .out_dir(out_dir)
217        .compile_protos(proto_files, &[proto_dir.to_path_buf()])
218        .unwrap();
219}

221/// Adds IOTA license headers to the tonic-generated files and rewrites the
222/// `super::google` paths prost emits, which do not resolve from the module the
223/// generated files are included into.
224fn add_license_headers(out_dir: &Path) {
225    let google_import_regex = regex::Regex::new(r"(?:super::)+google").unwrap();
226
227    for entry in std::fs::read_dir(out_dir).unwrap() {
228        let entry = entry.unwrap();
229        let path = entry.path();
230
231        // ignore google proto files
232        if path.to_str().unwrap().contains("google") {
233            continue;
234        }
235
236        if path.extension().and_then(|s| s.to_str()) == Some("rs")
237            && !path
238                .file_name()
239                .unwrap()
240                .to_str()
241                .unwrap()
242                .contains("field_info")
243        {
244            let mut content = std::fs::read_to_string(&path).unwrap();
245
246            // Add license header if missing
247            if !content.starts_with("// Copyright") {
248                content = format!(
249                    "// Copyright (c) Mysten Labs, Inc.\n// Modifications Copyright (c) 2026 IOTA Stiftung\n// SPDX-License-Identifier: Apache-2.0\n\n{content}"
250                );
251            }
252
253            // Replace all occurrences of super::google with crate::google (any number of
254            // super::)
255            content = google_import_regex
256                .replace_all(&content, "crate::google")
257                .to_string();
258            std::fs::write(&path, content).unwrap();
259        }
260    }
261}

263/// Expands the configured roots to every message that transitively contains one
264/// of them through a message-typed field.
265///
266/// A container that currently derives `Eq`/`Hash` must lose them too, otherwise
267/// the derive would require `Eq`/`Hash` on a field whose type no longer has
268/// them. Walking the containment graph here keeps [`STRIP_EQ_HASH_MESSAGES`] to
269/// the intended roots instead of a hand-maintained list of every container.
270fn eq_hash_strip_closure(pool: &DescriptorPool, roots: &[&str]) -> BTreeSet<String> {
271    // Reverse containment edges, keyed by fully-qualified name: message -> the
272    // messages that hold it in a field.
273    let mut containers: HashMap<String, Vec<String>> = HashMap::new();
274    for message in pool.all_messages() {
275        for field in message.fields() {
276            if let Kind::Message(field_type) = field.kind() {
277                containers
278                    .entry(field_type.full_name().to_owned())
279                    .or_default()
280                    .push(message.full_name().to_owned());
281            }
282        }
283    }
284
285    let mut closure = BTreeSet::new();
286    let mut pending: Vec<String> = roots
287        .iter()
288        .map(|root| {
289            assert!(
290                pool.get_message_by_name(root).is_some(),
291                "STRIP_EQ_HASH_MESSAGES names `{root}`, which is not a message in the proto set"
292            );
293            (*root).to_owned()
294        })
295        .collect();
296    while let Some(message) = pending.pop() {
297        if closure.insert(message.clone())
298            && let Some(parents) = containers.get(&message)
299        {
300            pending.extend(parents.iter().cloned());
301        }
302    }
303    closure
304}

306/// Removes `Eq` and `Hash` from the derive list of each given message.
307///
308/// See [`STRIP_EQ_HASH_MESSAGES`] for why. Each message is resolved through the
309/// descriptor pool to its own generated file (`<package>.rs`) and struct name,
310/// so a name shared by another package cannot be matched by mistake. Within
311/// that file the derive directly above the struct is rewritten: `Eq` and `Hash`
312/// are dropped and every other derive is kept, whatever the set or order. A
313/// message that no longer derives them (for example once it gains a
314/// non-hashable field) is left unchanged. The struct must appear exactly once
315/// in the file, so a nested type sharing the name fails the build rather than
316/// having the wrong struct stripped.
317fn remove_eq_hash_derives(out_dir: &Path, pool: &DescriptorPool, messages: &BTreeSet<String>) {
318    for message in messages {
319        let descriptor = pool
320            .get_message_by_name(message)
321            .expect("closure members are message names taken from the descriptor pool");
322        let struct_name = descriptor.name();
323        let path = out_dir.join(format!("{}.rs", descriptor.package_name()));
324
325        let derive = regex::Regex::new(&format!(
326            r"#\[derive\(([^)]*)\)\]\npub struct {} \{{",
327            regex::escape(struct_name)
328        ))
329        .unwrap();
330
331        let content = std::fs::read_to_string(&path)
332            .unwrap_or_else(|e| panic!("reading {} for `{message}`: {e}", path.display()));
333
334        let captures = derive.captures_iter(&content).collect::<Vec<_>>();
335        assert!(
336            captures.len() == 1,
337            "expected exactly one `pub struct {struct_name}` with a derive in {}, found {}; \
338             `{message}` needs module-aware matching",
339            path.display(),
340            captures.len()
341        );
342
343        let kept = captures[0][1]
344            .split(',')
345            .map(str::trim)
346            .filter(|derive| !matches!(*derive, "" | "Eq" | "Hash"))
347            .collect::<Vec<_>>()
348            .join(", ");
349        let replacement = format!("#[derive({kept})]\npub struct {struct_name} {{");
350
351        let content = derive.replace(&content, regex::NoExpand(&replacement));
352        std::fs::write(&path, content.as_bytes()).unwrap();
353    }
354}

356/// Fails the build when a run changed the generated files, which are committed.
357fn verify_generated_files_committed(out_dir: &Path) {
358    let status = std::process::Command::new("git")
359        .arg("diff")
360        .arg("--exit-code")
361        .arg("--")
362        .arg(out_dir)
363        .status();
364    match status {
365        Ok(status) if !status.success() => {
366            eprintln!("Generated protobuf files have uncommitted changes. Please commit them.");
367            std::process::exit(2); // Custom exit code for uncommitted changes
368        }
369        Err(error) => panic!("failed to run `git diff`: {error}"),
370        Ok(_) => {}
371    }
