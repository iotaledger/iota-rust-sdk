13/// Information about a transparent wrapper message (one that should be skipped
14/// in read_mask paths / field path builders).
15pub(crate) struct TransparentInfo {
16    /// True when the single inner field is a map field.
17    pub is_map: bool,
18    /// Full protobuf type name of the inner message (e.g.
19    /// ".iota.grpc.v1.event.Event"), None for maps.
20    pub inner_full_type_name: Option<String>,
21}
22
23/// Parses the descriptor pool to find all messages with the
24/// `field_mask_transparent` option set to true. Returns a map keyed by full
25/// protobuf type name (e.g. `.iota.grpc.v1.event.Events`) with information
26/// about the inner field that should be surfaced through the transparent
27/// wrapper.
28pub fn parse_transparent_messages_from_pool(
29    pool: &prost_reflect::DescriptorPool,
30) -> HashMap<String, TransparentInfo> {
31    let mut map = HashMap::new();
32
33    let transparent_ext = match pool.get_extension_by_name("iota.grpc.field_mask_transparent") {
34        Some(ext) => ext,
35        None => return map,
36    };
37
38    for message in pool.all_messages() {
39        let options = message.options();
40        if !options.has_extension(&transparent_ext) {
41            continue;
42        }
43
44        let is_transparent = options
45            .get_extension(&transparent_ext)
46            .as_bool()
47            .unwrap_or(false);
48
49        if !is_transparent {
50            continue;
51        }
52
53        // The wrapper must contain exactly one field.
54        let fields: Vec<_> = message.fields().collect();
55        if fields.len() != 1 {
56            eprintln!(
57                "Warning: transparent message {} has {} fields, expected 1 — skipping",
58                message.full_name(),
59                fields.len()
60            );
61            continue;
62        }
63
64        let inner_field = fields[0].clone();
65
66        // Detect whether the inner field is a map field (a repeated field whose
67        // message type has map_entry = true).
68        let is_map = match inner_field.kind() {
69            Kind::Message(ref msg) => msg.is_map_entry(),
70            _ => false,
71        };
72
73        let inner_full_type_name = match inner_field.kind() {
74            Kind::Message(ref msg) if !is_map => Some(format!(".{}", msg.full_name())),
75            _ => None,
76        };
77
78        let full_type_name = format!(".{}", message.full_name());
79        map.insert(
80            full_type_name,
81            TransparentInfo {
82                is_map,
83                inner_full_type_name,
84            },
85        );
86    }
87
88    map
89}

91/// What the generator needs to turn a field's protobuf type into Rust, shared
92/// by every message of the package being generated.
93struct PackageContext<'a> {
94    /// Package being generated, e.g. `iota.grpc.v1.command`.
95    package: &'a str,
96    context: &'a Context,
97    boxed_types: &'a [String],
98}
99
100impl PackageContext<'_> {
101    fn message_path(&self, full_type_name: &str) -> TokenStream {
102        self.context
103            .message_path(&self.context.resolve(full_type_name), self.package)
104    }
105
106    fn builder_path(&self, full_type_name: &str) -> TokenStream {
107        self.context
108            .builder_path(&self.context.resolve(full_type_name), self.package)
109    }
110
111    fn is_map_entry(&self, full_type_name: &str) -> bool {
112        self.context.resolve(full_type_name).is_map_entry()
113    }
114
115    fn transparent(&self, full_type_name: &str) -> Option<&TransparentInfo> {
116        self.context.transparent(full_type_name)
117    }
118
119    /// True when prost boxes the field, which the field constants have to
120    /// mirror by not recursing into it.
121    fn is_boxed_field(&self, message: &MessageDescriptor, field: &FieldDescriptor) -> bool {
122        let field_path = format!("{}.{}", message.full_name(), field.name());
123        self.boxed_types
124            .iter()
125            .any(|boxed_path| boxed_path.trim_start_matches('.') == field_path)
126    }
127}

129pub(crate) fn generate_field_info(context: &Context, out_dir: &Path, boxed_types: &[String]) {
130    for package in context.packages() {
131        if package.contains("google") {
132            continue;
133        }
134
135        let package_context = PackageContext {
136            package: &package,
137            context,
138            boxed_types,
139        };
140
141        let stream = generate_field_info_for_all_messages(
142            &package_context,
143            &context.messages_in_declaration_order(&package),
144        );
145
146        // Only generate file if there's actual content in the stream
147        if stream.is_empty() {
148            continue;
149        }
150
151        // Types of other packages are referenced through their full module path,
152        // so nothing but the traits has to be imported here.
153        let code = quote! {
154            mod _field_impls {
155                #![allow(clippy::wrong_self_convention)]
156
157                use super::*;
158
159                use crate::field::MessageFields;
160                use crate::field::MessageField;
161
162                #stream
163            }
164
165            pub use _field_impls::*;
166        };
167
168        let ast: syn::File = syn::parse2(code).expect("not a valid tokenstream");
169        let code = prettyplease::unparse(&ast);
170
171        // Add IOTA license header
172        let mut buf = String::new();
173        buf.push_str("// Copyright (c) Mysten Labs, Inc.\n");
174        buf.push_str("// Modifications Copyright (c) 2026 IOTA Stiftung\n");
175        buf.push_str("// SPDX-License-Identifier: Apache-2.0\n");
176        buf.push('\n');
177        buf.push_str(&code);
178
179        let file_name = format!("{package}.field_info.rs");
180        std::fs::write(out_dir.join(file_name), &buf).unwrap();
181    }
182}

184// Helper function to recursively generate field info for all messages including
185// nested ones
186fn generate_field_info_for_all_messages(
187    context: &PackageContext<'_>,
188    messages: &[MessageDescriptor],
189) -> TokenStream {
190    let mut stream = TokenStream::new();
191
192    // First pass: Generate nested modules first so they're defined before being
193    // used
194    for message in messages {
195        // Skip map entry messages
196        if message.is_map_entry() {
197            continue;
198        }
199
200        // Generate nested modules for nested messages
201        let children: Vec<_> = message.child_messages().collect();
202        if !children.is_empty() {
203            let module_name = quote::format_ident!("{}", to_snake(message.name()));
204            let nested_content = generate_field_info_for_all_messages(context, &children);
205
206            if !nested_content.is_empty() {
207                stream.extend(quote! {
208                    pub mod #module_name {
209                        use super::*;
210
211                        #nested_content
212                    }
213                });
214            }
215        }
216    }
217
218    // Second pass: Generate top-level messages after nested modules are defined
219    for message in messages {
220        // Skip map entry messages
221        if message.is_map_entry() {
222            continue;
223        }
224
225        stream.extend(generate_field_info_for_message(context, message));
226    }
227
228    stream
229}

231fn generate_field_info_for_message(
232    context: &PackageContext<'_>,
233    message: &MessageDescriptor,
234) -> TokenStream {
235    let message_path = context.message_path(message.full_name());
236
237    let constants = generate_field_constants(context, message, &message_path);
238    let oneof_constants = generate_oneof_name_constants(message, &message_path);
239    let message_fields_impl = generate_message_fields_impl(message, &message_path);
240    let field_path_builders = generate_field_path_builders_impl(context, message, &message_path);
241
242    quote! {
243        #constants
244        #oneof_constants
245        #message_fields_impl
246        #field_path_builders
247    }
248}

250/// Generates `pub const {NAME}_ONEOF: &'static str = "{name}";` constants for
251/// each real `oneof` declaration in a message. Synthetic oneofs created by the
252/// proto3-optional feature are excluded.
253fn generate_oneof_name_constants(
254    message: &MessageDescriptor,
255    message_path: &TokenStream,
256) -> TokenStream {
257    let mut consts = TokenStream::new();
258
259    for oneof in message.oneofs().filter(|oneof| !oneof.is_synthetic()) {
260        let name = oneof.name();
261        let ident = quote::format_ident!("{}_ONEOF", name.to_ascii_uppercase());
262        consts.extend(quote! {
263            pub const #ident: &'static str = #name;
264        });
265    }
266
267    if consts.is_empty() {
268        return TokenStream::new();
269    }
270
271    quote! {
272        impl #message_path {
273            #consts
274        }
275    }
276}

278fn generate_field_constants(
279    context: &PackageContext<'_>,
280    message: &MessageDescriptor,
281    message_path: &TokenStream,
282) -> TokenStream {
283    let mut field_consts = TokenStream::new();
284
285    for field in message.fields() {
286        field_consts.extend(generate_field_constant(context, message, &field));
287    }
288
289    quote! {
290        impl #message_path {
291            #field_consts
292        }
293    }
294}

296fn generate_message_fields_impl(
297    message: &MessageDescriptor,
298    message_path: &TokenStream,
299) -> TokenStream {
300    let mut field_refs = TokenStream::new();
301
302    for field in message.fields() {
303        field_refs.extend(generate_field_reference(&field));
304    }
305
306    // Collect real (non-synthetic) oneof names for the ONEOFS constant.
307    let real_oneof_names: Vec<String> = message
308        .oneofs()
309        .filter(|oneof| !oneof.is_synthetic())
310        .map(|oneof| oneof.name().to_owned())
311        .collect();
312
313    let oneofs_impl = if real_oneof_names.is_empty() {
314        TokenStream::new()
315    } else {
316        let names = real_oneof_names.iter().map(|name| {
317            quote! { #name, }
318        });
319        quote! {
320            const ONEOFS: &'static [&'static str] = &[
321                #(#names)*
322            ];
323        }
324    };
325
326    quote! {
327        impl MessageFields for #message_path {
328            const FIELDS: &'static [&'static MessageField] = &[
329                #field_refs
330            ];
331            #oneofs_impl
332        }
333    }
334}

336fn generate_field_constant(
337    context: &PackageContext<'_>,
338    message: &MessageDescriptor,
339    field: &FieldDescriptor,
340) -> TokenStream {
341    let descriptor = field.field_descriptor_proto();
342    let ident = quote::format_ident!("{}_FIELD", field.name().to_ascii_uppercase());
343    let name = field.name();
344    let json_name = field.json_name();
345    let number = descriptor.number();
346
347    // Check if the field is optional in the proto definition
348    let is_proto3_optional = descriptor.proto3_optional.unwrap_or(false);
349
350    let (is_map, message_fields) = match field.kind() {
351        // we skip google types
352        Kind::Message(_) if !descriptor.type_name().contains("google") => {
353            let full_type_name = descriptor.type_name();
354
355            // Check for circular references that need to be broken:
356            // 1. Self-reference
357            // 2. Map entry types
358            // 3. Fields that are boxed AND create circular dependencies in the message
359            //    graph
360            let is_circular_reference = context.is_boxed_field(message, field)
361                && context
362                    .context
363                    .has_reference_cycle(message.full_name(), full_type_name);
364
365            if full_type_name.trim_start_matches('.') == message.full_name()
366                || is_circular_reference
367            {
368                (quote! { false }, quote! { None })
369            } else if context.is_map_entry(full_type_name) {
370                (quote! { true }, quote! { None })
371            } else if let Some(info) = context.transparent(full_type_name) {
372                // Field points to a transparent wrapper: flatten through it.
373                if info.is_map {
374                    // The wrapper's inner field is a map → treat this field as a map directly.
375                    (quote! { true }, quote! { None })
376                } else if let Some(inner_full_name) = &info.inner_full_type_name {
377                    // The wrapper's inner field is a repeated message → use the inner type's
378                    // FIELDS.
379                    let inner = context.message_path(inner_full_name);
380                    (quote! { false }, quote! { Some(#inner::FIELDS) })
381                } else {
382                    (quote! { false }, quote! { None })
383                }
384            } else {
385                let field_message = context.message_path(full_type_name);
386                (quote! { false }, quote! { Some(#field_message::FIELDS) })
387            }
388        }
389        _ => (quote! { false }, quote! { None }),
390    };
391
392    quote! {
393        pub const #ident: &'static MessageField = &MessageField {
394            name: #name,
395            json_name: #json_name,
396            number: #number,
397            is_optional: #is_proto3_optional,
398            is_map: #is_map,
399            message_fields: #message_fields,
400        };
401    }
402}

404fn generate_field_reference(field: &FieldDescriptor) -> TokenStream {
405    let ident = quote::format_ident!("{}_FIELD", field.name().to_ascii_uppercase());
406
407    quote! {
408        Self::#ident,
409    }
410}

412fn generate_field_path_builders_impl(
413    context: &PackageContext<'_>,
414    message: &MessageDescriptor,
415    message_path: &TokenStream,
416) -> TokenStream {
417    let builder_ident = quote::format_ident!("{}FieldPathBuilder", message.name());
418
419    let mut field_chain_methods = TokenStream::new();
420
421    for field in message.fields() {
422        field_chain_methods.extend(generate_field_chain_methods(
423            context,
424            message,
425            message_path,
426            &field,
427        ));
428    }
429
430    quote! {
431        impl #message_path {
432            pub fn path_builder() -> #builder_ident {
433                #builder_ident::new()
434            }
435        }
436
437        pub struct #builder_ident {
438            path: Vec<&'static str>,
439        }
440
441        impl #builder_ident {
442            #[allow(clippy::new_without_default)]
443            pub fn new() -> Self {
444                Self {
445                    path: Default::default(),
446                }
447            }
448
449            #[doc(hidden)]
450            pub fn new_with_base(base: Vec<&'static str>) -> Self {
451                Self { path: base }
452            }
453
454            pub fn finish(self) -> String {
455                self.path.join(".")
456            }
457
458            #field_chain_methods
459        }
460    }
461}

463fn generate_field_chain_methods(
464    context: &PackageContext<'_>,
465    message: &MessageDescriptor,
466    message_path: &TokenStream,
467    field: &FieldDescriptor,
468) -> TokenStream {
469    let descriptor = field.field_descriptor_proto();
470    let field_const = quote::format_ident!("{}_FIELD", field.name().to_ascii_uppercase());
471    let name = if field.name() == "type" {
472        quote::format_ident!("r#{}", field.name())
473    } else {
474        quote::format_ident!("{}", field.name())
475    };
476
477    let leaf_method = quote! {
478        pub fn #name(mut self) -> String {
479            self.path.push(#message_path::#field_const.name);
480            self.finish()
481        }
482    };
483
484    // we need to ignore google types, because we don't generate builders for them
485    if !matches!(field.kind(), Kind::Message(_)) || descriptor.type_name().contains("google") {
486        return leaf_method;
487    }
488
489    let full_type_name = descriptor.type_name();
490
491    if full_type_name.trim_start_matches('.') == message.full_name()
492        || context.is_map_entry(full_type_name)
493    {
494        return leaf_method;
495    }
496
497    // A field pointing at a transparent wrapper chains into the wrapped type.
498    let target_type_name = match context.transparent(full_type_name) {
499        Some(info) => match &info.inner_full_type_name {
500            // The wrapper wraps a map, so the field is a leaf.
501            _ if info.is_map => return leaf_method,
502            Some(inner_full_name) => inner_full_name.as_str(),
503            None => return leaf_method,
504        },
505        None => full_type_name,
506    };
507
508    let return_type = context.builder_path(target_type_name);
509
510    quote! {
511        pub fn #name(mut self) -> #return_type {
512            self.path.push(#message_path::#field_const.name);
513            #return_type::new_with_base(self.path)
514        }
515    }
